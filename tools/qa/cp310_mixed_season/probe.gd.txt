extends "res://qa.gd"
const Mixed=Store.Owner.Mixed
const PACK := "6cff977a06ce5271bb245ef3747bb4ce26d9d2d3d0cf00d5914112072d7392da"
const PAID := "3271bb28f1e63d1836c981b9996da9ff192aded0e8f7750f63a18eaf159e141c"
const UNPAID := "ef3d4e3ea1d8bc899d7c16eb60d45345a9b6019fb03cc3d698d0847bde47b72e"
const WIRE := "6db193e8d0e55c2aae15b4d76ab859dd3d26574ca0f0ef52d8d1d5d6a9a3a487"

func _run() -> void:
	var args:=OS.get_cmdline_user_args()
	if not check(args.size()==3 and args[0]=="cp310" and args[1] in ["pack","paid","settle"]
			and Store.LeagueAdmission.starter_player() and not Store.cross_league_preview()
			and not OS.has_feature("editor") and not FileAccess.file_exists(ORIGINAL)
			and not FileAccess.file_exists(Store.PATH+".bak") and not FileAccess.file_exists(Store.PATH+".tmp"),"isolated no-env native primary only"): return
	var mode: String=args[1]
	var raw:=Store.raw()
	if not check(args[2]=={"pack":PACK,"paid":PAID,"settle":UNPAID}[mode]
			and Store.Owner._hash(raw)==args[2],"exact recovered raw fixture"): return
	var native: Script=Adapter.Ordinary
	var before: int=native.restore_calls
	var app:=App.new()
	add_child(app)
	await get_tree().process_frame
	printerr("CP310_STAGE cold ",mode)
	var tick:=Time.get_ticks_usec()
	app.reload_save()
	printerr("CP310_STAGE cold_done usec=",Time.get_ticks_usec()-tick)
	if not check(app._owned() and Mixed.season_owner_scope(app.career_owner) and native.restore_calls-before==3
			and Store.raw()==raw and not FileAccess.file_exists(Store.PATH+".bak")
			and app.career_owner.common.aggregate.currency==183 and app.career_owner.common.aggregate.titles.is_empty()
			and app.league_entry_state("equipment")=="locked"
			and app.season.career.start_basis.archive_origin.manifest_sha256==WIRE,"current native replay, three strict predecessors and no write"): return
	var s: SeasonState=app.season
	if mode=="pack":
		var buy: Dictionary=s.build._events[-1]
		if not check(buy.op=="pack_open" and s.round_index==1 and s.results.size()==3 and s.cash()==10
				and s.build.pack_pending() and not Mixed.game_entry_scope(app.career_owner),"paid unresolved stock pack reopens, no launch"): return
		if not check(not app.commit_shop(buy) and Store.raw()==raw and not FileAccess.file_exists(Store.PATH+".bak"),"duplicate debit rejected on fresh owner"): return
		await _finish(app)
		return
	if mode=="settle":
		if not check(s.round_index==1 and s.results.size()==3 and s.cash()==10
				and s.physical.pending.reports.size()==2,"original actual two reports remain unpaid"): return
		printerr("CP310_STAGE settle")
		app.resume_round()
		if not check(Store.Owner._hash(Store.raw())==PAID and FileAccess.get_file_as_bytes(Store.PATH+".bak")==raw
				and app.retained==null and not is_instance_valid(app.ai_runner),"same native settlement digest, exact backup, no replayed physics"): return
		s=app.season
	var paid:=Store.raw()
	var backup:=FileAccess.get_file_as_bytes(Store.PATH+".bak")
	if not check(s.round_index==2 and s.results.size()==6 and s.cash()==22 and s.physical.reports.size()==4
			and s.physical.pending.is_empty() and Store.Owner.Common._starter_dugout_rewards_match(s)
			and Mixed.game_entry_scope(app.career_owner) and s.pending_fixture().id==6,"6 authentic results, 12 rewards, exact next native game available"): return
	app.resume_round()
	if not check(Store.raw()==paid and FileAccess.get_file_as_bytes(Store.PATH+".bak")==backup
			and not FileAccess.file_exists(Store.PATH+".tmp"),"no duplicate settle or writes"): return
	if mode=="paid":
		var history:=Store.standard_history(app.career_owner,paid)
		if not check(history.size()==3 and history.map(func(x): return x.results.size())==[33,33,33]
				and history.map(func(x): return x.receipt.total)==[72,39,72],"99 older scores and exact earned receipts"): return
		var replay: SeasonState=Adapter.OrdinaryRuntime.create_common(s.career.start_basis,"dugout")
		if not check(replay!=null,"detached exact draft for negative"): return
		for id: String in s.picks:
			if not check(replay.choose_player(id),"exact authored pick in negative"): return
		var altered: Dictionary=Store.Owner._bytes_value(paid).ordinary
		altered.physical.reports[0].request="f".repeat(64)
		altered=Store.Owner._bytes_value(Store._native_encoded(altered))
		if not check(Mixed._replay_current(replay,altered)==null and Store.raw()==paid and native.restore_calls-before==3
				and not FileAccess.file_exists(Store.PATH+".bak"),"changed current request refused with no extra prior reads or writes"): return
	await _finish(app)
