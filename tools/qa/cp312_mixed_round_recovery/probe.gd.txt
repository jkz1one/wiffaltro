extends "res://qa.gd"
const PAID := "bb8df3bf660fbcdf4b519277c951e6230f60841178d07eb2b40a0f0eaae61372"
const UNPAID := "7483fac70bd35fba8c0a005c165af448d3a23ea592dc18b94fe3a8d97becf841"

func _button(root: Node,label: String) -> Button:
	if root is Button and root.text==label and root.is_visible_in_tree(): return root
	for child: Node in root.get_children():
		var result:=_button(child,label)
		if result!=null: return result
	return null

func _exact(raw: PackedByteArray,backup: PackedByteArray) -> bool:
	return Store.raw()==raw and FileAccess.get_file_as_bytes(Store.PATH+".bak")==backup 		and not FileAccess.file_exists(Store.PATH+".tmp")

func _run() -> void:
	var mode:=OS.get_cmdline_user_args()[1]
	var raw:=Store.raw()
	var backup:=FileAccess.get_file_as_bytes(Store.PATH+".bak")
	var sha:=Store.Owner._hash(raw)
	if not check(not backup.is_empty() and not raw.is_empty() and (mode=="corrupt" or sha==(PAID if mode=="worker" else UNPAID))
			and Store.LeagueAdmission.starter_fast_worker() and not FileAccess.file_exists(ORIGINAL)
			and _exact(raw,backup),"exact isolated official native input"): return
	var app:=App.new()
	add_child(app)
	await get_tree().process_frame
	var native: Script=Adapter.Ordinary
	var old_calls: int=native.restore_calls
	printerr("CP312_STAGE attempt ",mode)
	if mode=="worker":
		var click:=_button(app,"CONTINUE SAVED CAREER")
		if not check(click!=null,"real packaged Continue control"): return
		click.pressed.emit()
		await get_tree().process_frame
		if not check(app.mixed_continue_running() and _button(app,"CANCEL CAREER CHECK")!=null,"responsive worker issued"): return
		var start:=Time.get_ticks_msec()
		while app.mixed_continue_running() and Time.get_ticks_msec()-start<180000:
			await get_tree().process_frame
		if not check(native.restore_calls==old_calls and not app.mixed_continue_running(),"no parent historical replay"): return
	else:
		app.reload_save()
	if mode=="corrupt":
		if not check(not app._owned() and app.career_owner==null and _exact(raw,backup)
				and native.restore_calls-old_calls==3,"changed request refused using real native restore; bytes untouched"): return
		await _finish(app)
		return
	if not check(app._owned() and app.season.league_rules.get("league")=="dugout"
			and app.career_owner.common.aggregate.currency==183 and _exact(raw,backup)
			and app.season.build.pack_pending()==false,"native owned byte-inert restore, prior upgrade retained"): return
	if mode=="pending":
		if not check(app.season.round_index==2 and app.season.results.size()==6 and app.season.cash()==22
				and app.season.physical.pending.reports.size()==2
				and native.restore_calls-old_calls==3,"terminal unpaid recovered with both original CPU reports"): return
		printerr("CP312_STAGE consume retained reports")
		app.resume_round()
		if not check(not is_instance_valid(app.ai_runner) and Store.Owner._hash(Store.raw())==PAID
				and FileAccess.get_file_as_bytes(Store.PATH+".bak")==raw,"exact original paid bytes, without replaying games"): return
	if not check(app.season.round_index==3 and app.season.results.size()==9 and app.season.player_results.size()==3
			and app.season.physical.reports.size()==6 and app.season.physical.pending.is_empty()
			and app.season.cash()==34 and app.season.pending_fixture().id==9
			and Store.Owner.Common._starter_dugout_rewards_match(app.season),"Round 3 exact results, payouts, and next fixture"): return
	var paid:=Store.raw()
	var prior:=FileAccess.get_file_as_bytes(Store.PATH+".bak")
	app.resume_round()
	if not check(_exact(paid,prior),"repeat settlement byte-inert"): return
	await _finish(app)
