extends "res://qa.gd"
const CHECKS="user://cp314-boundaries"

func _button(root: Node,label: String) -> Button:
	if root is Button and root.text==label and root.is_visible_in_tree(): return root
	for child: Node in root.get_children():
		var result:=_button(child,label)
		if result!=null: return result
	return null

func _custody(label: String) -> bool:
	var sha:=FileAccess.get_sha256(Store.PATH)
	var f:=FileAccess.open(CHECKS.path_join(label+".ready"),FileAccess.WRITE)
	if f==null: return check(false,"custody marker opens")
	f.store_string(sha)
	f.close()
	printerr("CP314_STAGE durable boundary ",label," ",sha)
	var deadline:=Time.get_ticks_msec()+30000
	while Time.get_ticks_msec()<deadline:
		if FileAccess.file_exists(CHECKS.path_join(label+".ack")):
			return check(FileAccess.get_file_as_string(CHECKS.path_join(label+".ack"))==sha,"verified persistent custody "+label)
		await get_tree().process_frame
	return check(false,"custody acknowledgment timeout "+label)

func _run() -> void:
	var args:=OS.get_cmdline_user_args()
	var mode:String=args[1]
	var base_round:int=int(args[2])
	var first:int=base_round*3
	var pin:String=args[3]
	if not check(args[2]==str(base_round) and base_round>=2 and base_round<=8 and pin.length()==64,"bounded regular round and exact input pin"): return
	var raw:=Store.raw()
	var backup:=FileAccess.get_file_as_bytes(Store.PATH+".bak")
	if not check(not raw.is_empty() and Store.Owner._hash(raw)==pin
			and Store.LeagueAdmission.starter_fast_worker() and not FileAccess.file_exists(ORIGINAL)
			and not FileAccess.file_exists(Store.PATH+".tmp"),"isolated official Starter slot"): return
	var app:=App.new()
	add_child(app)
	await get_tree().process_frame
	var native: Script=Adapter.Ordinary
	var old_calls:int=native.restore_calls
	if mode=="cold":
		printerr("CP314_STAGE cold exact native App reopen")
		var start:=Time.get_ticks_msec()
		app.reload_save()
		printerr("CP314_STATUS cold elapsed_ms=",Time.get_ticks_msec()-start," owned=",app._owned()," native_restore_calls=",native.restore_calls-old_calls)
	else:
		var click:=_button(app,"CONTINUE SAVED CAREER")
		if not check(click!=null,"real Continue control"): return
		click.pressed.emit()
		await get_tree().process_frame
		if not check(app.mixed_continue_running() and _button(app,"CANCEL CAREER CHECK")!=null,"responsive worker issued"): return
		printerr("CP314_STAGE responsive Continue")
		var start:=Time.get_ticks_msec()
		while app.mixed_continue_running() and Time.get_ticks_msec()-start<325000:
			await get_tree().process_frame
		printerr("CP314_STATUS continue elapsed_ms=",Time.get_ticks_msec()-start," outstanding=",app.mixed_continue_running(),
			" owned=",app._owned()," parent_restore_calls=",native.restore_calls-old_calls)
		if not check(not app.mixed_continue_running(),"responsive worker terminal within QA observation"): return
		if not check(native.restore_calls==old_calls,"zero parent historical replay"): return
	if not check(app._owned() and not FileAccess.file_exists(Store.PATH+".tmp") and app.season.league_rules.get("league")=="dugout"
			and app.career_owner.common.aggregate.currency==183 and Store.raw()==raw
			and FileAccess.get_file_as_bytes(Store.PATH+".bak")==backup,"native owned byte-inert restore"): return
	if mode=="cold":
		if not check(app.season.round_index==base_round+1 and app.season.results.size()==first+3 and app.season.player_results.size()==base_round+1
				and app.season.physical.reports.size()==2*(base_round+1) and app.season.pending_fixture().id==first+3 and app.season.physical.pending.is_empty(),"cold settled regular round and exact next fixture"): return
		app.resume_round()
		if not check(Store.raw()==raw and FileAccess.get_file_as_bytes(Store.PATH+".bak")==backup,"settlement not repeated on cold reopen"): return
		await _finish(app)
		return
	if not check(app.season.round_index==base_round and app.season.results.size()==first and app.season.player_results.size()==base_round and app.season.physical.reports.size()==base_round*2
			and not app.season.build.pack_pending() and app.season.physical.pending.is_empty() and app.season.pending_fixture().id==first,"exact earned regular baseline and next human fixture"): return
	printerr("CP314_STAGE physical human ",first)
	var action:=await _play(app)
	if not check(action=="human-%d"%first and FileAccess.get_file_as_bytes(Store.PATH+".bak")==raw,"authentic human saved against baseline"): return
	if not await _custody(action): return
	for n: int in [first+1,first+2]:
		var prior:=Store.raw()
		printerr("CP314_STAGE native NPC ",n)
		action=await _resume(app)
		if not check(action=="npc-%d"%n and FileAccess.get_file_as_bytes(Store.PATH+".bak")==prior,"authentic NPC "+str(n)+" and exact backup"): return
		if not await _custody(action): return
	var unpaid:=Store.raw()
	printerr("CP314_STAGE settle retained authored reports")
	action=await _resume(app)
	if not check(action=="settle-%d"%(first+3) and app.season.round_index==base_round+1 and FileAccess.get_file_as_bytes(Store.PATH+".bak")==unpaid,"single native settlement with exact unpaid backup"): return
	if not await _custody(action): return
	var paid:=Store.raw()
	app.resume_round()
	if not check(Store.raw()==paid and FileAccess.get_file_as_bytes(Store.PATH+".bak")==unpaid,"live repeat settlement byte-inert"): return
	await _finish(app)
