extends Node
## Actual human-side automated gameplay plus two real AI games, no score fixtures.
const Report = preload("res://src/gameplay/season/successor_league_v1/physical_report.gd")
const Capture = preload("res://src/presentation/season/successor_owner_live_v1/capture_human.gd")

const Runtime = preload("res://src/gameplay/season/successor_owner_live_v1/runtime.gd")
const Codec = preload("res://src/gameplay/season/successor_owner_live_v1/codec.gd")
const Save = preload("res://src/gameplay/season/successor_owner_live_v1/save.gd")
const App = preload("res://src/presentation/season/successor_owner_live_v1/app.gd")
const Fixtures = preload("res://src/gameplay/season/successor_owner_live_v1/physical_fixtures.gd")
const Shop = preload("res://src/presentation/season/successor_owner_live_v1/shop_window.gd")
const Progress = preload("res://src/gameplay/season/successor_career_v1/career_progress.gd")
const OldRuntime = preload("res://src/gameplay/season/successor_owner_v1/runtime.gd")
const OldSave = preload("res://src/gameplay/season/successor_owner_v1/save.gd")

func _ready() -> void:
	call_deferred("run")

func check(ok: bool, label: String) -> bool:
	if not ok:
		printerr("FAIL: ", label)
		get_tree().quit(1)
	else:
		print("CHECK: ", label)
	return ok

func starts_valid(season: SeasonState) -> bool:
	var base: ClubCareer = ClubCareer.new()
	var build: SeasonBuild = season.build
	return build._gear_progress.enabled and build._sponsor_progress.enabled \
		and ClubCareer.same(build._gear_progress.start, base.gear_counts()) \
		and ClubCareer.same(build._sponsor_progress.start, base.sponsor_state()) \
		and build._order_start == base.order_access() and build._rain_start == base.rain_access() \
		and build._transfer_start == base.transfer_access() and build._supply_start == base.supply_count() \
		and build._checkout_start == SeasonLateCheckout.access(base) \
		and build._association_start == SeasonAssociation.access(base) \
		and build._freezer_start == SeasonFreezers.access(base) \
		and build._sides_start == SeasonLeftRight.access(base) \
		and build._jump_start == SeasonJumpstart.access(base) \
		and build._sure_start == SeasonSureShot.access(base) \
		and build._major.start == SeasonDoubleMajor.access(base) \
		and build._abilities.start == SeasonAbilities.access(base) \
		and build._copy.start == SeasonCarbonCopy.access(base) \
		and build._field_start == SeasonFieldSupply.access(base) \
		and build._batch_start == SeasonSmallBatch.access(base) \
		and build._book._exotic_start == []

func owner_valid(season: SeasonState) -> bool:
	if season == null or season.career == null or not season.league_career or season.career.current != 1 or season.career.runs.size() != 1 or not starts_valid(season):
		return false
	var row: Dictionary = season.career.runs[0]
	var build: SeasonBuild = season.build
	return row.id == 1 and row.seed == season.season_seed and row.tier == 0 and row.league == "dugout" \
		and row.status == "active" and row.receipt.is_empty() \
		and ClubCareer.same(row.proof, ClubSeasonRecord.capture(season)) \
		and ClubCareer.same(row.player_progress, Progress.capture(season)) \
		and ClubCareer.same(row.gear, build._gear_progress.games) \
		and ClubCareer.same(row.sponsors, build._sponsor_progress.games) \
		and ClubCareer.same(row.collection, ClubCollection.acquired(build)) \
		and row.order_rerolls == build._paid_rerolls and row.rain_earned == build._rain_earned \
		and row.transfer_earned == build._transfer_earned and row.supplies_used == build._supply_used \
		and row.checkout_earned == build._checkout_earned and row.association_earned == build._association_earned \
		and row.freezer_earned == build._freezer_earned and row.sides_earned == build._sides_earned \
		and row.jump_earned == build._jump_earned and row.sure_earned == build._sure_earned \
		and row.major_earned == build._major.earned and row.sky_outs == build._abilities.earned \
		and row.copy_earned == build._copy.earned and row.field_outs == build._field_outs \
		and ClubCareer.same(row.batch_used, build._batch_used) \
		and ClubCareer.same(season.career._assets, ClubAssetLedger.new(season.season_seed, [0], [], {}).to_data())

func retain(label: String) -> bool:
	return DirAccess.make_dir_recursive_absolute("user://" + label) == OK and DirAccess.copy_absolute(Save.PATH, "user://" + label + "/" + Save.PATH.get_file()) == OK

func original_valid() -> bool:
	return OldSave.PATH != Save.PATH and OldSave.restore() != null and FileAccess.get_sha256(OldSave.PATH) == FileAccess.get_file_as_string("user://original-owner.sha256")

func run() -> void:
	if not check(OS.get_environment("WIFFALTRO_ISOLATED_TEST") == "1", "isolated runner required"):
		return
	var app: SeasonApp = load("res://src/presentation/season/successor_owner_ui_v1/app.gd").new() if OS.get_environment("WIFFALTRO_TEST_FACADE") == "1" else App.new()
	print("APP_UNDER_TEST: ", app.get_script().resource_path)
	add_child(app)
	if OS.get_cmdline_user_args().has("reopen"):
		app.reload_save()
		if not check(app.season != null and app.season.results.size() == 3 and app.season.player_results.size() == 1, "cold settled round reopen"):
			return
		if not check(app.season.build._visit.pack_status == "used" and app.season.make_match() != null, "cold paid pack and next fixture"):
			return
		if not check(owner_valid(app.season) and original_valid(), "cold exact active career and unchanged draft-owner slot"):
			return
		if not check(ClubCareer.same(Codec.capture(app.season), JSON.parse_string(FileAccess.get_file_as_string(Save.PATH))), "cold exact published disk checkpoint"):
			return
		print("PASS: cold actual owned round and earned shop with exact career progression; no completion or Dynasty claim")
		app.queue_free()
		get_tree().quit()
		return
	var human: Array = []
	if OS.get_cmdline_user_args().has("resume-round"):
		app.reload_save()
		if not check(app.season != null and app.season.league_live and not app.season.physical.pending.is_empty() and app.season.physical.pending.reports.size() == (2 if OS.get_cmdline_user_args().has("settle-only") else (0 if OS.get_cmdline_user_args().has("first-ai-only") else 1)), "cold pending human/AI report and explicit genuine owner"):
			return
		if not check(owner_valid(app.season) and original_valid(), "cold pending genuine owner and original slot"):
			return
		human = app.season.physical.pending.human.duplicate(true)
	else:
		var old: SeasonState = OldRuntime.create_fresh(1803)
		if not check(OldSave.save(old) and OldSave.PATH != Save.PATH, "preserve genuine draft-owner slot before live career"):
			return
		var marker: FileAccess = FileAccess.open("user://original-owner.sha256", FileAccess.WRITE)
		marker.store_string(FileAccess.get_sha256(OldSave.PATH))
		marker.close()
		app.create_fresh()
		if not check(app.season != null and app.get("retained") == null, "fresh round UI creation"):
			return
		if not check(owner_valid(app.season) and app.season.picks.is_empty(), "genuine live owner and all starts before first pick"):
			return
		print("SEASON_SEED: ", app.season.season_seed)
		for pick in range(4):
			app.call("draft", app.season.offers()[0])
			if not check(app.season.picks.size() == pick + 1, "round draft %d" % pick):
				return
		if not check(owner_valid(app.season), "exact owner and all starts after draft"):
			return
		app.play_next()
		if not check(is_instance_valid(app.lab) and app.lab._match_state != null, "actual App play route opens match"):
			return
		# Let the actual intro finish before installing the automated controller.
		# Otherwise its later human-ready prompt can supersede automation startup.
		for frame in range(600):
			await get_tree().physics_frame
			if not app.lab._match_presentation_director.blocks_gameplay():
				break
		if not check(not app.lab._match_presentation_director.blocks_gameplay(), "actual match intro completes"):
			return
		# Test driver uses the ordinary live controllers for both sides; no outcome edits.
		app.lab._automation = MatchAutomation.new()
		app.lab._automation.seed = Fixtures.seed_for(app.season, app.season.pending_fixture())
		app.lab._automation.presentation_enabled = true
		app.lab._automation.begin(app.lab)
		var finished: bool = false
		for frame in range(150000):
			await get_tree().physics_frame
			if app.lab._match_state.phase == MatchState.Phase.GAME_END and app.lab._automation != null and not app.lab._home_run.presentation_pending(app.lab):
				# Automation intentionally omits the human outro. Stop only the test
				# driver after the actual result, then use the normal presentation path.
				app.lab._automation = null
				PitchBatLabFeelSupport.begin_match_outro(app.lab)
			if app.lab._match_state.phase == MatchState.Phase.GAME_END and app.lab._match_presentation_director.mode == MatchPresentationDirector.Mode.OUTRO_HOLD:
				finished = true
				break
			if frame % 12000 == 0:
				print("HUMAN_GAME_PROGRESS: ", frame, " pa=", app.lab._match_state.plate_appearance_number, " phase=", app.lab._match_state.phase)
		if not check(finished, "real human-side game reaches completed outro"):
			return
		human = Capture.capture(app)
		if not check(not human.is_empty(), "actual human evidence capture"):
			return
		var evidence: FileAccess = FileAccess.open("user://round-human-evidence.json", FileAccess.WRITE)
		evidence.store_string(JSON.stringify(human, "", true, true))
		evidence.close()
		app.finish_game()
		if not check(app.get("retained") == null and not is_instance_valid(app.lab) and not app.season.physical.pending.is_empty(), "completed human game durable before AI jobs"):
			return
		if not check(app.season.results.is_empty() and app.season.build.cash() == 0, "pending round gives no premature reward"):
			return
		if not check(owner_valid(app.season) and original_valid(), "pending human preserves genuine owner without premature progression"):
			return
		if not check(retain("retained-pending-0"), "retain exact pending human evidence"):
			return
		if OS.get_cmdline_user_args().has("human-only"):
			print("PASS: actual owned Dugout human game durably pending; no premature reward")
			app.queue_free()
			get_tree().quit()
			return
	# Each real report is durably adopted before explicitly requesting the next job.
	for expected in [1, 2]:
		if app.season.physical.pending.reports.size() >= expected:
			continue
		app.call("resume_round")
		var arrived: bool = false
		for frame in range(150000):
			await get_tree().physics_frame
			if not app._busy:
				arrived = true
				break
			if frame % 12000 == 0:
				print("AI_GAME_PROGRESS: ", expected, ":", frame)
		if not check(arrived and app.get("retained") == null and app.season.physical.pending.get("reports", []).size() == expected, "actual AI report checkpoint %d: %s" % [expected, app.notice]):
			return
		if not check(app.season.physical.pending.reports[-1].report.version == 30, "explicit successor League physical report30"):
			return
		if not check(owner_valid(app.season), "pending AI preserves genuine owner %d" % expected):
			return
		if not check(retain("retained-pending-%d" % expected), "retain exact pending AI evidence %d" % expected):
			return
		var historical: Dictionary = app.season.physical.pending.reports[-1].report.duplicate(true)
		historical.version = 27
		if not check(not Report.valid(historical, app.season.build.pitch_catalog_signature()), "historical report family cannot replace current League proof"):
			return
		if expected == 1 and OS.get_cmdline_user_args().has("first-ai-only"):
			print("PASS: actual first AI report durable with genuine owner and unchanged progression")
			app.queue_free()
			get_tree().quit()
			return
	if OS.get_cmdline_user_args().has("second-ai-only"):
		print("PASS: actual second AI report durable before cold settlement")
		app.queue_free()
		get_tree().quit()
		return
	app.call("resume_round")
	if not check(app.get("retained") == null and app.season.results.size() == 3 and app.season.player_results.size() == 1 and app.season.physical.pending.is_empty(), "three-game live Dugout round settlement"):
		return
	if not check(owner_valid(app.season) and app.season.career.runs[0].player_progress.games.size() == 1, "settled actual human progress and every career field"):
		return
	if not check(retain("retained-settled-before-shop"), "retain exact settled evidence"):
		return
	var human_results: Array = app.season.results.filter(func(row: Dictionary) -> bool: return row.id == human[0])
	if not check(human_results.size() == 1, "exact actual human result"):
		return
	var earned: int = 18 if SeasonState._winner(human_results[0]) == 0 else 12
	if not check(app.season.build._bank.view().rewards.size() == 1 and app.season.build.cash() == earned, "one score-matched earned human reward"):
		return
	if not check(app.season.opponents.clubs.size() == 5, "all five AI clubs retained"):
		return
	for club: Dictionary in app.season.opponents.clubs.values():
		if not check(club.build._bank.view().rewards.size() == 1, "one reward per AI club"):
			return
	var settled: Dictionary = Codec.capture(app.season)
	app.call("resume_round")
	if not check(not settled.is_empty() and ClubCareer.same(Codec.capture(app.season), settled), "duplicate round resume is inert"):
		return
	var build: SeasonBuild = app.season.build
	app.call("shop_command", {"id":"open:%d"%build.revision(),"rev":build.revision(),"op":"open"})
	build = app.season.build
	if not check(build._visit.open and not build._visit.cards.is_empty(), "generated earned shop opens"):
		return
	var cash_before: int = build.cash()
	var price: int = build.pack_price()
	var open_command: Dictionary = {"id":"pack:%d"%build.revision(),"rev":build.revision(),"op":"pack_open"}
	app.call("shop_command", open_command)
	if not check(app.get("retained") == null and app.season.build.pack_pending() and app.season.build.cash() == cash_before - price, "real earned pack payment"):
		return
	build = app.season.build
	var selected: Dictionary = {}
	for item: String in build._visit.cards:
		var targets: Array[Dictionary] = build.targets(item)
		if not targets.is_empty():
			selected = {"id":"pick:%d"%build.revision(),"rev":build.revision(),"op":"pack_pick","item":item,"player":targets[0].player,"pitch":targets[0].pitch,"replace":targets[0].replace}
			break
	if not check(not selected.is_empty(), "generated legal pack target"):
		return
	app.open_shop()
	var window: SeasonShopWindow
	for child: Node in app.menu.get_children():
		if child.get_script() == Shop:
			window = child
	if not check(window != null and window.theme != null and window._scroll != null and window._confirm != null and window._body.get_child_count() > 0, "actual owned full shop window and pack targets"):
		return
	app.open_shop()
	var windows: int = 0
	for child: Node in app.menu.get_children():
		if child is SeasonShopWindow:
			windows += 1
	if not check(windows == 1, "owned shop repeated open remains singleton"):
		return
	var before_profile: Dictionary = build.player(selected.player).duplicate(true)
	var book_revision: int = build._book.revision()
	var before_selection: Dictionary = Codec.capture(app.season)
	window._preview(selected, "Isolated actual earned pack selection")
	if not check(not window._pending.is_empty() and ClubCareer.same(Codec.capture(app.season), before_selection), "owned shop confirmation preview spends nothing"):
		return
	window._confirm.hide()
	var saved_bytes: PackedByteArray = FileAccess.get_file_as_bytes(Save.PATH)
	if not check(DirAccess.make_dir_absolute(Save.PATH + ".tmp") == OK, "isolated earned-shop save failure"):
		return
	window._commit()
	if not check(app.get("retained") != null and ClubCareer.same(Codec.capture(app.season), before_selection) and FileAccess.get_file_as_bytes(Save.PATH) == saved_bytes, "failed purchase save preserves published owner and slot"):
		return
	var exact_candidate: SeasonState = app.get("retained")
	var exact_snapshot: Dictionary = Codec.capture(exact_candidate)
	if not check(not exact_snapshot.is_empty() and not window._notice.begins_with("Saved."), "failed shop never reports success and retains exact owned candidate"):
		return
	if not check(DirAccess.remove_absolute(Save.PATH + ".tmp") == OK, "release isolated save-failure fixture"):
		return
	var retry_button: Button
	for child: Node in window._body.get_children():
		if child is Button and child.text == "RETRY EXACT SAVE":
			retry_button = child
	if not check(retry_button != null, "retry remains reachable in exclusive owned shop"):
		return
	retry_button.pressed.emit()
	if not check(app.get("retained") == null and app.season == exact_candidate and ClubCareer.same(Codec.capture(app.season), exact_snapshot) and owner_valid(app.season), "exact earned purchase retry atomically promotes career"):
		return
	var after_profile: Dictionary = app.season.build.player(selected.player)
	var stats_delta: int = 0
	var mastery_delta: int = 0
	for stat: String in before_profile.stats:
		stats_delta += int(after_profile.stats[stat]) - int(before_profile.stats[stat])
	for pitch: String in after_profile.mastery:
		mastery_delta += int(after_profile.mastery[pitch]) - int(before_profile.mastery.get(pitch, 0))
	if not check(stats_delta >= 0 and mastery_delta >= 0 and stats_delta + mastery_delta == 1 and app.season.build._book.revision() == book_revision + 1, "exactly one earned development level applied"):
		return
	var purchased: Dictionary = Codec.capture(app.season)
	if not check(not purchased.is_empty() and app.get("retained") == null and app.season.build._visit.pack_status == "used", "paid pack selection durably applied"):
		return
	app.call("shop_command", selected)
	if not check(ClubCareer.same(Codec.capture(app.season), purchased), "repeated purchase command grants nothing twice"):
		return
	if not check(owner_valid(app.season) and original_valid(), "paid development retained in genuine career with no permanent credit and original slot unchanged"):
		return
	for change in ["receipt", "progress", "start", "owner"]:
		var wrong: Dictionary = purchased.duplicate(true)
		match change:
			"receipt": wrong.checkpoint.career.runs[0].receipt = {"total": 100}
			"progress": wrong.checkpoint.career.runs[0].player_progress.games = []
			"start": wrong.checkpoint.player_access.append("player.invalid")
			"owner": wrong.checkpoint.career.current = 2
		if not check(Codec.restore(wrong) == null and ClubCareer.same(Codec.capture(app.season), purchased), "owned round rejects forged " + change):
			return
	print("PASS: actual owned human+two AI games settled once, exact career progression, earned pack and save retry; no completion or Dynasty claim")
	app.queue_free()
	await get_tree().process_frame
	get_tree().quit()
