const START="f54c3ee3b99bbd419cf614630b49f0bb06af53eb87c9d3ffaf090b21bab43330"

class SelectiveContact extends BatterApproachModel:
	var native_power:int=0
	var centered_power:int=0
	var locked_power:int=0
	var converted:int=0
	var delivered_conversions:int=0
	var delivered_swings:int=0
	var changed_plan:bool=false

	func plan_swing(pitch: PitchDefinition,position: Vector3,velocity: Vector3,
			batter: PlayerDefinition,balls: int,strikes: int,decision_seed: int,
			batting_hand: int,elapsed: float) -> Dictionary:
		var intent: Dictionary=super.plan_swing(pitch,position,velocity,batter,balls,strikes,decision_seed,batting_hand,elapsed)
		changed_plan=false
		if not intent.swing or not intent.use_power: return intent
		native_power+=1
		# Existing estimate derived from current/past visible motion. No
		# target, outcome, umpire, future solver state or opponent inputs.
		var read: Vector2=intent.plate_read
		if locked_swing!=&"":
			locked_power+=1
		elif strikes<2 and absf(read.x)<=0.19 and absf(read.y-ZONE_CENTER_Y_M)<=0.23:
			centered_power+=1
		else:
			intent.use_power=false
			changed_plan=true
			converted+=1
		return intent

	func track_pitch(pitch: PitchDefinition,state: PitchState,batter: PlayerDefinition,balls: int,strikes: int,
			decision_seed: int,batting_hand: int,contact_profile: SwingProfileDefinition,power_profile: SwingProfileDefinition,
			disclosed: Dictionary={}) -> Dictionary:
		var decision: Dictionary=super.track_pitch(pitch,state,batter,balls,strikes,decision_seed,batting_hand,
			contact_profile,power_profile,disclosed)
		# Observational only. Controller still submits the native decision.
		if not decision.is_empty() and decision.swing:
			delivered_swings+=1
			if changed_plan and not decision.use_power: delivered_conversions+=1
		return decision

func _pending(app: Node) -> bool:
	var reference:=FileAccess.get_file_as_bytes(REFERENCE)
	var witness: Variant=JSON.parse_string(FileAccess.get_file_as_string(RAW))
	if Store.Owner._hash(reference)!=START or not witness is Dictionary or witness.source!=START: return false
	var a: Dictionary=Store.Owner._bytes_value(reference)
	var after: Dictionary=Store.Owner._bytes_value(Store.raw())
	var pending: Dictionary=after.ordinary.physical.pending
	if pending.get("reports")!=[] or not Store.Codec.same(JSON.parse_string(JSON.stringify(app.season.physical.pending.get("human"),"",true,true)),witness.human) or pending.human[0]!=31:
		return false
	after.ordinary.physical.pending={}
	return app.season.results.size()==30 and Store.Codec.same(a,after)

func _button(node: Node,label: String) -> Button:
	if node.is_queued_for_deletion(): return null
	if node is Button and node.text==label and node.is_visible_in_tree(): return node
	for child: Node in node.get_children():
		var found:=_button(child,label)
		if found!=null: return found
	return null
