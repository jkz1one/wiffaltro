 static func digest(bytes:PackedByteArray) -> String:
  var hasher=HashingContext.new()
  if hasher.start(HashingContext.HASH_SHA256)!=OK or hasher.update(bytes)!=OK:return ""
  return hasher.finish().hex_encode()

 func _ai_finished(report:Dictionary, runner:Node=null) -> void:
  emitted_at=Time.get_ticks_msec()
  print("CP127_STAGE admission original_runner_callback")
  if is_instance_valid(runner) and runner==ai_runner and _busy and _npc_owner_matches() and fixture_id in [19,20] and _paired_npc_binding.get("fixture")==fixture_id and not report.is_empty():
   var file=FileAccess.open(evidence_path(),FileAccess.WRITE)
   if file!=null:
    # Test-local forensic output only; preserves native types and never grants authority.
    file.store_var({"fixture":fixture_id,"request":_paired_npc_binding.request,"primary":digest(_npc_expected),"backup":digest(FileAccess.get_file_as_bytes(Store.PATH+".bak")),"report":report},false)
    file.flush()
    file.close()
    preserved=FileAccess.file_exists(evidence_path())
  print("CP127_REPORT_RETAINED ",preserved)
  super._ai_finished(report,runner)
  admitted_at=Time.get_ticks_msec()

 func _ai_failed(reason:String, runner:Node=null) -> void:
  injury=reason
  print("CP127_RUNNER_FAILED ",reason)
  super._ai_failed(reason,runner)

func unpaid_signed(season:SeasonState) -> bool:
 var frame:Dictionary=season.physical.pending
 if not Binding.bound_shape(frame) or frame.human[0]!=18 or not frame.human[3].has(RECRUIT) or frame.human[3].has(RELEASED):return false
 var pristine:SeasonState=season.fork_live()
 pristine.physical.pending.clear()
 return signed_exact(pristine)

func opening_pair(season:SeasonState,fixture:Dictionary,side:int) -> Dictionary:
 var club:int=fixture.away if side==0 else fixture.home
 var last:Array=season.results.filter(func(row):return row.round==5 and (row.away==club or row.home==club))
 if last.size()!=1 or not last[0].doubleheader_evidence[0 if last[0].away==club else 1].pair.debt.values().any(func(n):return n>0):return {}
 if SeasonDoubleheaderRules.first(int(fixture.id))!=-1:return {}
 return {"game":int(fixture.id),"first":-1,"debt":{}}

func npc_debt(season:SeasonState,row:Dictionary,opening:Array) -> bool:
 if not Live.whole(row.fixture,19,20):return false
 var fixture:Dictionary=Authored.authored_phase(6).npc[int(row.fixture)-19]
 for side:int in range(2):
  var expected:Dictionary=opening_pair(season,fixture,side)
  if expected.is_empty() or not ClubCareer.same(row.report.pair[side],expected):return false
  for player:String in row.report.workload[side].initial:
   var entry:Dictionary=row.report.workload[side].initial[player]
   if not PhysicalRecoveryFlow.near(entry.initial,entry.capacity):return false
   if not opening.is_empty() and not PhysicalRecoveryFlow.near(entry.initial,opening[side][player]):return false
 return true
