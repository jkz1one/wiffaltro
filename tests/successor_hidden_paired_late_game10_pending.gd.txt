 static func digest(bytes:PackedByteArray) -> String:
  var hasher=HashingContext.new()
  if hasher.start(HashingContext.HASH_SHA256)!=OK or hasher.update(bytes)!=OK:return ""

func game10_debt(season:SeasonState,fixture:Dictionary,side:int) -> Dictionary:
 var club:int=int(fixture.away if side==0 else fixture.home)
 var first:int=SeasonDoubleheaderRules.first(int(fixture.id))
 var played:Array=season.results.filter(func(row):return row.id==first and (row.away==club or row.home==club))
 if played.size()!=1:return {}
 var envelope:Dictionary=played[0].doubleheader_evidence[0 if played[0].away==club else 1]
 var workload=preload("res://src/gameplay/season/successor_league_general_v2/physical_workload.gd")
 var debt:Dictionary=workload.net_debt(envelope.proof,envelope.pair.debt)
 if not debt.values().any(func(n):return n>0):return {}

func game10_rosters(season:SeasonState,state:MatchState,fixture:Dictionary,proofs:Array) -> bool:
 if not Live.same(fixture,Authored.authored_phase(9).human):return false
 var team:TeamMatchState=state.home_team
 var roster:Array[String]=season.build.roster()
 if not Live.same(team.roster.map(func(player):return String(player.definition.id)),roster) or roster[1]!=RECRUIT or roster.has(RELEASED):return false
 if not Live.same(team.roster[1].definition.fielding,RecruitCatalog.fresh(RECRUIT,"late").stats.fielding) or team.roster[0].definition.fielding!=SeasonPlayerCatalog.profile("player.mika_reed").stats.fielding+1:return false
 if proofs.is_empty() and (team.current_fielder()!=team.roster[1] or not Live.same(team.tactics.held,SeasonTacticalCatalog.held(season.build.view().wallet))):return false
 for side:int in range(2):
  team=state.away_team if side==0 else state.home_team
  var expected:Dictionary=game10_debt(season,fixture,side)
  if expected.is_empty() or not ClubCareer.same(team.league_pair,expected):return false
  if not proofs.is_empty() and not ClubCareer.same(proofs[side].pair,expected):return false
  for player:PlayerMatchState in team.roster:
   var id:String=String(player.definition.id)
   var start:float=maxf(0.0,player.stamina_max-float(expected.debt.get(id,0.0)))
   if proofs.is_empty() and not PhysicalRecoveryFlow.near(player.stamina_remaining,start):return false
   if not proofs.is_empty() and (not proofs[side].proof.initial.has(id) or not PhysicalRecoveryFlow.near(proofs[side].proof.initial[id].initial,start)):return false
 return true
