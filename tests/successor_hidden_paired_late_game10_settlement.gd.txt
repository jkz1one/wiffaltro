func run() -> void:
 var args=OS.get_cmdline_user_args()

 var app=Game10ObservedApp.new()
 get_tree().root.add_child(app)
 var bytes:PackedByteArray=Store.raw()
 var backup:PackedByteArray=FileAccess.get_file_as_bytes(Store.PATH+".bak")
 var gates:int=Live.paired_history_calls
 var replays:int=PairHistory.Collector.replay_calls
 print("CP127_STAGE cold")
 var started:int=Time.get_ticks_msec()
 app.reload_save()
 print("G10_SETTLE_COLD_MS=",Time.get_ticks_msec()-started," replays=",PairHistory.Collector.replay_calls-replays," historical=",PairHistory.historical_humans_checked)
 var mode:String=args[0]
 if mode.begins_with("tamper"):
  check(app.season==null and exact(bytes,backup),"corrupt "+mode+" refuses without writes")
 elif check(app.season!=null and app._owned() and exact(bytes,backup),"fresh owned cacheless process"):
  if mode=="settle":
   if check(history(gates,replays,9) and unpaid_game10(app.season),"original unpaid 27/28/29; exactly one historical batch"):
    settle_game10(app,bytes,backup)
  elif mode=="paid":
   check(history(gates,replays,10) and paid_game10(app.season),"paid sealed bracket independently checks every receipt and score")
   sealed_semis(app,bytes,backup)
  else:check(false,"unexpected proof mode")
 app.free()
 if failures==0:print("CP127_PASS")
 get_tree().quit(0 if failures==0 else 1)

 var owner=app.career_owner
 var source:SeasonState=app.season
 var pending:Dictionary=source.physical.pending.duplicate(true)
 if not check(Live.same(pending.reports.map(func(row):return row.fixture),[28,29]) and game10_npc_debt(source,pending.reports[0]) and game10_npc_debt(source,pending.reports[1]),"exact real complete pending, paired carry"):
  return
 var owner_pin:Dictionary=Store.RawGuard.capture(owner)
 check(Store._prepare_paired_settlement(owner.fork(),bytes,9)==null and Store._prepare_paired_settlement(owner,backup,9)==null,"foreign and stale settlement rejected")
 var obstruction:String=ProjectSettings.globalize_path(Store.PATH+".tmp")
 if not check(DirAccess.make_dir_absolute(obstruction)==OK,"isolated write obstruction"):return
 var gates:int=Live.paired_history_calls
 var replays:int=PairHistory.Collector.replay_calls
 var resolutions:int=Live.Physical.resolve_calls
 print("CP127_STAGE candidate")
 var started:int=Time.get_ticks_msec()
 app._settle_paired_late_regular()
 print("G10_SETTLEMENT_CANDIDATE_MS=",Store.paired_candidate_ms," ADMISSION_MS=",Store.paired_admission_ms," TOTAL_MS=",Time.get_ticks_msec()-started)
 if not check(app.retained!=null and app.retained.get_script()==Live.Runtime,"owned exact concrete sealed paid season prepared"):return
 var target:SeasonState=app.retained
 check(Live.paired_history_calls-gates==1 and PairHistory.Collector.replay_calls-replays==12 and Live.Physical.resolve_calls-resolutions==2 and PairHistory.historical_humans_checked==10,"one pending batch, one independent paid batch, only two existing regular reports resolved")
 check(paid_game10(target) and Live.same(target.results.slice(0,27),source.results) and Live.same(target.player_results.slice(0,9),source.player_results) and Live.same(target.physical.reports,source.physical.reports+pending.reports),"retains exact history and binds results to sealed end")
 check(Live.same(target.player_results[-1].performance,pending.human[3]) and Live.same(target.player_results[-1].pitching_current,pending.human[9]) and Live.same(target.player_results[-1].doubleheader_evidence,pending.human[13]),"paid human proof remains exact")
 app._settle_paired_late_regular()
 check(app.retained==target and app.season==source and app.career_owner==owner and Store.RawGuard.matches(owner,owner_pin) and exact(bytes,backup),"held previous owner and both files survive; no duplicate")
 if not check(DirAccess.remove_absolute(obstruction)==OK,"remove only isolated obstruction"):return
 gates=Live.paired_history_calls
 replays=PairHistory.Collector.replay_calls
 print("CP127_STAGE retry")
 started=Time.get_ticks_msec()
 app.retry_save()
 print("G10_SETTLEMENT_RETRY_MS=",Time.get_ticks_msec()-started)
 if not check(app.retained==null and app.season==target and Live.paired_history_calls==gates and PairHistory.Collector.replay_calls==replays and FileAccess.get_file_as_bytes(Store.PATH+".bak")==bytes and Store.RawGuard.matches(owner,owner_pin),"held retry commits exact one authorized payout, no second validation"):return
 sealed_semis(app,Store.raw(),bytes)

func paid_game10(season:SeasonState) -> bool:
 if not Live._sealed_regular_end_shape(season) or not Binding._unplayed_semifinals(season):return false
 if not Live.same(season.playoff_seeds,[0,4,1,3]) or not Live.same(season.semifinals,[{"id":30,"round":10,"away":3,"home":0},{"id":31,"round":10,"away":1,"home":4}]):return false
 var own:SeasonBuild=season.build
 if own.roster()[1]!=RECRUIT or own.roster().has(RELEASED) or not Live.same(own.roster_for_game(27),own.roster()) or own.player("player.mika_reed").stats.fielding!=SeasonPlayerCatalog.profile("player.mika_reed").stats.fielding+1:return false
 for old:Dictionary in season.player_results.slice(0,6):
  if not old.club_roster.has(RELEASED) or old.club_roster.has(RECRUIT):return false
 var last:Dictionary=season.player_results[-1]
 if last.id!=27 or not last.club_roster.has(RECRUIT) or not last.performance.has(RECRUIT):return false
 var receipts:Array=[]
 for club:int in range(6):
  var build:SeasonBuild=own if club==0 else season.opponents.clubs[str(club)].build
  var rows:Array=season.results.slice(27).filter(func(row):return row.away==club or row.home==club)
  if rows.size()!=1:return false
  var result:Dictionary=rows[0]
  var game:int=int(result.id)
  var win:bool=(result.away_runs>result.home_runs)==(result.away==club)
  var paid:Array=build._events.filter(func(event):return event.get("op")=="successor_reward" and Reward.unwrap(event).get("game")==game)
  if paid.size()!=1 or Reward.unwrap(paid[0]).get("win")!=win:return false
  var bank:SeasonOwnership=build._bank.empty_fork()
  var payout:int=-1
  var sweep:int=-1
  for event:Dictionary in build._bank._events:
   var before:int=bank.cash()
   var before_rewards:Dictionary=bank.view().rewards.duplicate(true)
   var receipt:Dictionary=bank.commit(event)
   if not receipt.ok or receipt.replayed:return false
   if event.get("game")!=game:continue
   if event.op!="reward" or event.id!="game:%d"%game or event.win!=win or payout!=-1:return false
   sweep=SeasonDoubleheaderRules.bonus(season.league_rules,game,win,before_rewards)
   payout=bank.cash()-before
   if club==0 and (before!=122 or own.cash()!=before+payout):return false
  if payout!=(18 if win else 12)+sweep or sweep<0 or not Live.same(bank.view(),build._bank.view()):return false
  receipts.append([club,game,win,payout,sweep])
 print("G10_NATIVE_RECEIPTS ",JSON.stringify(receipts)," HUMAN_CASH=",own.cash())
 print("G10_FINAL_STANDINGS ",JSON.stringify(season.standings())," BRACKET=",JSON.stringify(season.semifinals))
 return receipts.size()==6
