func played_debt(season:SeasonState, row:Dictionary, opening:Array=[]) -> bool:
 var fixture:Dictionary={}
 for authored:Dictionary in SeasonDoubleheaderRules.schedule():
  if authored.id==row.fixture:fixture=authored
 if fixture.is_empty():return false
 var first:int=SeasonDoubleheaderRules.first(fixture.id)
 if first not in [1,2]:return false
 var prior:Dictionary=season.results[first]
 for side:int in range(2):
  var club:int=int(fixture.away if side==0 else fixture.home)
  var historical:Dictionary=prior.doubleheader_evidence[0 if prior.away==club else 1]
  var debt:Dictionary=Work.net_debt(historical.proof,historical.pair.debt)
  var report:Dictionary=row.report
  # Match the frozen Report.matches JSON correspondence, not raw float delta.
  if not ClubCareer.same(report.pair[side],{"game":fixture.id,"first":first,"debt":debt}):return false
  if not debt.values().any(func(n):return n>0):return false
  for player:String in report.workload[side].initial:
   var entry:Dictionary=report.workload[side].initial[player]
   var reduced:float=maxf(0.0,float(entry.capacity)-float(debt.get(player,0.0)))
   if not PhysicalRecoveryFlow.near(entry.initial,reduced):return false
   if not opening.is_empty() and not PhysicalRecoveryFlow.near(entry.initial,opening[side][player]):return false
 return true

 return settled(historical)
