func _button(node: Node,label: String) -> Button:
	if node.is_queued_for_deletion(): return null
	if node is Button and node.text==label and node.is_visible_in_tree(): return node
	for child: Node in node.get_children():
		var hit:=_button(child,label)
		if hit!=null: return hit
	return null
func need(ok: bool,label: String) -> bool:
	if check(ok,label): return true
	get_tree().quit(1)
	return false

func _prior(app: App,original: Dictionary) -> bool:
	var s: SeasonState=app.season
	var current: Dictionary=Store.Owner._bytes_value(Store.raw())
	var materialized: Dictionary=current.ordinary.duplicate(true)
	materialized.physical.pending={}
	return Live.same(materialized,original.ordinary) and s.round_index==3 and s.build.cash()==54 and Live.unpurchased_next_reverse_kind(s)=="browsed-visit3-pitching-bought" \
		and Live.same(s.build._bank.view().rewards,{"0":true,"3":true,"6":true}) and Live.same(s.results,original.ordinary.state.results) \
		and Live.same(s.player_results,original.ordinary.state.player_results) and Live.same(s.physical.reports,original.ordinary.physical.reports) \
		and Live.same(app.career_owner.common,original.common)
func _pending(app: App) -> bool:
	var pending: Dictionary=app.season.physical.pending
	return Pair.bound_shape(pending) and pending.human[0]==9 and pending.reports.is_empty()
func _reverse_debt(season: SeasonState,state: MatchState,fixture: Dictionary,proof: Array) -> bool:
	var prior: Dictionary=season.results[6]
	if prior.id!=6 or prior.away!=fixture.home or prior.home!=fixture.away: return false
	var teams: Array=[state.away_team,state.home_team]
	for side: int in range(2):
		var team: TeamMatchState=teams[side]
		var first: Dictionary=prior.doubleheader_evidence[1-side]
		var debt: Dictionary=Work.net_debt(first.proof,first.pair.debt)
		if not debt.values().any(func(value): return float(value)>0.0): return false
		if not Live.same(team.league_pair,{"game":9,"first":6,"debt":debt}): return false
		if not proof.is_empty() and not Live.same(proof[side].pair,team.league_pair): return false
		for player: PlayerMatchState in team.roster:
			var id:=String(player.definition.id)
			var available:=maxf(0.0,player.stamina_max-float(debt.get(id,0.0)))
			if player.pitch_count!=0 or player.pitching_finished or player.stamina_max<=0.0 \
					or not is_equal_approx(player.stamina_remaining,available): return false
			if not proof.is_empty():
				var entry: Dictionary=proof[side].proof.initial[id]
				if not is_equal_approx(entry.initial,available) or not is_equal_approx(entry.capacity,player.stamina_max): return false
	return true


func _pitch_evidence(human: Array) -> bool:
	var key: String="player.mika_reed"
	var away: Dictionary=human[13][0]
	var start: Dictionary=away.proof.initial[key]
	var debt: float=float(away.pair.debt[key])
	if not is_equal_approx(start.capacity,191.16) or absf(debt-161.50694100062051)>1e-9 \
			or absf(float(start.initial)-maxf(0.0,float(start.capacity)-debt))>1e-9: return false
	var count:=0
	var first_before:=0.0
	var elapsed_paid:=0.0
	for pitch: Dictionary in away.proof.events:
		if pitch.player!=key or pitch.kind!="release": continue
		if count==0: first_before=pitch.before
		if pitch.paid<=0.0 or pitch.after>=pitch.before: return false
		count+=1
		elapsed_paid+=float(pitch.paid)
	print("PAID_PITCH_MIKA_DEBT ",debt," capacity=",start.capacity," starting_available=",start.initial,
		" first_release_before=",first_before," releases=",count," release_cost=",elapsed_paid)
	return count>0 and absf(first_before-float(start.initial))<1e-9 and elapsed_paid>0.0
