func played_reset(season:SeasonState,row:Dictionary,opening:Array=[]) -> bool:
 var fixture:Dictionary={}
 for authored:Dictionary in SeasonDoubleheaderRules.schedule():
  if authored.id==row.fixture:fixture=authored
 if fixture.is_empty() or fixture.round!=4 or SeasonDoubleheaderRules.first(fixture.id)!=-1:return false
 for side:int in range(2):
  var club:int=int(fixture.away if side==0 else fixture.home)
  var prior:Array=season.results.filter(func(item):return item.round==3 and (item.away==club or item.home==club))
  if prior.size()!=1:return false
  var historic:Dictionary=prior[0].doubleheader_evidence[0 if prior[0].away==club else 1]
  if not historic.pair.debt.values().any(func(n):return n>0):return false
  var report:Dictionary=row.report
  if not ClubCareer.same(report.pair[side],{"game":fixture.id,"first":-1,"debt":{}}):return false
  for player:String in report.workload[side].initial:
   var entry:Dictionary=report.workload[side].initial[player]
   if not PhysicalRecoveryFlow.near(entry.initial,entry.capacity):return false
   if not opening.is_empty() and not PhysicalRecoveryFlow.near(entry.initial,opening[side][player]):return false
 return true
