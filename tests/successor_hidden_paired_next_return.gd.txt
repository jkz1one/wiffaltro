func unpaid_next_return(season:SeasonState) -> bool:
 if season.round_index!=3 or not Binding.bound_shape(season.physical.pending) or season.physical.pending.human[0]!=9:return false
 var paid:SeasonState=season.fork_live()
 if paid==null:return false
 paid.physical.pending.clear()

func exact_debt(season:SeasonState,fixture:Dictionary,side:int) -> Dictionary:
 var club:int=int(fixture.away if side==0 else fixture.home)
 var first:int=SeasonDoubleheaderRules.first(fixture.id)
 if first not in [6,7,8]:return {}
 var own:Dictionary=season.results[first]
 if own.away!=fixture.home or own.home!=fixture.away:return {}
 var observed:Dictionary=own.doubleheader_evidence[0 if own.away==club else 1]
 var debt:Dictionary=Work.net_debt(observed.proof,observed.pair.debt)
 var earlier:Array=season.results.filter(func(row):return row.round==1 and (row.away==club or row.home==club))
 if earlier.size()!=1:return {}
 var retired:Dictionary=earlier[0].doubleheader_evidence[0 if earlier[0].away==club else 1].pair.debt
 if not debt.values().any(func(n):return n>0) or not retired.values().any(func(n):return n>0) or ClubCareer.same(debt,retired):return {}
 return {"game":fixture.id,"first":first,"debt":debt}

func return_loadout(season:SeasonState,state:MatchState,fixture:Dictionary,proofs:Array,opening:bool=true) -> bool:
 if fixture.id!=9 or fixture.away!=0 or fixture.home!=4 or SeasonDoubleheaderRules.first(fixture.id)!=6:return false
 var picked:Dictionary=season.build._events[4]
 if picked.get("player")!="player.mika_reed":return false
 var loadout:bool=false
 for side:int in range(2):
  var team:TeamMatchState=state.away_team if side==0 else state.home_team
  var expected:Dictionary=exact_debt(season,fixture,side)
  if expected.is_empty() or not ClubCareer.same(team.league_pair,expected) or (not proofs.is_empty() and not ClubCareer.same(proofs[side].pair,expected)):return false
  for player:PlayerMatchState in team.roster:
   var id:String=String(player.definition.id)
   var initial:float=maxf(0.0,player.stamina_max-float(expected.debt.get(id,0.0)))
   if opening and not PhysicalRecoveryFlow.near(player.stamina_remaining,initial):return false
   if not proofs.is_empty() and not PhysicalRecoveryFlow.near(proofs[side].proof.initial[id].initial,initial):return false
   if side==0 and id==picked.player:
    var stat:String=season.build.shop_item(picked.item).family
    loadout=stat=="fielding" and player.definition.fielding==season.build.player(id).stats[stat] and player.definition.fielding==SeasonPlayerCatalog.profile(id).stats[stat]+1
 return loadout

func report_debt(season:SeasonState,row:Dictionary,opening:Array=[]) -> bool:
 var fixture:Dictionary={}
 for authored:Dictionary in SeasonDoubleheaderRules.schedule():
  if authored.id==row.fixture:fixture=authored
 if fixture.is_empty() or fixture.round!=3 or int(row.fixture) not in [10,11]:return false
 for side:int in range(2):
  var expected:Dictionary=exact_debt(season,fixture,side)
  if expected.is_empty() or not ClubCareer.same(row.report.pair[side],expected):return false
  for player:String in row.report.workload[side].initial:
   var entry:Dictionary=row.report.workload[side].initial[player]
   var initial:float=maxf(0.0,float(entry.capacity)-float(expected.debt.get(player,0.0)))
   if not PhysicalRecoveryFlow.near(entry.initial,initial):return false
   if not opening.is_empty() and not PhysicalRecoveryFlow.near(entry.initial,opening[side][player]):return false
 return true
