extends "res://tests/successor_hidden_paired_fourth_first_settlement.gd"

const ShopSource=preload("res://src/gameplay/season/successor_league_general_v2/shop.gd")
const OPEN7={"id":"visit7-open:19","rev":19,"op":"open"}

func run() -> void:
 var args=OS.get_cmdline_user_args()
 if OS.get_environment("WIFFALTRO_ISOLATED_TEST")!="1" or args.size()!=1:get_tree().quit(2);return
 var app=App.new()
 get_tree().root.add_child(app)
 var bytes:PackedByteArray=Store.raw()
 var backup:PackedByteArray=FileAccess.get_file_as_bytes(Store.PATH+".bak")
 var gates:int=Live.paired_history_calls
 var calls:int=PairHistory.Collector.replay_calls
 var started:int=Time.get_ticks_msec()
 print("CP127_STAGE cold")
 app.reload_save()
 print("VISIT7_COLD_MS=",Time.get_ticks_msec()-started," hist=",PairHistory.historical_humans_checked," builds=",PairHistory.Collector.replay_calls-calls)
 if args[0].begins_with("tamper"):
  check(app.season==null and exact(bytes,backup),"fresh App rejects "+args[0]+" without writing either file")
 elif check(app.season!=null and app._owned() and exact(bytes,backup) and Live.paired_history_calls-gates==1 and PairHistory.Collector.replay_calls-calls==6 and checked_paid(),"fresh App authenticates one entire paid signed history batch"):
  if args[0]=="open":
   if check(paid_exact(app.season),"exact paid cp135 entry: Game18, Dakota's real slot and closed visit7"):
    open_cutoff(app,bytes,backup)
  elif args[0]=="cold":
   check(exact_cutoff(app.season),"original seeded normal stock/pack present; visit7 has no recruit or new signing")
   rejects_at_open(app,bytes,backup)
  else:
   check(false,"unexpected mode")
 app.free()
 if failures==0:print("CP127_PASS")
 get_tree().quit(0 if failures==0 else 1)

func open_cutoff(app,bytes:PackedByteArray,backup:PackedByteArray) -> void:
 var owner=app.career_owner
 var original:SeasonState=app.season
 var pin:Dictionary=Store.RawGuard.capture(owner)
 verify_closed(app,bytes,backup)
 var fake:Dictionary=OPEN7.duplicate()
 fake.recruit={"id":"recruit:7"}
 check(Store._prepare_paired_visit7_shop_open(owner.fork(),bytes)==null and Store._prepare_paired_visit7_shop_open(owner,backup)==null and Store._prepare_paired_shop(owner,bytes,fake,7)==null and Store._prepare_paired_visit6_shop_open(owner,bytes)==null and exact(bytes,backup),"foreign owner, stale bytes, forged open and wrong visit refuse")
 var obstruction:String=ProjectSettings.globalize_path(Store.PATH+".tmp")
 if not check(DirAccess.make_dir_absolute(obstruction)==OK,"isolated exact-write obstruction"):return
 var gates:int=Live.paired_history_calls
 var calls:int=PairHistory.Collector.replay_calls
 print("CP127_STAGE candidate")
 app._open_paired_visit7_shop()
 print("VISIT7_CANDIDATE_MS=",Store.paired_shop_candidate_ms," ADMISSION_MS=",Store.paired_shop_admission_ms)
 if not check(app.retained!=null and Live.paired_history_calls-gates==1 and PairHistory.Collector.replay_calls-calls==6 and checked_paid(),"actual native fixed open retained after one full history check"):return
 var next:SeasonState=app.retained
 check(exact_cutoff(next),"natural seventh visit has ordinary stock and an exact zero-chance empty recruit")
 check(Live.same(next.build._bank._events,original.build._bank._events) and Live.same(next.build.roster(),original.build.roster()) and Live.same(next.build._contracts,original.build._contracts) and Live.same(next.build._book._events,original.build._book._events) and Live.same(next.build._game_rosters,original.build._game_rosters) and Live.same(next.results,original.results) and Live.same(next.player_results,original.player_results) and Live.same(next.physical.reports,original.physical.reports),"existing signed receipt, roster, contracts and all seven paid rounds are exact")
 app._open_paired_visit7_shop()
 check(app.retained==next and app.season==original and app.career_owner==owner and Store.RawGuard.matches(owner,pin) and exact(bytes,backup),"held duplicate and obstructed write preserve exact owner and both files")
 if not check(DirAccess.remove_absolute(obstruction)==OK,"remove only isolated obstruction"):return
 print("CP127_STAGE retry")
 gates=Live.paired_history_calls;calls=PairHistory.Collector.replay_calls
 var started:int=Time.get_ticks_msec()
 app.retry_save()
 print("VISIT7_RETRY_MS=",Time.get_ticks_msec()-started)
 if not check(app.retained==null and app.season==next and Live.paired_history_calls==gates and PairHistory.Collector.replay_calls==calls and Store.RawGuard.matches(owner,pin) and FileAccess.get_file_as_bytes(Store.PATH+".bak")==bytes,"exact held ticket writes once with no further validation"):return
 rejects_at_open(app,Store.raw(),bytes)

func exact_cutoff(season:SeasonState) -> bool:
 var b:SeasonBuild=season.build
 if season.round_index!=7 or season.results.size()!=21 or season.player_results.size()!=7 or season.player_results[-1].id!=18 or season.physical.reports.size()!=14 or not season.physical.pending.is_empty():return false
 if b.cash()!=80 or b.revision()!=20 or not Live.visit7_open_command(b._events[-1]) or not Live._signed_visit6_prefix(b):return false
 if b._market!=0 or not b._appeared or not Live.whole(b._visit.get("number"),7,7) or b._visit.get("open")!=true or not Live._visit7_recruit_cutoff(b):return false
 if RecruitCatalog.appearance_chance(7,true)!=0.0 or RecruitCatalog.appearance_chance(7,false)!=0.0 or RecruitCatalog.stage_for_visit(7)!="":return false
 var rng:RandomNumberGenerator=b._rng(1000)
 print("VISIT7_ORIGINAL_SEEDED_RECRUIT_ROLL=",rng.randf()," eligible_chance=0")
 print("VISIT7_STOCK=",b._visit.offers.size()," PACK_CARDS=",b._visit.cards.size())
 return Live.same(b._visit.offers,b._offers(0)) and Live.same(b._visit.cards,ShopSource.pack(b._book,b.roster(),b._rng(-1))) and not Live._archived_visit6_offer(b).is_empty()

func rejects_at_open(app,bytes:PackedByteArray,backup:PackedByteArray) -> void:
 var owner=app.career_owner
 var s:SeasonState=app.season
 var pin:Dictionary=Store.RawGuard.capture(owner)
 var stale:Dictionary={"id":"visit7-stale:20","rev":20,"op":"sign","offer":"recruit:6","replace":"player.mika_reed"}
 var forged:Dictionary={"id":"visit7-forgery:20","rev":20,"op":"sign","offer":"recruit:7","replace":"player.mika_reed"}
 var offered:String=String(s.build._visit.offers.keys()[0])
 var buy:Dictionary={"id":"visit7-buy:20","rev":20,"op":"buy","offer":offered}
 app._sign_paired_visit6_recruit()
 for command:Dictionary in [EXACT_SIGN,stale,forged,buy]:
  check(Store._prepare_paired_shop(owner,bytes,command,7)==null and not app.commit_shop(command),"owned/public refuses "+command.id)
 for command:Dictionary in [stale,forged]:
  var isolated:SeasonState=s.fork_live()
  check(isolated.build.candidate(command)==null,"unchanged frozen signer independently refuses "+command.id)
 check(Store._prepare_paired_visit6_sign(owner,bytes)==null and Store._prepare_paired_visit7_shop_open(owner,bytes)==null and Store._prepare_paired_visit7_shop_open(owner.fork(),bytes)==null and Store._prepare_paired_visit7_shop_open(owner,backup)==null,"old sign, repeat open, wrong owner and stale expected bytes refuse")
 app._open_paired_visit7_shop()
 app._open_paired_visit6_shop()
 sealed_paid(app,bytes,backup)
 check(app.season==s and app.career_owner==owner and Store.RawGuard.matches(owner,pin) and s.build.cash()==80 and exact(bytes,backup),"all refused actions leave primary, backup, Cash and roster untouched")
