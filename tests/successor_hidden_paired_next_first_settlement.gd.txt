func run() -> void:
 var args=OS.get_cmdline_user_args()
 if OS.get_environment("WIFFALTRO_ISOLATED_TEST")!="1" or args.size()!=1:get_tree().quit(2);return
 var app=App.new()
 get_tree().root.add_child(app)
 var bytes=Store.raw()
 var backup=FileAccess.get_file_as_bytes(Store.PATH+".bak")
 var started:int=Time.get_ticks_msec()
 var gates:int=Live.paired_history_calls
 app.reload_save()
 print("NEXT_PAID_COLD_MS=",Time.get_ticks_msec()-started," HISTORY_CALLS=",Live.paired_history_calls-gates)
 if args[0].begins_with("tamper"):
  check(app.season==null and exact(bytes,backup),"fresh App refuses "+args[0]+" without altering either file")
 elif not check(app.season!=null and exact(bytes,backup) and Live.paired_history_calls-gates==1,"fresh App authenticates exact privately owned bytes"):
  get_tree().quit(1);return
 elif args[0]=="cold":
  check(PairHistory.checked_sides==18 and PairHistory.checked_reports==6 and paid_next_first(app.season),"cold confirms all nine original outcomes, six NPC archives and eighteen paid sides")
  closed_next(app,bytes,backup)
 elif args[0]=="settle":
  settle_next(app,bytes,backup)
 else:
  check(false,"unexpected mode")
 app.free()
 if failures==0:print("HIDDEN_PAIRED_NEXT_FIRST_SETTLEMENT_PASS")
 get_tree().quit(0 if failures==0 else 1)

func settle_next(app,bytes:PackedByteArray,backup:PackedByteArray) -> void:
 var owner=app.career_owner
 var source:SeasonState=app.season
 var pending:Dictionary=source.physical.pending.duplicate(true)
 if not check(next_unpaid(source) and pending.human[0]==6 and pending.reports.size()==2 and pending.reports[0].fixture==7 and pending.reports[1].fixture==8,"genuine complete fixture 6/7/8 pending loaded, no matches rerun"):return
 var owner_raw:Dictionary=Store.RawGuard.capture(owner)
 check(Store._prepare_paired_next_first_settlement(owner.fork(),bytes)==null and Store._prepare_paired_next_first_settlement(owner,backup)==null and Store._prepare_paired_return_settlement(owner,bytes)==null and exact(bytes,backup),"foreign owner, stale primary and wrong leg refuse")
 verify_closed(app,bytes,backup)
 var obstruction:String=ProjectSettings.globalize_path(Store.PATH+".tmp")
 if not check(DirAccess.make_dir_absolute(obstruction)==OK,"isolated synthetic write obstruction"):return
 var gates:int=Live.paired_history_calls
 var started:int=Time.get_ticks_msec()
 app._settle_paired_next_first_leg()
 print("NEXT_PAID_CANDIDATE_MS=",Store.paired_candidate_ms," ADMISSION_MS=",Store.paired_admission_ms," BLOCKED_TOTAL_MS=",Time.get_ticks_msec()-started)
 if not check(app.retained!=null and app.retained.get_script()==Live.Runtime and Live.paired_history_calls-gates==1 and PairHistory.checked_sides==18 and PairHistory.checked_reports==6,"one independently validated concrete round3 candidate"):return
 var next:SeasonState=app.retained
 check(paid_next_first(next),"paid six new native receipts at 18/12 without any sweep")
 check(Live.same(next.results.slice(0,6),source.results) and Live.same(next.player_results.slice(0,2),source.player_results) and Live.same(next.physical.reports,source.physical.reports+pending.reports),"two prior legs and exact new report/request bytes retained")
 check(Live.same(next.build._events.slice(0,7),source.build._events) and Live.same(next.build._bank._events.slice(0,3),source.build._bank._events) and next.build.player("player.mika_reed").stats.fielding==source.build.player("player.mika_reed").stats.fielding,"existing upgrade and first-pair ledger unchanged")
 var result:Dictionary=next.player_results[2]
 check(Live.same([result.id,result.away_runs,result.home_runs],pending.human.slice(0,3)) and Live.same(result.pitching_current,pending.human[9]) and Live.same(result.doubleheader_evidence,pending.human[13]),"actual 3–2 human win, called proof and fresh stamina match the committed row")
 app._settle_paired_next_first_leg()
 check(app.retained==next and app.career_owner==owner and app.season==source and Store.RawGuard.matches(owner,owner_raw) and exact(bytes,backup),"failed write and repeat retain the same candidate without changing owner or files")
 if not check(DirAccess.remove_absolute(obstruction)==OK,"remove only the owned scratch obstruction"):return
 gates=Live.paired_history_calls
 started=Time.get_ticks_msec()
 app.retry_save()
 print("NEXT_PAID_RETRY_WRITE_PROMOTE_MS=",Time.get_ticks_msec()-started)
 if not check(app.retained==null and app.season==next and app.career_owner!=owner and Live.paired_history_calls==gates and FileAccess.get_file_as_bytes(Store.PATH+".bak")==bytes and Store.RawGuard.matches(owner,owner_raw),"exact held ticket promotes once with no additional history validation"):return
 var saved:PackedByteArray=Store.raw()
 check(Store.prepare_and_write(app.career_owner,owner.fork(),saved)==null and Store.prepare_and_write(app.career_owner,app.career_owner.fork(),saved)==null and not Store.save(app.career_owner,saved),"generic writer cannot rewrite or rewind the private owner")
 closed_next(app,saved,bytes)

 app._play_paired_human(3)
 app._run_paired_npc(9)
 app._run_paired_npc(10)

func paid_next_first(season:SeasonState) -> bool:
 if season.round_index!=3 or season.phase!=SeasonState.Phase.REGULAR or season.results.size()!=9 or season.player_results.size()!=3 or season.physical.cursor!=6 or season.physical.reports.size()!=6 or not season.physical.pending.is_empty():return false
 if not Live.same(season.results.slice(6,9).map(func(row):return [row.id,row.away_runs,row.home_runs]),[[6,2,3],[7,0,8],[8,11,15]]):return false
 if season.build.cash()!=52 or season.build.player("player.mika_reed").stats.fielding!=SeasonPlayerCatalog.profile("player.mika_reed").stats.fielding+1:return false
 var new_receipts:int=0
 var preserved_bonuses:Array=[]
 for club:int in range(6):
  var build:SeasonBuild=season.build if club==0 else season.opponents.clubs[str(club)].build
  var expected:Dictionary={}
  for row:Dictionary in season.results:
   if row.away!=club and row.home!=club:continue
   expected[str(int(row.id))]=(row.away_runs>row.home_runs)==(row.away==club)
   if row.round==2:
    var present:Dictionary=row.doubleheader_evidence[0 if row.away==club else 1]
    if not Live.same(present.pair,{"game":row.id,"first":-1,"debt":{}}):return false
  var bank:SeasonOwnership=build._bank.empty_fork()
  var seen:Dictionary={}
  for event:Dictionary in build._bank._events:
   var before:Dictionary=bank.view().rewards.duplicate(true)
   var cash:int=bank.cash()
   var receipt:Dictionary=bank.commit(event)
   if not receipt.ok or receipt.replayed:return false
   if event.op!="reward":continue
   var id:String=str(int(event.game))
   var bonus:int=SeasonDoubleheaderRules.bonus(season.league_rules,event.game,event.win,before)
   if event.id!="game:"+id or expected.get(id)!=event.win or seen.has(id) or bank.cash()-cash!=(18 if event.win else 12)+bonus:return false
   seen[id]=true
   if bonus>0:preserved_bonuses.append([club,int(event.game),bonus])
   if event.game>=6:
    if bonus!=0:return false
    new_receipts+=1
  if seen.size()!=3 or not Live.same(build._bank.view().rewards,expected) or not ClubCareer.same(bank.view(),build._bank.view()):return false
 return new_receipts==6 and Live.same(preserved_bonuses,[[0,3,6],[3,5,6],[4,4,6]])
