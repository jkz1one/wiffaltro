extends SceneTree
## Unit contract only. Golden fixture order is explicit, not generated using the
## implementation's rotation. Does not qualify live admission, playoffs or saves.
const Selector = preload("res://selector.gd")
const ORDINARY = [[0, 0, 0, 5], [15, 5, 5, 0], [1, 0, 4, 1], [16, 5, 1, 4], [2, 0, 2, 3], [17, 5, 3, 2], [3, 1, 4, 0], [18, 6, 0, 4], [4, 1, 5, 3], [19, 6, 3, 5], [5, 1, 2, 1], [20, 6, 1, 2], [6, 2, 0, 3], [21, 7, 3, 0], [7, 2, 2, 4], [22, 7, 4, 2], [8, 2, 5, 1], [23, 7, 1, 5], [9, 3, 2, 0], [24, 8, 0, 2], [10, 3, 3, 1], [25, 8, 1, 3], [11, 3, 5, 4], [26, 8, 4, 5], [12, 4, 0, 1], [27, 9, 1, 0], [13, 4, 5, 2], [28, 9, 2, 5], [14, 4, 3, 4], [29, 9, 4, 3]]
const PAIRED = [[0, 0, 0, 5], [3, 1, 5, 0], [1, 0, 4, 1], [4, 1, 1, 4], [2, 0, 2, 3], [5, 1, 3, 2], [6, 2, 4, 0], [9, 3, 0, 4], [7, 2, 5, 3], [10, 3, 3, 5], [8, 2, 2, 1], [11, 3, 1, 2], [12, 4, 0, 3], [15, 5, 3, 0], [13, 4, 2, 4], [16, 5, 4, 2], [14, 4, 5, 1], [17, 5, 1, 5], [18, 6, 2, 0], [21, 7, 0, 2], [19, 6, 3, 1], [22, 7, 1, 3], [20, 6, 5, 4], [23, 7, 4, 5], [24, 8, 0, 1], [27, 9, 1, 0], [25, 8, 5, 2], [28, 9, 2, 5], [26, 8, 3, 4], [29, 9, 4, 3]]
var failures: int = 0


func check(condition: bool, label: String) -> void:
	if not condition:
		failures += 1
		push_error(label)


func fixtures(rows: Array) -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	for row: Array in rows:
		result.append({"id": row[0], "round": row[1], "away": row[2], "home": row[3]})
	return result


func invariants(games: Array[Dictionary], offset: int) -> void:
	check(games.size() == 30, "thirty fixtures")
	var ids: Dictionary = {}
	var directed_pairs: Dictionary = {}
	var rounds: Dictionary = {}
	for game: Dictionary in games:
		check(not ids.has(game.id), "unique ID")
		ids[game.id] = true
		var key: String = "%d:%d" % [game.away, game.home]
		check(not directed_pairs.has(key), "one fixture per directed pair")
		directed_pairs[key] = true
		check(game.id >= 0 and game.id < 30, "no playoff IDs")
		check(game.away != game.home, "no self match")
		if not rounds.has(game.round):
			rounds[game.round] = []
		rounds[game.round].append(game.away)
		rounds[game.round].append(game.home)
	var keys: Array = rounds.keys()
	keys.sort()
	check(keys == range(10), "ten regular rounds")
	for clubs: Array in rounds.values():
		clubs.sort()
		check(clubs == range(6), "each club once per round")
	for i: int in range(0, 30, 2):
		var first: Dictionary = games[i]
		var second: Dictionary = games[i + 1]
		check(second.round == first.round + offset, "return round offset")
		check(second.away == first.home and second.home == first.away, "reverse venues")


func _initialize() -> void:
	var ordinary: Array[Dictionary] = fixtures(ORDINARY)
	var paired: Array[Dictionary] = fixtures(PAIRED)
	for flag: bool in [false, true]:
		var expected: Array[Dictionary] = paired if flag else ordinary
		var selected: Array[Dictionary] = Selector.select(flag, ordinary)
		check(selected == expected, "exact ordered golden fixtures")
		invariants(selected, 1 if flag else 5)
		check(Selector.select(flag, ordinary) == selected, "determinism")
		seed(932026)
		var expected_random: int = randi()
		seed(932026)
		Selector.select(flag, ordinary)
		check(randi() == expected_random, "no global RNG consumption")
		selected[0].away = 99
		selected.pop_back()
		check(ordinary == fixtures(ORDINARY), "input unchanged")
		check(Selector.select(flag, ordinary) == expected, "no aliases or retained result")
	for invalid: Variant in [null, 0, 1, 0.0, 1.0, "false", "true", "doubleheader", {}, [], {"league": "doubleheader"}]:
		check(Selector.select(invalid, ordinary).is_empty(), "reject non-boolean")
	var nested: Array[Dictionary] = [{"custom": {"values": [1, 2]}}]
	var copied: Array[Dictionary] = Selector.select(false, nested)
	copied[0].custom.values[0] = 9
	check(nested[0].custom.values == [1, 2], "ordinary deep copy")
	check(Selector.select(false, []).is_empty(), "empty ordinary preserved")
	check(Selector.select(true, []) == paired, "paired independent of ordinary")
	if failures == 0:
		print("SUCCESSOR_DOUBLEHEADER_SELECTOR_PASS")
	quit(0 if failures == 0 else 1)
