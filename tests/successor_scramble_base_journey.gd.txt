## Actual Common V4 Scramble Base opening, using the pre-existing private engineering factory; not earned admission.
const RULES=preload("res://src/gameplay/season/league_rules.gd")
const GENERAL=preload("res://src/gameplay/season/successor_league_general_v2/physical_report.gd")
const ROOT_SEED=42
const SCRAMBLE_ARCHIVE="user://scramble-base-steps"

func _run() -> void:
	var args:=OS.get_cmdline_user_args()
	if not check(OS.get_environment("WIFFALTRO_ISOLATED_TEST")=="1"
			and OS.get_environment("WIFFALTRO_DYNASTY_ENGINEERING")=="1"
			and (OS.get_environment("WIFFALTRO_EARNED_LEAGUES_PREVIEW")=="1")== (args.size()==2 and args[0] in ["cold"])
			and args.size()==2 and args[0] in ["birth","step","cold"]
			and not FileAccess.file_exists(ORIGINAL),"isolated exact Scramble Common profile"): return
	var app=App.new()
	add_child(app)
	await get_tree().process_frame
	var action: String=""
	if args[0]=="birth":
		if not check(args[1].is_valid_int() and int(args[1])==ROOT_SEED and not Store.occupied()
				and not app.earned_league_preview(),"explicit private named engineering lane, no prior title"): return
		app.select_league("scramble")
		app.create_fresh(int(args[1]))
		if not check(_owned_scramble(app) and app.season.picks.is_empty()
				and app.career_owner.common.aggregate.titles.is_empty(),"actual Scramble Base Common App birth"): return
		var changed: int=0
		for index in range(4):
			if not check(_draft_copy(app),"exact transformed visible stats for each actual native offer"): return
			for id: String in app.season.offers():
				if not ClubCareer.same(SeasonPlayerCatalog.profile(id).stats,_expected(id).stats): changed+=1
			var previous:=Store.raw()
			app.draft(app.season.offers()[0])
			if not check(_owned_scramble(app) and app.season.picks.size()==index+1
					and FileAccess.get_file_as_bytes(Store.PATH+".bak")==previous,"native Scramble draft picks and backup"): return
		if not check(changed>0 and _capacities(app.season),"real transformed offers and six immutable season-root stat baselines"): return
		action="birth"
	else:
		var prior:=Store.raw()
		var backup:=FileAccess.get_file_as_bytes(Store.PATH+".bak")
		if not check(args[1].length()==64 and FileAccess.get_sha256(Store.PATH)==args[1] and not backup.is_empty(),
				"exact current generated slot and archive custody"): return
		app.reload_save()
		if not check(_owned_scramble(app) and _capacities(app.season)
				and Store.raw()==prior and FileAccess.get_file_as_bytes(Store.PATH+".bak")==backup,"strict cold reopen preserves all six capacities"): return
		if args[0]=="cold":
			if not check(_settled(app) and app.season.build.cash()>=12
					and app.league_entry_state("scramble")=="locked","earned pack price reachable after real Common round; no public entitlement"): return
			print("SCRAMBLE_BASE_COLD ",args[1]," cash=",app.season.build.cash())
			await _finish(app)
			return
		if not check(app.season.round_index==0,"bounded to opening round"): return
		action=await _play(app) if app.season.physical.pending.is_empty() else await _resume(app)
		if action.is_empty(): return
		if not check(app.retained==null and _owned_scramble(app) and Store.raw()!=prior
				and FileAccess.get_file_as_bytes(Store.PATH+".bak")==prior,"one real operation and exact backup"): return
		if action.begins_with("settle") and not check(_settled(app),"all three games and six awards settle"): return
	if _archive(action):
		print("SCRAMBLE_BASE_STEP ",action," ",FileAccess.get_sha256(Store.PATH))
		await _finish(app)

func _owned_scramble(app) -> bool:
	return app.career_owner!=null and app.career_owner.active=="ordinary" and app.career_owner.paused.is_empty() \
		and app._owned() and app.season.league_live and ClubCareer.same(app.season.league_rules,RULES.capture("scramble",0,ROOT_SEED))

func _capacities(season: SeasonState) -> bool:
	if season.opponents.clubs.size()!=5: return false
	for build: SeasonBuild in [season.build]+season.opponents.clubs.values().map(func(club: Dictionary)->SeasonBuild: return club.build):
		var expected: Dictionary=RULES.capacities(RULES.capture("scramble",0,ROOT_SEED))
		if expected!={"held":2,"sponsors":5} or not ClubCareer.same(build._league_rules,season.league_rules) \
			or not ClubCareer.same(build._bank.view().capacity,expected) or not build._book.context_matches(season.league_rules): return false
		for id: String in build.roster():
			if not ClubCareer.same(build.player(id).base,_expected(id).base): return false
	return true

# Independent frozen algorithm reference: no call to SeasonScramble.transform,
# profile or mapping and no mutation of authored catalog resources.
func _expected(id: String) -> Dictionary:
	var profile: Dictionary=SeasonPlayerCatalog.profile(id)
	var names: Array[String]=["contact","power","fielding","pitching"]
	var vector:=names.duplicate()
	for n: int in [3,2,1]:
		var idx: int=0
		var hashword: int=0
		while true:
			var hexcode:=("league-scramble-stat-permutation-v1\n%d\n%s\n%d\n%d"%[ROOT_SEED,id,n,idx]).sha256_text()
			hashword=hexcode.left(8).hex_to_int()
			if hashword<4294967296-(4294967296%(n+1)): break
			idx+=1
		var k: int=hashword%(n+1)
		var swap: String=vector[k]
		vector[k]=vector[n]
		vector[n]=swap
	var expected: Dictionary=profile.duplicate(true)
	for i: int in range(4):
		expected.stats[names[i]]=int(profile.stats[vector[i]])
		expected.base[names[i]]=int(profile.base[vector[i]])
	return expected

func _draft_copy(app: App) -> bool:
	var seen: Array[String]=[]
	for n: Node in app.menu.find_children("*","Label",true,false):
		if n is Label and not n.is_queued_for_deletion() and n.has_meta("stat_keys"): seen.append(n.text)
	if seen.size()!=app.season.offers().size(): return false
	for i: int in range(seen.size()):
		var id: String=app.season.offers()[i]
		var p: Dictionary=app.season.build.player(id)
		if not ClubCareer.same(p,_expected(id)): return false
		var values: PackedStringArray=[]
		for key: String in SeasonPlayerCatalog.STATS: values.append(key.capitalize()+": "+str(p.stats[key]))
		if seen[i]!=" · ".join(values): return false
	return true

func _live_roster(team: TeamMatchState,number: int,_fixture: Dictionary) -> bool:
	for player: PlayerMatchState in team.roster:
		var d: PlayerDefinition=player.definition
		var authored: Dictionary=_expected(String(d.id)).stats
		if not ClubCareer.same([d.contact,d.power,d.fielding,d.control],
				[authored.contact,authored.power,authored.fielding,authored.pitching]): return false
	print("SCRAMBLE_LIVE_TEAM ",number," ",team.roster.size()," transformed native players")
	return true

func _before_automated_play(lab: PitchBatLab, fixture: Dictionary) -> void:
	var state:=lab._match_state
	var caps: Dictionary=RULES.capacities(RULES.capture("scramble",0,ROOT_SEED))
	check(ClubCareer.same(state.league_rules,RULES.capture("scramble",0,ROOT_SEED))
		and ClubCareer.same(PhysicalLeagueReport.wallet(state.league_rules,state.away_team.league_ownership).get("capacity"),caps)
		and ClubCareer.same(PhysicalLeagueReport.wallet(state.league_rules,state.home_team.league_ownership).get("capacity"),caps)
		and _live_roster(state.away_team,int(fixture.away),fixture) \
		and _live_roster(state.home_team,int(fixture.home),fixture) \
		and not state.pitch_catalog.is_empty() and state.stadium.is_empty(),
		"both actual teams use transformed active player definitions, unchanged pitch catalog and ordinary field")
	print("SCRAMBLE_FIELD fixture=",fixture.id," catalog=",state.pitch_catalog," named_venue=",not state.stadium.is_empty())

func _settled(app) -> bool:
	var season: SeasonState=app.season
	if season.round_index!=1 or season.results.size()!=3 or season.player_results.size()!=1 \
			or not season.physical.pending.is_empty() or not season.physical.needed.is_empty() \
			or season.physical.reports.size()!=2: return false
	var seen: Dictionary={}
	for game: Dictionary in season.results:
		for i: int in [int(game.away),int(game.home)]:
			if seen.has(i): return false
			var build: SeasonBuild=season.build if i==0 else season.opponents.clubs[str(i)].build
			var award: Array=build._bank._events.filter(func(e: Dictionary)->bool: return e.op=="reward")
			var win: bool=SeasonState._winner(game)==i
			if award.size()!=1 or award[0].game!=game.id or award[0].win!=win: return false
			var cursor: SeasonOwnership=build._bank.empty_fork()
			var paid: int=0
			for e: Dictionary in build._bank._events:
				var before: int=cursor.cash()
				var native: Dictionary=cursor.commit(e)
				if not native.get("ok",false) or native.get("replayed",true): return false
				if e.op=="reward": paid=cursor.cash()-before
			if paid!=(18 if win else 12) or not ClubCareer.same(cursor.view(),build._bank.view()): return false
			print("SCRAMBLE_NATIVE_REWARD team=",i," game=",game.id," win=",win," cash=",paid)
			seen[i]=true
	for row: Dictionary in season.physical.reports:
		if not GENERAL.valid(row.report,season.build.pitch_catalog_signature()): return false
	return seen.size()==6 and _capacities(season)

func _archive(action: String) -> bool:
	var sha:=FileAccess.get_sha256(Store.PATH)
	var path:=SCRAMBLE_ARCHIVE.path_join(action+"-"+sha)
	if not check(sha.length()==64 and not DirAccess.dir_exists_absolute(path)
			and DirAccess.make_dir_recursive_absolute(path)==OK,"unique generated Scramble stage"): return false
	for suffix: String in ["",".bak"]:
		var dest:=path.path_join("owner.json"+suffix)
		if not check(DirAccess.copy_absolute(Store.PATH+suffix,dest)==OK
				and FileAccess.get_sha256(dest)==FileAccess.get_sha256(Store.PATH+suffix),"byte-exact Scramble stage "+suffix): return false
	return true
