func _ready() -> void:
	call_deferred("_run")

func _expect(ok: bool, label: String) -> bool:
	if ok: return true
	printerr("DETACHED D1 FAIL: ",label)
	get_tree().quit(1)
	return false

func _native_authority(wins: Array) -> Dictionary:
	var club: ClubCareer=Fixture.prefix(wins)
	if club==null: return {}
	var source: SeasonState=SeasonState.create(39102,false,true,true)
	source.physical=SeasonPhysicalFixtures.new()
	source.opponents._format=13
	if not ClubPlayerDraft.apply(source,ClubPlayers.owned(club)) or not club.start(source,0): return {}
	club._assets=ClubPlayers.assets(club).to_data()
	source.career=club
	source.home_property=StadiumLayoutSnapshot.capture(ClubPlayers.assets(club),true)
	source.npc_venues=SeasonVenues.capture_five(source.season_seed,club.current,0)
	source.final_venue=SeasonFinalVenue.capture_standard_base(source)
	for index: int in range(4):
		if not source.choose_player(source.offers()[0]): return {}
	var origin: Dictionary=SeasonSave.snapshot(source)
	return Authority.create(origin,"detached-d1-qa") if not origin.is_empty() else {}

func _binding(admission: Dictionary, tier: int) -> Dictionary:
	var grant: Dictionary=Detached.Grant.capture(admission)
	return {"owner_id":"detached-d1-qa","lineage_id":"detached-lineage","generation":0,"ordinal":1,
		"session_id":"fixture","seed":39201,"tier":tier,"league":"standard","format":Detached.Bound.LIVE_VERSION,
		"grant_sha256":Save.snapshot_reference(grant).sha256}

func _run() -> void:
	if not _expect(OS.get_environment("WIFFALTRO_ISOLATED_TEST")=="1" and not Common.occupied()
			and not FileAccess.file_exists("user://season-v1.json"),"scratch slots must start absent"): return
	var admission: Dictionary=_native_authority([{"league":"standard","tier":0}])
	var ineligible: Dictionary=_native_authority([])
	if not _expect(not admission.is_empty() and not ineligible.is_empty(),"isolated native title and no-title operands"): return
	var binding: Dictionary=_binding(admission,1)
	var profile: Dictionary=Replay.profile(1)
	if not _expect(Detached.create(ineligible,_binding(ineligible,1),profile)==null
			and Detached.create(admission,binding,Replay.profile(0))==null
			and Detached.create(admission,_binding(admission,0),profile)==null
			and Detached.create(Detached.Grant.capture(admission),binding,profile)==null,
			"no title, wrong profile, wrong binding tier and a raw grant refused"): return
	var forged: Dictionary=binding.duplicate(true)
	forged.grant_sha256="a".repeat(64)
	if not _expect(Detached.create(admission,forged,profile)==null,"foreign grant reference refused"): return
	var d1: SeasonState=Detached.create(admission,binding,profile)
	var base: SeasonState=Detached.Live.create(Detached.Grant.capture(admission),_binding(admission,0))
	if not _expect(d1!=null and base!=null and d1.league_rules.is_empty() and base.league_rules.is_empty()
			and d1.difficulty==base.difficulty and d1.build.d1_pricing_valid() and not base.build.d1_pricing_valid()
			and not d1.build.d3_sponsors_valid() and not d1.build.d4_sponsors_valid(),
			"native Base untouched; D1 is pricing tier, not tactical or alternate rules"): return
	if not _expect(base.npc_venues.venue_ordinal==0 and d1.npc_venues.venue_ordinal==1
			and base.final_venue.selection==SeasonFinalVenue.STANDARD_BASE_SELECTION
			and d1.final_venue.selection==SeasonFinalVenue.STANDARD_D1_SELECTION
			and ClubCareer.same(base.home_property,d1.home_property),
			"native D1 away/final venue policy differs; authored home retained"): return
	for index: int in range(4):
		if not _expect(d1.choose_player(d1.offers()[0]) and base.choose_player(base.offers()[0]),"native draft"): return
	if not _expect(Detached.matches(d1,profile) and d1.opponents.clubs.size()==5
			and base.opponents.clubs.size()==5 and d1.opponents.draft_rule.is_empty(),
			"native D1 draft and format 13 opponents"): return
	for id: String in d1.opponents.clubs:
		var cpu: SeasonBuild=d1.opponents.clubs[id].build
		if not _expect(cpu._market==12 and cpu.d1_pricing_valid()
				and not base.opponents.clubs[id].build.d1_pricing_valid(),"native NPC price market "+id): return
	var venue_ids: Array[String]=[]
	for id: String in ["1","2","3","4","5"]:
		venue_ids.append(String(d1.npc_venues.assignments[id].manifest.id))
	if not _expect(venue_ids==["AC-02","AY-02","AY-03","AC-01","AE-01"]
			and d1.npc_venues.quota=={"backyard":2,"expanded":1,"club":2,"mogul":0},"exact authored seed-39201 assignments"): return
	var fixture: Dictionary=d1.pending_fixture()
	var field: FieldDefinition=SeasonState.field_for_fixture(fixture,d1)
	var match_state: MatchState=d1.make_match()
	if not _expect(field!=null and match_state!=null and match_state.league_rules.is_empty()
			and d1.results.is_empty(),"native D1 fixture/field constructs without simulating"): return
	var final: Dictionary=d1.final_venue
	d1.final_venue=base.final_venue.duplicate(true)
	if not _expect(not Detached.matches(d1,profile),"Base-venue substitution refused"): return
	d1.final_venue=final
	if not _expect(Detached.matches(d1,profile) and not Common.occupied()
			and not FileAccess.file_exists("user://season-v1.json"),"no saves written"): return
	print("STANDARD_D1_DETACHED_PASS ",JSON.stringify({"home":d1.home_property.tier,
		"away_ids":venue_ids,"final":d1.final_venue.selection,"opening_field":field.id,"draft":d1.picks}))
	get_tree().quit()
