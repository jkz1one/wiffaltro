func real_roster_start(season:SeasonState,state:MatchState,fixture:Dictionary,proofs:Array) -> bool:
 if not Live.same(fixture,Authored.authored_phase(6).human):return false
 var roster:Array[String]=season.build.roster()
 var home:TeamMatchState=state.home_team
 var wallet:Dictionary=season.build.view().wallet
 if not Live.same(home.roster.map(func(player):return String(player.definition.id)),roster) or roster[1]!=RECRUIT or roster.has(RELEASED):return false
 if not Live.same(home.roster[1].definition.fielding,RecruitCatalog.fresh(RECRUIT,"late").stats.fielding) or home.roster[0].definition.fielding!=SeasonPlayerCatalog.profile("player.mika_reed").stats.fielding+1:return false
 if proofs.is_empty() and (home.current_fielder()!=home.roster[1] or home.fielder_index!=season.fielder_index or not Live.same(home.tactics.held,SeasonTacticalCatalog.held(wallet)) or not Live.same(state.gear_usage.equipped,SeasonReclamation.receipts(wallet))):return false
 var expected:Dictionary={"game":18,"first":-1,"debt":{}}
 for side:int in range(2):
  var club:int=int(fixture.away if side==0 else fixture.home)
  var team:TeamMatchState=state.away_team if side==0 else home
  var prior:Array=season.results.filter(func(row):return row.round==5 and (row.away==club or row.home==club))
  if prior.size()!=1 or not prior[0].doubleheader_evidence[0 if prior[0].away==club else 1].pair.debt.values().any(func(n):return n>0):return false
  if not Live.same(team.league_pair,expected) or (not proofs.is_empty() and not Live.same(proofs[side].pair,expected)):return false
  for player:PlayerMatchState in team.roster:
   var id:String=String(player.definition.id)
   if proofs.is_empty() and not PhysicalRecoveryFlow.near(player.stamina_remaining,player.stamina_max):return false
   if not proofs.is_empty() and (not proofs[side].proof.initial.has(id) or not PhysicalRecoveryFlow.near(proofs[side].proof.initial[id].initial,player.stamina_max)):return false
 return true
