func graph(value: Variant, seen: Dictionary = {}) -> Variant:
 if value is Object:
  var identity=value.get_instance_id()
  if seen.has(identity):return {"ref":seen[identity]}
  seen[identity]=seen.size()
  var fields={}
  for property: Dictionary in value.get_property_list():
   if int(property.usage)&PROPERTY_USAGE_SCRIPT_VARIABLE:fields[property.name]=graph(value.get(property.name),seen)
  return {"type":value.get_class(),"fields":fields}
 if value is Dictionary:
  var result={}
  for key:Variant in value:result[key]=graph(value[key],seen)
  return result
 if value is Array:
  var result=[]
  for entry:Variant in value:result.append(graph(entry,seen))
  return result
 return var_to_str(value)
