func run() -> void:
 var args=OS.get_cmdline_user_args()
 if OS.get_environment("WIFFALTRO_ISOLATED_TEST")!="1" or args.size()!=1:get_tree().quit(2);return
 var app=App.new()
 get_tree().root.add_child(app)
 var bytes=Store.raw()
 var backup=FileAccess.get_file_as_bytes(Store.PATH+".bak")
 var started:int=Time.get_ticks_msec()
 var gates:int=Live.paired_history_calls
 var resolves:int=Live.Physical.resolve_calls
 app.reload_save()
 var pending_reports:int=Live.Physical.resolve_calls-resolves
 print("THIRD_PAIR_FIRST_PAID_COLD_MS=",Time.get_ticks_msec()-started," HISTORY_CALLS=",Live.paired_history_calls-gates," PENDING_RESOLVES=",pending_reports)
 if args[0].begins_with("tamper"):
  check(app.season==null and exact(bytes,backup),"fresh App refuses "+args[0]+" without altering either file")
 elif not check(app.season!=null and exact(bytes,backup) and Live.paired_history_calls-gates==1,"fresh App authenticates exact privately owned bytes"):
  get_tree().quit(1);return
 elif args[0]=="cold":
  check(pending_reports==0,"paid cold has no pending report work")
  check(PairHistory.checked_sides==30 and PairHistory.checked_reports==10 and paid_third_first(app.season),"cold confirms all fifteen original outcomes, ten NPC archives and thirty paid sides")
  closed_third_first(app,bytes,backup)
 elif args[0]=="settle":
  check(pending_reports==2,"complete pending cold checked both reports")
  settle_third_first(app,bytes,backup)
 else:
  check(false,"unexpected mode")
 app.free()
 if failures==0:print("HIDDEN_PAIRED_THIRD_FIRST_SETTLEMENT_PASS")
 get_tree().quit(0 if failures==0 else 1)

func paid_third_first(season:SeasonState) -> bool:
 if season.round_index!=5 or season.phase!=SeasonState.Phase.REGULAR or season.results.size()!=15 or season.player_results.size()!=5 or season.physical.cursor!=10 or season.physical.reports.size()!=10 or not season.physical.pending.is_empty():return false
 if not Live.same(season.results.slice(12).map(func(row):return [row.id,row.away_runs,row.home_runs]),[[12,7,1],[13,3,5],[14,10,17]]):return false
 if season.build.cash()!=82 or season.build._visit.get("number")!=5 or season.build._visit.get("open")!=false or season.build.player("player.mika_reed").stats.fielding!=SeasonPlayerCatalog.profile("player.mika_reed").stats.fielding+1:return false
 var receipts:int=0
 var bonuses:Array=[]
 for club:int in range(6):
  var build:SeasonBuild=season.build if club==0 else season.opponents.clubs[str(club)].build
  var bank:SeasonOwnership=build._bank.empty_fork()
  var seen:Dictionary={}
  var exact_game:Array=season.results.filter(func(row):return row.id>=12 and (row.away==club or row.home==club))
  if exact_game.size()!=1:return false
  var row:Dictionary=exact_game[0]
  var win:bool=(row.away_runs>row.home_runs)==(row.away==club)
  # NPC Build reward command IDs are native ai:*; independent bank IDs are game:*.
  var paid:Array=build._events.filter(func(event):return event.get("op")=="successor_reward" and Reward.unwrap(event).get("game")==row.id)
  if paid.size()!=1:return false
  var command:Dictionary=Reward.unwrap(paid[0])
  if command.get("game")!=row.id or command.get("win")!=win:return false
  for event:Dictionary in build._bank._events:
   var before:Dictionary=bank.view().rewards.duplicate(true)
   var cash:int=bank.cash()
   var receipt:Dictionary=bank.commit(event)
   if not receipt.ok or receipt.replayed:return false
   if event.op!="reward":continue
   var id:String=str(int(event.game))
   var bonus:int=SeasonDoubleheaderRules.bonus(season.league_rules,event.game,event.win,before)
   if seen.has(id) or event.id!="game:"+id or bank.cash()-cash!=(18 if event.win else 12)+bonus:return false
   seen[id]=true
   if bonus>0:bonuses.append([club,int(event.game),bonus])
   if event.game>=12:
    if event.game!=row.id or event.win!=win or bonus!=0:return false
    receipts+=1
  if seen.size()!=5 or not Live.same(bank.view(),build._bank.view()):return false
 return receipts==6 and Live.same(bonuses,[[0,3,6],[3,5,6],[3,10,6],[4,4,6]])
