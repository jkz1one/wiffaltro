 var original=full.to_data().duplicate(true)

 var observer=func(prefix: SeasonBuild) -> bool:
  if prefix._events.size()==1: holder.build=prefix._fork()
  return true
 var validated=Build.restore(original,full._seed,full._initial_roster,full._pool,full._blocked,-1,observer)

  row.erase("op")
  if not check(b.commit(SeasonOpponentPolicy.command(b,op,row)).ok,"authentic paid "+op): get_tree().quit(1);return

 check(b.commit(command).ok and ClubCareer.same(after,b.to_data()),"exact request retry is idempotent")
 check(SeasonOpponentBudget.commit(b,4) and ClubCareer.same(after,b.to_data()),"budget wrapper retry is idempotent")

 bad.events.back().game=-1

  if p._events.size()==before.events.size(): seen.prefix=p._fork()
  return true

  var reopened=Build.restore(saved,b._seed,b._initial_roster,b._pool,b._blocked)
  check(reopened!=null and ClubCareer.same(after,reopened.to_data()),"separate cold saved grant replay equals authentic reconstructed branch")
 else:
  var file=FileAccess.open(saved_path,FileAccess.WRITE)
  if file==null: check(false,"synthetic grant file writable")
  else: file.store_string(JSON.stringify(after,"",true,true));file.close()
 check(ClubCareer.same(original,full.to_data()) and FileAccess.get_file_as_bytes(path)==bytes,"source pair journal and save unchanged")
