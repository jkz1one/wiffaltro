extends "res://transition.gd"
## Private source-engineering transport, no gameplay slot modification.
const Flow=preload(G+"standard_d1_witness_process.gd")
const EXACT="1a226548fad8932770868dcb286d1466a30976d571c2238ed78964fefb02b082"
const PRIOR="427deabb3f82a95afe829323955cea7ad39d36067078686d522268cc1bc91293"

func _ready() -> void:
	call_deferred("_run")

func _need(ok: bool, label: String) -> bool:
	if ok: return true
	printerr("D1 WITNESS PROCESS FAIL ",label)
	get_tree().quit(1)
	return false

func _wire(data: Dictionary) -> PackedByteArray:
	return ("Godot Engine v4.6.3 - https://godotengine.org\n\n"+Flow.FRAME+Flow.canonical(data)+"\n").to_utf8_buffer()

func _run() -> void:
	var args:=OS.get_cmdline_user_args()
	if not _need(OS.get_environment("WIFFALTRO_ISOLATED_TEST")=="1" and Flow.supported()
			and args.size()==1 and args[0] in ["replay","cancel","forged","stale"],"isolated mode"): return
	if not _need(Flow._revision()==EXACT and FileAccess.get_sha256(S.PATH+".bak")==PRIOR
			and not FileAccess.file_exists("user://season-v1.json"),"actual generated eliminated control"): return
	var job:=Flow.new()
	add_child(job)
	var started:=Time.get_ticks_msec()
	if not _need(not job.start(PRIOR,39201) and job.start(EXACT,39201)
			and not job.start(EXACT,39201),"wrong revision and overlap refused"): return
	var launch_ms:=Time.get_ticks_msec()-started
	var pid: int=job._pid
	var request: Dictionary=job._request.duplicate(true)
	# Shape-positive protocol fixture only, never fed to the App or an admission.
	var trial: Dictionary={"request":request,"outcome":"qualified","transition":_shape_sample(O.create_ordinary(42,"standard"))}
	trial.request.seed=trial.transition.basis.seed
	request=trial.request
	var wire:=_wire(trial)
	if not _need(not Flow._decode(wire,request,0).is_empty()
			and Flow._decode(wire,request,Flow.REFUSED_EXIT).is_empty(),"synthetic frame parser only with exact child exit"): return
	var foreign:=request.duplicate()
	foreign.token="00000000000000000000000000000000"
	var too_large:=PackedByteArray()
	too_large.resize(Flow.MAX_FRAME+1)
	if not _need(Flow._decode(wire,foreign,0).is_empty() and Flow._decode(wire+wire,request,0).is_empty()
			and Flow._decode(too_large,request,0).is_empty(),"foreign epoch, duplicate/replay and oversize frames reject"): return
	var outcome: Array[String]=[]
	job.completed.connect(func(result: String, proof: Dictionary)->void:
		outcome.append(result if proof.is_empty() else "unexpected-proof"))
	if args[0]=="cancel":
		job.cancel()
	elif args[0]=="forged":
		var folder:=PhysicalJobFiles.directory(job._request.token)
		if not _need(DirAccess.make_dir_recursive_absolute(folder)==OK
				and PhysicalJobFiles.write(folder.path_join("response.json"),trial) and OS.kill(pid)==OK,
				"planted legacy reply cannot impersonate a pipe or exit code"): return
		PhysicalJobFiles.clean(job._request.token)
	elif args[0]=="stale":
		job.set_process(false)
		var deadline:=Time.get_ticks_msec()+120000
		while not job._output.get_string_from_utf8().contains(Flow.FRAME) and Time.get_ticks_msec()<deadline:
			if not _need(job._drain(),"drain private child reply"): return
			await get_tree().process_frame
		if not _need(job._output.get_string_from_utf8().contains(Flow.FRAME),"genuine response before revision change"): return
		if not _need(DirAccess.rename_absolute(S.PATH,S.PATH+".qa-moved")==OK,"simulate moved primary only in isolated copy"): return
		job.set_process(true)
	var frames: int=0
	while outcome.is_empty():
		await get_tree().process_frame
		frames+=1
	if args[0]=="stale" and not _need(DirAccess.rename_absolute(S.PATH+".qa-moved",S.PATH)==OK,"restore copied test primary"): return
	var expected: String={"replay":"refused","cancel":"cancelled","forged":"invalid","stale":"stale"}[args[0]]
	if not _need(outcome==[expected] and not job.running
			and job._stdout==null and job._stderr==null,"single correct outcome, stopped child and removed files"): return
	if not _need(Flow._revision()==EXACT and FileAccess.get_sha256(S.PATH+".bak")==PRIOR
			and not FileAccess.file_exists(S.PATH+".tmp"),"unchanged primary and predecessor"): return
	if args[0]=="replay" and not _need(frames>10,"menu-equivalent frame heartbeat continued while child replays"): return
	print("STANDARD_D1_WITNESS_PROCESS_PASS mode=",args[0]," outcome=",expected,
		" launch_ms=",launch_ms," total_ms=",Time.get_ticks_msec()-started," parent_frames=",frames)
	get_tree().quit()
