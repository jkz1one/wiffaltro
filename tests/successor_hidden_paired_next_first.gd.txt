func run() -> void:
 var args=OS.get_cmdline_user_args()
 if OS.get_environment("WIFFALTRO_ISOLATED_TEST")!="1" or args.size()!=1:get_tree().quit(2);return
 var app=App.new()
 get_tree().root.add_child(app)
 var bytes=Store.raw()
 var backup=FileAccess.get_file_as_bytes(Store.PATH+".bak")
 var started=Time.get_ticks_msec()
 var gates=Live.paired_history_calls
 app.reload_save()
 print("NEXT_FIRST_COLD_MS=",Time.get_ticks_msec()-started," HISTORY_CALLS=",Live.paired_history_calls-gates)
 if args[0].begins_with("tamper"):
  check(app.season==null and exact(bytes,backup),"fresh cold rejects "+args[0]+" and leaves both files intact")
 elif not check(app.season!=null and exact(bytes,backup) and Live.paired_history_calls-gates==1,"fresh App opens exact paid pair or next pending bytes"):
  get_tree().quit(1);return
 elif args[0]=="cold":
  var pending:Dictionary=app.season.physical.pending
  check(next_unpaid(app.season) and pending.human[0]==6 and pending.reports.is_empty(),"cold retains exact next-pair first human evidence without new payout or NPC report")
  var pristine:SeasonState=app.season.fork_live()
  pristine.physical.pending.clear()
  var opening:Dictionary=Binding.start(pristine)
  check(not opening.is_empty() and Live.same(opening.binding,pending.binding) and reset_and_loadout(pristine,opening.state,opening.fixture,pending.human[13]),"cold recomputes exact binding and both fresh stamina budgets and the exact earned upgrade")
  app._play_paired_next_first()
  app._run_paired_npc(7)
  app._run_paired_npc(8)
  verify_closed(app,bytes,backup)
  check(not is_instance_valid(app.lab) and not is_instance_valid(app.ai_runner) and exact(bytes,backup),"duplicate next first or any NPC cannot run")
 elif args[0]=="play":
  await play_next_first(app,bytes,backup)
 else:
  check(false,"unexpected mode")
 app.free()
 if failures==0:print("HIDDEN_PAIRED_NEXT_FIRST_PASS")
 get_tree().quit(0 if failures==0 else 1)

func next_unpaid(season:SeasonState) -> bool:
 if season.round_index!=2 or not Binding.bound_shape(season.physical.pending):return false
 var paid:SeasonState=season.fork_live()
 if paid==null:return false
 paid.physical.pending.clear()
 return settled_pair(paid)

func reset_and_loadout(season:SeasonState,state:MatchState,fixture:Dictionary,proofs:Array,opening:bool=true) -> bool:
 if fixture.id!=6 or fixture.away!=4 or fixture.home!=0 or SeasonDoubleheaderRules.first(fixture.id)!=-1:return false
 var picked:Dictionary=season.build._events[4]
 if picked.get("player")!="player.mika_reed":return false
 var loadout:bool=false
 for side:int in range(2):
  var club:int=int(fixture.away if side==0 else fixture.home)
  var team:TeamMatchState=state.away_team if side==0 else state.home_team
  var prior:Array=season.results.filter(func(row):return row.round==1 and (row.away==club or row.home==club))
  if prior.size()!=1:return false
  var past:Dictionary=prior[0].doubleheader_evidence[0 if prior[0].away==club else 1]
  if not past.pair.debt.values().any(func(n):return n>0):return false
  var expected:Dictionary={"game":6,"first":-1,"debt":{}}
  if not Live.same(team.league_pair,expected) or (not proofs.is_empty() and not Live.same(proofs[side].pair,expected)):return false
  for player:PlayerMatchState in team.roster:
   var id:String=String(player.definition.id)
   if opening and not PhysicalRecoveryFlow.near(player.stamina_remaining,player.stamina_max):return false
   if not proofs.is_empty() and not PhysicalRecoveryFlow.near(proofs[side].proof.initial[id].initial,player.stamina_max):return false
   if club==0 and id==picked.player:
    var stat:String=season.build.shop_item(picked.item).family
    loadout=stat=="fielding" and player.definition.fielding==season.build.player(id).stats[stat] and player.definition.fielding==SeasonPlayerCatalog.profile(id).stats[stat]+1
 return loadout
