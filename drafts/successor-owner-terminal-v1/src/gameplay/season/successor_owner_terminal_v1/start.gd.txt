extends RefCounted
## Exact fresh Dugout0 birth, with replay-bound first-season terminal settlement.
const Catalog = preload("res://src/gameplay/season/successor_league_v1/catalog_context.gd")
const Progress = preload("res://src/gameplay/season/successor_career_v1/career_progress.gd")

static func apply(season: SeasonState) -> ClubCareer:
	var career: ClubCareer = load("res://src/gameplay/season/successor_owner_terminal_v1/career.gd").fresh()
	return career if apply_to(season, career) else null

static func apply_to(season: SeasonState, career: ClubCareer) -> bool:
	if not _pristine(season, career):
		return false
	var prefix: ClubCareer = ClubCareer.new()
	season.build._gear_progress.enabled = true
	season.build._gear_progress.start = prefix.gear_counts()
	season.build._sponsor_progress.enabled = true
	season.build._sponsor_progress.start = prefix.sponsor_state()
	season.build._order_start = prefix.order_access()
	season.build._rain_start = prefix.rain_access()
	season.build._transfer_start = prefix.transfer_access()
	season.build._supply_start = prefix.supply_count()
	season.build._checkout_start = SeasonLateCheckout.access(prefix)
	season.build._association_start = SeasonAssociation.access(prefix)
	season.build._freezer_start = SeasonFreezers.access(prefix)
	season.build._sides_start = SeasonLeftRight.access(prefix)
	season.build._jump_start = SeasonJumpstart.access(prefix)
	season.build._sure_start = SeasonSureShot.access(prefix)
	season.build._major.start = SeasonDoubleMajor.access(prefix)
	season.build._abilities.start = SeasonAbilities.access(prefix)
	season.build._copy.start = SeasonCarbonCopy.access(prefix)
	season.build._field_start = SeasonFieldSupply.access(prefix)
	season.build._batch_start = SeasonSmallBatch.access(prefix)
	if not season.build._book.install_exotic_start([]):
		return false
	var ledger: ClubAssetLedger = fresh_assets(season.season_seed)
	if ledger == null or not ClubPlayerDraft.apply(season, ledger.owned()):
		return false
	var initial: Dictionary = row(season)
	if initial.is_empty():
		return false
	career._pitch_catalog = Catalog.signature()
	career.current = 1
	career.runs.append(initial)
	career._assets = ledger.to_data()
	StadiumPennantRewards.adopt(career)
	StadiumPlaqueRewards.adopt(career)
	season.career = career
	season.league_career = true
	return matches(season)

static func fresh_assets(seed_value: int) -> ClubAssetLedger:
	# Exact empty first-run income/history/award basis; no purchased or earned assets.
	return ClubAssetLedger.new(seed_value, [0], [], {})

static func row(season: SeasonState) -> Dictionary:
	var details: Dictionary = Progress.capture(season)
	if details.is_empty():
		return {}
	return {
			"id": 1,
			"seed": season.season_seed,
			"league": season.league_rules.get("league", "standard"),
			"tier": 0,
			"status": "active",
			"proof": ClubSeasonRecord.capture(season),
			"receipt": {},
			"gear": season.build._gear_progress.games.duplicate(true),
			"sponsors": season.build._sponsor_progress.games.duplicate(true),
			"order_rerolls": season.build._paid_rerolls,
			"rain_earned": season.build._rain_earned,
			"transfer_earned": season.build._transfer_earned,
			"checkout_earned": season.build._checkout_earned,
			"association_earned": season.build._association_earned,
			"freezer_earned": season.build._freezer_earned,
			"sides_earned": season.build._sides_earned,
			"jump_earned": season.build._jump_earned,
			"sure_earned": season.build._sure_earned,
			"major_earned": season.build._major.earned,
			"sky_outs": season.build._abilities.earned,
			"copy_earned": season.build._copy.earned,
			"field_outs": season.build._field_outs,
			"batch_used": season.build._batch_used.duplicate(),
			"collection": ClubCollection.acquired(season.build),
			"supplies_used": season.build._supply_used,
			"player_progress": details,
			"league_rules": season.league_rules.duplicate(true)
		}

static func _pristine(season: SeasonState, career: ClubCareer) -> bool:
	return _stage(season) and season.phase == SeasonState.Phase.DRAFT \
		and season.build.revision() == 0 and season.build._book._events.is_empty() \
		and season.results.is_empty() and season.player_results.is_empty() \
		and season.physical.pending.is_empty() and season.physical.reports.is_empty() \
		and season.build._book._exotic_current.is_empty() \
		and season.picks.is_empty() and season.career == null and not season.league_career \
		and season.player_access.is_empty() and career != null \
		and career.get_script() == load("res://src/gameplay/season/successor_owner_terminal_v1/career.gd") \
		and career.current == 0 and career.runs.is_empty() and career._assets.is_empty() \
		and career._pennants.is_empty() and career._plaques.is_empty() and career._pitch_catalog.is_empty() \
		and not season.build._gear_progress.enabled and not season.build._sponsor_progress.enabled

static func _stage(season: SeasonState) -> bool:
	return season != null and season.get_script() == load("res://src/gameplay/season/successor_owner_terminal_v1/runtime.gd") \
		and season.build != null and season.build.get_script() == load("res://src/gameplay/season/successor_league_v1/build.gd") \
		and season.build.call("pitch_catalog_successor") \
		and season.build._book._exotic_start.is_empty() \
		and season.build._market == 0 \
		and season.league_live and ClubCareer.same(season.league_rules, LeagueRules.capture("dugout")) \
		and season.physical != null \
		and season.picks.size() <= 4 \
		and ((season.picks.size() < 4 and season.phase == SeasonState.Phase.DRAFT) \
			or (season.picks.size() == 4 and season.phase in [SeasonState.Phase.REGULAR, SeasonState.Phase.SEMIFINAL, SeasonState.Phase.FINAL, SeasonState.Phase.COMPLETE])) \
		and season.inventory_rules.is_empty() and season._closed_access.is_empty() \
		and season._dynasty_start.is_empty() and season._dynasty_owner_binding.is_empty() \
		and season._dynasty_career_session.is_empty() and season._standard_live_context.is_empty() \
		and season._standard_bound_grant.is_empty() and season._league_home_context.is_empty() \
		and season.home_property.is_empty() and season.owned_surface_night.is_empty() \
		and season.npc_venues.is_empty() and season.final_venue.is_empty()

static func structural_matches(season: SeasonState) -> bool:
	# Only identity/immutable birth context. Safe during owner-aware replay, when
	# the synchronized row can still describe the preceding committed checkpoint.
	if not _stage(season) or not season.league_career or season.career == null:
		return false
	var career: ClubCareer = season.career
	if career.get_script() != load("res://src/gameplay/season/successor_owner_terminal_v1/career.gd") \
		or career.current != 1 or career.runs.size() != 1 or career._pitch_catalog != Catalog.signature() \
		or not ClubCareer.same(_starts(season.build), expected_starts()):
		return false
	var run: Dictionary = career.runs[0]
	if not SeasonOwnership._whole(run.get("id"), 1, 1) \
		or not SeasonOwnership._whole(run.get("seed"), season.season_seed, season.season_seed) \
		or run.get("league") != "dugout" or not SeasonOwnership._whole(run.get("tier"), 0, 0) \
		or not career.call("_row_valid", run) or not ClubCareer.same(run.get("league_rules"), season.league_rules):
		return false
	var ledger: ClubAssetLedger = fresh_assets(season.season_seed)
	if season.player_access != ledger.owned() or not ClubPlayerDraft.frozen_matches(season):
		return false
	# During replay the committed row may still precede newly terminal gameplay.
	# It must nevertheless remain an exact valid active/settled owned state.
	return load("res://src/gameplay/season/successor_owner_terminal_v1/settlement.gd").matches(career)


static func base_matches(season: SeasonState) -> bool:
	return structural_matches(season)

static func matches(season: SeasonState) -> bool:
	if not structural_matches(season):
		return false
	return season.career.call("exact_row_matches", season)
