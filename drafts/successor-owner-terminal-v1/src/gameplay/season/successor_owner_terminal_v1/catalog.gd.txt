static func game_signature() -> String:
	return Game.signature()

static func signature() -> String:
	var game: String = game_signature()
	if game.is_empty():
		return ""
	var evidence: String = Evidence.signature()
	if evidence.is_empty():
		return ""
	var parts: Array[String] = [FORMAT, game, evidence]
	for file: String in FILES:
		var digest: String = FileAccess.get_sha256("res://src/gameplay/season/successor_owner_terminal_v1/" + file)
		if digest.is_empty():
			return ""
		parts.append(digest)
	for path: String in PRESENTATION_FILES + SHARED_FILES:
		var digest: String = FileAccess.get_sha256(path)
		if digest.is_empty():
			return ""
		parts.append(path + ":" + digest)
	return "|".join(parts).sha256_text()
