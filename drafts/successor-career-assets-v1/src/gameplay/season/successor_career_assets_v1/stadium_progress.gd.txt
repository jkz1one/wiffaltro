extends RefCounted
## Pure closed-prefix stadium aggregates, not career/asset/receipt authority.
## Caller must validate/replay the owner history before using any aggregate.
const Progress = preload("res://src/gameplay/season/successor_career_v1/career_progress.gd")

static func capture(runs: Variant) -> Array:
	if not runs is Array:
		return []
	var projected: ClubCareer = ClubCareer.new()
	for value: Variant in runs:
		if not value is Dictionary:
			return []
		# Preserve the original closed-prefix boundary. Later rows are not examined
		# here; rejecting illegal owner-history order belongs to the caller.
		if value.get("status") == "active":
			break
		if value.get("status") not in ["completed", "abandoned"] \
			or not value.get("proof") is Dictionary or not value.proof.get("scores") is Array:
			return []
		var summary: Dictionary = ClubSeasonRecord.analyze(value.proof)
		if summary.is_empty() or (value.status == "completed") != (summary.finish != "unfinished"):
			return []
		var row: Dictionary = value.duplicate(true)
		var details: Variant = row.get("player_progress")
		if details != null:
			if not Progress.valid(details, row.proof.scores):
				return []
			# Validated in-memory aggregate projection only; never a saved record.
			details.erase("format")
			details.erase("pitch_catalog")
			details.erase("exotic")
			details.version = ClubPlayerLegend.VERSION
			var historical: Array = []
			for id: String in details.development.lessons:
				if SeasonPlayerCatalog.RECIPES.values().has(id):
					historical.append(id)
			details.development.lessons = historical
		projected.runs.append(row)
	# Exact existing regular-season wins, positive-only optional stat fields,
	# null/unknown evidence behavior, and completed/abandoned season counts.
	return StadiumCareerProgress.capture(projected)

static func valid(rows: Variant, credits: Array) -> bool:
	return StadiumCareerProgress.valid(rows, credits)

static func metrics(rows: Array, through: int) -> Dictionary:
	return StadiumCareerProgress.metrics(rows, through)
