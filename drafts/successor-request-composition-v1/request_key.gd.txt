const Frozen=preload("res://src/gameplay/season/successor_league_general_v2/physical_fixtures.gd")
# Exact V4 route representations. The fixture's authored lane selects a
# single representation; a report or request digest never selects it.
static func for_materialized_return(season: SeasonState, fixture: Dictionary, state: MatchState, field: FieldDefinition) -> String:
	if fixture.get("round")!=1 or fixture.get("id") not in [4,5]: return ""
	return _for_materialized_wallet(season,fixture,state,field)

static func for_materialized_next_first(season: SeasonState, fixture: Dictionary, state: MatchState, field: FieldDefinition) -> String:
	if fixture.get("round")!=2 or fixture.get("id") not in [7,8]: return ""
	return _for_materialized_wallet(season,fixture,state,field)

static func for_materialized_next_return(season: SeasonState, fixture: Dictionary, state: MatchState, field: FieldDefinition) -> String:
	if fixture.get("round")!=3 or fixture.get("id") not in [9,10,11]: return ""
	return _for_materialized_wallet(season,fixture,state,field)

# Hydration represents three derived receipt arrays as JSON numbers. Recreate
# exactly that form from independently authenticated state, then restore it.
static func _for_materialized_wallet(season: SeasonState, fixture: Dictionary, state: MatchState, field: FieldDefinition) -> String:
	if state==null or field==null or not LeagueRules.doubleheader(state.league_rules): return ""
	var teams: Array[TeamMatchState]=[state.away_team,state.home_team]
	for team: TeamMatchState in teams:
		if team==null or not team.league_ownership.get("events") is Array or (team.ai_budget.has("before") and not team.ai_budget.before is Array): return ""
	var originals: Array=[]
	for team: TeamMatchState in teams:
		originals.append([team.league_ownership,team.ai_tactical_initial,team.ai_budget])
		var wallet: Dictionary=team.league_ownership.duplicate()
		wallet.events=_materialized_receipts(wallet.events)
		var tactics: Array[Dictionary]=[]
		tactics.assign(_materialized_receipts(team.ai_tactical_initial))
		var budget: Dictionary=team.ai_budget.duplicate()
		if budget.has("before"): budget.before=_materialized_receipts(budget.before)
		team.league_ownership=wallet
		team.ai_tactical_initial=tactics
		team.ai_budget=budget
	var hash: String=for_context(season,fixture,state,field)
	for side: int in range(2):
		teams[side].league_ownership=originals[side][0]
		teams[side].ai_tactical_initial=originals[side][1]
		teams[side].ai_budget=originals[side][2]
	return hash

static func _materialized_receipts(value: Array) -> Array:
	return JSON.parse_string(JSON.stringify(value,"",true,true))

static func for_context(season: SeasonState, fixture: Dictionary, state: MatchState, field: FieldDefinition) -> String:
	var rows: Array = []
	for team: TeamMatchState in [state.away_team, state.home_team]:
		var players: Array = []
		for player: PlayerMatchState in team.roster:
			players.append(
				{
					"definition": Frozen._resource(player.definition,
						season.build != null and (season.build.pitch_catalog_current() or season.build.call("pitch_catalog_successor"))),
					"capacity": player.stamina_max,
					"initial": player.stamina_remaining
				}
			)
		var row: Dictionary = {
			"name": team.display_name,
			"pitcher": team.pitcher_index,
			"fielder": team.fielder_index,
			"players": players
		}
		if team.ai_sponsor_choices:
			row["choice_policy"] = 1
		if not team.ai_tactical_hitter.is_empty():
			row["batting_supplies"] = {
				"featured": team.ai_tactical_hitter,
				"held": team.ai_tactical_initial.duplicate(true)
			}
		if team.ai_heat:
			row["heat_policy"] = 1
		if not team.ai_recovery_pitcher.is_empty():
			row["recovery_policy"] = {"version": 1, "pitcher": team.ai_recovery_pitcher}
		if team.ai_base:
			row["base_policy"] = 1
		if not team.ai_budget.is_empty():
			row["budget_policy"] = team.ai_budget.duplicate(true)
		if not state.league_rules.is_empty():
			row["league_ownership"] = team.league_ownership.duplicate(true)
			if LeagueRules.dynasty(state.league_rules):
				row["dynasty_start"] = team.dynasty_start.duplicate(true)
			if LeagueRules.doubleheader(state.league_rules):
				row["league_pair"] = team.league_pair.duplicate(true)
		rows.append(row)
	var resolver: String = (
		load("res://src/gameplay/season/successor_league_general_v2/physical_core.gd").BASE_RESOLVER
		if (state.away_team.ai_base or state.home_team.ai_base)
		else (
			load("res://src/gameplay/season/successor_league_general_v2/physical_core.gd").BUDGET_RESOLVER
			if (
				not state.away_team.ai_budget.is_empty() or not state.home_team.ai_budget.is_empty()
			)
			else (
				load("res://src/gameplay/season/successor_league_general_v2/physical_core.gd").RECOVERY_RESOLVER
				if (
					not state.away_team.ai_recovery_pitcher.is_empty()
					or not state.home_team.ai_recovery_pitcher.is_empty()
				)
				else (
					load("res://src/gameplay/season/successor_league_general_v2/physical_core.gd").HEAT_RESOLVER
					if (state.away_team.ai_heat or state.home_team.ai_heat)
					else (
						load("res://src/gameplay/season/successor_league_general_v2/physical_core.gd").TACTICAL_RESOLVER
						if (
							not state.away_team.ai_tactical_hitter.is_empty()
							or not state.home_team.ai_tactical_hitter.is_empty()
						)
						else (
							load("res://src/gameplay/season/successor_league_general_v2/physical_core.gd").CHOICE_RESOLVER
							if (
								state.away_team.ai_sponsor_choices
								or state.home_team.ai_sponsor_choices
							)
							else load("res://src/gameplay/season/successor_league_general_v2/physical_core.gd").RESOLVER
						)
					)
				)
			)
		)
	)
	var request: Dictionary = {
		"resolver": resolver,
		"seed": Frozen.seed_for(season, fixture),
		"field": String(field.id),
		"quality": state.ai_tactical_quality,
		"fixture": fixture,
		"teams": rows
	}
	if not state.league_rules.is_empty():
		request["league_rules"] = state.league_rules.duplicate(true)
		request.resolver = (
			PhysicalDoubleheaderReport.RESOLVER
			if LeagueRules.doubleheader(state.league_rules)
			else load("res://src/gameplay/season/successor_league_general_v2/physical_league.gd").RESOLVER
		)
	if not state.stadium.is_empty():
		if StadiumLayoutSnapshot.is_terrain(state.stadium):
			if (
				not StadiumLayoutSnapshot.terrain_valid(state.stadium)
				or state.stadium.version != StadiumLayoutSnapshot.TERRAIN_FIXTURE_VERSION
			):
				return ""
			request.resolver = PhysicalStadiumReport.TERRAIN_RESOLVER
		elif StadiumSurfaceNightSnapshot.is_family(state.stadium):
			request.resolver = StadiumSurfaceNightReport.RESOLVER
			if state.stadium.get("version") == StadiumSurfaceNightSnapshot.TIME_FIXTURE_VERSION:
				request.resolver = StadiumSurfaceNightReport.TIME_RESOLVER
			var proof: Dictionary = StadiumSurfaceNightSnapshot.proof(
				state.stadium,
				season.owned_surface_night,
				StadiumSurfaceNightSnapshot.ledger(season)
			)
			if proof.is_empty():
				return ""
			request["night"] = proof
		elif StadiumNpcSnapshot.is_public(state.stadium):
			request.resolver = PhysicalStadiumReport.npc_resolver(state.stadium)
			if request.resolver.is_empty():
				return ""
		else:
			request.resolver = (
				StadiumPoolOwnedReport.RESOLVER
				if (state.stadium.get("version") == StadiumPoolOwnedSnapshot.FIXTURE_VERSION)
				else PhysicalStadiumReport.RESOLVER
			)
		request["property"] = StadiumLayoutSnapshot.normalized(state.stadium)
	if not state.pitch_catalog.is_empty():
		request["pitch_catalog"] = state.pitch_catalog
		request.resolver = load("res://src/gameplay/season/successor_league_general_v2/physical_report.gd").RESOLVER
	return JSON.stringify(request, "", true, true).sha256_text()
