static func capture() -> Dictionary:
	var pitches: Dictionary = _pitches()
	var players: Dictionary = _players()
	var fields: Dictionary = {}
	for path: String in ["res://src/content/fields/commons_park.tres", "res://src/content/fields/starter_field.tres"]:
		var source: FieldDefinition = load(path) as FieldDefinition
		if source == null:
			return {}
		var value: Dictionary = _resource(source)
		if value.is_empty() or fields.has(value.id):
			return {}
		fields[value.id] = value
	if pitches.is_empty() or players.is_empty():
		return {}
	var domains: Dictionary = {}
	var digests: Dictionary = {}
	for key: String in ["pitches", "players", "fields"]:
		var projection: Dictionary = pitches if key == "pitches" else (players if key == "players" else fields)
		var digest: String = Canonical.digest(projection)
		if digest.is_empty():
			return {}
		domains[key] = {"projection": projection, "digest": digest}
		digests[key] = digest
	var identity: String = Canonical.digest({"schema": SCHEMA, "partial": true, "coverage": COVERAGE, "domains": digests})
	return {} if identity.is_empty() else {"schema": SCHEMA, "partial": true, "coverage": COVERAGE, "domains": domains, "authored_content_id": identity}

static func _pitches() -> Dictionary:
	var manifest: ContentManifest = load(Base.MANIFEST_PATH) as ContentManifest
	if manifest == null or manifest.pitches.size() != 43:
		return {}
	var definitions: Dictionary = {}
	var manifest_order: Array = []
	for source: PitchDefinition in manifest.pitches:
		var row: Dictionary = _resource(source)
		if row.is_empty() or definitions.has(row.id):
			return {}
		definitions[row.id] = row
		manifest_order.append(row.id)
	for paths: Dictionary in [WorkingPitchCatalog.RETIRED_PATHS, Provider.NEW_PATHS]:
		for id: String in paths:
			var source: PitchDefinition = load(paths[id]) as PitchDefinition
			if source == null:
				return {}
			var row: Dictionary = _resource(source)
			if row.is_empty() or row.id != id or definitions.has(id):
				return {}
			definitions[id] = row
	var active: Array = Provider.ids()
	var known: Array = Provider.known_ids()
	if active.size() != 41 or known.size() != 50 or definitions.size() != 50:
		return {}
	var seen: Dictionary = {}
	for id: String in known:
		if not definitions.has(id) or seen.has(id):
			return {}
		seen[id] = true
	seen = {}
	var prices: Dictionary = {}
	for id: String in active:
		if not definitions.has(id) or seen.has(id):
			return {}
		seen[id] = true
		prices[id] = Provider.lesson_price(id)
	var historical: Dictionary = {}
	for id: String in WorkingPitchCatalog.HISTORICAL_CANONICAL_PATHS:
		var source: PitchDefinition = load(WorkingPitchCatalog.HISTORICAL_CANONICAL_PATHS[id]) as PitchDefinition
		if source == null:
			return {}
		var row: Dictionary = _resource(source)
		if row.is_empty() or row.id != id:
			return {}
		historical[id] = row
	return {"definitions": definitions, "known_order": known, "active_order": active,
		"v6_manifest_order": manifest_order, "historical_canonical": historical,
		"retired_active": Provider.RETIRED_ACTIVE.duplicate(true),
		"reserved_exotics": Provider.RESERVED_EXOTICS.duplicate(true),
		"historical_aliases": WorkingPitchCatalog.ALIASES.duplicate(true),
		"retired_aliases": WorkingPitchCatalog.RETIRED_ALIASES.duplicate(true),
		"lesson_prices": prices, "candidate_acquisition": Provider.ACQUISITION.duplicate(true)}

static func _players() -> Dictionary:
	var manifest: ContentManifest = load("res://src/content/manifest/content_manifest.tres") as ContentManifest
	if manifest == null:
		return {}
	var definitions: Dictionary = {}
	var order: Array = []
	for source: PlayerDefinition in manifest.players:
		var row: Dictionary = _resource(source)
		if row.is_empty() or definitions.has(row.id):
			return {}
		definitions[row.id] = row
		order.append(row.id)
	var profiles: Dictionary = {}
	for id: String in SeasonPlayerCatalog.ids():
		if not definitions.has(id):
			return {}
		var profile: Dictionary = SeasonPlayerCatalog.profile(id)
		if profile.is_empty():
			return {}
		profiles[id] = profile.duplicate(true)
	return {"definitions": definitions, "manifest_order": order,
		"rows": SeasonPlayerCatalog.ROWS.duplicate(true),
		"recipes": SeasonPlayerCatalog.RECIPES.duplicate(true),
		"stats": SeasonPlayerCatalog.STATS.duplicate(true),
		"stat_cap": SeasonPlayerCatalog.STAT_CAP, "initial_profiles": profiles}

# Exact scripts and declared fields prevent silent omission after schema expansion.
# Resource paths/editor metadata are deliberately excluded; nested authored values
# are retained (historical player repertoire may share IDs with newer recipes).
static func _resource(source: Resource, depth: int = 0) -> Dictionary:
	if source == null or depth > 4:
		return {}
	var script = source.get_script()
	var allowed: Array[String] = []
	if script == PITCH_SCRIPT:
		allowed = PITCH_FIELDS
	elif script == DELIVERY_SCRIPT:
		allowed = DELIVERY_FIELDS
	elif script == PLAYER_SCRIPT:
		allowed = PLAYER_FIELDS
	elif script == FIELD_SCRIPT:
		allowed = FIELD_FIELDS
	else:
		return {}
	var found: Dictionary = {}
	var result: Dictionary = {}
	for property: Dictionary in source.get_property_list():
		if not (int(property.usage) & PROPERTY_USAGE_SCRIPT_VARIABLE):
			continue
		if not (int(property.usage) & PROPERTY_USAGE_STORAGE):
			continue
		var name: String = String(property.name)
		if not allowed.has(name) or found.has(name):
			return {}
		found[name] = true
		var value: Variant = source.get(name)
		if typeof(value) != int(property.type):
			return {}
		if value is Resource:
			var child: Dictionary = _resource(value, depth + 1)
			if child.is_empty():
				return {}
			result[name] = child
		elif value is Array:
			var items: Array = []
			for item: Variant in value:
				if item is Resource:
					var child: Dictionary = _resource(item, depth + 1)
					if child.is_empty():
						return {}
					items.append(child)
				elif item is String or item is StringName:
					items.append(String(item))
				else:
					return {}
			result[name] = items
		elif value is Vector2:
			result[name] = {"vector2": [value.x, value.y]}
		elif value is Vector3:
			result[name] = {"vector3": [value.x, value.y, value.z]}
		elif value is StringName:
			result[name] = String(value)
		elif value is bool or value is int or value is float or value is String or value is Dictionary:
			result[name] = value.duplicate(true) if value is Dictionary else value
		else:
			return {}
	if found.size() != allowed.size() or not result.has("id") or String(result.id).is_empty():
		return {}
	return {} if Canonical.encode(result).is_empty() else result
