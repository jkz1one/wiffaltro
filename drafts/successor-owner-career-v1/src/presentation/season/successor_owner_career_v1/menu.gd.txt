extends SeasonMenu
const Save = preload("res://src/gameplay/season/successor_owner_career_v1/save.gd")
func render() -> void:
	_screen("successor", "Owned career", "Original club progression • separate save • existing saves unchanged")
	if not app.notice.is_empty(): _label(_body,app.notice,18)
	if app.get("retained")!=null:
		_button(_body,"RETRY EXACT SAVE",Callable(app,"retry_save"))
		return
	if app.season==null:
		_next_options()
		var occupied: bool = FileAccess.file_exists(Save.PATH) or FileAccess.file_exists(Save.PATH + ".bak")
		_button(_body,"CONTINUE COMPLETED CAREER",Callable(app,"create_fresh")).disabled = occupied or app.predecessor.is_empty()
		_button(_body,"ABANDON IMPORTED SEASON AND START NEW",Callable(app,"request_abandon")).disabled = occupied or app.predecessor.is_empty()
		_button(_body,"CONTINUE SAVED SEASON",Callable(app,"reload_save"))
		_button(_body,"IMPORT PREVIOUS SAVE (KEEPS ORIGINAL)",Callable(app,"import_previous_save")).disabled = occupied
		if not occupied and FileAccess.file_exists(Save.PATH+".tmp"):
			_button(_body,"RECOVER INTERRUPTED IMPORT",Callable(app,"recover_interrupted_import"))
		return
	if app.season.phase==SeasonState.Phase.DRAFT:
		for id: String in app.season.offers():
			var definition: PlayerDefinition = ContentDB.get_player(StringName(id))
			var profile: Dictionary = app.season.build.player(id)
			_button(_body, "%s  %s" % [definition.display_name, str(profile.stats)], Callable(app,"draft").bind(id))
		return
	for id: String in app.season.teams[0].roster:
		var definition: PlayerDefinition = app.season.build.definition(id)
		var pitches: Array[String] = []
		for pitch: PitchDefinition in definition.starting_pitches:
			pitches.append(pitch.display_name)
		_label(_body,"%s: %s" % [definition.display_name, ", ".join(pitches)],18)
	if not app.season.physical.pending.is_empty():
		_button(_body,"RESUME SAVED ROUND",Callable(app,"resume_round"))
	elif app.season.phase!=SeasonState.Phase.COMPLETE:
		_button(_body,"PLAY NEXT GAME",Callable(app,"play_next"))
	else:
		_label(_body,"Season complete. Earned results remain saved.",20)
		_next_options()
		_button(_body,"START NEXT SEASON",Callable(app,"next_season"))
	if app.season.phase != SeasonState.Phase.COMPLETE and app.season.physical.pending.is_empty():
		_next_options()
		_button(_body,"ABANDON AND START NEW SEASON",Callable(app,"request_abandon"))
	if app.season.shop_available():
		_button(_body,"SHOP",Callable(app,"open_pitch_shop"))

func _next_options() -> void:
	var choices: Array = app.next_season_choices()
	for choice: Dictionary in choices:
		if not choice.implemented: continue # Locked integration remains documented, not purchasable.
		var selected: bool = choice.league == app.next_league and choice.tier == app.next_tier
		var title: String = ("✓ " if selected else "") + String(choice.league).capitalize() + " · " + ("Base" if choice.tier == 0 else "Tier %d" % choice.tier)
		_button(_body,title,Callable(app,"select_next_season").bind(choice.league,choice.tier))
