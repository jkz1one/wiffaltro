func _enter_tree() -> void:
	_quit_tree = get_tree()
	_previous_auto_quit = _quit_tree.auto_accept_quit
	_quit_tree.auto_accept_quit = false

func _exit_tree() -> void:
	_loading_generation += 1 # Invalidate any inherited deferred paint continuation.
	_cancel_exit()
	if is_instance_valid(_quit_tree):
		_quit_tree.auto_accept_quit = _previous_auto_quit
	_quit_tree = null

func _notification(what: int) -> void:
	if what == NOTIFICATION_WM_CLOSE_REQUEST:
		_request_quit()

func _exit_blocked() -> bool:
	if _busy or Save.saving or is_instance_valid(ai_runner):
		notice = "A game or checkpoint is still running. Wait before leaving."
		return true
	if retained != null or not retained_report.is_empty() or not retained_human.is_empty():
		notice = "Unsaved progress is retained. Retry the exact checkpoint before leaving."
		return true
	if is_instance_valid(lab) and lab._match_state != null and lab._match_state.phase == MatchState.Phase.GAME_END:
		notice = "Save the completed game before leaving."
		return true
	return false

func ask_leave_game() -> void:
	_request_exit("leave")

func leave_game() -> void:
	# Direct inherited callbacks cannot bypass the same confirmation boundary.
	_request_exit("leave")

func _request_quit() -> void:
	_request_exit("quit")

func _request_exit(mode: String) -> void:
	if mode not in ["leave", "quit"] or _exit_blocked():
		return
	if not is_instance_valid(lab):
		if mode == "quit":
			_cancel_exit()
			_quit_now()
		return
	if lab._match_state == null or not is_instance_valid(_exit_dialog):
		return
	if not _exit_mode.is_empty():
		return # Repeated/different requests cannot retarget an open confirmation.
	_exit_mode = mode
	_exit_season = season
	_exit_lab = lab
	_exit_state = lab._match_state
	_exit_dialog.dialog_text = "Quit? This unfinished game will restart. Saved progress stays." if mode == "quit" else "Leave this unfinished game? It will restart. Saved progress stays."
	_exit_dialog.popup_centered(Vector2i(560, 180))
	_exit_dialog.get_cancel_button().grab_focus()

func _cancel_exit() -> void:
	_exit_mode = ""
	_exit_season = null
	_exit_lab = null
	_exit_state = null
	if is_instance_valid(_exit_dialog):
		_exit_dialog.hide()

func _confirm_exit() -> void:
	var mode: String = _exit_mode
	var owner: SeasonState = _exit_season
	var matches: bool = mode in ["leave", "quit"] and season == _exit_season \
		and is_instance_valid(lab) and lab == _exit_lab and lab._match_state == _exit_state
	_cancel_exit() # Consume once before cleanup or any potential callback.
	if not matches or _exit_blocked():
		return
	if not _close_match(false):
		notice = "The original match could not close safely. It remains available."
		return
	if season != owner or is_instance_valid(lab) or _exit_blocked():
		return # Cleanup may have invoked callbacks; never quit a replacement owner.
	if mode == "quit":
		_quit_now()
	else:
		show_season()

func _quit_now() -> void:
	# Sole final OS action; tests may spy on it without bypassing any exit guard.
	get_tree().quit()

func _ready() -> void:
	_exit_dialog = ConfirmationDialog.new()
	_exit_dialog.title = "Leave unfinished game"
	_exit_dialog.dialog_autowrap = true
	_exit_dialog.confirmed.connect(_confirm_exit)
	_exit_dialog.canceled.connect(_cancel_exit)
	_exit_dialog.close_requested.connect(_cancel_exit)
	add_child(_exit_dialog)
	menu = Page.new()
	add_child(menu)
	menu.build(self)
	_dialog = ConfirmationDialog.new()
	add_child(_dialog)
	_continue = Button.new()
	_continue.text = "SAVE COMPLETED GAME"
	_continue.position = Vector2(24,24)
	_continue.pressed.connect(finish_game)
	add_child(_continue)
	_continue.hide()
	loadout = SeasonLoadoutUI.new()
	loadout.app = self
	add_child(loadout)
	loadout.process_mode = Node.PROCESS_MODE_DISABLED
	loadout.entry.hide() # Legacy equipment transactions are not routed into this new domain.
	show_season()

func _process(_delta: float) -> void:
	_continue.visible = is_instance_valid(lab) and lab._match_state!=null and lab._match_state.phase==MatchState.Phase.GAME_END and lab._match_presentation_director.mode==MatchPresentationDirector.Mode.OUTRO_HOLD and not _busy
	_continue.disabled = _busy

func show_season() -> void:
	if is_instance_valid(menu):
		menu.call("render")

func _available() -> bool:
	return not _busy and retained==null and not is_instance_valid(lab) and not Save.saving

func _clone() -> SeasonState:
	return Codec.clone(season) if season!=null else null

func create_fresh() -> void:
	# Initial import only. Never discover, replace or relabel a legacy save.
	if not _available() or season != null or predecessor.is_empty() or FileAccess.file_exists(Save.PATH) or FileAccess.file_exists(Save.PATH+".bak"):
		return
	_start_next(predecessor.duplicate(true),PackedByteArray())

func next_season() -> void:
	if not _available() or season == null or season.phase != SeasonState.Phase.COMPLETE or not season.physical.pending.is_empty():
		return
	var before: Dictionary = Codec.guard_snapshot(season)
	var bytes: PackedByteArray = FileAccess.get_file_as_bytes(Save.PATH)
	var disk: Variant = JSON.parse_string(bytes.get_string_from_utf8())
	if before.is_empty() or bytes.is_empty() or not ClubCareer.same(before,disk):
		notice = "The completed save changed. Reload before starting another season."
		show_season()
		return
	_start_next(before,bytes)

# Presentation choices are informational only. Runtime.create_next independently
# authenticates the complete archive and re-derives permission before any write.
func next_season_choices() -> Array:
	var rows: Variant = []
	if season != null and season.phase == SeasonState.Phase.COMPLETE:
		rows = season.career.runs
	elif season == null and predecessor.get("checkpoint") is Dictionary:
		var career_data: Variant = predecessor.checkpoint.get("career")
		if career_data is Dictionary: rows = career_data.get("runs",[])
	if not rows is Array: return []
	return load("res://src/gameplay/season/successor_owner_career_v1/start.gd").available_specs(rows)

func select_next_season(league: String, tier: int) -> void:
	if not _available() or (season != null and season.phase != SeasonState.Phase.COMPLETE): return
	for choice: Dictionary in next_season_choices():
		if choice.league == league and choice.tier == tier and choice.implemented:
			next_league = league
			next_tier = tier
			show_season()
			return

func _start_next(input: Dictionary, expected_primary: PackedByteArray) -> void:
	var seed_value: int = int(Time.get_unix_time_from_system()) & 0x7fffffff
	var candidate: SeasonState = Runtime.create_next(input,seed_value,next_league,next_tier)
	if candidate == null:
		notice = "The completed career could not advance safely. Existing saves remain unchanged."
		show_season()
		return
	_retain(candidate,expected_primary)

func reload_save() -> void:
	if not _available():
		return
	var candidate: SeasonState = Save.restore()
	if candidate==null:
		notice = Save.last_error
	else:
		season = candidate
		profile = "standard" if season.league_rules.is_empty() else String(season.league_rules.league)
		notice = ""
	show_season()

func draft(id: String) -> void:
	if not _available() or season==null or season.phase!=SeasonState.Phase.DRAFT or not season.offers().has(id):
		return
	var candidate: SeasonState = _clone()
	if candidate!=null and candidate.choose_player(id):
		_retain(candidate)

func _retain(candidate: SeasonState, expected_primary: Variant = null) -> void:
	# Equality guard only, not authentication. Save strictly authenticates candidate.
	if season != null and Codec.guard_snapshot(season).is_empty(): return
	if candidate==null or retained!=null:
		return
	retained = candidate
	retained_owner = season
	retained_before = Codec.guard_snapshot(season).duplicate(true) if season!=null else {}
	retained_lab = lab
	retained_state = lab._match_state if is_instance_valid(lab) else null
	retained_saved = false
	retained_saved_snapshot.clear()
	retained_saved_bytes = PackedByteArray()
	retained_primary_expected = expected_primary.duplicate() if expected_primary is PackedByteArray else expected_primary
	retry_save()

func retry_save() -> void:
	if _busy or retained == null: return
	if retained_saved and (retained_saved_snapshot.is_empty() or retained_saved_bytes.is_empty() or not FileAccess.file_exists(Save.PATH) or FileAccess.get_file_as_bytes(Save.PATH) != retained_saved_bytes or not ClubCareer.same(Codec.guard_snapshot(retained),retained_saved_snapshot)):
		notice = "Saved candidate changed; the exact saved snapshot must be recovered before promotion."
		return
	if _busy or retained==null or season!=retained_owner or lab!=retained_lab or (is_instance_valid(lab) and lab._match_state!=retained_state) or (season!=null and not ClubCareer.same(Codec.guard_snapshot(season),retained_before)):
		return
	_busy = true
	if not retained_saved:
		retained_saved = Save.save(retained,retained_primary_expected)
		if retained_saved:
			var saved_bytes: PackedByteArray = FileAccess.get_file_as_bytes(Save.PATH)
			var disk: Variant = JSON.parse_string(saved_bytes.get_string_from_utf8())
			var snapshot: Dictionary = Codec.guard_snapshot(retained)
			if not disk is Dictionary or snapshot.is_empty() or not ClubCareer.same(disk,snapshot):
				notice = "Saved candidate differs from its durable checkpoint; promotion is blocked."
				_busy = false
				return
			retained_saved_snapshot = snapshot.duplicate(true)
			retained_saved_bytes = saved_bytes.duplicate()
	if not retained_saved:
		notice = Save.last_error
		_busy = false
		show_season()
		return
	# The original completed lab remains attached until its exact candidate is durable.
	if is_instance_valid(lab) and not _close_match():
		notice = "Saved. Original match cleanup must finish before continuing."
		_busy = false
		return
	if season!=retained_owner or (season!=null and not ClubCareer.same(Codec.guard_snapshot(season),retained_before)):
		notice = "Saved candidate retained; original owner changed during cleanup."
		_busy = false
		return
	if not FileAccess.file_exists(Save.PATH) or FileAccess.get_file_as_bytes(Save.PATH) != retained_saved_bytes or not ClubCareer.same(Codec.guard_snapshot(retained),retained_saved_snapshot):
		notice = "Saved candidate changed during cleanup; promotion is blocked."
		_busy = false
		return
	season = retained
	retained = null
	retained_saved = false
	retained_owner = null
	retained_lab = null
	retained_state = null
	retained_report.clear()
	_clear_npc_pins()
	retained_before.clear()
	retained_saved_snapshot.clear()
	retained_saved_bytes = PackedByteArray()
	retained_primary_expected = null
	retained_human.clear()
	_busy = false
	_result_recorded = false
	_result_saved = false
	notice = ""
	show_season()

func play_next() -> void:
	if not _available() or season==null or not Codec.scope(season) or not season.physical.pending.is_empty():
		return
	var fixture: Dictionary = season.pending_fixture()
	var state: MatchState = season.make_match()
	var field: FieldDefinition = Runtime.field_for_fixture(fixture,season)
	if state==null or field==null or not state.stadium.is_empty() or not state.sure_shot.enable_current_evidence():
		notice = "This fixture could not be prepared."
		show_season()
		return
	_fixture_id = int(fixture.id)
	_result_recorded = false
	_result_saved = false
	_open_match(state,int(fixture.home)==0,true,field.id)

func _played_field_matches() -> bool:
	if season==null or not Codec.scope(season) or not is_instance_valid(lab) or lab._match_state==null:
		return false
	var fixture: Dictionary = season.pending_fixture()
	var expected: FieldDefinition = Runtime.field_for_fixture(fixture,season)
	return expected!=null and lab._field_definition==expected and lab._field_id==expected.id and lab._match_state.stadium.is_empty() and ClubCareer.same(lab._match_state.league_rules, season.league_rules) and _fixture_id==int(fixture.get("id",-1)) and lab._match_state.pitch_catalog==season.build.pitch_catalog_signature()

func finish_game() -> void:
	if _busy:
		return
	if retained!=null:
		retry_save()
		return
	var human: Array = Capture.capture(self)
	if human.is_empty():
		notice = "The actual completed game could not be captured. It remains open."
		return
	var candidate: SeasonState = _clone()
	if candidate==null:
		return
	candidate.physical.pending = {"human":human.duplicate(true),"reports":[]}
	retained_human = human.duplicate(true)
	_retain(candidate)

func _commit_result() -> bool:
	finish_game()
	return false

func _checkpoint() -> bool:
	return false # No inherited legacy transaction may write a new-domain season.

func resume_round() -> void:
	if not _available() or season==null or season.physical.pending.is_empty():
		return
	if not retained_report.is_empty():
		_checkpoint_report()
		return
	projection = Pending.project(season)
	if projection.has("error"):
		notice = str(projection.error)
		show_season()
		return
	if not projection.has("fixture"):
		_retain(projection.candidate)
		return
	var candidate: SeasonState = projection.candidate
	var fixture: Dictionary = projection.fixture
	var state: MatchState = Fixtures.match_for(candidate,fixture)
	var field: FieldDefinition = Runtime.field_for_fixture(fixture,candidate)
	if state==null or field==null:
		return
	_npc_owner = season
	_npc_before = Codec.guard_snapshot(season).duplicate(true)
	_npc_lab = lab
	_npc_state = lab._match_state if is_instance_valid(lab) else null
	if _npc_before.is_empty():
		_clear_npc_pins()
		return
	ai_runner = Runner.new()
	add_child(ai_runner)
	ai_runner.finished.connect(_ai_finished.bind(ai_runner))
	ai_runner.failed.connect(_ai_failed.bind(ai_runner))
	_busy = true
	if not ai_runner.start(state,Fixtures.seed_for(candidate,fixture),field.id):
		_ai_failed("The physical fixture could not start.",ai_runner)

func _npc_owner_matches() -> bool:
	return _npc_owner != null and season == _npc_owner and not _npc_before.is_empty() \
		and lab == _npc_lab and (not is_instance_valid(lab) or lab._match_state == _npc_state) \
		and ClubCareer.same(Codec.guard_snapshot(season),_npc_before)

func _clear_npc_pins() -> void:
	_npc_owner = null
	_npc_before.clear()
	_npc_lab = null
	_npc_state = null

func _ai_finished(report: Dictionary, runner: Node = null) -> void:
	# Only the currently bound runner may complete this job. Foreign or repeated
	# callbacks must not clear busy state, stop a newer runner or replace evidence.
	if runner == null or not is_instance_valid(runner) or runner != ai_runner or not _busy:
		return
	if retained != null or not retained_report.is_empty(): return
	if not _npc_owner_matches():
		_ai_failed("Completed physical job belongs to a changed owner; checkpoint was not changed.",runner)
		return
	_busy = false
	ai_runner.queue_free()
	ai_runner = null
	retained_report = report.duplicate(true)
	_checkpoint_report()

func _checkpoint_report() -> void:
	if _busy or retained != null or retained_report.is_empty() or not _npc_owner_matches():
		return
	# Fixed operation rederives the fixture from the strict matching primary.
	# Neither projection.candidate nor a caller-provided request is authority.
	_busy = true
	var transaction: Dictionary = Save.append_npc_report(season,retained_report.duplicate(true))
	_busy = false
	if transaction.candidate == null:
		notice = transaction.error
		show_season()
		return
	retained = transaction.candidate
	retained_owner = _npc_owner
	retained_before = _npc_before.duplicate(true)
	retained_lab = _npc_lab
	retained_state = _npc_state
	retained_saved = transaction.saved
	retained_saved_snapshot.clear()
	retained_saved_bytes = PackedByteArray()
	var expected: Variant = transaction.expected_primary
	retained_primary_expected = expected.duplicate() if expected is PackedByteArray else expected
	if not retained_saved:
		notice = transaction.error
		show_season()
		return # Explicit retry only; retain the exact report and failed candidate.
	var disk_bytes: PackedByteArray = FileAccess.get_file_as_bytes(Save.PATH)
	var disk: Variant = JSON.parse_string(disk_bytes.get_string_from_utf8())
	var snapshot: Dictionary = Codec.guard_snapshot(retained)
	if disk_bytes.is_empty() or not disk is Dictionary or snapshot.is_empty() or not ClubCareer.same(disk,snapshot):
		notice = "Saved physical report differs from its durable checkpoint; promotion is blocked."
		return
	retained_saved_snapshot = snapshot.duplicate(true)
	retained_saved_bytes = disk_bytes.duplicate()
	if not _npc_owner_matches():
		notice = "Saved physical report retained; original owner changed before promotion."
		return
	retry_save() # Saved pins prohibit a second write; normal guarded promotion only.

func _ai_failed(reason: String, runner: Node = null) -> void:
	if runner == null or not is_instance_valid(runner) or runner != ai_runner or not _busy:
		return
	_busy = false
	notice = reason
	ai_runner.queue_free()
	ai_runner = null
	_clear_npc_pins()
	show_season()

func open_pitch_shop() -> void:
	open_shop()

func open_shop() -> void:
	if not _available() or season==null or not Codec.scope(season) or not season.physical.pending.is_empty() or not season.shop_available():
		return
	if not season.build._visit.open:
		var revision: int = season.build.revision()
		if not commit_shop({"id":"shop-open:%d"%revision,"rev":revision,"op":"open"}):
			return
	for child: Node in menu.get_children():
		if child is SeasonShopWindow:
			child.popup_centered()
			return
	var window: SeasonShopWindow = Shop.new()
	window.app = self
	menu.add_child(window)
	window.popup_centered()

func shop_command(command: Dictionary) -> void:
	commit_shop(command)

func commit_shop(command: Dictionary) -> bool:
	if not _available() or season==null or not Codec.scope(season) or not season.physical.pending.is_empty() or not season.shop_available():
		return false
	if not SeasonBuild.SHOP_OPS.has(str(command.get("op", ""))):
		return false
	# Handled requests are not new persistence operations, even identical replays.
	var request_id: Variant = command.get("id")
	if request_id is String and not request_id.is_empty() and season.build._requests.has(request_id):
		return false
	var expected_primary: Variant = null
	if command.get("op") in ["pack_open","pack_pick"]:
		# Fixed pack path owns its one strict disk reconstruction and original command.
		# Its result never serves as a reusable validation token or an arbitrary writer.
		var prior_owner: SeasonState = season
		var prior_lab: PitchBatLab = lab
		var prior_state: MatchState = lab._match_state if is_instance_valid(lab) else null
		var prior_snapshot: Dictionary = Codec.guard_snapshot(season)
		if prior_snapshot.is_empty(): return false
		_busy = true
		var transaction: Dictionary = Save.commit_shop(season,command)
		_busy = false
		expected_primary = transaction.expected_primary
		if transaction.handled:
			if transaction.candidate == null:
				notice = transaction.error
				return false
			retained = transaction.candidate
			retained_owner = prior_owner
			retained_lab = prior_lab
			retained_state = prior_state
			retained_before = prior_snapshot.duplicate(true)
			retained_saved = transaction.saved
			retained_saved_snapshot.clear()
			retained_saved_bytes = PackedByteArray()
			retained_primary_expected = expected_primary.duplicate() if expected_primary is PackedByteArray else expected_primary
			if not retained_saved:
				# Do not automatically repeat a refused write. Explicit retry uses the
				# ordinary independent strict Save, never this earlier fixed-call state.
				notice = transaction.error
				show_season()
				return false
			var disk_bytes: PackedByteArray = FileAccess.get_file_as_bytes(Save.PATH)
			var disk: Variant = JSON.parse_string(disk_bytes.get_string_from_utf8())
			var snapshot: Dictionary = Codec.guard_snapshot(retained)
			if disk_bytes.is_empty() or not disk is Dictionary or snapshot.is_empty() or not ClubCareer.same(disk,snapshot):
				notice = "Saved purchase differs from its durable checkpoint; promotion is blocked."
				return false
			retained_saved_snapshot = snapshot.duplicate(true)
			retained_saved_bytes = disk_bytes.duplicate()
			if season != prior_owner or lab != prior_lab or (is_instance_valid(lab) and lab._match_state != prior_state) or not ClubCareer.same(Codec.guard_snapshot(prior_owner),prior_snapshot):
				notice = "Saved purchase retained; original owner changed before promotion."
				return false
			var paid: SeasonState = retained
			retry_save() # Durable pins are installed first; this cannot save a second time.
			return season == paid and retained == null
		# Only a missing primary or valid lineup-only difference uses strict fallback.
		if season != prior_owner or lab != prior_lab or (is_instance_valid(lab) and lab._match_state != prior_state) or not ClubCareer.same(Codec.guard_snapshot(prior_owner),prior_snapshot):
			return false
	var candidate: SeasonState = _clone()
	if candidate==null:
		notice = "The current shop checkpoint could not be validated."
		return false
	var next: SeasonBuild = candidate.build.candidate(command)
	if next==null:
		notice = candidate.build.last_error
		return false
	# Recruitment must replace the outgoing player's lineup position as well as
	# the build roster; never promote either until this exact candidate is durable.
	candidate.adopt_build(next)
	if candidate.build != next:
		notice = "The shop change could not be adopted; original progress remains available."
		return false
	_retain(candidate,expected_primary)
	return season==candidate and retained==null
