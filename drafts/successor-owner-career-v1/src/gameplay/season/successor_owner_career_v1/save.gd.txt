extends RefCounted
## Independent slot. Strict capture authenticates the complete detached envelope
## once; exact JSON equality/readback cannot introduce different semantic data.
const Codec = preload("codec.gd")
const PATH: String = "user://successor-owned-career-live-v3.json"
const MAX_BYTES: int = 32 * 1024 * 1024 # Retained predecessor plus active run; never prune history.
static var saving: bool = false
static var last_error: String = ""
# Private process-owned validated state. No serialized/caller validation tokens.
# These guards protect detached UI mutation and disk replacement, not arbitrary
# injected script execution in an owner-controlled offline process.
static var _session: SeasonState
static var _session_bytes: PackedByteArray = PackedByteArray()
static var _session_snapshot: Dictionary = {}
static var session_hits: int = 0 # Diagnostic only, never authority.

static func _forget_session() -> void:
	_session = null
	_session_bytes = PackedByteArray()
	_session_snapshot = {}

static func _clean_fork(season: SeasonState) -> SeasonState:
	if season == null or not season.physical.pending.is_empty(): return null
	var before: Dictionary = Codec.guard_snapshot(season)
	if before.is_empty(): return null
	# This existing exhaustive fork deliberately clears pending; zero-pending is
	# mandatory here. Every mutable runtime/build/career container is detached.
	var candidate: SeasonState = load("res://src/gameplay/season/successor_owner_career_v1/pending.gd")._fork_replayed_state(season)
	return candidate if candidate != null and ClubCareer.same(Codec.guard_snapshot(candidate),before) else null

static func _remember_session(valid: SeasonState, bytes: PackedByteArray) -> void:
	_forget_session()
	if bytes.is_empty() or not FileAccess.file_exists(PATH) or FileAccess.get_file_as_bytes(PATH) != bytes: return
	var parser: JSON = JSON.new()
	if parser.parse(bytes.get_string_from_utf8()) != OK: return
	var candidate: SeasonState = _clean_fork(valid)
	if candidate == null: return
	var snapshot: Dictionary = Codec.guard_snapshot(candidate)
	if not ClubCareer.same(snapshot,parser.data): return
	_session = candidate
	_session_bytes = bytes.duplicate()
	_session_snapshot = snapshot.duplicate(true)

static func _session_candidate(bytes: PackedByteArray, owner: Dictionary) -> SeasonState:
	if _session == null or bytes != _session_bytes or not ClubCareer.same(owner,_session_snapshot) \
		or not ClubCareer.same(Codec.guard_snapshot(_session),_session_snapshot):
		_forget_session()
		return null
	var candidate: SeasonState = _clean_fork(_session)
	if candidate != null: session_hits += 1
	return candidate

static func save(season: SeasonState, expected_primary: Variant = null) -> bool:
	if saving or load("res://src/gameplay/season/async_save/async_save_ui.gd").locked \
		or load("res://src/gameplay/season/async_save/async_save_coordinator.gd").leased:
		return false
	saving = true
	var result: bool = _save(season,expected_primary)
	saving = false
	return result

static func _save(season: SeasonState, expected_primary: Variant = null) -> bool:
	last_error = "The owned checkpoint could not be saved. Original progress remains available."
	if not _primary_matches(expected_primary): return false
	for suffix: String in ["", ".bak", ".tmp"]:
		if DirAccess.dir_exists_absolute(PATH+suffix): return false
	if FileAccess.file_exists(PATH+".tmp"): return false
	var old_exists: bool = FileAccess.file_exists(PATH)
	var old: PackedByteArray = FileAccess.get_file_as_bytes(PATH) if old_exists else PackedByteArray()
	if old_exists and (old.is_empty() or old.size() > MAX_BYTES): return false
	# Validate old bytes before authenticating the new candidate. No callbacks,
	# engine mutation, or caller-provided proof occurs after strict capture.
	var old_valid: bool = false
	if old_exists:
		var parser: JSON = JSON.new()
		if parser.parse(old.get_string_from_utf8()) == OK:
			old_valid = Codec.restore(parser.data) != null
	# Keep the independently reconstructed runtime as well as its snapshot.
	# Caller-hidden caches are not authenticated by serialized equality alone.
	var authenticated: SeasonState = Codec.clone(season)
	if authenticated == null: return false
	var data: Dictionary = Codec.guard_snapshot(authenticated)
	if data.is_empty(): return false
	var bytes: PackedByteArray = JSON.stringify(data,"",true,true).to_utf8_buffer()
	if bytes.is_empty() or bytes.size() > MAX_BYTES: return false
	var parsed: JSON = JSON.new()
	if parsed.parse(bytes.get_string_from_utf8()) != OK or not ClubCareer.same(parsed.data,data): return false
	if not ClubCareer.same(Codec.guard_snapshot(season),data): return false
	if not _primary_matches(expected_primary): return false
	# Recheck slot guards after validation; never replace an uncertain temp file.
	for suffix: String in ["", ".bak", ".tmp"]:
		if DirAccess.dir_exists_absolute(PATH+suffix): return false
	if FileAccess.file_exists(PATH+".tmp") or FileAccess.file_exists(PATH) != old_exists \
		or (old_exists and FileAccess.get_file_as_bytes(PATH) != old): return false
	var file: FileAccess = FileAccess.open(PATH+".tmp",FileAccess.WRITE)
	if file == null: return false
	file.store_buffer(bytes)
	file.flush()
	var error: Error = file.get_error()
	file.close()
	if error != OK or FileAccess.get_file_as_bytes(PATH+".tmp") != bytes: return false
	if FileAccess.file_exists(PATH) != old_exists or (old_exists and FileAccess.get_file_as_bytes(PATH) != old): return false
	if old_valid and DirAccess.copy_absolute(PATH,PATH+".bak") != OK: return false
	if DirAccess.rename_absolute(PATH+".tmp",PATH) != OK or FileAccess.get_file_as_bytes(PATH) != bytes: return false
	_remember_session(authenticated,bytes)
	last_error = ""
	return true

# Explicit recovery only for a new-slot import interrupted before promotion.
# Never replace an existing primary/backup, and never discard uncertain temp data.
static func recover_interrupted_import() -> bool:
	if saving or load("res://src/gameplay/season/async_save/async_save_ui.gd").locked \
		or load("res://src/gameplay/season/async_save/async_save_coordinator.gd").leased: return false
	saving = true
	var result: bool = _recover_interrupted_import()
	saving = false
	return result

static func _recover_interrupted_import() -> bool:
	last_error = "Interrupted save could not be recovered. All files are preserved."
	for suffix: String in ["", ".bak", ".tmp"]:
		if DirAccess.dir_exists_absolute(PATH+suffix): return false
	if FileAccess.file_exists(PATH) or FileAccess.file_exists(PATH+".bak") or not FileAccess.file_exists(PATH+".tmp"): return false
	var file: FileAccess = FileAccess.open(PATH+".tmp",FileAccess.READ)
	if file == null or file.get_length() > MAX_BYTES: return false
	var bytes: PackedByteArray = file.get_buffer(file.get_length())
	file.close()
	var parser: JSON = JSON.new()
	if bytes.is_empty() or parser.parse(bytes.get_string_from_utf8()) != OK: return false
	var valid: SeasonState = Codec.restore(parser.data)
	if valid == null or not ClubCareer.same(Codec.guard_snapshot(valid),parser.data): return false
	if FileAccess.file_exists(PATH) or FileAccess.file_exists(PATH+".bak") \
		or DirAccess.dir_exists_absolute(PATH) or DirAccess.dir_exists_absolute(PATH+".bak") \
		or FileAccess.get_file_as_bytes(PATH+".tmp") != bytes: return false
	if DirAccess.rename_absolute(PATH+".tmp",PATH) != OK or FileAccess.get_file_as_bytes(PATH) != bytes: return false
	last_error = ""
	return true

static func restore() -> SeasonState:
	_forget_session()
	var primary: PackedByteArray = PackedByteArray()
	var file: FileAccess = FileAccess.open(PATH,FileAccess.READ)
	if file != null:
		if file.get_length() <= MAX_BYTES: primary = file.get_buffer(file.get_length())
		file.close()
	var result: SeasonState = _read(PATH)
	if result != null: _remember_session(result,primary)
	else: result = _read(PATH+".bak")
	last_error = "" if result != null else "No valid owned continuation checkpoint is available."
	return result

static func _read(path: String) -> SeasonState:
	var file: FileAccess = FileAccess.open(path,FileAccess.READ)
	if file == null or file.get_length() > MAX_BYTES: return null
	var parser: JSON = JSON.new()
	if parser.parse(file.get_as_text()) != OK: return null
	return Codec.restore(parser.data)

# Closed synchronous pack-or-NPC dispatcher. No API accepts a validated envelope
# or proof. Each private candidate starts with strict primary-disk restoration;
# only an original pack operation or one validated pending report may change it.
static func commit_draft(owner: SeasonState, player: String) -> Dictionary:
	if saving or load("res://src/gameplay/season/async_save/async_save_ui.gd").locked \
		or load("res://src/gameplay/season/async_save/async_save_coordinator.gd").leased:
		return {"handled":true,"candidate":null,"saved":false,"error":"Save is busy.","expected_primary":null}
	saving = true
	var roots: Array[Resource] = Codec._operation_content_roots()
	var result: Dictionary = _commit_fixed(owner,{"op":"draft","player":player},null)
	roots.clear()
	saving = false
	last_error = result.error
	return result

static func commit_shop(owner: SeasonState, command: Dictionary) -> Dictionary:
	var fallback: Dictionary = {"handled":false,"candidate":null,"saved":false,"error":"","expected_primary":null}
	if command.get("op") not in ["pack_open","pack_pick"]: return fallback
	if saving or load("res://src/gameplay/season/async_save/async_save_ui.gd").locked \
		or load("res://src/gameplay/season/async_save/async_save_coordinator.gd").leased:
		return {"handled":true,"candidate":null,"saved":false,"error":"Save is busy.","expected_primary":null}
	saving = true
	var roots: Array[Resource] = Codec._operation_content_roots()
	var result: Dictionary = _commit_shop(owner,command)
	roots.clear()
	saving = false
	if result.handled: last_error = result.error
	return result

static func _commit_shop(owner: SeasonState, command: Dictionary) -> Dictionary:
	return _commit_fixed(owner,command,null)

# Raw report only. Missing or different primary is a hard rejection, not a
# fallback that might write over newer progress. No supplied fixture or proof.
static func append_npc_report(owner: SeasonState, raw_report: Dictionary) -> Dictionary:
	if saving or load("res://src/gameplay/season/async_save/async_save_ui.gd").locked \
		or load("res://src/gameplay/season/async_save/async_save_coordinator.gd").leased:
		return {"handled":true,"candidate":null,"saved":false,"error":"Save is busy.","expected_primary":null}
	saving = true
	var roots: Array[Resource] = Codec._operation_content_roots()
	var result: Dictionary = _commit_fixed(owner,{},raw_report)
	roots.clear()
	saving = false
	last_error = result.error
	return result

# Operation selection is internal; each branch uses a canonical private fork
# or strict primary replay and validates its raw action. No caller validation token.
static func _commit_fixed(owner: SeasonState, command: Dictionary, raw_report: Variant) -> Dictionary:
	var npc: bool = raw_report != null
	var draft: bool = not npc and command.get("op") == "draft"
	var fallback: Dictionary = {"handled":false,"candidate":null,"saved":false,"error":"","expected_primary":null}
	var rejected: Dictionary = {"handled":true,"candidate":null,"saved":false,"error":"Invalid owned shop transaction.","expected_primary":null}
	if owner == null or not Codec.scope(owner): return rejected
	var request_id: Variant = command.get("id")
	if npc:
		rejected.error = "The physical report cannot be appended to this owned checkpoint."
		if owner.physical.pending.is_empty() or not raw_report is Dictionary or raw_report.is_empty() \
			or not _primitive_report_graph(raw_report): return rejected
		# The frozen report core accepts typed ownership arguments before its late
		# wrapper checks. Reject malformed elements before reaching that boundary.
		if raw_report.get("ownership") is Array:
			for entry: Variant in raw_report.ownership:
				if not entry is Dictionary: return rejected
	else:
		if draft:
			rejected.error = "This draft choice cannot be saved from the current checkpoint."
			if owner.phase != SeasonState.Phase.DRAFT or not command.get("player") is String or not owner.offers().has(command.player): return rejected
		elif command.get("op") not in ["pack_open","pack_pick"] or not owner.shop_available(): return rejected
		if not owner.physical.pending.is_empty(): return rejected
		# Pack commands are flat primitive records, with original exact validation.
		for key: Variant in command:
			if not key is String or typeof(command[key]) not in [TYPE_NIL,TYPE_BOOL,TYPE_INT,TYPE_FLOAT,TYPE_STRING]: return rejected
		if request_id is String and not request_id.is_empty() and owner.build._requests.has(request_id):
			rejected.error = "This shop request has already been handled."
			return rejected
	var before: Dictionary = Codec.guard_snapshot(owner)
	if before.is_empty(): return rejected
	# Only absence or benign unsaved lineup changes permit strict fallback. A
	# different/corrupt existing checkpoint is never silently overwritten here.
	if not FileAccess.file_exists(PATH):
		if npc or draft or DirAccess.dir_exists_absolute(PATH): return rejected
		fallback.expected_primary = PackedByteArray()
		return fallback
	var old: PackedByteArray = FileAccess.get_file_as_bytes(PATH)
	rejected.expected_primary = old.duplicate()
	if old.is_empty() or old.size() > MAX_BYTES: return rejected
	var parser: JSON = JSON.new()
	if parser.parse(old.get_string_from_utf8()) != OK: return rejected
	if not ClubCareer.same(parser.data,before):
		if not npc and not draft and _lineup_only(parser.data,before):
			fallback.expected_primary = old.duplicate()
			return fallback
		rejected.error = "The saved checkpoint differs from this owner. Reload before purchasing."
		return rejected
	# Only fixed draft/pack commands may fork the private canonical session.
	# NPC reports retain strict replay; all misses use the unchanged strict reader.
	var session_allowed: bool = not npc and (draft or command.get("op") in ["pack_open","pack_pick"])
	var candidate: SeasonState = _session_candidate(old,before) if session_allowed else null
	if candidate == null: candidate = Codec.restore(parser.data)
	if candidate == null: return rejected
	if not ClubCareer.same(Codec.guard_snapshot(candidate),before) or not ClubCareer.same(Codec.guard_snapshot(owner),before):
		return rejected
	if npc:
		var pending = load("res://src/gameplay/season/successor_owner_career_v1/pending.gd")
		var fixtures = load("res://src/gameplay/season/successor_owner_career_v1/physical_fixtures.gd")
		if not pending._pending_shape(candidate): return rejected
		# This fork's settled prefix was strictly replayed above in this call.
		# Its projection may compute rewards, but it is never persisted/published.
		# Forks retain authored Resources; local roots preserve lifetime, not a
		# generic immutability proof. No caller Objects/callbacks enter; fresh
		# source/state guards below must still match before persistence.
		var discovery: SeasonState = pending._fork_replayed_state(candidate)
		if discovery == null: return rejected
		var projection: Dictionary = pending._project_candidate(discovery,candidate.physical.pending.duplicate(true))
		if projection.has("error") or not projection.get("fixture") is Dictionary or not projection.get("candidate") is SeasonState: return rejected
		var fixture: Dictionary = projection.fixture
		var request: String = fixtures.request_key(projection.candidate,fixture)
		if request.is_empty(): return rejected
		candidate.physical.pending.reports.append({"fixture":int(fixture.id),"request":request,"report":raw_report.duplicate(true)})
		var expected_pending: Dictionary = candidate.physical.pending.duplicate(true)
		# Validate the entire augmented pending archive through the unchanged full
		# report/correspondence/receipt path on another disposable replay fork.
		if not pending._valid_pending_replayed(candidate) or not ClubCareer.same(candidate.physical.pending,expected_pending): return rejected
		var changed: Dictionary = Codec.guard_snapshot(candidate)
		if changed.is_empty(): return rejected
		var unchanged: Dictionary = changed.duplicate(true)
		unchanged.active.checkpoint.physical.pending = before.active.checkpoint.physical.pending.duplicate(true)
		if not ClubCareer.same(unchanged,before): return rejected
	elif draft:
		if candidate.phase != SeasonState.Phase.DRAFT or not candidate._choose_local(command.player): return rejected
		var after: Dictionary = Codec.guard_snapshot(candidate)
		for key: String in ["origin","origin_close","closed"]:
			if not ClubCareer.same(after.get(key),before.get(key)): return rejected
	else:
		if not candidate.physical.pending.is_empty() or not candidate.shop_available() or candidate.build._requests.has(request_id): return rejected
		var next: SeasonBuild = candidate.build.candidate(command.duplicate(true))
		if next == null:
			rejected.error = candidate.build.last_error
			return rejected
		candidate.adopt_build(next)
		if candidate.build != next: return rejected # Adoption synchronized, or rolled back.
	# From here a valid private operation exists. Every I/O refusal retains this
	# exact candidate; any later retry must use ordinary strict Save.save.
	var result: Dictionary = {"handled":true,"candidate":candidate,"saved":false,"expected_primary":old.duplicate(),
		"error":"The owned checkpoint could not be saved. The exact candidate is retained for retry."}
	var data: Dictionary = Codec.guard_snapshot(candidate)
	if data.is_empty() or not ClubCareer.same(Codec.guard_snapshot(owner),before): return result
	# Equality is not a generic authentication shortcut: the only changed object
	# here is a canonical detached candidate after a fixed command or validated NPC pending addition.
	for key: String in ["version","format","owner_source","pitch_catalog"]:
		if not ClubCareer.same(data.get(key),before.get(key)): return result
	# NPC schema validation above precedes serialization. The full envelope
	# contains the raw report, so its existing cap also bounds report JSON bytes;
	# no redundant stringify of arbitrary unvalidated/deep input is needed.
	var bytes: PackedByteArray = JSON.stringify(data,"",true,true).to_utf8_buffer()
	if bytes.is_empty() or bytes.size() > MAX_BYTES: return result
	var parsed: JSON = JSON.new()
	if parsed.parse(bytes.get_string_from_utf8()) != OK or not ClubCareer.same(parsed.data,data): return result
	if not ClubCareer.same(Codec.guard_snapshot(candidate),data) or not ClubCareer.same(Codec.guard_snapshot(owner),before): return result
	# Deliberately after candidate creation: prewrite faults retain the exact pack/report candidate,
	# just as the ordinary App transaction does. Never overwrite uncertain files.
	for suffix: String in ["", ".bak", ".tmp"]:
		if DirAccess.dir_exists_absolute(PATH+suffix): return result
	if FileAccess.file_exists(PATH+".tmp") or not FileAccess.file_exists(PATH) or FileAccess.get_file_as_bytes(PATH) != old: return result
	var file: FileAccess = FileAccess.open(PATH+".tmp",FileAccess.WRITE)
	if file == null: return result
	file.store_buffer(bytes)
	file.flush()
	var error: Error = file.get_error()
	file.close()
	if error != OK or FileAccess.get_file_as_bytes(PATH+".tmp") != bytes: return result
	if not FileAccess.file_exists(PATH) or FileAccess.get_file_as_bytes(PATH) != old: return result
	if DirAccess.copy_absolute(PATH,PATH+".bak") != OK: return result
	if DirAccess.rename_absolute(PATH+".tmp",PATH) != OK or FileAccess.get_file_as_bytes(PATH) != bytes: return result
	_remember_session(candidate,bytes)
	result.saved = true
	result.error = ""
	return result

# Optional compare-and-swap expectation only, never a validation proof. Empty
# bytes explicitly pin absence; null preserves the original unpinned Save API.
static func _primary_matches(expected: Variant) -> bool:
	if expected == null: return true
	if not expected is PackedByteArray: return false
	if DirAccess.dir_exists_absolute(PATH): return false
	if expected.is_empty(): return not FileAccess.file_exists(PATH)
	return FileAccess.file_exists(PATH) and FileAccess.get_file_as_bytes(PATH) == expected

static func _lineup_only(disk: Variant, live: Dictionary) -> bool:
	if not disk is Dictionary or not disk.get("active") is Dictionary or not live.get("active") is Dictionary \
		or not disk.active.get("checkpoint") is Dictionary or not live.active.get("checkpoint") is Dictionary:
		return false
	var a: Dictionary = disk.active.checkpoint
	var b: Dictionary = live.active.checkpoint
	for row: Dictionary in [a,b]:
		if not row.get("teams") is Array or row.teams.size() != 6 or not row.teams[0] is Dictionary \
			or not row.teams[0].get("roster") is Array or row.teams[0].roster.size() != 4 \
			or not SeasonOwnership._whole(row.get("starter_index"),0,3) \
			or not SeasonOwnership._whole(row.get("fielder_index"),0,3) \
			or row.starter_index == row.fielder_index:
			return false
		var ids: Array = []
		for id: Variant in row.teams[0].roster:
			if not id is String or id.is_empty() or ids.has(id): return false
			ids.append(id)
	for id: String in a.teams[0].roster:
		if not b.teams[0].roster.has(id): return false
	var compared: Dictionary = disk.duplicate(true)
	compared.active.checkpoint.teams[0].roster = b.teams[0].roster.duplicate()
	compared.active.checkpoint.starter_index = b.starter_index
	compared.active.checkpoint.fielder_index = b.fielder_index
	return ClubCareer.same(compared,live)

# Shape/safety preflight only, never report validity or persistence authority.
# Each visited value/key and each UTF-8 string byte consumes at least one JSON
# byte. These conservative work bounds cannot exclude a report that fits inside
# the existing MAX_BYTES complete envelope. No fast-path depth limit is admission.
static func _primitive_report_graph(value: Variant) -> bool:
	var remaining: int = MAX_BYTES
	var string_bytes: int = MAX_BYTES
	# Frames are [value, exiting]. Only current ancestors identify cycles: shared
	# acyclic containers are permitted and charged again, as JSON encodes them.
	var stack: Array = [[value,false]]
	var ancestors: Array = []
	while not stack.is_empty():
		var frame: Array = stack.pop_back()
		var current: Variant = frame[0]
		if frame[1]:
			ancestors.pop_back()
			continue
		remaining -= 1
		if remaining < 0: return false
		match typeof(current):
			TYPE_NIL, TYPE_BOOL, TYPE_INT:
				pass
			TYPE_FLOAT:
				if not is_finite(current): return false
			TYPE_STRING:
				if current.length() > string_bytes: return false
				string_bytes -= current.to_utf8_buffer().size()
				if string_bytes < 0: return false
			TYPE_ARRAY, TYPE_DICTIONARY:
				for ancestor: Variant in ancestors:
					if is_same(current,ancestor): return false
				var children: int = current.size() * (2 if current is Dictionary else 1)
				# Include pending entry frames in the work bound; exit frames merely
				# unwind ancestors and do not represent serialized data.
				var pending: int = stack.size() - ancestors.size()
				if children > remaining - pending: return false
				ancestors.append(current)
				stack.append([current,true])
				if current is Dictionary:
					for key: Variant in current:
						if not key is String: return false
						stack.append([current[key],false])
						stack.append([key,false])
				else:
					for child: Variant in current:
						stack.append([child,false])
			_:
				return false
	return true
