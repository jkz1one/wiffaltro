extends RefCounted
## Full archive authentication. Flat snapshots/guards are equality checks only.
const Catalog = preload("catalog.gd")
const Start = preload("start.gd")
const Boundary = preload("res://src/gameplay/season/successor_owner_terminal_v1/codec.gd")
const Origin = preload("res://src/gameplay/season/successor_owner_general_v1/codec.gd")
const Primitive = preload("res://src/gameplay/season/successor_owner_general_v1/save.gd")

static func scope(season: SeasonState) -> bool:
	return Start.structural(season) and Start.base_matches(season)

static func _snapshot(season: SeasonState) -> Dictionary:
	# Exactly the original durable season fields, excluding separately stored career.
	# origin/origin_close/closed are the ONLY extra runtime fields.
	if not scope(season) or season._heat_replay_team != null or season._doubleheader_replay_build != null \
		or season._pool_replay_ledger != null or season._npc_replay_run != -1 \
		or season.physical.projecting or season.physical.cursor != season.physical.reports.size(): return {}
	for property: Dictionary in season.get_property_list():
		if int(property.usage) & PROPERTY_USAGE_SCRIPT_VARIABLE:
			if not Boundary.FIELDS.has(property.name) and not Boundary.OBJECTS.has(property.name) \
				and property.name not in ["origin","origin_close","closed"]: return {}
	var state: Dictionary = {}
	for field: String in Boundary.FIELDS:
		var value: Variant = season.get(field)
		state[field] = value.duplicate(true) if value is Array or value is Dictionary else value
	# Dot assignment creates StringName keys for these new fields. The primitive
	# envelope contract requires ordinary String keys, just like JSON restores.
	state["ownership"] = season.ownership.to_data()
	state["build"] = season.build.to_data()
	state["development"] = season.build._book.to_data()
	state["build_view"] = season.build.view()
	state["opponents"] = season.opponents.to_data()
	state["physical"] = season.physical.to_data()
	return {} if state.build.is_empty() or state.development.is_empty() else state

static func _bounded(value: Variant) -> bool:
	# Reuse frozen primitive eligibility only, NEVER its report/owner authority.
	# Cycles/custom objects cannot enter JSON; final encoded envelope owns the cap.
	return _json_depth_safe(value) and Primitive._primitive_report_graph(value) and JSON.stringify(value,"",true,true).to_utf8_buffer().size() <= Catalog.MAX_BYTES

static func _json_depth_safe(value: Variant) -> bool:
	# The closed nonrecursive schema is far shallower than this. Refuse hostile
	# primitive nesting before native JSON serialization; never truncate a save.
	var stack: Array = [[value,0]]
	var remaining: int = Catalog.MAX_BYTES
	while not stack.is_empty():
		if remaining <= 0: return false
		remaining -= 1
		var item: Array = stack.pop_back()
		if item[1] > 128: return false
		var current: Variant = item[0]
		if current is Array or current is Dictionary:
			if current.size() > remaining-stack.size(): return false
			for child: Variant in current.values() if current is Dictionary else current:
				if child is Array or child is Dictionary: stack.append([child,item[1]+1])
	return true

static func _envelope(season: SeasonState, state: Dictionary) -> Dictionary:
	if state.is_empty() or not season.get("origin") is Dictionary or not season.get("closed") is Array \
		or season.closed.size() >= ClubCareer.MAX_RUNS: return {}
	var source: String = Catalog.signature()
	var game: String = Catalog.Game.signature()
	if source.is_empty() or game.is_empty(): return {}
	var value: Dictionary = {"version":Catalog.VERSION,"format":Catalog.FORMAT,"owner_source":source,
		"pitch_catalog":game,"origin":season.origin,"origin_close":season.origin_close,"closed":season.closed,
		"active":{"id":season.career.current,"checkpoint":state},"career":season.career.to_data()}
	return value.duplicate(true) if _bounded(value) else {}

static func guard_snapshot(season: SeasonState) -> Dictionary:
	# NONAUTHORITY: fresh source + whole archive equality, no replay or grants.
	var roots: Array[Resource] = _operation_content_roots()
	var result: Dictionary = _envelope(season,_snapshot(season))
	roots.clear()
	return result

static func envelope_valid(value: Variant) -> bool:
	return value is Dictionary and SeasonOwnership._keys(value,["version","format","owner_source","pitch_catalog","origin","origin_close","closed","active","career"]) \
		and SeasonOwnership._whole(value.version,Catalog.VERSION,Catalog.VERSION) and value.format == Catalog.FORMAT \
		and value.owner_source is String and not value.owner_source.is_empty() and value.owner_source == Catalog.signature() \
		and value.pitch_catalog is String and not value.pitch_catalog.is_empty() and value.pitch_catalog == Catalog.Game.signature() \
		and value.origin is Dictionary and value.origin_close in ["completed","abandoned"] and value.closed is Array and value.closed.size() < ClubCareer.MAX_RUNS \
		and value.active is Dictionary and SeasonOwnership._keys(value.active,["id","checkpoint"]) \
		and SeasonOwnership._whole(value.active.id,1,ClubCareer.MAX_RUNS) and value.active.checkpoint is Dictionary \
		and value.career is Dictionary and _bounded(value)

static func _operation_content_roots() -> Array[Resource]:
	return Origin._operation_content_roots()

static func clone(season: SeasonState) -> SeasonState:
	var roots: Array[Resource] = _operation_content_roots()
	var value: Dictionary = guard_snapshot(season)
	var result: SeasonState = null
	if not value.is_empty(): result = load("res://src/gameplay/season/successor_owner_career_v1/decoder.gd").restore(value)
	roots.clear()
	return result

static func capture(season: SeasonState) -> Dictionary:
	var roots: Array[Resource] = _operation_content_roots()
	var value: Dictionary = guard_snapshot(season)
	var result: Dictionary = {}
	if not value.is_empty():
		var checked: SeasonState = load("res://src/gameplay/season/successor_owner_career_v1/decoder.gd").restore(value)
		if checked != null and ClubCareer.same(guard_snapshot(checked),value): result = value
	roots.clear()
	return result

static func restore(value: Variant) -> SeasonState:
	var roots: Array[Resource] = _operation_content_roots()
	var result: SeasonState = null
	if envelope_valid(value): result = load("res://src/gameplay/season/successor_owner_career_v1/decoder.gd").restore(value)
	roots.clear()
	return result
