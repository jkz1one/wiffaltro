# Informational only: public constructors independently replay the whole archive.
# Keep unlocked-but-not-integrated tiers visible, without admitting unsupported play.
static func available_specs(runs: Array) -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	for row: Variant in runs:
		if not row is Dictionary or row.get("status") not in ["completed","abandoned"]: return result
	var progress: Dictionary = Policy.evaluate(runs)
	if progress.is_empty(): return result
	for league: String in progress.available_leagues:
		for tier in range(5):
			if tier > 0 and not progress.titles.get(ClubLeagueCareer.title_key(league,tier-1),false): continue
			result.append({"league":league,"tier":tier,"implemented":league != "dynasty" and tier == 0})
	return result

static func specification(seed: int, league: String, tier: int) -> Dictionary:
	if not SeasonOwnership._whole(seed,0,2147483647) or not SeasonOwnership._whole(tier,0,4) \
		or league not in LeagueUnlocks.IDS or league == "dynasty" or tier != 0: return {}
	var rules: Dictionary = {} if league == "standard" and tier == 0 else LeagueRules.capture(league,tier,seed if league == "scramble" else null)
	var live: bool = not rules.is_empty()
	if not Ordinary.supported(rules,live): return {}
	return {"league":league,"tier":tier,"seed":seed,"rules":rules,"live":live}

static func admitted(basis: Dictionary, spec: Dictionary) -> bool:
	if spec.is_empty() or not basis.get("runs") is Array: return false
	for option: Dictionary in available_specs(basis.runs):
		if option.implemented and option.league == spec.league and option.tier == spec.tier: return true
	return false

static func state_spec(season: SeasonState) -> Dictionary:
	if season == null: return {}
	var league: String = str(season.league_rules.get("league","standard"))
	var tier: int = int(season.league_rules.get("tier",0))
	var spec: Dictionary = specification(season.season_seed,league,tier)
	return spec if not spec.is_empty() and ClubCareer.same(spec.rules,season.league_rules) and spec.live == season.league_live else {}

# Decoder-only synchronous birth. Public APIs must authenticate the full archive;
# neither this basis nor the resulting live object grants caller authority.
static func _create(seed_value: int, basis: Dictionary, league: String = "dugout", tier: int = 0) -> SeasonState:
	var spec: Dictionary = specification(seed_value,league,tier)
	if not _basis_shape(basis) or not admitted(basis,spec): return null
	var season: SeasonState = load("res://src/gameplay/season/successor_owner_career_v1/runtime.gd").new()
	season.season_seed = seed_value
	season.league_rules = spec.rules.duplicate(true)
	season.league_live = spec.live
	season.build = GameBuild.new(seed_value)
	if not season.build.enable_successor_pitch_catalog() or (season.league_live and not season.build.enable_league_rules(season.league_rules)):
		return null
	season.opponents = GameOpponents.new()
	season.opponents._format = 13
	season.physical = GamePhysical.new()
	var rng: RandomNumberGenerator = RandomNumberGenerator.new()
	rng.seed = seed_value
	season.draft_pool = ClubPlayerCatalog.draft_pool(rng,PitchBatLab.DEBUG_PLAYER_ID,false,SeasonState.LEGACY_IDS)
	for index in range(6):
		season.teams.append({"name":SeasonState.TEAM_NAMES[index],"roster":[],"draw":rng.randf()})
	season._build_schedule()
	load("res://src/gameplay/season/successor_league_general_v2/doubleheader.gd").install_schedule(season)
	if not _install(season.build,basis) or not ClubPlayerDraft.apply(season,basis.player_access):
		return null
	var career: ClubCareer = load("res://src/gameplay/season/successor_owner_career_v1/career.gd").new()
	career.current = int(basis.target_run)
	career.start_basis = basis.duplicate(true)
	career._assets = basis.assets.duplicate(true)
	career._pennants = load("res://src/gameplay/season/successor_owner_career_v1/settlement.gd").birth_pennants(basis)
	if career._pennants.is_empty(): return null
	career._plaques = basis.plaques.duplicate(true)
	career._pitch_catalog = load("res://src/gameplay/season/successor_owner_career_v1/catalog.gd").Game.signature()
	career.runs.assign(basis.runs.duplicate(true))
	career.runs.append({})
	season.career = career
	season.league_career = true
	return season if career.sync(season) else null

static func _install(build: SeasonBuild, basis: Dictionary) -> bool:
	var s: Dictionary = basis.starts
	build._gear_progress.enabled = true
	build._gear_progress.start = s.gear.duplicate(true)
	build._sponsor_progress.enabled = true
	build._sponsor_progress.start = s.sponsors.duplicate(true)
	build._order_start = s.order
	build._rain_start = s.rain
	build._transfer_start = s.transfer
	build._supply_start = s.supply
	build._checkout_start = s.checkout
	build._association_start = s.association
	build._freezer_start = s.freezer
	build._sides_start = s.sides
	build._jump_start = s.jump
	build._sure_start = s.sure
	build._major.start = s.major
	build._abilities.start = s.abilities
	build._copy.start = s.copy
	build._field_start = s.field
	build._batch_start = s.batch.duplicate(true)
	build._pool.assign(basis.player_access)
	return build._book.install_exotic_start(basis.exotic_history)

static func structural(season: SeasonState) -> bool:
	if season == null or season.get_script() != load("res://src/gameplay/season/successor_owner_career_v1/runtime.gd") \
		or season.ownership == null \
		or season.build == null or season.build.get_script() != GameBuild \
		or season.opponents == null or season.opponents.get_script() != GameOpponents \
		or season.physical == null or season.physical.get_script() != GamePhysical \
		or season.career == null or season.career.get_script() != load("res://src/gameplay/season/successor_owner_career_v1/career.gd") \
		or not season.league_career or state_spec(season).is_empty() \
		or season.picks.size() > 4 \
		or season.build._book == null or season.build._bank == null \
		or season.build._gear_progress == null or season.build._sponsor_progress == null \
		or season.build._major == null or season.build._abilities == null or season.build._copy == null:
		return false
	# Structural consistency also runs inside exact replay, where heat-team and
	# projecting are valid private transients. Public Codec checks those separately.
	if season.phase == SeasonState.Phase.DRAFT:
		if season.picks.size() >= 4 or not season.results.is_empty() or not season.player_results.is_empty(): return false
	elif season.picks.size() != 4:
		return false
	elif season.phase == SeasonState.Phase.REGULAR:
		if not SeasonOwnership._whole(season.round_index,0,9): return false
	elif season.phase in [SeasonState.Phase.SEMIFINAL,SeasonState.Phase.FINAL,SeasonState.Phase.COMPLETE]:
		if season.round_index != 10: return false
	else:
		return false
	if not season.inventory_rules.is_empty() or not season._closed_access.is_empty() \
		or not season._dynasty_start.is_empty() or not season._dynasty_owner_binding.is_empty() \
		or not season._dynasty_career_session.is_empty() or not season._league_home_context.is_empty() \
		or not season._standard_live_context.is_empty() or not season._standard_bound_grant.is_empty() \
		or not season.home_property.is_empty() or not season.npc_venues.is_empty() \
		or not season.final_venue.is_empty() or not season.owned_surface_night.is_empty() \
		or not Ordinary.build_matches(season.build,season.league_rules) \
		or season.build._market != 0 or season.opponents._format != 13:
		return false
	# Permission belongs to _create after full archive replay. This hot structural
	# path is not authority and must not re-evaluate all closed history per pitch.
	return base_matches(season) and _opponents_match(season)

static func _opponents_match(season: SeasonState) -> bool:
	if not season.league_rules.is_empty(): return season._league_opponent_context_consistent()
	for club: Dictionary in season.opponents.clubs.values():
		var build: SeasonBuild = club.get("build")
		if build == null or build.get_script() != GameBuild or not Ordinary.build_matches(build,{}) \
			or build._market != 12 or not build._dynasty_start.is_empty(): return false
	return true

# Exact correspondence to this owner's rederived start basis, not independent
# authentication of that basis. Public creation/decoder independently reconstructs the complete closed prefix.
static func base_matches(season: SeasonState) -> bool:
	if season == null or season.build == null or season.career == null \
		or season.career.get_script() != load("res://src/gameplay/season/successor_owner_career_v1/career.gd") \
		or season.build.get_script() != GameBuild or season.build._book == null \
		or season.build._book.get_script() != preload("res://src/gameplay/season/successor_league_general_v2/development.gd") \
		or season.build._gear_progress == null or season.build._sponsor_progress == null \
		or season.build._major == null or season.build._abilities == null or season.build._copy == null:
		return false
	var career: ClubCareer = season.career
	if not _basis_shape(career.start_basis):
		return false
	var s: Dictionary = career.start_basis.starts
	if not season.build._gear_progress.enabled or not season.build._sponsor_progress.enabled \
		or not ClubCareer.same(season.build._gear_progress.start,s.gear) \
		or not ClubCareer.same(season.build._sponsor_progress.start,s.sponsors) \
		or not ClubCareer.same(season.build._book._exotic_start,career.start_basis.exotic_history) \
		or not ClubCareer.same(season.build._pool,career.start_basis.player_access):
		return false
	if not ClubCareer.same(season.build._order_start,s.order):
		return false
	if not ClubCareer.same(season.build._rain_start,s.rain):
		return false
	if not ClubCareer.same(season.build._transfer_start,s.transfer):
		return false
	if not ClubCareer.same(season.build._supply_start,s.supply):
		return false
	if not ClubCareer.same(season.build._checkout_start,s.checkout):
		return false
	if not ClubCareer.same(season.build._association_start,s.association):
		return false
	if not ClubCareer.same(season.build._freezer_start,s.freezer):
		return false
	if not ClubCareer.same(season.build._sides_start,s.sides):
		return false
	if not ClubCareer.same(season.build._jump_start,s.jump):
		return false
	if not ClubCareer.same(season.build._sure_start,s.sure):
		return false
	if not ClubCareer.same(season.build._major.start,s.major):
		return false
	if not ClubCareer.same(season.build._abilities.start,s.abilities):
		return false
	if not ClubCareer.same(season.build._copy.start,s.copy):
		return false
	if not ClubCareer.same(season.build._field_start,s.field):
		return false
	if not ClubCareer.same(season.build._batch_start,s.batch):
		return false
	return _prefix_matches(career) \
		and (season.phase == SeasonState.Phase.COMPLETE or (ClubCareer.same(career._assets,career.start_basis.assets) \
		and ClubCareer.same(career._pennants,load("res://src/gameplay/season/successor_owner_career_v1/settlement.gd").birth_pennants(career.start_basis)) \
		and ClubCareer.same(career._plaques,career.start_basis.plaques))) \
		and ClubPlayerDraft.frozen_matches(season) \
		and ClubCareer.same(season.player_access,career.start_basis.player_access)

# Structural shape only. Public Codec/constructor still replays the complete archive.
static func _basis_shape(basis: Dictionary) -> bool:
	for key: String in ["starts","assets","pennants","plaques","derived"]:
		if not basis.get(key) is Dictionary: return false
	if not basis.get("runs") is Array or basis.runs.is_empty() or basis.runs.size() >= ClubCareer.MAX_RUNS \
		or not SeasonOwnership._whole(basis.get("target_run"),basis.runs.size()+1,basis.runs.size()+1) \
		or not basis.get("exotic_history") is Array or not basis.get("player_access") is Array:
		return false
	var s: Dictionary = basis.starts
	if not s.get("gear") is Dictionary or not s.get("sponsors") is Dictionary or not s.get("batch") is Array:
		return false
	for key: String in ["gear_enabled","sponsor_enabled","order","rain","transfer","checkout","association","freezer","sides","jump","sure","major","copy","field"]:
		if not s.get(key) is bool:
			return false
	return SeasonOwnership._whole(s.get("supply"),0,3) and SeasonOwnership._whole(s.get("abilities"),0,3)

static func matches(season: SeasonState) -> bool:
	return structural(season)

static func _prefix_matches(career: ClubCareer) -> bool:
	var basis: Dictionary = career.start_basis
	return career.current == basis.target_run and career.runs.size() == career.current \
		and ClubCareer.same(career.runs.slice(0,career.current-1),basis.runs) \
		and career._pitch_catalog == load("res://src/gameplay/season/successor_owner_career_v1/catalog.gd").Game.signature()
