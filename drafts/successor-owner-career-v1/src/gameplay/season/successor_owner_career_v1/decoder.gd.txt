## Rebuild every run from one frozen origin. Serialized starts never grant authority.

const Origin = preload("res://src/gameplay/season/successor_owner_general_v1/codec.gd")
const Prefix = preload("prefix.gd")
const GameBuild = preload("res://src/gameplay/season/successor_league_general_v2/build.gd")

static func _completed(season: SeasonState) -> bool:
	return season != null and season.phase == SeasonState.Phase.COMPLETE and season.career != null \
		and season.career.current >= 1 and season.career.current == season.career.runs.size() \
		and season.career.runs[-1].get("status") == "completed" and season.physical.pending.is_empty()

static func restore(value: Dictionary) -> SeasonState:
	var roots: Array[Resource] = Codec._operation_content_roots()
	var result: SeasonState = _restore(value)
	roots.clear()
	return result

static func _restore(value: Dictionary) -> SeasonState:
	if not Codec.envelope_valid(value): return null
	var previous: SeasonState = Origin.restore(value.origin)
	if not _completed(previous): return null
	if previous.career.current + value.closed.size() + 1 > ClubCareer.MAX_RUNS: return null
	for record: Variant in value.closed:
		previous = _record(record,previous.career)
		if not _completed(previous): return null
	var result: SeasonState = _record(value.active,previous.career)
	if result == null: return null
	result.origin = value.origin.duplicate(true)
	result.closed = value.closed.duplicate(true)
	var actual: Dictionary = Codec.guard_snapshot(result)
	return result if not actual.is_empty() and ClubCareer.same(actual,value) else null

static func create_next(value: Dictionary, seed: int, league: String = "dugout", tier: int = 0) -> SeasonState:
	# Public archive boundary: strict predecessor replay, never caller basis/proof.
	if seed < 0 or seed > 2147483647 or not Codec._bounded(value): return null
	var roots: Array[Resource] = Codec._operation_content_roots()
	var source: String = Catalog.signature()
	var previous: SeasonState
	var origin: Dictionary = {}
	var closed: Array = []
	if value.get("version") == Origin.Catalog.VERSION:
		previous = Origin.restore(value)
		origin = value.duplicate(true) if previous != null else {}
	elif value.get("version") == Catalog.VERSION:
		previous = Codec.restore(value)
		if previous != null:
			origin = value.origin.duplicate(true)
			closed = value.closed.duplicate(true)
			closed.append(value.active.duplicate(true))
	var result: SeasonState = null
	if _completed(previous) and previous.career.current < ClubCareer.MAX_RUNS:
		var basis: Dictionary = Prefix._derive(previous.career)
		if not basis.is_empty():
			result = Start._create(seed,basis,league,tier)
			if result != null:
				result.origin = origin
				result.closed = closed
				var guard: Dictionary = Codec.guard_snapshot(result)
				if guard.is_empty() or source.is_empty() or guard.owner_source != source: result = null
	roots.clear()
	return result

static func _record(record: Variant, previous: ClubCareer) -> SeasonState:
	if not record is Dictionary or not SeasonOwnership._keys(record,["id","checkpoint"]) \
		or not SeasonOwnership._whole(record.id,previous.current+1,previous.current+1) or not record.checkpoint is Dictionary: return null
	var basis: Dictionary = Prefix._derive(previous)
	return null if basis.is_empty() else _run(record.checkpoint,basis)

static func _run(state: Dictionary, basis: Dictionary) -> SeasonState:
	if state.has("career") or not SeasonOwnership._whole(state.get("season_seed"),0,2147483647) \
		or not state.get("picks") is Array or state.picks.size() > 4 or not state.get("player_results") is Array \
		or not state.get("build") is Dictionary or not state.get("teams") is Array or state.teams.size() != 6: return null
	if not state.get("league_rules") is Dictionary or not state.get("league_live") is bool: return null
	var rules: Dictionary = state.league_rules
	var requested_league: Variant = rules.get("league","standard")
	var requested_tier: Variant = rules.get("tier",0)
	if not requested_league is String or not SeasonOwnership._whole(requested_tier,0,4): return null
	var spec: Dictionary = Start.specification(int(state.season_seed),requested_league,int(requested_tier))
	if spec.is_empty() or not ClubCareer.same(spec.rules,rules) or spec.live != state.league_live: return null
	var season: SeasonState = Start._create(int(state.season_seed),basis,requested_league,int(requested_tier))
	if season == null: return null
	for id: Variant in state.picks:
		if not id is String or not season._choose_local(id):
			return null
	var roster: Array[String] = []
	if season.picks.size() == 4:
		roster.assign(season.picks)
	var restored: SeasonBuild = GameBuild.restore(state.build, season.season_seed, roster, season.build._pool, season.recruit_blocked())
	if restored == null:
		return null
	# A valid paid Build alone cannot authorize a career start. Validate every
	# start/access/owner field against the genuine authenticated predecessor birth first.
	var birth_build: SeasonBuild = season.build
	season.build = restored
	if not Start.base_matches(season):
		return null
	season.build = birth_build
	season.physical = Physical.from_data(state.get("physical"), Catalog.Game.signature())
	if season.physical == null:
		return null
	for saved: Variant in state.player_results:
		if not saved is Dictionary or not _result(season, restored, saved):
			return null
	if season.physical.cursor != season.physical.reports.size() \
		or not GameHistory._build_history_valid(season, restored):
		return null
	restored.migrate()
	season.build = restored
	var lineup: Variant = state.teams[0].get("roster") if state.teams[0] is Dictionary else null
	if not lineup is Array:
		return null
	if season.picks.size() == 4:
		if lineup.size() != 4:
			return null
		var unique: Array = []
		for id: Variant in lineup:
			if not id is String or not restored.roster().has(id) or unique.has(id):
				return null
			unique.append(id)
		season.teams[0].roster = unique
	elif not lineup.is_empty():
		return null
	if not SeasonOwnership._whole(state.get("starter_index"), 0, 3) \
		or not SeasonOwnership._whole(state.get("fielder_index"), 0, 3) \
		or state.starter_index == state.fielder_index:
		return null
	season.starter_index = int(state.starter_index)
	season.fielder_index = int(state.fielder_index)
	if not season.career.sync(season):
		return null
	if not season.physical.pending.is_empty() \
		and not load("res://src/gameplay/season/successor_owner_career_v1/pending.gd")._valid_pending_replayed(season):
		return null
	season.physical.projecting = false
	var actual: Dictionary = load("res://src/gameplay/season/successor_owner_career_v1/codec.gd")._snapshot(season)
	return season if not actual.is_empty() and ClubCareer.same(actual,state) else null

static func _result(season: SeasonState, restored: SeasonBuild, saved: Dictionary) -> bool:
	if saved.has("pitching_current") and (not saved.get("pitching") is Dictionary
		or not load("res://src/gameplay/season/successor_league_general_v2/called_proof.gd").accepts(restored, saved.pitching_current, saved.pitching)):
		return false
	for key: String in ["id", "away_runs", "home_runs"]:
		if not SeasonOwnership._whole(saved.get(key), 0, 9999):
			return false
	var performance: Variant = saved.get("performance", {})
	if not performance is Dictionary or (saved.has("performance") and performance.is_empty()):
		return false
	var fixture: Dictionary = season.pending_fixture()
	if fixture.is_empty():
		return false
	var index: int = int(fixture.away if fixture.home == 0 else fixture.home)
	if not saved.get("ai_tactics") is Dictionary \
		or not SeasonOwnership._keys(saved.ai_tactics, [str(index)]) \
		or not saved.ai_tactics[str(index)] is Dictionary or not saved.get("tactics", []) is Array:
		return false
	var tactics: Dictionary = saved.ai_tactics[str(index)]
	var roster: Array = restored.roster_for_game(int(saved.id))
	if roster.size() != 4:
		return false
	roster = SeasonSave._home_replay_roster(tactics, roster)
	if roster.is_empty():
		return false
	season.teams[0].roster = roster
	# Preserve the General paired replay route; ordinary play uses its unchanged
	# paid-prefix reconstruction rather than a caller-validated shortcut.
	if LeagueRules.doubleheader(season.league_rules):
		season._doubleheader_replay_build = restored
		var ok: bool = load("res://src/gameplay/season/successor_league_general_v2/doubleheader.gd").replay_result(season,saved,performance,tactics,saved.get("tactics",[]))
		season._doubleheader_replay_build = null
		if not ok or not season.career.sync(season): return false
	elif saved.has("doubleheader_evidence") or not GameReplay._ordinary_result(season, restored, saved, performance, tactics, saved.get("tactics", [])):
		return false
	season._heat_replay_team = null
	# Original decoder retains optional exact proof fields after their validators
	# have consumed them. Complete envelope recapture still rejects added fields.
	for key: String in ["used_gear", "tactics", "batting", "stances", "fielding", "pitching_current", "pitching", "field_supply", "frozen"]:
		if saved.has(key):
			if key in ["used_gear", "tactics", "stances", "fielding", "frozen"] and not saved[key] is Array:
				return false
			if key in ["batting", "pitching_current", "pitching", "field_supply"] and not saved[key] is Dictionary:
				return false
			if key in ["stances", "frozen"] and not SeasonLeftRight.own_halves(saved[key], season.player_results[-1].home == 0):
				return false
			if key == "fielding" and not SeasonLeftRight.own_halves(saved[key], season.player_results[-1].home != 0):
				return false
			if key == "pitching" and not SeasonSureShot.own_halves(saved[key], season.player_results[-1].home != 0):
				return false
			season.player_results[-1][key] = saved[key].duplicate(true)
			for played: Dictionary in season.results:
				if played.id == saved.id:
					played[key] = saved[key].duplicate(true)
	return true
