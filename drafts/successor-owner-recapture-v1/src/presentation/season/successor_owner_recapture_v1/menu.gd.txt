extends SeasonMenu
func render() -> void:
	_screen("successor", "Dugout Club Career", "Recapture-owner staging • separate save • legacy clubs unchanged")
	if app.get("retained")!=null:
		_button(_body,"RETRY EXACT SAVE",Callable(app,"retry_save"))
		return
	if app.season==null:
		_button(_body,"NEW DUGOUT CLUB",Callable(app,"create_fresh"))
		_button(_body,"CONTINUE DUGOUT CLUB",Callable(app,"reload_save"))
		return
	if app.season.phase==SeasonState.Phase.DRAFT:
		for id: String in app.season.offers():
			var definition: PlayerDefinition = ContentDB.get_player(StringName(id))
			var profile: Dictionary = app.season.build.player(id)
			_button(_body, "%s  %s" % [definition.display_name, str(profile.stats)], Callable(app,"draft").bind(id))
		return
	for id: String in app.season.teams[0].roster:
		var definition: PlayerDefinition = app.season.build.definition(id)
		var pitches: Array[String] = []
		for pitch: PitchDefinition in definition.starting_pitches:
			pitches.append(pitch.display_name)
		_label(_body,"%s: %s" % [definition.display_name, ", ".join(pitches)],18)
	if not app.season.physical.pending.is_empty():
		_button(_body,"RESUME SAVED ROUND",Callable(app,"resume_round"))
	elif app.season.phase!=SeasonState.Phase.COMPLETE:
		_button(_body,"PLAY NEXT GAME",Callable(app,"play_next"))
	else:
		# Read only the published season. A retained unsaved terminal candidate
		# is handled above and must never present its proposed award as granted.
		var career: ClubCareer = app.season.career
		if career != null and not career.runs.is_empty() and career.runs[-1].get("status") == "completed":
			var receipt: Dictionary = career.runs[-1].get("receipt", {})
			if not receipt.is_empty():
				var outcome: String = {"missed":"Missed playoffs", "semifinal":"Semifinal finish", "runner_up":"Runner-up", "champion":"Champion"}.get(receipt.finish, receipt.finish)
				_label(_body, "Season saved: %s" % outcome, 20)
				_label(_body, "%d regular-season wins • %d Club Cash earned" % [int(receipt.wins), int(receipt.total)], 18)
				_label(_body, "Finish %d + record %d + first title %d" % [int(receipt.finish_award), int(receipt.record_bonus), int(receipt.first_bonus)], 16)
		_label(_body, "Next-career and Dynasty starts are not available in this staging entry.", 16)
	if app.season.shop_available():
		_button(_body,"SHOP",Callable(app,"open_pitch_shop"))
