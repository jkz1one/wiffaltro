extends RefCounted
const V7Catalog = preload("catalog_context.gd")
const V7Access = preload("exotic_access.gd")
const V7Eligibility = preload("exotic_eligibility.gd")

const BUILD_VERSION: int = 109
const DEVELOPMENT_VERSION: int = 9
const FORMAT: String = "successor-dugout-live-v7-r1"

static func recipe(id: String) -> PitchDefinition:
	# Presentation/evidence identity only. Learning and ownership remain book-scoped.
	return V7Catalog.get_recipe(id)

static func lesson(id: String) -> Dictionary:
	var pitch: PitchDefinition = recipe(id)
	var price: int = V7Catalog.lesson_price(id)
	if price < 0 and pitch != null:
		price = V7Access.price(id)
	if pitch == null or pitch.delivery_profile == null or price < 0:
		return {}
	return {"name": pitch.display_name + " Lesson", "price": price, "op": "learn",
		"recipe": id, "family": "lesson"}

static func display_item(id: String) -> Dictionary:
	return lesson(id.trim_prefix("lesson.")) if id.begins_with("lesson.") \
		else DevelopmentShopCatalog.item(id)

static func signature(base: String) -> String:
	return _signature_for(base, V7Catalog.signature())

static func _signature_for(base: String, source: String) -> String:
	return ("successor-dugout-live-v7-r1:" + base + ":" + source).sha256_text()

static func _descriptor_for(source: String) -> Dictionary:
	return {} if source.is_empty() else {"version": V7Catalog.VERSION, "purpose": V7Catalog.PURPOSE, "signature": source}

static func _descriptor_valid_for(value: Variant, source: String) -> bool:
	return value is Dictionary and SeasonOwnership._keys(value, ["version", "purpose", "signature"]) \
		and SeasonOwnership._whole(value.version, V7Catalog.VERSION, V7Catalog.VERSION) \
		and value.purpose is String and value.purpose == V7Catalog.PURPOSE \
		and value.signature is String and not value.signature.is_empty() \
		and value.signature == source

static func wrap_build(data: Dictionary, exotic_start: Array = []) -> Dictionary:
	var source: String = V7Catalog.signature()
	var result: Dictionary = data.duplicate(true)
	result["successor_format"] = FORMAT
	result["exotic_start"] = exotic_start.duplicate(true)
	result["base_version"] = data.version
	result.version = BUILD_VERSION
	result["pitch_catalog"] = source
	result["catalog_context"] = _descriptor_for(source)
	result.catalog = _signature_for(data.catalog, source)
	return result

static func unwrap_build(value: Dictionary) -> Dictionary:
	var source: String = V7Catalog.signature()
	if value.get("successor_format") != FORMAT:
		return {}
	if not _descriptor_valid_for(value.get("catalog_context"), source) or not V7Access.valid_history(value.get("exotic_start")):
		return {}
	if not is_version(value.get("version"), BUILD_VERSION) \
		or not SeasonOwnership._whole(value.get("base_version"), 45, 45) \
		or LeagueRules.normalized(value.get("league_rules")) != LeagueRules.capture("dugout") \
		or not value.get("pitch_catalog") is String \
		or value.pitch_catalog != source:
		return {}
	var base: String = SeasonBuild._signature(int(value.base_version), value.get("league_rules"))
	if base.is_empty() or not value.get("catalog") is String or value.catalog != _signature_for(base, source):
		return {}
	var result: Dictionary = value.duplicate(true)
	result.version = int(value.base_version)
	result.catalog = base
	result.erase("exotic_start")
	result.erase("base_version")
	result.erase("successor_format")
	result.erase("pitch_catalog")
	result.erase("catalog_context")
	return result

static func wrap_development(data: Dictionary, exotic_start: Array = [], exotic_current: Dictionary = {}) -> Dictionary:
	var source: String = V7Catalog.signature()
	var result: Dictionary = data.duplicate(true)
	result["successor_format"] = FORMAT
	result["exotic_current"] = exotic_current.duplicate(true)
	result["exotic_start"] = exotic_start.duplicate(true)
	result["base_version"] = data.version
	result.version = DEVELOPMENT_VERSION
	result["pitch_catalog"] = source
	result["catalog_context"] = _descriptor_for(source)
	result.catalog = _signature_for(data.catalog, source)
	return result

static func unwrap_development(value: Dictionary) -> Dictionary:
	var source: String = V7Catalog.signature()
	if value.get("successor_format") != FORMAT:
		return {}
	if not value.get("exotic_current") is Dictionary or (not value.exotic_current.is_empty() and not V7Eligibility.valid(value.exotic_current)):
		return {}
	if not _descriptor_valid_for(value.get("catalog_context"), source) or not V7Access.valid_history(value.get("exotic_start")):
		return {}
	if not is_version(value.get("version"), DEVELOPMENT_VERSION) \
		or not SeasonOwnership._whole(value.get("base_version"), 1, 1) \
		or not value.get("pitch_catalog") is String \
		or value.pitch_catalog != source \
		or not value.get("catalog") is String \
		or value.catalog != _signature_for(SeasonPlayerCatalog.signature(), source):
		return {}
	var result: Dictionary = value.duplicate(true)
	result.version = int(value.base_version)
	result.catalog = SeasonPlayerCatalog.signature()
	result.erase("exotic_current")
	result.erase("exotic_start")
	result.erase("base_version")
	result.erase("successor_format")
	result.erase("pitch_catalog")
	result.erase("catalog_context")
	return result

static func base_build_version(value: Variant) -> int:
	if not value is Dictionary or not SeasonOwnership._whole(value.get("version"), 1, BUILD_VERSION):
		return -1
	if is_version(value.version, BUILD_VERSION):
		return int(value.base_version) if SeasonOwnership._whole(value.get("base_version"), 45, 45) else -1
	return int(value.version)

static func is_version(value: Variant, expected: int) -> bool:
	return SeasonOwnership._whole(value, expected, expected)
