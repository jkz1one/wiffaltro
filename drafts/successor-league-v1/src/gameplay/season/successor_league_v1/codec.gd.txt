extends RefCounted
## Explicit live Dugout tier0 boundary. No career, owner or migration authority.
const FORMAT: String = "successor-dugout-live-v7-r1"
const ADMISSION: int = 109
const Catalog = preload("res://src/gameplay/season/successor_league_v1/catalog_context.gd")
const Progression = preload("res://src/gameplay/season/successor_league_v1/progression.gd")

static func scope(season: SeasonState) -> bool:
	return season != null and season.get_script() == load("res://src/gameplay/season/successor_league_v1/runtime.gd") \
		and season.build != null and season.build.get_script() == load("res://src/gameplay/season/successor_league_v1/build.gd") \
		and season.build.call("pitch_catalog_successor") and season.build._format == 45 \
		and season.build._market == 0 and season.career == null \
		and season.league_live and not season.league_career \
		and season.league_rules == LeagueRules.capture("dugout") and season.inventory_rules.is_empty() \
		and season.build.league_context_valid() and season.build._league_rules == season.league_rules \
		and season._dynasty_start.is_empty() and season._closed_access.is_empty() \
		and season._dynasty_owner_binding.is_empty() and season._dynasty_career_session.is_empty() \
		and season._standard_live_context.is_empty() and season._standard_bound_grant.is_empty() \
		and season._league_home_context.is_empty() and season.home_property.is_empty() \
		and season.owned_surface_night.is_empty() and season.npc_venues.is_empty() \
		and season.final_venue.is_empty() and season.player_access.is_empty() \
		and season.opponents != null and season.opponents.get_script() == load("res://src/gameplay/season/successor_league_v1/opponents.gd") \
		and season.opponents._format == 13 and season.physical != null \
		and season.physical.get_script() == load("res://src/gameplay/season/successor_league_v1/physical_fixtures.gd") \
		and season._league_inventory_context_consistent() and _domain_builds_valid(season)

static func _domain_builds_valid(season: SeasonState) -> bool:
	if season.build._book.get_script() != load("res://src/gameplay/season/successor_league_v1/development.gd") \
		or not season.build._book._exotic_start.is_empty():
		return false
	if season.phase == SeasonState.Phase.DRAFT:
		return season.picks.size() < 4 and season.opponents.clubs.is_empty()
	if season.picks.size() != 4 or season.opponents.clubs.size() != 5:
		return false
	for club: Dictionary in season.opponents.clubs.values():
		var build: SeasonBuild = club.get("build")
		if build == null or build.get_script() != load("res://src/gameplay/season/successor_league_v1/build.gd") \
			or not build.call("pitch_catalog_successor") \
			or build._book.get_script() != load("res://src/gameplay/season/successor_league_v1/development.gd") \
			or not build._book._exotic_start.is_empty():
			return false
	return true

static func leaf_valid(value: Variant, pitch_catalog: String) -> bool:
	if pitch_catalog.is_empty() or pitch_catalog != Catalog.signature() or not value is Dictionary \
		or not SeasonOwnership._whole(value.get("version"), 57, 57) \
		or not value.get("build") is Dictionary:
		return false
	for key: String in ["career", "home_property", "owned_surface_night", "npc_venues", "final_venue", "player_access", "league_rules", "inventory_rules"]:
		if value.has(key):
			return false
	return value.get("results") is Array \
		and not Progression.unwrap_build(value.build).is_empty() and value.build.get("exotic_start") == []

static func envelope_valid(value: Variant) -> bool:
	return value is Dictionary and SeasonOwnership._keys(value, ["version", "format", "pitch_catalog", "league_rules", "league_live", "checkpoint"]) \
		and SeasonOwnership._whole(value.version, ADMISSION, ADMISSION) \
		and value.league_live is bool and value.league_live \
		and LeagueRules.normalized(value.league_rules) == LeagueRules.capture("dugout") \
		and value.format is String and value.format == FORMAT and value.pitch_catalog is String \
		and leaf_valid(value.checkpoint, value.pitch_catalog)

static func _operation_content_roots() -> Array[Resource]:
	# Strong authored Resource references for this synchronous operation only.
	# No signature, validation, decoded-state or result cache survives the call.
	var roots: Array[Resource] = SeasonSave._expanded_operation_content_roots()
	for path: String in preload("res://src/gameplay/pitching/working_pitch_catalog_v7.gd").NEW_PATHS.values():
		roots.append(load(path))
	return roots

static func capture(season: SeasonState) -> Dictionary:
	var roots: Array[Resource] = _operation_content_roots()
	var result: Dictionary = _capture(season)
	roots.clear()
	return result

static func _capture(season: SeasonState) -> Dictionary:
	if not scope(season) or not load("res://src/gameplay/season/successor_league_v1/exotic_access.gd").current_matches(season):
		return {}
	var historical_shell: Dictionary = SeasonSave._ac02_passive_15_snapshot(season)
	if not SeasonOwnership._keys(historical_shell, ["version", "league_rules", "checkpoint"]) \
		or not SeasonOwnership._whole(historical_shell.get("version"), 76, 76) \
		or not ClubCareer.same(historical_shell.league_rules, season.league_rules):
		return {}
	# Reuse only the passive inner representation, never the historical decoder.
	var payload: Dictionary = historical_shell.checkpoint
	var result: Dictionary = {"version": ADMISSION, "format": FORMAT, "pitch_catalog": Catalog.signature(), "league_rules": season.league_rules.duplicate(true), "league_live": true, "checkpoint": payload}
	return result if envelope_valid(result) else {}

static func restore(value: Variant) -> SeasonState:
	var roots: Array[Resource] = _operation_content_roots()
	var result: SeasonState = _restore(value)
	roots.clear()
	return result

static func _restore(value: Variant) -> SeasonState:
	if not envelope_valid(value):
		return null
	var season: SeasonState = load("res://src/gameplay/season/successor_league_v1/decoder.gd")._ac02_passive_15_private_decode_payload_terrain(
		value.checkpoint, {}, false, LeagueRules.normalized(value.league_rules), true, {}, null, {}, null, [], {}, {}, null,
		value.pitch_catalog, ADMISSION)
	if not scope(season) or not load("res://src/gameplay/season/successor_league_v1/exotic_access.gd").current_matches(season):
		return null
	var recaptured: Dictionary = capture(season)
	return season if not recaptured.is_empty() and ClubCareer.same(recaptured, value) else null
