## Actual paired qualification, one durable boundary per process. Inherited
## helpers provide UI search only; Equipment run/play/finish are overridden.
const PairContext = preload("res://src/gameplay/season/successor_league_general_v2/doubleheader_build.gd")
const PairCapture = preload("res://src/gameplay/season/successor_league_general_v2/doubleheader.gd")
const Workload = preload("res://src/gameplay/season/successor_league_general_v2/physical_workload.gd")
const Request = preload("res://src/gameplay/season/successor_league_general_v2/physical_request.gd")
const PAIR_SEED: int = 1001

func finish_step(app: SeasonApp, action: String, complete: bool = false) -> void:
	var payload: Dictionary = Codec.capture(app.season)
	if not check(app.get("retained") == null and not app._busy and not payload.is_empty() \
		and ClubCareer.same(payload,JSON.parse_string(FileAccess.get_file_as_string(Save.PATH))),"exact durable Doubleheader boundary " + action):
		return
	var directory: String = "user://doubleheader-evidence"
	if not check(DirAccess.make_dir_recursive_absolute(directory) == OK,"private paired evidence directory"):
		return
	var index: int = 0
	while FileAccess.file_exists(directory + "/%04d.json" % index):
		index += 1
	if not check(DirAccess.copy_absolute(Save.PATH,directory + "/%04d.json" % index) == OK,"preserve actual boundary without overwrite"):
		return
	var meta: Dictionary = {"step":index,"action":action,"kind":action,"human_games":app.season.player_results.size(),"results":app.season.results.size(),"complete":complete}
	var file: FileAccess = FileAccess.open(directory + "/%04d.meta.json" % index,FileAccess.WRITE)
	if not check(file != null,"paired metadata writable"):
		return
	file.store_string(JSON.stringify(meta,"",true,true))
	file.close()
	print("DOUBLEHEADER_STEP_DONE: ",JSON.stringify(meta,"",true,true))
	print("PASS: actual Doubleheader boundary ",action)
	app.queue_free()
	get_tree().quit()

func exact_profile(season: SeasonState) -> bool:
	return season != null and Codec.scope(season) and season.career == null and season.league_live \
		and season.season_seed == PAIR_SEED and ClubCareer.same(season.league_rules,LeagueRules.capture("doubleheader"))

func run() -> void:
	if not check(OS.get_environment("WIFFALTRO_ISOLATED_TEST") == "1","isolated actual Doubleheader run"):
		return
	var app = App.new()
	app.profile = "doubleheader"
	add_child(app)
	if not FileAccess.file_exists(Save.PATH):
		if not check(not FileAccess.file_exists(Save.PATH+".bak"),"pair birth cannot replace backup"):
			return
		var fresh: SeasonState = Runtime.create_closed_league(PAIR_SEED,LeagueRules.capture("doubleheader"),true)
		if not check(exact_profile(fresh),"fixed-seed public paired birth"):
			return
		app._retain(fresh)
		finish_step(app,"birth")
		return
	app.reload_save()
	if not check(exact_profile(app.season),"cold exact Doubleheader seed/profile"):
		return
	if OS.get_cmdline_user_args().has("verify-complete"):
		if final_checks(app):
			finish_step(app,"cold-complete",true)
		return
	if app.season.phase == SeasonState.Phase.DRAFT:
		app.draft(app.season.offers()[0])
		finish_step(app,"draft-%d" % app.season.picks.size())
		return
	if not app.season.physical.pending.is_empty():
		await pending_step(app)
		return
	var games: int = app.season.player_results.size()
	if games == 2:
		if not app.season.build._visit.open:
			app.open_pitch_shop()
			if not check(app.season.build._visit.open and app.season.build._visit.number == 2,"second leg independently opens its earned shop"):
				return
			finish_step(app,"second-leg-shop-open")
			return
		if final_checks(app):
			finish_step(app,"pair-complete",true)
		return
	if games == 1 and not between_leg_shop(app):
		return
	await play_human(app)

func pending_step(app: SeasonApp) -> void:
	var before: Dictionary = Codec.capture(app.season)
	var projection: Dictionary = Pending.project(app.season)
	if not check(not projection.has("error") and ClubCareer.same(Codec.capture(app.season),before),"pending projection preserves exact state"):
		return
	var count: int = app.season.physical.pending.reports.size()
	var second_settlement: bool = not projection.has("fixture") and app.season.player_results.size() == 1
	var primary: PackedByteArray = FileAccess.get_file_as_bytes(Save.PATH)
	var backup: PackedByteArray = FileAccess.get_file_as_bytes(Save.PATH+".bak")
	if second_settlement and not check(not FileAccess.file_exists(Save.PATH+".tmp") and not DirAccess.dir_exists_absolute(Save.PATH+".tmp") \
		and DirAccess.make_dir_absolute(Save.PATH+".tmp") == OK,"isolated second-leg settlement prewrite refusal"):
		return
	app.resume_round()
	if projection.has("fixture"):
		var finished: bool = false
		for frame in range(150000):
			await get_tree().physics_frame
			if not app._busy:
				finished = true
				break
			if frame % 12000 == 0:
				print("DOUBLEHEADER_NPC: ",frame)
		if not check(finished and app.get("retained") == null and app.season.physical.pending.reports.size() == count+1,"actual NPC report durably appended once"):
			return
		finish_step(app,"npc-%d" % int(projection.fixture.id))
		return
	if second_settlement:
		var retained: SeasonState = app.get("retained")
		if not check(retained != null and ClubCareer.same(Codec.capture(app.season),before) \
			and FileAccess.get_file_as_bytes(Save.PATH) == primary and FileAccess.get_file_as_bytes(Save.PATH+".bak") == backup,"second-leg refusal retains old owner, primary, backup and completed report history"):
			return
		var expected: Dictionary = Codec.capture(retained)
		if not check(not expected.is_empty() and DirAccess.remove_absolute(Save.PATH+".tmp") == OK,"remove only own temporary-path fault"):
			return
		app.retry_save()
		if not check(app.get("retained") == null and app.season == retained and app.notice.is_empty() \
			and ClubCareer.same(Codec.capture(app.season),expected),"exact retained second-leg candidate adopted, no recomputation or duplicate payout"):
			return
		app.retry_save()
		if not check(ClubCareer.same(Codec.capture(app.season),expected),"repeated retry changes neither payout nor history"):
			return
	if not check(app.get("retained") == null and app.season.physical.pending.is_empty() \
		and app.season.results.size() == app.season.player_results.size()*3,"each actual leg settles all three fixtures"):
		return
	finish_step(app,"settlement-%d" % app.season.player_results.size())

func between_leg_shop(app: SeasonApp) -> bool:
	var build: SeasonBuild = app.season.build
	if not build._visit.open:
		app.open_pitch_shop()
		if not check(app.season.build._visit.open,"first-leg earned shop opens before return leg"):
			return false
		finish_step(app,"between-leg-shop-open")
		return false
	if build._visit.pack_status == "sealed":
		var before_cash: int = build.cash()
		var price: int = build.pack_price()
		if not check(price == 8 and before_cash >= price,"original ordinary pack affordable from actual first-leg earnings"):
			return false
		app.shop_command({"id":"pair-pack:%d"%build.revision(),"rev":build.revision(),"op":"pack_open"})
		if not check(app.get("retained") == null and app.season.build.pack_pending() and app.season.build.cash() == before_cash-price,"actual between-leg pack paid once"):
			return false
		finish_step(app,"between-leg-pack-paid")
		return false
	if build.pack_pending():
		var selected: Dictionary = {}
		for item: String in build._pack_choices():
			var targets: Array[Dictionary] = build.targets(item)
			if build.shop_item(item).get("op") == "stat" and not targets.is_empty():
				selected = {"id":"pair-pick:%d"%build.revision(),"rev":build.revision(),"op":"pack_pick","item":item,"player":targets[0].player,"pitch":targets[0].pitch,"replace":targets[0].replace}
				break
		if not check(not selected.is_empty(),"naturally revealed eligible stat upgrade; no fabricated stock or fallback"):
			return false
		var old: Dictionary = build.player(selected.player).duplicate(true)
		var old_cash: int = build.cash()
		var stat: String = build.shop_item(selected.item).family
		app.shop_command(selected)
		var changed: Dictionary = app.season.build.player(selected.player)
		for key: String in SeasonPlayerCatalog.STATS:
			if not check(changed.stats[key] == old.stats[key]+(1 if key == stat else 0),"actual between-leg upgrade targets exact stat " + key):
				return false
		if not check(app.get("retained") == null and not app.season.build.pack_pending() and app.season.build.cash() == old_cash,"one applied choice, no recharge"):
			return false
		var after: Dictionary = Codec.capture(app.season)
		app.shop_command(selected)
		if not check(ClubCareer.same(Codec.capture(app.season),after),"repeated between-leg choice grants nothing twice"):
			return false
		finish_step(app,"between-leg-upgrade")
		return false
	return check(build._visit.pack_status == "used","between-leg paid choice already durable")

func pristine_request_negatives(season: SeasonState) -> bool:
	var before: Dictionary = Codec.capture(season)
	var pristine: MatchState = season.make_match()
	if not check(pristine != null and Request.capture(pristine) != null,"fresh actual paired request admitted before mutation tests"):
		return false
	for kind: String in ["workload initial","workload events","wrong pair","wrong debt"]:
		var source: MatchState = season.make_match()
		if not check(source != null,"independent pristine negative operand"):
			return false
		var team: TeamMatchState = source.away_team
		var player: String = String(team.roster[0].definition.id)
		match kind:
			"workload initial": team.recovery.workload_initial[player] = {"capacity":1.0,"initial":1.0}
			"workload events": team.recovery.workload_events.append({"kind":"release","player":player})
			"wrong pair": team.league_pair.game = int(team.league_pair.game)+1
			"wrong debt": team.league_pair.debt[player] = 999999.0
		if not check(Request.capture(source) == null,"pristine request rejects injected " + kind):
			return false
	return check(ClubCareer.same(Codec.capture(season),before),"negative request operands do not mutate durable season")

func play_human(app: SeasonApp) -> void:
	var fixture: Dictionary = app.season.pending_fixture()
	var games: int = app.season.player_results.size()
	if not check(games in [0,1] and not fixture.is_empty() and (fixture.home == 0 or fixture.away == 0),"next actual human leg"):
		return
	if games == 1 and not check(SeasonDoubleheaderRules.first(fixture.id) == app.season.player_results[0].id,"return fixture paired to exact first human leg"):
		return
	if not pristine_request_negatives(app.season):
		return
	var expected_pair: Dictionary = PairContext.pair_for(app.season.build,fixture.id)
	app.play_next()
	if not check(is_instance_valid(app.lab) and app.lab._match_state != null,"actual App opens paired human match"):
		return
	var state: MatchState = app.lab._match_state
	var team: TeamMatchState = state.home_team if fixture.home == 0 else state.away_team
	if not check(state.balls == 0 and state.strikes == 0 and state.outs == 0 and state.inning == 1 and state.plate_appearance_number == 1 \
		and ClubCareer.same(team.league_pair,expected_pair),"new-leg counts reset and exact player-keyed carry installed"):
		return
	for player: PlayerMatchState in team.roster:
		var id: String = String(player.definition.id)
		if not check(is_equal_approx(player.stamina_remaining,maxf(0.0,player.stamina_max-float(expected_pair.debt.get(id,0.0)))),"actual initial stamina derives player debt " + id):
			return
	for frame in range(600):
		await get_tree().physics_frame
		if not app.lab._match_presentation_director.blocks_gameplay():
			break
	if not check(not app.lab._match_presentation_director.blocks_gameplay(),"paired intro completes"):
		return
	app.lab._automation = MatchAutomation.new()
	app.lab._automation.seed = Fixtures.seed_for(app.season,fixture)
	app.lab._automation.presentation_enabled = true
	app.lab._automation.begin(app.lab)
	var finished: bool = false
	for frame in range(150000):
		await get_tree().physics_frame
		if state.phase == MatchState.Phase.GAME_END and app.lab._automation != null and not app.lab._home_run.presentation_pending(app.lab):
			app.lab._automation = null
			PitchBatLabFeelSupport.begin_match_outro(app.lab)
		if state.phase == MatchState.Phase.GAME_END and app.lab._match_presentation_director.mode == MatchPresentationDirector.Mode.OUTRO_HOLD:
			finished = true
			break
		if frame % 12000 == 0:
			print("DOUBLEHEADER_HUMAN: fixture=",fixture.id," frame=",frame," pa=",state.plate_appearance_number)
	var workloads: Array = PairCapture.capture(state)
	if not check(finished and workloads.size() == 2 and not workloads[0].is_empty() and not workloads[1].is_empty() \
		and not Capture.capture(app).is_empty(),"actual two-team workload and called-sidecar-capable human capture"):
		return
	var count: int = app.season.results.size()
	var cash: int = app.season.build.cash()
	app.finish_game()
	if not check(app.get("retained") == null and not is_instance_valid(app.lab) and not app.season.physical.pending.is_empty() \
		and app.season.results.size() == count and app.season.build.cash() == cash,"actual human leg durable pending, no premature reward"):
		return
	finish_step(app,"human-%d"%int(fixture.id))

func final_checks(app: SeasonApp) -> bool:
	var season: SeasonState = app.season
	if not check(season.player_results.size() == 2 and season.results.size() == 6 and season.physical.pending.is_empty() \
		and season.physical.reports.size() == 4,"two actual human and four NPC fixtures settled, no pending jobs"):
		return false
	var builds: Array[SeasonBuild] = [season.build]
	for club in range(1,6):
		builds.append(season.opponents.clubs[str(club)].build)
	if not check(builds.size() == 6,"all six clubs retained"):
		return false
	for club in range(builds.size()):
		var build: SeasonBuild = builds[club]
		for fixture: Dictionary in season.schedule:
			if fixture.round < 2 and (fixture.home == club or fixture.away == club):
				if not check(build._bank.view().rewards.has(str(fixture.id)),"correct actual fixture reward for club %d game %d" % [club,fixture.id]):
					return false
		var bank: SeasonOwnership = build._bank.empty_fork()
		var count: int = 0
		for event: Dictionary in build._bank._events:
			var before: int = bank.cash()
			var rewards: Dictionary = bank.view().rewards.duplicate(true)
			var result: Dictionary = bank.commit(event)
			if not check(result.ok and not result.replayed,"independent exact paid bank journal replay"):
				return false
			if event.op == "reward":
				var bonus: int = SeasonDoubleheaderRules.bonus(build._league_rules,event.game,event.win,rewards)
				if not check(bank.cash()-before == (18 if event.win else 12)+bonus,"each actual reward pays ordinary income plus only original second-leg sweep"):
					return false
				count += 1
		if not check(count == 2 and build._bank.view().rewards.size() == 2 and ClubCareer.same(bank.view(),build._bank.view()),"every club has two exact paid rewards; opponent failures cannot silently skip"):
			return false
	var payload: Dictionary = Codec.capture(season)
	for row: Dictionary in season.player_results:
		if not check(row.get("doubleheader_evidence") is Array and row.doubleheader_evidence.size() == 2 \
			and row.get("pitching_current") is Dictionary,"actual human workload and current called sidecar retained"):
			return false
	for kind: String in ["pair","debt","workload","receipt"]:
		var wrong: Dictionary = payload.duplicate(true)
		var evidence: Dictionary = wrong.checkpoint.results[1].doubleheader_evidence[0]
		match kind:
			"pair": evidence.pair.first = -1
			"debt": evidence.pair.debt[season.build.roster()[0]] = 999999.0
			"workload": evidence.proof.remaining = {}
			"receipt": evidence.proof.events.append({"kind":"recovery","receipt":"not-owned","player":season.build.roster()[0],"pa":1})
		if not check(Codec.restore(wrong) == null,"cold replay rejects tampered actual second-leg " + kind):
			return false
	var reward_index: int = -1
	var game: int = int(payload.checkpoint.results[1].id)
	for index in range(payload.checkpoint.build.events.size()):
		var event: Dictionary = payload.checkpoint.build.events[index]
		if event.get("op") == "successor_reward" and event.get("command",{}).get("game") == game:
			reward_index = index
			break
	if not check(reward_index >= 0 and payload.checkpoint.build.events[reward_index].command.get("pitching_current") is Dictionary,"exact actual second reward retains current sidecar"):
		return false
	for kind: String in ["drop result","change result","drop reward","change reward"]:
		var wrong: Dictionary = payload.duplicate(true)
		var target: Dictionary = wrong.checkpoint.results[1] if kind.ends_with("result") else wrong.checkpoint.build.events[reward_index].command
		if kind.begins_with("drop"):
			target.erase("pitching_current")
		else:
			target.pitching_current.version = -1
		if not check(Codec.restore(wrong) == null,"independent current sidecar tamper rejected: " + kind):
			return false
	return check(ClubCareer.same(Codec.capture(season),payload) and ClubCareer.same(payload,JSON.parse_string(FileAccess.get_file_as_string(Save.PATH))),"cold final pair exact; no full season or Dynasty claim")
