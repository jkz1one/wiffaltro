extends "res://tests/successor_general_birth.gd"
## Actual human-side automated gameplay plus two real AI games, no score fixtures.
const Report = preload("res://src/gameplay/season/successor_league_general_v2/physical_report.gd")
const App = preload("res://src/presentation/season/successor_league_general_v2/app.gd")
const Fixtures = preload("res://src/gameplay/season/successor_league_general_v2/physical_fixtures.gd")
const Capture = preload("res://src/presentation/season/successor_league_general_v2/capture_human.gd")

func run() -> void:
	if not check(OS.get_environment("WIFFALTRO_ISOLATED_TEST") == "1", "isolated runner required"):
		return
	var app: SeasonApp = App.new()
	var selected_profile: String = requested_profile()
	app.set("profile", selected_profile)
	add_child(app)
	if OS.get_cmdline_user_args().has("reopen"):
		app.reload_save()
		if not check(app.season != null and app.season.results.size() == 3 and app.season.player_results.size() == 1, "cold settled round reopen"):
			return
		if not check(app.season.build._visit.pack_status == "used" and app.season.make_match() != null, "cold paid pack and next fixture"):
			return
		if not check(profile(app.season, selected_profile) and ClubCareer.same(Codec.capture(app.season), JSON.parse_string(FileAccess.get_file_as_string(Save.PATH))), "cold paid exact requested-profile disk capture"):
			return
		var before_new: PackedByteArray = FileAccess.get_file_as_bytes(Save.PATH)
		app.call("create_profile", "standard" if selected_profile != "standard" else "dugout")
		if not check(FileAccess.get_file_as_bytes(Save.PATH) == before_new and profile(app.season,selected_profile), "UI new profile cannot overwrite existing season"):
			return
		app.reload_save()
		if not check(app.get("profile") == selected_profile, "UI reload restores exact requested profile identity"):
			return
		print("PASS: cold settled round, earned pack purchase and next match")
		app.queue_free()
		get_tree().quit()
		return
	var human: Array = []
	if OS.get_cmdline_user_args().has("resume-round"):
		app.reload_save()
		if not check(profile(app.season, selected_profile) and not app.season.physical.pending.is_empty(), "cold pending human report and requested standalone profile"):
			return
		human = app.season.physical.pending.human.duplicate(true)
	else:
		app.create_fresh()
		if not check(app.season != null and app.get("retained") == null, "fresh round UI creation"):
			return
		print("SEASON_SEED: ", app.season.season_seed)
		for pick in range(4):
			app.call("draft", app.season.offers()[0])
			if not check(app.season.picks.size() == pick + 1, "round draft %d" % pick):
				return
		app.play_next()
		if not check(is_instance_valid(app.lab) and app.lab._match_state != null, "actual App play route opens match"):
			return
		# Let the actual intro finish before installing the automated controller.
		# Otherwise its later human-ready prompt can supersede automation startup.
		for frame in range(600):
			await get_tree().physics_frame
			if not app.lab._match_presentation_director.blocks_gameplay():
				break
		if not check(not app.lab._match_presentation_director.blocks_gameplay(), "actual match intro completes"):
			return
		# Test driver uses the ordinary live controllers for both sides; no outcome edits.
		app.lab._automation = MatchAutomation.new()
		app.lab._automation.seed = Fixtures.seed_for(app.season, app.season.pending_fixture())
		app.lab._automation.presentation_enabled = true
		app.lab._automation.begin(app.lab)
		var finished: bool = false
		for frame in range(150000):
			await get_tree().physics_frame
			if app.lab._match_state.phase == MatchState.Phase.GAME_END and app.lab._automation != null and not app.lab._home_run.presentation_pending(app.lab):
				# Automation intentionally omits the human outro. Stop only the test
				# driver after the actual result, then use the normal presentation path.
				app.lab._automation = null
				PitchBatLabFeelSupport.begin_match_outro(app.lab)
			if app.lab._match_state.phase == MatchState.Phase.GAME_END and app.lab._match_presentation_director.mode == MatchPresentationDirector.Mode.OUTRO_HOLD:
				finished = true
				break
			if frame % 12000 == 0:
				print("HUMAN_GAME_PROGRESS: ", frame, " pa=", app.lab._match_state.plate_appearance_number, " phase=", app.lab._match_state.phase)
		if not check(finished, "real human-side game reaches completed outro"):
			return
		human = Capture.capture(app)
		if not check(not human.is_empty(), "actual human evidence capture"):
			return
		var evidence: FileAccess = FileAccess.open("user://round-human-evidence.json", FileAccess.WRITE)
		evidence.store_string(JSON.stringify(human, "", true, true))
		evidence.close()
		app.finish_game()
		if not check(app.get("retained") == null and not is_instance_valid(app.lab) and not app.season.physical.pending.is_empty(), "completed human game durable before AI jobs"):
			return
		if not check(app.season.results.is_empty() and app.season.build.cash() == 0, "pending round gives no premature reward"):
			return
		if not check(profile(app.season, selected_profile) and ClubCareer.same(Codec.capture(app.season), JSON.parse_string(FileAccess.get_file_as_string(Save.PATH))), "human pending exact requested-profile disk capture"):
			return
		if OS.get_cmdline_user_args().has("human-only"):
			print("PASS: actual B2 selected-profile human game durably pending; no premature reward")
			app.queue_free()
			get_tree().quit()
			return
	# Each real report is durably adopted before explicitly requesting the next job.
	for expected in range(app.season.physical.pending.reports.size() + 1, 3):
		app.call("resume_round")
		var arrived: bool = false
		for frame in range(150000):
			await get_tree().physics_frame
			if not app._busy:
				arrived = true
				break
			if frame % 12000 == 0:
				print("AI_GAME_PROGRESS: ", expected, ":", frame)
		if not check(arrived and app.get("retained") == null and app.season.physical.pending.get("reports", []).size() == expected, "actual AI report checkpoint %d: %s" % [expected, app.notice]):
			return
		if not check(app.season.physical.pending.reports[-1].report.version == 114, "explicit successor League physical report114"):
			return
		var historical: Dictionary = app.season.physical.pending.reports[-1].report.duplicate(true)
		historical.version = 27
		if not check(not Report.valid(historical, app.season.build.pitch_catalog_signature()), "historical report family cannot replace current League proof"):
			return
		if OS.get_cmdline_user_args().has("one-ai"):
			if not check(ClubCareer.same(Codec.capture(app.season), JSON.parse_string(FileAccess.get_file_as_string(Save.PATH))), "actual NPC checkpoint exact disk capture"):
				return
			print("PASS: one real NPC report durable; settlement remains pending")
			app.queue_free()
			get_tree().quit()
			return
	app.call("resume_round")
	if not check(app.get("retained") == null and app.season.results.size() == 3 and app.season.player_results.size() == 1 and app.season.physical.pending.is_empty(), "three-game B2 selected-profile round settlement"):
		return
	var human_results: Array = app.season.results.filter(func(row: Dictionary) -> bool: return row.id == human[0])
	if not check(human_results.size() == 1, "exact actual human result"):
		return
	var own: int = human_results[0].home_runs if human_results[0].home == 0 else human_results[0].away_runs
	var rival: int = human_results[0].away_runs if human_results[0].home == 0 else human_results[0].home_runs
	var earned: int = (18 if own > rival else 12) if selected_profile == "standard" else LeagueRules.game_income(app.season.league_rules, own > rival, own, rival)
	if not check(app.season.build._bank.view().rewards.size() == 1 and app.season.build.cash() == earned, "one score-matched earned human reward"):
		return
	if not check(app.season.opponents.clubs.size() == 5, "all five AI clubs retained"):
		return
	for club: Dictionary in app.season.opponents.clubs.values():
		if not check(club.build._bank.view().rewards.size() == 1, "one reward per AI club"):
			return
	for index in range(6):
		var owner_build: SeasonBuild = app.season.build if index == 0 else app.season.opponents.clubs[str(index)].build
		var bank: SeasonOwnership = owner_build._bank
		var prefix: Dictionary = bank.to_data()
		prefix.events = []
		var replay: SeasonOwnership = SeasonOwnership.from_data(prefix, bank._catalog)
		if not check(replay != null, "original bank replay setup for payout audit"):
			return
		var reward_count: int = 0
		for event: Dictionary in bank._events:
			var cash: int = replay.cash()
			if not check(replay.commit(event).ok, "exact actual bank journal replay"):
				return
			if event.op == "reward":
				reward_count += 1
				var game: Dictionary = app.season.results.filter(func(row: Dictionary) -> bool: return row.id == event.game)[0]
				var scored: int = game.home_runs if game.home == index else game.away_runs
				var conceded: int = game.away_runs if game.home == index else game.home_runs
				var expected_income: int = (18 if scored > conceded else 12) if selected_profile == "standard" else LeagueRules.game_income(app.season.league_rules, scored > conceded, scored, conceded)
				if not check(replay.cash() - cash == expected_income, "actual exact replacement base payout without stacking club%d" % index):
					return
		if not check(reward_count == 1 and ClubCareer.same(replay.to_data(), bank.to_data()), "exact once reward and full journal club%d" % index):
			return
	var settled: Dictionary = Codec.capture(app.season)
	app.call("resume_round")
	if not check(not settled.is_empty() and ClubCareer.same(Codec.capture(app.season), settled), "duplicate round resume is inert"):
		return
	var build: SeasonBuild = app.season.build
	app.call("open_pitch_shop")
	var visible_shop: bool = false
	for child: Node in app.menu.get_children():
		if child is SeasonShopWindow and child.visible:
			visible_shop = true
	if not check(visible_shop, "actual full-shop presentation route reachable; not visual acceptance"):
		return
	build = app.season.build
	if not check(build._visit.open and not build._visit.cards.is_empty(), "generated earned shop opens"):
		return
	var cash_before: int = build.cash()
	var price: int = build.pack_price()
	var open_command: Dictionary = {"id":"pack:%d"%build.revision(),"rev":build.revision(),"op":"pack_open"}
	app.call("shop_command", open_command)
	if not check(app.get("retained") == null and app.season.build.pack_pending() and app.season.build.cash() == cash_before - price, "real earned pack payment"):
		return
	build = app.season.build
	var selected: Dictionary = {}
	for item: String in build._visit.cards:
		var targets: Array[Dictionary] = build.targets(item)
		if not targets.is_empty():
			selected = {"id":"pick:%d"%build.revision(),"rev":build.revision(),"op":"pack_pick","item":item,"player":targets[0].player,"pitch":targets[0].pitch,"replace":targets[0].replace}
			break
	if not check(not selected.is_empty(), "generated legal pack target"):
		return
	var before_profile: Dictionary = build.player(selected.player).duplicate(true)
	var book_revision: int = build._book.revision()
	app.call("shop_command", selected)
	var after_profile: Dictionary = app.season.build.player(selected.player)
	var stats_delta: int = 0
	var mastery_delta: int = 0
	for stat: String in before_profile.stats:
		stats_delta += int(after_profile.stats[stat]) - int(before_profile.stats[stat])
	for pitch: String in after_profile.mastery:
		mastery_delta += int(after_profile.mastery[pitch]) - int(before_profile.mastery.get(pitch, 0))
	if not check(stats_delta >= 0 and mastery_delta >= 0 and stats_delta + mastery_delta == 1 and app.season.build._book.revision() == book_revision + 1, "exactly one earned development level applied"):
		return
	var purchased: Dictionary = Codec.capture(app.season)
	if not check(not purchased.is_empty() and app.get("retained") == null and app.season.build._visit.pack_status == "used", "paid pack selection durably applied"):
		return
	app.call("shop_command", selected)
	if not check(ClubCareer.same(Codec.capture(app.season), purchased), "repeated purchase command grants nothing twice"):
		return
	if not check(ClubCareer.same(purchased, JSON.parse_string(FileAccess.get_file_as_string(Save.PATH))) and profile(app.season, selected_profile), "paid checkpoint exact profile and disk"):
		return
	print("PASS: three actual B2 selected-profile games settled once, earned pack paid/applied once, durable checkpoint")
	app.queue_free()
	await get_tree().process_frame
	get_tree().quit()
