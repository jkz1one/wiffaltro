extends RefCounted
## Snapshot only fresh matches; callers retain all original resources and state.
# gdlint: disable=max-returns


static func capture(source: MatchState) -> MatchState:
	if (
		source == null
		or not source.stadium.is_empty()
		or LeagueRules.normalized(source.league_rules) != LeagueRules.capture("dugout")
		or not load("res://src/gameplay/season/successor_league_general_v2/pitch_evidence.gd").context_valid(source.pitch_catalog)
		or source.away_team == null
		or source.home_team == null
		or source.phase != MatchState.Phase.PRE_PITCH
	):
		return null
	if (
		not is_finite(source.ai_tactical_quality)
		or source.ai_tactical_quality < 0.0
		or source.ai_tactical_quality > 1.0
	):
		return null
	if (
		source.inning != 1
		or not source.top_half
		or source.plate_appearance_number != 1
		or source.balls != 0
		or source.strikes != 0
		or source.outs != 0
		or not source.between_batters
		or source.elapsed_seconds != 0.0
		or not source.performance.players.is_empty()
		or not source.sides.rows.is_empty()
		or not source.sure_shot.releases.is_empty()
		or not source.sure_shot.calls.is_empty()
		or not source.cold.appearances.is_empty()
		or not source.clean_outs.rows.is_empty()
		or not source.ai_choice_events.is_empty()
		or not source._ai_choice_prepared.is_empty()
		or not source.base_flow.ready.is_empty()
		or not source.base_flow.results.is_empty()
		or not source.base_flow.tags.is_empty()
		or not source.base_flow.advances.is_empty()
		or not source.frozen_contacts.is_empty()
		or source.bases.occupied_count() != 0
		or source.gear_usage.started
		or not source.gear_usage.first_pitch.is_empty()
	):
		return null
	if not load("res://src/gameplay/season/successor_league_general_v2/pitch_evidence.gd").context_valid(source.pitch_catalog) \
		or source.away_team.pitch_catalog != source.pitch_catalog \
		or source.home_team.pitch_catalog != source.pitch_catalog:
		return null
	if not source.stadium.is_empty() and not StadiumLayoutSnapshot.valid(source.stadium):
		return null
	if source.league_rules.is_empty():
		if (
			not source.away_team.league_rules.is_empty()
			or not source.home_team.league_rules.is_empty()
		):
			return null
	elif (
		LeagueRules.normalized(source.league_rules).is_empty()
		or source.away_team.league_rules != source.league_rules
		or source.home_team.league_rules != source.league_rules
		or (source.pitch_catalog.is_empty() and not source.stadium.is_empty())
	):
		return null
	if (
		not source.away_team.dynasty_start.is_empty()
		and not source.home_team.dynasty_start.is_empty()
	):
		return null
	if LeagueRules.equipment(source.league_rules):
		var equipped: Array[String] = []
		for team: TeamMatchState in [source.away_team, source.home_team]:
			if not team.ai_base:
				var owned: Dictionary = load("res://src/gameplay/season/successor_league_general_v2/physical_league.gd").wallet(
					team.league_rules, team.league_ownership
				)
				if owned.is_empty():
					return null
				equipped.append_array(SeasonReclamation.receipts(owned))
		equipped.sort()
		var submitted: Array[String] = source.gear_usage.equipped.duplicate()
		submitted.sort()
		if submitted != equipped:
			return null
	var away: TeamMatchState = _team(source.away_team)
	var home: TeamMatchState = _team(source.home_team)
	if away == null or home == null:
		return null
	var ids: Array = []
	for team: TeamMatchState in [away, home]:
		for player: PlayerMatchState in team.roster:
			if ids.has(player.definition.id):
				return null
			ids.append(player.definition.id)
	var result: MatchState = MatchState.create(away, home)
	result.league_rules = source.league_rules.duplicate(true)
	result.pitch_catalog = source.pitch_catalog
	result.stadium = (
		StadiumLayoutSnapshot.normalized(source.stadium) if not source.stadium.is_empty() else {}
	)
	result.away_team.field_supply.receipt = source.away_team.field_supply.receipt
	result.home_team.field_supply.receipt = source.home_team.field_supply.receipt
	result.gear_usage.equipped.assign(source.gear_usage.equipped.duplicate())
	result.ai_tactical_quality = source.ai_tactical_quality
	result.optics_mode = source.optics_mode
	result.jumpstart_mode = source.jumpstart_mode
	result.cornerstone_anchored = source.cornerstone_anchored
	return result


static func _team(source: TeamMatchState) -> TeamMatchState:
	if source == null or not source.is_valid_roster() or source.runs != 0:
		return null
	if (
		source.batting_index != 0
		or source.pitcher_index not in range(4)
		or source.fielder_index not in range(4)
		or source.fielder_index == source.pitcher_index
		or not source.tactics.consumed.is_empty()
		or not source.field_supply.evidence().is_empty()
		or source.field_supply.clean != 0
		or source.sure_shot_locked
		or source.encore_used
		or source.strikecraft_uses != 0
		or source.strikecraft_refunded != 0.0
	):
		return null
	if not source.inventory_actions.is_empty():
		return null
	if (
		not source.recovery.workload_events.is_empty()
		or not source.recovery.workload_initial.is_empty()
		or not source.recovery.workload_input.is_empty()
	):
		return null
	if source.ai_base and source.ai_budget.is_empty():
		return null
	if not LeagueRules.doubleheader(source.league_rules) and not source.league_pair.is_empty():
		return null
	if not source.league_rules.is_empty():
		if not load("res://src/gameplay/season/successor_league_general_v2/physical_league.gd").team_valid(source):
			return null
	elif not source.league_ownership.is_empty() or not source.dynasty_start.is_empty():
		return null
	elif (
		not source.ai_budget.is_empty()
		and (
			source.ai_recovery_pitcher.is_empty()
			or not PhysicalBudgetEvidence.shape(
				{"version": 4, "budget": source.ai_budget, "initial": source.ai_tactical_initial},
				12 if source.ai_base else 11
			)
		)
	):
		return null
	if source.ai_heat and source.ai_tactical_hitter.is_empty():
		return null
	if (
		not source.ai_recovery_pitcher.is_empty()
		and (
			not source.ai_heat
			or not source.recovery.ready.is_empty()
			or not source.recovery.costs.is_empty()
			or not source.recovery.refunds.is_empty()
		)
	):
		return null
	if not source.ai_tactical_hitter.is_empty():
		var roster: Array = source.roster.map(
			func(player: PlayerMatchState) -> String: return String(player.definition.id)
		)
		if (
			not source.ai_sponsor_choices
			or not roster.has(source.ai_tactical_hitter)
			or not ClubCareer.same(source.ai_tactical_initial, source.tactics.held)
		):
			return null
		if not source.ai_recovery_pitcher.is_empty() and not roster.has(source.ai_recovery_pitcher):
			return null
	var definitions: Array[PlayerDefinition] = []
	for player: PlayerMatchState in source.roster:
		if (
			player == null
			or player.definition == null
			or player.definition.id.is_empty()
			or player.definition.starting_pitches.is_empty()
			or player.pitch_count != 0
			or player.pitching_finished
			or player.first_batter_completed
			or not is_finite(player.stamina_max)
			or player.stamina_max <= 0.0
			or not is_finite(player.stamina_remaining)
			or player.stamina_remaining < 0.0
			or player.stamina_remaining > player.stamina_max
		):
			return null
		definitions.append(player.definition.duplicate(true) as PlayerDefinition)
	var result: TeamMatchState = TeamMatchState.create(source.display_name, definitions)
	result.league_rules = source.league_rules.duplicate(true)
	result.pitch_catalog = source.pitch_catalog
	result.league_ownership = source.league_ownership.duplicate(true)
	result.league_pair = source.league_pair.duplicate(true)
	result.dynasty_start = (
		{}
		if source.dynasty_start.is_empty()
		else preload("res://src/gameplay/season/dynasty_start_profile.gd").normalize(
			source.dynasty_start
		)
	)
	result.dynasty_start.make_read_only()
	result.ai_sponsor_choices = source.ai_sponsor_choices
	result.track_bases = source.track_bases
	result.ai_base = source.ai_base
	result.ai_budget = source.ai_budget.duplicate(true)
	result.ai_heat = source.ai_heat
	result.ai_recovery_pitcher = source.ai_recovery_pitcher
	result.ai_tactical_hitter = source.ai_tactical_hitter
	result.ai_tactical_initial.assign(source.ai_tactical_initial.duplicate(true))
	result.pitcher_index = source.pitcher_index
	result.fielder_index = source.fielder_index
	result.copy_source = source.copy_source
	result.scouted_recipe = source.scouted_recipe
	result.tactics.held.assign(source.tactics.held.duplicate(true))
	result.tactics.insured_receipt = source.tactics.insured_receipt
	result.tactics.track_walks = source.tactics.track_walks
	result.field_supply.retired = source.field_supply.retired
	result.field_supply.capacity = source.field_supply.capacity
	result.field_supply.reserved = source.field_supply.reserved
	result.field_supply.receipt = source.field_supply.receipt
	for index in range(4):
		result.roster[index].stamina_max = source.roster[index].stamina_max
		result.roster[index].stamina_remaining = source.roster[index].stamina_remaining
		result.roster[index].batting_hand_override = source.roster[index].batting_hand_override
	return result
