extends RefCounted
## Ordinary tier0 boundary only, never career, higher-tier or owner authority.
const LEAGUES: Array[String] = ["dugout", "sponsorship", "earnings", "academy", "equipment", "doubleheader"]
static func supported(rules: Variant, live: Variant) -> bool:
	if not rules is Dictionary or not live is bool:
		return false
	if rules.is_empty():
		return not live # Bare Standard remains distinct from explicit League profiles.
	if not live:
		return false
	# Scramble is one fixed stat-permutation profile, not a random League.
	# Its root season seed is part of the exact normalized identity. Runtime,
	# decoder and whole-season scope separately bind it to the actual season.
	if rules.get("league") == "scramble":
		var normalized: Dictionary = LeagueRules.normalized(rules)
		return not normalized.is_empty() and normalized.tier == 0 \
			and ClubCareer.same(rules, LeagueRules.capture("scramble", 0, normalized.season_seed))
	for id: String in LEAGUES:
		if ClubCareer.same(rules, LeagueRules.capture(id)):
			return true
	return false

static func build_matches(build: SeasonBuild, rules: Dictionary) -> bool:
	if build == null:
		return false
	if rules.is_empty():
		return build._format == 41 and build._league_rules.is_empty() and build.pricing_context_valid()
	return build._format == 45 and build.league_context_valid() and ClubCareer.same(build._league_rules, rules)
