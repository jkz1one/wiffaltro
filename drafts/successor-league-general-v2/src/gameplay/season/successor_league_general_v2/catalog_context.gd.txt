static func signature() -> String:
	var source: String = Provider.signature()
	if source.is_empty():
		return ""
	var legacy: String = load("res://src/gameplay/season/successor_league_v1/catalog_context.gd").signature()
	if legacy.is_empty():
		return ""
	var parts: Array[String] = [PURPOSE, source, legacy]
	for path: String in FILES:
		var digest: String = FileAccess.get_sha256(path)
		if digest.is_empty():
			return ""
		parts.append(digest)
	return "|".join(parts).sha256_text()

static func descriptor() -> Dictionary:
	var source: String = signature()
	return {} if source.is_empty() else {"version": VERSION, "purpose": PURPOSE, "signature": source}

static func valid_descriptor(value: Variant) -> bool:
	if not value is Dictionary or not SeasonOwnership._keys(value, ["version", "purpose", "signature"]):
		return false
	return SeasonOwnership._whole(value.version, VERSION, VERSION) \
		and value.purpose is String and value.purpose == PURPOSE \
		and value.signature is String and not value.signature.is_empty() \
		and value.signature == signature()
