extends "res://src/presentation/season/successor_league_general_v2/app.gd"
## Additive reroll-only transaction route. All other commands and visuals inherit
## unchanged General114 behavior. No save/catalog identity substitution or cache.
const Transaction = preload("res://src/gameplay/season/successor_shop_transaction_v1/transaction.gd")

func commit_shop(command: Dictionary) -> bool:
	if command.get("op") != "reroll":
		return super.commit_shop(command)
	var roots: Array[Resource] = Codec._operation_content_roots()
	var result: bool = _commit_reroll(command)
	roots.clear()
	return result

func _commit_reroll(command: Dictionary) -> bool:
	if not _available() or season == null or not Codec.scope(season) \
		or not season.physical.pending.is_empty() or not season.shop_available():
		return false
	# A previously handled request is not a new persistence operation. Even an
	# identical replay must not rotate the backup; conflicting reuse also rejects.
	var request_id: Variant = command.get("id")
	if request_id is String and not request_id.is_empty() and season.build._requests.has(request_id):
		return false
	var owner: SeasonState = season
	var original_lab: PitchBatLab = lab
	var original_state: MatchState = lab._match_state if is_instance_valid(lab) else null
	var before: Dictionary = Codec.capture(owner)
	if before.is_empty():
		notice = "The current shop checkpoint could not be validated."
		return false
	_busy = true
	var outcome: Dictionary = Transaction.commit(owner,command.duplicate(true))
	var candidate: SeasonState = outcome.get("candidate")
	if candidate == null:
		_busy = false
		notice = str(outcome.get("error","The reroll could not be committed."))
		return false
	# Retain even a durable candidate whose pinned owner no longer matches.
	# Never call _retain: it resets saved=false and would write a second time.
	retained = candidate
	retained_owner = owner
	retained_before = before.duplicate(true)
	retained_lab = original_lab
	retained_state = original_state
	retained_saved = outcome.get("saved",false) == true
	notice = str(outcome.get("error","The reroll could not be saved."))
	_busy = false
	if not retained_saved:
		show_season() # Explicit retry only; keep the exact failed candidate.
		return false
	notice = "Saved reroll retained until the original owner can be promoted."
	retry_save() # Existing identity/cleanup guards; saved=true skips writing.
	return season == candidate and retained == null
