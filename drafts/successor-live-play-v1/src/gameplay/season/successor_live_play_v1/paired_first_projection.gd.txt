# Geometry is reusable; only these exact private phases are qualified.
const HUMAN_PHASES: Array[int]=[0,1,2,3,4,5,6]
const REPORT_PHASES: Array[int]=[0,1,2,3,4,5,6]
const SETTLEMENT_PHASES: Array[int]=[0,1,2,3,4,5,6]
# Exact authored regular advancement, independent of UI eligibility.
# State identity, native replay and the owned CAS still gate every operation.
const LATE_REGULAR: Dictionary={7:{"revision":20,"reports":2,"settle":true},8:{"revision":23,"reports":2,"settle":true},9:{"revision":26,"reports":2,"settle":true}}

static func late_revision(round_index: int) -> int:
	return int(LATE_REGULAR.get(round_index,{}).get("revision",-1))

static func human_enabled(round_index: int) -> bool:
	return round_index==10 or round_index in HUMAN_PHASES or LATE_REGULAR.has(round_index)

static func settlement_enabled(round_index: int) -> bool:
	return round_index==10 or round_index in SETTLEMENT_PHASES or LATE_REGULAR.get(round_index,{}).get("settle",false)

static func _round_fixtures(round_index: int) -> Dictionary:
	return Request.authored_phase(round_index) if human_enabled(round_index) else {}

# Qualified report count is independent of the authored number of NPC games.
static func report_limit(round_index: int) -> int:
	return 2 if round_index==10 or round_index in REPORT_PHASES else int(LATE_REGULAR.get(round_index,{}).get("reports",0))

static func bound_shape(value: Variant) -> bool:
	if not Live.keys(value,["human","reports","binding"]) or not value.human is Array or value.human.size()!=14 or not value.reports is Array or value.reports.size()>2: return false
	if Live.whole(value.human[0],30,30):
		for index: int in range(value.reports.size()):
			if not _bound_report(value.reports[index],31+index): return false
		return _binding_shape(value.binding)
	var fixtures: Dictionary={}
	for round_index: int in HUMAN_PHASES+LATE_REGULAR.keys():
		var authored: Dictionary=_round_fixtures(round_index)
		if not authored.is_empty() and Live.whole(value.human[0],authored.human.id,authored.human.id):
			if value.reports.size()>report_limit(round_index): return false
			fixtures=authored
			break
	if fixtures.is_empty(): return false
	for index: int in range(value.reports.size()):
		var row: Variant=value.reports[index]
		var id: int=fixtures.npc[index].id
		if not _bound_report(row,id): return false
	return _binding_shape(value.binding)

static func _bound_report(row: Variant, id: int) -> bool:
	return Live.keys(row,["fixture","request","report"]) and Live.whole(row.fixture,id,id) and row.request is String and row.request.length()==64 and row.request.is_valid_hex_number(false) and row.report is Dictionary

static func _binding_shape(binding: Variant) -> bool:
	return Live.keys(binding,["version","request"]) and Live.whole(binding.version,1,1) and binding.request is String and binding.request.length()==64 and binding.request.is_valid_hex_number(false)

static func start(source: SeasonState) -> Dictionary:
	if not Live.scope(source) or not Live.same(source.league_rules,LeagueRules.capture("doubleheader")) or not human_enabled(source.round_index) or source.picks.size()!=4 or not source.physical.pending.is_empty(): return {}
	var semifinal: bool=source.round_index==10
	if semifinal:
		if not Live._sealed_regular_end_shape(source): return {}
	elif source.phase!=SeasonState.Phase.REGULAR: return {}
	if Live.CommonStart.detached_earned_doubleheader(source.career.start_basis):
		if not Live.earned_regular_play_ready(source): return {}
	else:
		if source.round_index==1 and not Live.return_session_ready(source): return {}
		if source.round_index==2 and not Live.next_first_session_ready(source): return {}
		if source.round_index==4 and not Live._unpurchased_completed_legs(source.build,4).is_empty(): return {}
		if source.round_index==6 and not Live._signed_visit6_journal(source.build): return {}
		if LATE_REGULAR.has(source.round_index) and not Live._paired_late_start(source): return {}
	# Saved JSON hydrates whole IDs as floats. Bind to the semantically equal
	# frozen-authored row so warm and cold hash exactly the same primitives.
	var authored: Dictionary=_semifinal_fixtures(source) if semifinal else _round_fixtures(source.round_index)
	if authored.is_empty(): return {}
	var fixture: Dictionary=authored.human
	if not Live.same(fixture,source.pending_fixture()): return {}
	var field: FieldDefinition=SeasonState.field_for_fixture(fixture,source)
	var state: MatchState=GeneralOps._source_public_make_match(source)
	if field==null or state==null or not state.stadium.is_empty() or not state.sure_shot.enable_current_evidence(): return {}
	var hash: String=Request._for_materialized_wallet(source,fixture,state,field) if semifinal else Request.for_live_regular(source,fixture,state,field)
	if hash.is_empty(): return {}
	var club: TeamMatchState=state.home_team if fixture.home==0 else state.away_team
	var format: String="private-semifinal-human-start-v1" if semifinal else "private-late-regular-human-start-v1" if LATE_REGULAR.has(source.round_index) else ["private-first-human-start-v1","private-return-human-start-v1","private-next-first-human-start-v1","private-next-return-human-start-v1","private-pair3-first-human-start-v1","private-pair3-return-human-start-v1","private-pair4-first-human-start-v1"][source.round_index]
	if source.round_index==1:
		var unpurchased: String=Live.unpurchased_return_kind(source.build)
		if not unpurchased.is_empty(): format="private-unpurchased-return-"+unpurchased+"-v1"
	if source.round_index==2:
		var unpurchased: String=Live.unpurchased_next_pair_kind(source)
		if not unpurchased.is_empty(): format="private-unpurchased-next-first-"+unpurchased+"-v1"
	if source.round_index==3:
		var unpurchased: String=Live.unpurchased_next_reverse_kind(source)
		if not unpurchased.is_empty(): format="private-unpurchased-next-reverse-"+unpurchased+"-v1"
	var live: Dictionary={"format":format,"request":hash,"equipped":state.gear_usage.equipped,"held":club.tactics.held,"field_supply":[club.field_supply.receipt,club.field_supply.retired,club.field_supply.capacity],"scout":club.scouted_recipe,"copy":club.copy_source,"insurance":club.tactics.insured_receipt,"track_walks":club.tactics.track_walks}
	return {"state":state,"field":field,"fixture":fixture.duplicate(true),"binding":{"version":1,"request":JSON.stringify(live,"",true,true).sha256_text()}}

static func verify_bound(source: SeasonState) -> bool:
	return not _inspect_bound(source).is_empty()

# Read only the exact next slot. Job and Store admission independently replay it.
static func next_npc_fixture(source: SeasonState) -> Dictionary:
	if not Live.scope(source) or report_limit(source.round_index)==0 or not bound_shape(source.physical.pending): return {}
	if source.round_index==10:
		var playoff: Dictionary=_semifinal_fixtures(source)
		if playoff.is_empty(): return {}
		if source.physical.pending.reports.is_empty(): return playoff.other
		if source.physical.pending.reports.size()==1 and _unpaid_human_eliminated(source.physical.pending.human,playoff.human):
			return {"id":32} # Ordinal only: only the native detached transition authors its geometry.
		return {}
	var authored: Dictionary=_round_fixtures(source.round_index)
	var index: int=source.physical.pending.reports.size()
	if authored.is_empty() or index>=report_limit(source.round_index) or index>=authored.npc.size() or source.physical.pending.human[0]!=authored.human.id: return {}
	return authored.npc[index]

# Only a detached match start is returned, never the projected season/payout.
static func npc_job(source: SeasonState, fixture_id: int) -> Dictionary:
	var next: Dictionary=next_npc_fixture(source)
	if next.is_empty() or fixture_id!=next.id: return {}
	return _inspect_bound(source,false,true).get("job",{})

# There is no reusable projection token or caller-supplied candidate.
static func materialize_settlement(source: SeasonState, leg: int=0) -> SeasonState:
	if not settlement_enabled(leg): return null
	if leg==10:
		if not Live.scope(source) or source.round_index!=10 or source.phase!=SeasonState.Phase.SEMIFINAL or not bound_shape(source.physical.pending) or source.physical.pending.reports.size()!=2: return null
		return _inspect_bound(source,true).get("settlement")
	var authored: Dictionary=_round_fixtures(leg)
	if authored.is_empty() or not Live.scope(source) or source.round_index!=leg or not bound_shape(source.physical.pending) or source.physical.pending.reports.size()!=authored.npc.size(): return null
	return _inspect_bound(source,true).get("settlement")

static func _bound_inputs(source: SeasonState) -> Dictionary:
	if not Live.scope(source) or not bound_shape(source.physical.pending): return {}
	var pending: Dictionary=source.physical.pending.duplicate(true)
	var pristine: SeasonState=source.fork_live()
	if pristine==null: return {}
	pristine.physical.pending.clear()
	var opening: Dictionary=start(pristine)
	if opening.is_empty() or not Live.same(opening.binding,pending.binding) or opening.fixture.id!=pending.human[0]: return {}
	return {"pristine":pristine,"pending":pending,"fixture":opening.fixture}

static func _inspect_bound(source: SeasonState, produce_settlement: bool = false, preview_job: bool = false) -> Dictionary:
	var bound:=_bound_inputs(source)
	if bound.is_empty(): return {}
	if source.round_index==10:
		return _inspect_semifinal(bound.pristine,bound.pending,bound.fixture,preview_job,produce_settlement)
	if source.round_index>=2 and human_enabled(source.round_index):
		return _inspect_followup(bound.pristine,bound.pending,bound.fixture,preview_job,produce_settlement)
	if source.round_index==1:
		return _inspect_return(bound.pristine,bound.pending.human,bound.pending.reports,preview_job,produce_settlement)
	return _inspect(bound.pristine,bound.pending.human,bound.pending.reports,produce_settlement)

# Fixed cold archive replay only. Starts with a newly reconstructed canonical
# earned opening, replays each physical suffix and actual Shop command, and
# returns nothing until the unchanged full validator checks all paid history.
# No writer/individual game entry takes a caller-authored prefix or trust flag.
static func replay_earned_regular_archive(opening: SeasonState, journal: Array, current_pending: Dictionary) -> Dictionary:
	if not Live.scope(opening) or not Live.CommonStart.detached_earned_doubleheader(opening.career.start_basis) \
			or opening.round_index!=0 or not opening.physical.pending.is_empty() or journal.size()>Live.EARNED_PAIRED_PAID_LIMIT: return {}
	var named=load("res://src/gameplay/season/successor_dynasty_owner_v1/earned_named_base.gd")
	var draft: SeasonState=opening
	for row: Variant in journal:
		if not Live.keys(row,["pending","shop"]) or not row.pending is Dictionary or not row.shop is Array or row.shop.size()>128: return {}
		var leg: int=draft.round_index
		if leg>0:
			draft=named._paired_replay_boundary(draft)
			if draft==null: return {}
		draft.physical.pending=row.pending.duplicate(true)
		if leg<2:
			draft=materialize_settlement(draft,leg)
		else:
			var bound:=_bound_inputs(draft)
			if bound.is_empty() or bound.pending.reports.size()!=_round_fixtures(leg).npc.size() \
					or not _followup_input(bound.pristine,bound.pending,false,true): return {}
			draft=_inspect_followup_physical(bound.pristine,bound.pending,bound.fixture,false,true).get("settlement")
		if draft==null or draft.round_index!=leg+1 or not named._paired_current(draft): return {}
		for command: Variant in row.shop:
			if not command is Dictionary: return {}
			draft=named.paired_shop_step(draft,command)
			if draft==null: return {}
	if not current_pending.is_empty():
		if draft.round_index>0:
			draft=named._paired_replay_boundary(draft)
			if draft==null: return {}
		draft.physical.pending=current_pending.duplicate(true)
	# Ordinary.capture verifies the complete archived paid prefix, any live
	# held result, exact workload/source and all native opponent reports.
	var final: Dictionary=Live.capture(draft)
	return {} if final.is_empty() else {"draft":draft,"current":final}

# Frozen Pending queues its already shape-checked archive before recording
# the human result. Only this exact private projection can carry that queue.
static func npc_queue(season: SeasonState) -> bool:
	if season==null or season.get_script()!=PrivateFirstFork or season.physical==null or not season.physical.projecting or season.physical.cursor!=0: return false
	if season.physical.reports.size() not in [1,2]: return false
	for index: int in range(season.physical.reports.size()):
		var row: Dictionary=season.physical.reports[index]
		if not Live.keys(row,["fixture","request","report"]) or not Live.whole(row.fixture,index+1,index+1): return false
	return true

# Exact non-persistable type and state after the original human-only projection.
# The mutable projecting field alone never admits any materialized owner.
static func projected_npc_fixture(season: SeasonState, fixture: Dictionary) -> Dictionary:
	if season==null or not Live.same(season.league_rules,LeagueRules.capture("doubleheader")) or season.picks.size()!=4: return {}
	if season.get_script()==PrivateEarnedCpuBracket:
		return _projected_earned_cpu_postseason(season,fixture)
	if season.get_script()==PrivateEarnedClubSemifinal:
		return _projected_earned_club_cpu_final(season,fixture) if season.phase==SeasonState.Phase.FINAL \
			else _projected_earned_club_other_semifinal(season,fixture)
	if season.round_index==10:
		return _projected_final_npc(season,fixture) if season.phase==SeasonState.Phase.FINAL else _projected_semifinal_npc(season,fixture)
	if season.phase!=SeasonState.Phase.REGULAR: return {}
	var leg: int=season.round_index
	if not ((leg==0 and season.get_script()==PrivateFirstFork) or (leg==1 and season.get_script()==PrivateReturnFork) or (leg>=2 and report_limit(leg)>0 and season.get_script()==PrivateFollowupFork)): return {}
	var fixtures: Dictionary=_round_fixtures(leg)
	var human: int=fixtures.human.id
	var archive: int=fixtures.prior_npcs.size()
	if season.physical==null or not season.physical.projecting or not season.physical.pending.is_empty() or season.player_results.size()!=fixtures.prior_humans.size()+1 or not season.build._bank.view().rewards.has(str(human)): return {}
	var progress: int=season.physical.cursor-archive
	if progress<0 or progress>=report_limit(leg) or progress>=fixtures.npc.size() or season.physical.reports.size()<archive+progress: return {}
	var played: Array=fixtures.prior_results+[human]
	for index: int in range(progress): played.append(fixtures.npc[index].id)
	if not Live.same(season.results.map(func(row):return row.get("id")),played): return {}
	var next: Dictionary=fixtures.npc[progress]
	return next if Live.same(next,fixture) else {}

class PrivateFollowupFork extends PrivateFirstFork:
	func _begin_playoffs() -> bool:
		var codec=load("res://src/gameplay/season/successor_live_play_v1/codec.gd")

			return super._begin_playoffs()

		for pair: int in range(2):
			semifinals.append({"id":30+pair,"round":10,"home":playoff_seeds[pair],"away":playoff_seeds[3-pair]})
		phase=Phase.SEMIFINAL
		return true

static func _projected_earned_cpu_postseason(season: SeasonState, fixture: Dictionary) -> Dictionary:
	if season==null or season.get_script()!=PrivateEarnedCpuBracket or season.round_index!=10 \
			or season.champion!=-1 or season.player_results.size()!=10 or season.playoff_seeds.size()!=4 \
			or 0 in season.playoff_seeds or season.semifinals.size()!=2 or season.physical==null \
			or not season.physical.projecting or not season.physical.pending.is_empty(): return {}
	var cursor: int=season.physical.cursor
	if cursor<20 or cursor>22 or season.results.size()!=10+cursor or season.physical.reports.size()<cursor \
			or season.physical.reports.size()>23: return {}
	var authored: Dictionary
	if cursor<22:
		if season.phase!=SeasonState.Phase.SEMIFINAL or not season.final_fixture.is_empty(): return {}
		authored=season.semifinals[cursor-20]
	else:
		if season.phase!=SeasonState.Phase.FINAL: return {}
		authored=season.final_fixture
	if not Live.whole(authored.get("id"),10+cursor,10+cursor) \
			or not Live.whole(authored.get("round"),10+int(cursor==22),10+int(cursor==22)) \
			or not Live.whole(authored.get("home"),1,5) or not Live.whole(authored.get("away"),1,5) \
			or authored.home==authored.away or (cursor==22 and authored.get("neutral")!=true): return {}
	var canonical: Dictionary={"id":int(authored.id),"round":int(authored.round),"home":int(authored.home),"away":int(authored.away)}
	if cursor==22: canonical.neutral=true
	return canonical if Live.same(authored,canonical) and Live.same(fixture,canonical) else {}

# Only the strict entry above or the fixed installed matching persisted
# revision may reach the suffix projection. Newly submitted physical
# reports still pass their original exact independent resolver checks.
static func _project_earned_cpu_bracket(source: SeasonState, reports: Array) -> Dictionary:
	if source==null or not Live.scope(source) or not Live.CommonStart.detached_earned_doubleheader(source.career.start_basis) \
			or not _unplayed_semifinals(source) or source.playoff_seeds.has(0) or reports.size()>3 \
			or source.physical.reports.size()!=20: return {}
	for index: int in range(reports.size()):
		if not reports[index] is Dictionary or not _bound_report(reports[index],30+index): return {}
	var candidate: SeasonState=_detached(source,PrivateEarnedCpuBracket)
	if candidate==null or Live.scope(candidate): return {}
	candidate.phase=SeasonState.Phase.REGULAR
	candidate.playoff_seeds.clear()
	candidate.semifinals.clear()
	candidate.physical.projecting=true
	candidate.physical.reports.append_array(reports.duplicate(true))
	var complete: bool=candidate._begin_playoffs()
	if not Live.same(candidate.playoff_seeds,source.playoff_seeds) or not Live.same(candidate.semifinals,source.semifinals) \
			or not Live.same(candidate.results.slice(0,30),source.results) or not Live.same(candidate.player_results,source.player_results) \
			or not Live.same(candidate.physical.reports,source.physical.reports+reports) \
			or candidate.physical.cursor!=20+reports.size() or not candidate.physical.error.is_empty() \
			or not Live.same(candidate.build._events,source.build._events) \
			or not Live.same(candidate.build._bank.view(),source.build._bank.view()): return {}
	if reports.size()<3:
		if complete or candidate.champion!=-1 or candidate.results.size()!=30+reports.size(): return {}
		var needed:=_projected_earned_cpu_postseason(candidate,candidate.physical.needed)
		if needed.is_empty(): return {}
		var job:=_job(candidate,needed)
		if not job.is_empty(): job.fixture=needed
		return job
	if not complete or candidate.results.size()!=33 or not candidate.physical.needed.is_empty() \
			or _complete_geometry(candidate).is_empty() or not GeneralPending._receipts(candidate): return {}
	candidate.physical.projecting=false
	var settled:=_concrete_settlement(candidate)
	if settled.is_empty(): return {}
	settled.current=Live.capture(settled.settlement)
	return settled if not settled.current.is_empty() else {}

# Only earned Doubleheader's fully paid regular archive reaches this
# geometry. Leave the earlier sealed (1-v-4 only) private draft unchanged.
static func _earned_club_semifinal_pair(source: SeasonState) -> Dictionary:
	if source==null or source.career==null or not Live.scope(source) \
			or not Live.CommonStart.detached_earned_doubleheader(source.career.start_basis) \
			or not _unplayed_semifinals(source) or source.physical.reports.size()!=20: return {}
	var position: int=source.playoff_seeds.find(0)
	if position<0 or position>3: return {}
	var slot: int=mini(position,3-position)
	var seeds: Array=source.playoff_seeds
	var human: Dictionary={"id":30+slot,"round":10,"home":int(seeds[slot]),"away":int(seeds[3-slot])}
	var other_slot: int=1-slot
	var other: Dictionary={"id":30+other_slot,"round":10,"home":int(seeds[other_slot]),"away":int(seeds[3-other_slot])}
	if 0 not in [human.home,human.away] or 0 in [other.home,other.away] \
			or not Live.same(source.pending_fixture(),human) or not Live.same(source.semifinals[other_slot],other): return {}
	return {"human":human,"other":other,"seed":position+1}

		"binding":{"version":1,"request":JSON.stringify(live,"",true,true).sha256_text()}}

static func earned_club_semifinal_held_shape(value: Variant) -> bool:

# Full detached validation. The installed writer uses the suffix only

# The other semifinal derives from the exact held human result. No slot,

# Only authenticate this one receipt. Resolving a report on the detached
# fork must not settle either semifinal, advance to the final or pay.
static func _earned_club_other_semifinal_held_bound(source: SeasonState,value: Variant,row: Variant) -> bool:
	var fork:=_earned_club_semifinal_fork_bound(source,value)
	if fork.is_empty() or not _bound_report(row,int(fork.fixture.id)): return false
	var candidate: SeasonState=fork.candidate
	var fixture: Dictionary=fork.fixture
	var needed: String=Physical.request_key(candidate,fixture)
	if needed.is_empty() or needed!=row.request: return false
	candidate.physical.reports.append(row.duplicate(true))
	var played: Dictionary=candidate.physical.resolve(candidate,fixture)
	return not played.is_empty() and Live.whole(played.get("id"),int(fixture.id),int(fixture.id)) \
		and candidate.physical.cursor==21 and candidate.physical.error.is_empty() and candidate.physical.needed.is_empty() \
		and candidate.phase==SeasonState.Phase.SEMIFINAL and candidate.results.size()==31 \
		and candidate.champion==-1 and candidate.final_fixture.is_empty() and not Live.scope(candidate)

static func _earned_club_semifinal_fork_bound(source: SeasonState,value: Variant) -> Dictionary:
	if not earned_club_semifinal_held_shape(value): return {}
	var game:=_earned_club_semifinal_opening_bound(source)
	if game.is_empty() or game.fixture.id!=value.human[0] or not Live.same(game.binding,value.binding): return {}
	var candidate: SeasonState=_detached(source,PrivateEarnedClubSemifinal)
	if candidate==null or Live.scope(candidate): return {}
	var shape: Dictionary={"human":value.human,"reports":[]}
	if not GeneralPending._doubleheader_shape(shape,Live.Catalog.game_signature()): return {}
	var projected:=GeneralPending._project_candidate(candidate,shape)
	if projected.has("error") or projected.get("candidate")!=candidate or not Live.same(projected.get("fixture"),game.other) \
			or not Live.same(candidate.physical.needed,game.other) or not Live.same(candidate.semifinals,source.semifinals) \
			or not Live.same(candidate.playoff_seeds,source.playoff_seeds) \
			or not Live.same(candidate.results.slice(0,30),source.results) \
			or not Live.same(candidate.player_results.slice(0,10),source.player_results) \
			or not Live.same(candidate.physical.reports,source.physical.reports) \
			or _projected_earned_club_other_semifinal(candidate,game.other).is_empty(): return {}
	var played: Dictionary=candidate.player_results.back()
	if played.id!=int(value.human[0]) or played.away_runs!=value.human[1] \
			or played.home_runs!=value.human[2] or not Live.same(played,candidate.results.back()): return {}
	return {"candidate":candidate,"fixture":game.other}

# Frozen General settlement owns both match rewards. On the isolated fork
# it may stop for the CPU final or open the club's actual final, never invent a
# report or expose a partially paid saved season.
static func earned_club_semifinals_paid(source: SeasonState,held: Variant,other: Variant) -> Dictionary:
	if source==null or not Live.scope(source) or not Live._paid_history_valid(source): return {}
	return _earned_club_semifinals_paid_bound(source,held,other)

static func _earned_club_semifinals_paid_bound(source: SeasonState,held: Variant,other: Variant) -> Dictionary:
	if not earned_club_semifinal_held_shape(held): return {}
	var game:=_earned_club_semifinal_opening_bound(source)
	if game.is_empty() or game.fixture.id!=held.human[0] or not Live.same(game.binding,held.binding) \
			or not _bound_report(other,int(game.other.id)): return {}
	var candidate: SeasonState=_detached(source,PrivateEarnedClubSemifinal)
	if candidate==null or Live.scope(candidate): return {}
	# Frozen General uses typed array membership when taking its two
	# winners. Reuse the exact authored integral rows (already compared to
	# the parsed save), rather than modifying any frozen/legacy rules.
	candidate.semifinals[int(game.fixture.id)-30]=game.fixture.duplicate()
	candidate.semifinals[int(game.other.id)-30]=game.other.duplicate()
	var shape: Dictionary={"human":held.human,"reports":[other]}
	if not GeneralPending._doubleheader_shape(shape,Live.Catalog.game_signature()): return {}
	var projected:=GeneralPending._project_candidate(candidate,shape)
	if projected.has("error") or projected.get("candidate")!=candidate or Live.scope(candidate) \
			or candidate.phase!=SeasonState.Phase.FINAL or candidate.round_index!=10 or candidate.champion!=-1 \
			or candidate.results.size()!=32 or candidate.player_results.size()!=11 or candidate.physical.cursor!=21 \
			or not candidate.physical.error.is_empty() or not GeneralPending._receipts(candidate) \
			or not Live.same(candidate.playoff_seeds,source.playoff_seeds) or not Live.same(candidate.semifinals,source.semifinals) \
			or not Live.same(candidate.results.slice(0,30),source.results) \
			or not Live.same(candidate.player_results.slice(0,10),source.player_results) \
			or not Live.same(candidate.physical.reports,source.physical.reports+[other]): return {}
	var managed: Dictionary=candidate.results[30]
	var sibling: Dictionary=candidate.results[31]
	if not Live.whole(managed.get("id"),int(game.fixture.id),int(game.fixture.id)) \
			or not Live.whole(sibling.get("id"),int(game.other.id),int(game.other.id)) \
			or not Live.same(managed,candidate.player_results[10]) \
			or managed.away_runs!=held.human[1] or managed.home_runs!=held.human[2]: return {}
	var wins: Array[int]=[SeasonState._winner(managed),SeasonState._winner(sibling)]
	if wins[0]==wins[1] or not source.playoff_seeds.has(wins[0]) or not source.playoff_seeds.has(wins[1]): return {}
	var club_won: bool=wins[0]==0
	wins.sort_custom(func(a: int,b: int):return source.playoff_seeds.find(a)<source.playoff_seeds.find(b))
	var final: Dictionary={"id":32,"round":11,"home":wins[0],"away":wins[1],"neutral":true}
	var reward: Dictionary=source.build._bank.view().rewards.duplicate(true)
	reward[str(int(managed.id))]=club_won
	if not Live.same(candidate.build._bank.view().rewards,reward) or not Live.same(candidate.final_fixture,final): return {}
	if club_won:
		if projected.has("fixture") or not candidate.physical.needed.is_empty() or candidate.physical.projecting: return {}
	elif not Live.same(projected.get("fixture"),final) or not Live.same(candidate.physical.needed,final) or not candidate.physical.projecting: return {}
	return {"fixture":final,"cash":candidate.cash(),"club_won":club_won,"candidate":candidate}

# This context is private to the authentic two-semifinal projection.
# Its type is never accepted by the live codec's persistence scope.
static func earned_club_managed_final_context(season: SeasonState) -> bool:
	if season==null or season.get_script()!=PrivateEarnedClubSemifinal or season.career==null \
			or not Live.CommonStart.detached_earned_doubleheader(season.career.start_basis) \
			or season.phase!=SeasonState.Phase.FINAL or season.round_index!=10 or season.picks.size()!=4 \
			or season.champion!=-1 or not season.playoff_seeds.has(0) or season.semifinals.size()!=2 \
			or season.results.size()!=32 or season.player_results.size()!=11 or season.physical==null \
			or not season.physical.pending.is_empty() or not season.physical.needed.is_empty() \
			or not season.physical.error.is_empty() or season.physical.cursor!=21 or season.physical.reports.size()!=21: return false
	var managed: Dictionary=season.player_results.back()
	if not Live.whole(managed.get("id"),30,31) or not Live.same(managed,season.results[30]) \
			or SeasonState._winner(managed)!=0 or not _bound_report(season.physical.reports[20],61-int(managed.id)): return false
	var reward: Dictionary=season.build._bank.view().rewards
	if reward.get(str(int(managed.id)))!=true or reward.has("32"): return false
	var final: Dictionary=season.final_fixture
	var rival: int=SeasonState._winner(season.results[31])
	if rival<1 or rival>5 or not season.playoff_seeds.has(rival): return false
	var club_home: bool=season.playoff_seeds.find(0)<season.playoff_seeds.find(rival)
	var authored: Dictionary={"id":32,"round":11,"away":rival if club_home else 0,"home":0 if club_home else rival,"neutral":true}
	return Live.same(final,authored) and Live.same(season.pending_fixture(),authored)

static func _earned_club_managed_final_opening(paid: Dictionary) -> Dictionary:
	var season: SeasonState=paid.get("candidate")
	if not paid.get("club_won",false) or not earned_club_managed_final_context(season) or season.physical.projecting \
			or not Live.same(paid.get("fixture"),season.final_fixture): return {}
	var fixture: Dictionary=season.final_fixture
	var state: MatchState=GeneralOps._source_public_make_match(season)
	var field: FieldDefinition=SeasonState.field_for_fixture(fixture,season)
	if state==null or field==null or not state.stadium.is_empty() or not state.sure_shot.enable_current_evidence(): return {}
	var hash:=Request._for_materialized_wallet(season,fixture,state,field)
	if hash.is_empty(): return {}
	var club: TeamMatchState=state.home_team if fixture.home==0 else state.away_team
	var binding: Dictionary={"format":"installed-earned-doubleheader-managed-final-v1","request":hash,
		"equipped":state.gear_usage.equipped,"held":club.tactics.held,
		"field_supply":[club.field_supply.receipt,club.field_supply.retired,club.field_supply.capacity],
		"scout":club.scouted_recipe,"copy":club.copy_source,"insurance":club.tactics.insured_receipt,
		"track_walks":club.tactics.track_walks}
	return {"state":state,"field":field,"fixture":fixture,"season":season,
		"binding":{"version":1,"request":JSON.stringify(binding,"",true,true).sha256_text()}}

static func earned_club_managed_final_held_shape(value: Variant) -> bool:

# Validate with native Final settlement on the new disposable paid fork.
# The owning operation retains only the evidence, not this paid candidate.
static func _earned_club_managed_final_held(paid: Dictionary,value: Variant) -> bool:
	if not earned_club_managed_final_held_shape(value): return false
	var opened:=_earned_club_managed_final_opening(paid)
	if opened.is_empty() or not Live.same(opened.binding,value.binding): return false
	var candidate: SeasonState=opened.season
	var before_results: Array=candidate.results.duplicate(true)
	var before_player: Array=candidate.player_results.duplicate(true)
	var before_reports: Array=candidate.physical.reports.duplicate()
	var expected_reward: Dictionary=candidate.build._bank.view().rewards.duplicate(true)
	var shape: Dictionary={"human":value.human,"reports":[]}
	if not GeneralPending._doubleheader_shape(shape,Live.Catalog.game_signature()): return false
	var projected:=GeneralPending._project_candidate(candidate,shape)
	if projected.has("error") or projected.has("fixture") or projected.get("candidate")!=candidate or Live.scope(candidate) \
			or candidate.phase!=SeasonState.Phase.COMPLETE or candidate.results.size()!=33 or candidate.player_results.size()!=12 \
			or candidate.physical.cursor!=21 or candidate.physical.projecting or not candidate.physical.needed.is_empty() \
			or not candidate.physical.error.is_empty() or not Live.same(candidate.results.slice(0,32),before_results) \
			or not Live.same(candidate.player_results.slice(0,11),before_player) or not Live.same(candidate.physical.reports,before_reports) \
			or not GeneralPending._receipts(candidate): return false
	var match_row: Dictionary=candidate.player_results.back()
	expected_reward["32"]=SeasonState._winner(match_row)==0
	return Live.whole(match_row.get("id"),32,32) and Live.same(match_row,candidate.results.back()) \
		and match_row.away_runs==value.human[1] and match_row.home_runs==value.human[2] \
		and candidate.champion==SeasonState._winner(match_row) and Live.same(candidate.build._bank.view().rewards,expected_reward)

# A CPU final consumes fixture 32 only after the native two-semifinal
# projection has proved elimination and emitted the exact final request.
static func _earned_club_cpu_final_job(paid: Dictionary) -> Dictionary:
	var candidate: SeasonState=paid.get("candidate")
	if candidate==null or candidate.get_script()!=PrivateEarnedClubSemifinal or paid.get("club_won",true) \
			or not paid.get("fixture") is Dictionary or candidate.phase!=SeasonState.Phase.FINAL \
			or not Live.same(candidate.physical.needed,paid.fixture): return {}
	var fixture: Dictionary=paid.fixture
	if _projected_earned_club_cpu_final(candidate,fixture).is_empty() or candidate.physical.reports.size()!=21: return {}
	var job:=_job(candidate,fixture)
	if not job.is_empty(): job.fixture=fixture
	return job

static func _earned_club_cpu_final_result(paid: Dictionary,row: Variant) -> Dictionary:
	if not _bound_report(row,32): return {}
	var job:=_earned_club_cpu_final_job(paid)
	if job.is_empty() or job.job.binding.request!=row.request: return {}
	var candidate: SeasonState=paid.candidate
	var fixture: Dictionary=job.fixture
	var before_results: Array=candidate.results.duplicate(true)
	var before_player: Array=candidate.player_results.duplicate(true)
	var before_reports: Array=candidate.physical.reports.duplicate()
	var before_wallet: Dictionary=candidate.build._bank.view()
	var before_build: Array=candidate.build._events.duplicate(true)
	candidate.physical.reports.append(row.duplicate(true))
	# Native _prepare_final runs only the final. It does not re-settle
	# either semifinal or grant the eliminated club another game result.
	if not candidate._prepare_final() or candidate.phase!=SeasonState.Phase.COMPLETE \
			or candidate.physical.cursor!=22 or not candidate.physical.needed.is_empty() \
			or not candidate.physical.error.is_empty() or candidate.results.size()!=33 \
			or not Live.same(candidate.results.slice(0,32),before_results) or not Live.same(candidate.player_results,before_player) \
			or not Live.same(candidate.physical.reports,before_reports+[row]) or not Live.same(candidate.final_fixture,fixture) \
			or not Live.same(candidate.build._bank.view(),before_wallet) or not Live.same(candidate.build._events,before_build) \
			or not GeneralPending._receipts(candidate): return {}
	var result: Dictionary=candidate.results.back()
	if not Live.whole(result.get("id"),32,32) or result.get("away")!=fixture.away \
			or result.get("home")!=fixture.home or result.get("neutral")!=true \
			or result.get("away_runs")==result.get("home_runs") or candidate.champion!=SeasonState._winner(result) \
			or (candidate.champion!=fixture.away and candidate.champion!=fixture.home): return {}
	candidate.physical.projecting=false
	return {"result":result,"champion":candidate.champion,"candidate":candidate}

static func _earned_club_cpu_final_close_scope(season: SeasonState) -> bool:
	if season==null or season.get_script()!=PrivateEarnedClubSemifinal or season.career==null \
			or not Live.CommonStart.detached_earned_doubleheader(season.career.start_basis) \
			or season.phase!=SeasonState.Phase.COMPLETE or season.round_index!=10 or season.picks.size()!=4 \
			or season.physical==null or season.physical.projecting or not season.physical.pending.is_empty() \
			or not season.physical.needed.is_empty() or not season.physical.error.is_empty() \
			or season.physical.cursor!=22 or season.physical.reports.size()!=22 or season.results.size()!=33 \
			or season.player_results.size()!=11 or not season.playoff_seeds.has(0) \
			or season.champion<1 or season.champion>5: return false
	var club: Dictionary=season.player_results.back()
	var final: Dictionary=season.results.back()
	return Live.whole(club.get("id"),30,31) and SeasonState._winner(club)!=0 \
		and (Live.whole(club.get("away"),0,0) or Live.whole(club.get("home"),0,0)) \
		and season.build._bank.view().rewards.get(str(int(club.id)))==false \
		and not season.build._bank.view().rewards.has("32") and Live.whole(final.get("id"),32,32) \
		and Live.whole(final.get("away"),1,5) and Live.whole(final.get("home"),1,5) and final.get("neutral")==true \
		and season.champion==SeasonState._winner(final) and _bound_report(season.physical.reports[21],32)

static func _projected_earned_club_cpu_final(season: SeasonState,fixture: Dictionary) -> Dictionary:
	if not season.physical.projecting or not season.physical.pending.is_empty() or season.round_index!=10 \

			or season.results.size()!=32 or season.player_results.size()!=11: return {}
	# Match the same request with its one expected resolver row present.
	if season.physical.reports.size()==22 and not _bound_report(season.physical.reports[21],32): return {}
	var saved: Dictionary=season.final_fixture
	if not Live.whole(saved.get("id"),32,32) or not Live.whole(saved.get("round"),11,11) \
			or not Live.whole(saved.get("home"),1,5) or not Live.whole(saved.get("away"),1,5) \
			or saved.home==saved.away or saved.get("neutral")!=true: return {}
	var canonical: Dictionary={"id":32,"round":11,"home":int(saved.home),"away":int(saved.away),"neutral":true}
	return canonical if Live.same(saved,canonical) and Live.same(fixture,canonical) else {}

# Just this non-persistable type may validate against the earned regular
# endpoint while Native General projects its human report. No materialized

static func earned_club_semifinal_projection(season: SeasonState) -> bool:

static func _prequeued_other_semifinal(season: SeasonState) -> bool:
	var human: Dictionary=season.pending_fixture()
	return Live.whole(human.get("id"),30,31) and _bound_report(season.physical.reports[20],61-int(human.id))

static func _projected_earned_club_other_semifinal(season: SeasonState,fixture: Dictionary) -> Dictionary:
	if season==null or season.get_script()!=PrivateEarnedClubSemifinal or season.phase!=SeasonState.Phase.SEMIFINAL \
			or season.round_index!=10 or season.champion!=-1 or not season.final_fixture.is_empty() \
			or season.results.size()!=31 or season.player_results.size()!=11 or season.semifinals.size()!=2 \
			or not season.physical.projecting or not season.physical.pending.is_empty() \
			or season.physical.cursor!=20 or season.physical.reports.size() not in [20,21]: return {}
	var human: Dictionary=season.pending_fixture()
	# JSON whole numbers are floats. Test the exact integer semantics
	# before building the request; Array.has distinguishes 0 from 0.0.
	if not Live.whole(human.get("id"),30,31) or not Live.whole(human.get("away"),0,5) \
			or not Live.whole(human.get("home"),0,5) \
			or not (Live.whole(human.get("away"),0,0) or Live.whole(human.get("home"),0,0)): return {}
	var id: int=int(human.id)
	# The same native resolver reconstructs its request after the one
	# matching receipt is staged. No other continuation or fixture enters.
	if season.physical.reports.size()==21 and not _bound_report(season.physical.reports[20],61-id): return {}
	if not Live.whole(season.results.back().get("id"),id,id) \
			or not Live.whole(season.player_results.back().get("id"),id,id) \
			or not season.build._bank.view().rewards.has(str(id)): return {}
	var row: Dictionary=season.semifinals[31-id]
	if not Live.whole(row.get("id"),61-id,61-id) or not Live.whole(row.get("round"),10,10) \
			or not Live.whole(row.get("away"),1,5) or not Live.whole(row.get("home"),1,5): return {}
	var clean: Dictionary={"id":int(row.id),"round":10,"away":int(row.away),"home":int(row.home)}
	return clean if Live.same(row,clean) and Live.same(fixture,clean) else {}

# Only frozen human result ingestion on the detached authenticated paid fork.
static func semifinal_projection(season: SeasonState) -> bool:
	return season!=null and season.get_script()==PrivateSemifinalFork and season.physical!=null and season.physical.projecting and season.physical.pending.is_empty()

static func _semifinal_fixtures(season: SeasonState) -> Dictionary:
	# Soft structural geometry is checked first. Only exact frozen standings
	# issue the integer request row; hydrated float representations are aliases.
	if not Live._results_valid(season) or season.teams.size()!=6: return {}
	for team: Dictionary in season.teams:
		if typeof(team.get("draw")) not in [TYPE_FLOAT,TYPE_INT] or not is_finite(float(team.draw)): return {}
	if not _unplayed_bracket(season): return {}
	var seeds: Array=season.playoff_seeds
	var human: Dictionary={"id":30,"round":10,"home":int(seeds[0]),"away":int(seeds[3])}
	var other: Dictionary={"id":31,"round":10,"home":int(seeds[1]),"away":int(seeds[2])}
	return {"human":human,"other":other} if (human.home==0 or human.away==0) else {}

static func _projected_semifinal_npc(season: SeasonState, fixture: Dictionary) -> Dictionary:
	if not semifinal_projection(season) or season.phase!=SeasonState.Phase.SEMIFINAL or season.round_index!=10 or season.semifinals.size()!=2 or season.physical.cursor!=20 or season.physical.reports.size() not in [20,21,22]: return {}
	if season.results.size()!=31 or season.player_results.size()!=11 or season.results[-1].get("id")!=30 or season.player_results[-1].get("id")!=30 or not season.build._bank.view().rewards.has("30"): return {}
	var saved: Dictionary=season.semifinals[1]
	if not Live.whole(saved.get("id"),31,31) or not Live.whole(saved.get("round"),10,10) or not Live.whole(saved.get("away"),1,5) or not Live.whole(saved.get("home"),1,5): return {}
	var other: Dictionary={"id":31,"round":10,"home":int(saved.home),"away":int(saved.away)}
	return other if Live.same(saved,other) and Live.same(fixture,other) else {}

static func _unpaid_human_eliminated(human: Array, fixture: Dictionary) -> bool:

static func _projected_final_npc(season: SeasonState, fixture: Dictionary) -> Dictionary:
	if not semifinal_projection(season) or season.phase!=SeasonState.Phase.FINAL or season.round_index!=10 or season.champion!=-1 or season.physical.cursor!=21 or season.physical.reports.size() not in [21,22]: return {}
	if season.results.size()!=32 or season.player_results.size()!=11 or season.results[30].get("id")!=30 or season.results[31].get("id")!=31 or season.build._bank.view().rewards.get("30")!=false: return {}
	var saved: Dictionary=season.final_fixture
	if not Live.whole(saved.get("id"),32,32) or not Live.whole(saved.get("round"),11,11) or not Live.whole(saved.get("away"),1,5) or not Live.whole(saved.get("home"),1,5) or saved.away==saved.home or saved.get("neutral")!=true: return {}
	var final: Dictionary={"id":32,"round":11,"home":int(saved.home),"away":int(saved.away),"neutral":true}
	return final if Live.same(saved,final) and Live.same(fixture,final) else {}

# Both played semifinals follow frozen native settlement on the detached
# fork. With no final row queued that code can only request the fixture.
static func _inspect_npc_final(source: SeasonState, candidate: SeasonState, pending: Dictionary, preview_job: bool) -> Dictionary:
	var semi: Array=[pending.reports[0]]
	var shape: Dictionary={"human":pending.human,"reports":semi}
	if not GeneralPending._doubleheader_shape(shape,Live.Catalog.game_signature()): return {}
	var projected: Dictionary=GeneralPending._project_candidate(candidate,shape)
	var final: Dictionary=_projected_final_npc(candidate,candidate.final_fixture)
	if projected.has("error") or projected.get("candidate")!=candidate or final.is_empty() or not Live.same(projected.get("fixture"),final) or not Live.same(candidate.physical.needed,final): return {}
	if not candidate.physical.error.is_empty() or not Live.same(candidate.results.slice(0,30),source.results) or not Live.same(candidate.player_results.slice(0,10),source.player_results) or not Live.same(candidate.physical.reports,source.physical.reports+semi): return {}
	var human: Dictionary=candidate.player_results[-1]
	if not Live.same(candidate.results[30],human) or human.away_runs!=pending.human[1] or human.home_runs!=pending.human[2]: return {}
	var expected: Dictionary=source.build._bank.view().rewards.duplicate(true)
	expected["30"]=false
	if not Live.same(candidate.build._bank.view().rewards,expected) or not GeneralPending._receipts(candidate): return {}
	# Independently corroborate the native final order; no submitted report
	# names, scores or venue can author this geometry.
	var winners: Array[int]=[SeasonState._winner(candidate.results[30]),SeasonState._winner(candidate.results[31])]
	winners.sort_custom(func(a: int,b: int):return source.playoff_seeds.find(a)<source.playoff_seeds.find(b))
	if not Live.same(final,{"id":32,"round":11,"home":winners[0],"away":winners[1],"neutral":true}): return {}
	if preview_job: return _job(candidate,final)
	candidate.physical.reports.append(pending.reports[1].duplicate(true))
	var result: Dictionary=candidate.physical.resolve(candidate,final)
	if result.is_empty() or result.get("id")!=32 or candidate.physical.cursor!=22 or not candidate.physical.needed.is_empty() or not candidate.physical.error.is_empty(): return {}
	if candidate.phase!=SeasonState.Phase.FINAL or candidate.results.size()!=32 or candidate.champion!=-1: return {}
	return {"accepted":true} # Keep every playoff result and payout unapplied.

# The owning Store's one terminal operation takes both native reports.
# Frozen General performs both semifinal and final effects in one fork.
static func _inspect_postseason_settlement(source: SeasonState, candidate: SeasonState, pending: Dictionary) -> Dictionary:
	var shape: Dictionary={"human":pending.human,"reports":pending.reports}
	if not GeneralPending._doubleheader_shape(shape,Live.Catalog.game_signature()): return {}
	var projected: Dictionary=GeneralPending._project_candidate(candidate,shape)
	if projected.has("error") or projected.has("fixture") or projected.get("candidate")!=candidate or Live.scope(candidate): return {}
	if candidate.phase!=SeasonState.Phase.COMPLETE or candidate.round_index!=10 or candidate.results.size()!=33 or candidate.player_results.size()!=11 or candidate.physical.projecting or not candidate.physical.needed.is_empty() or not candidate.physical.error.is_empty() or candidate.physical.cursor!=22: return {}
	if not Live.same(candidate.results.slice(0,30),source.results) or not Live.same(candidate.player_results.slice(0,10),source.player_results) or not Live.same(candidate.physical.reports,source.physical.reports+pending.reports): return {}
	if _complete_geometry(candidate).is_empty() or not GeneralPending._receipts(candidate): return {}
	var human: Dictionary=candidate.player_results[-1]
	var expected: Dictionary=source.build._bank.view().rewards.duplicate(true)
	expected["30"]=false
	if not Live.same(human,candidate.results[30]) or human.away_runs!=pending.human[1] or human.home_runs!=pending.human[2] or not Live.same(candidate.build._bank.view().rewards,expected): return {}
	return _concrete_settlement(candidate)

# Author the exact finalized geometry from regular standings and actual
# semifinal winners. Called only on the sealed eliminated-owner format.
static func _complete_geometry(season: SeasonState) -> Dictionary:
	if season.phase!=SeasonState.Phase.COMPLETE or season.round_index!=10 or season.results.size()!=33 or season.teams.size()!=6 or not Live._results_valid(season): return {}
	for team: Dictionary in season.teams:
		if typeof(team.get("draw")) not in [TYPE_FLOAT,TYPE_INT] or not is_finite(float(team.draw)): return {}
	var seeds: Array=[]
	for row: Dictionary in season.standings().slice(0,4): seeds.append(int(row.team))
	var earned_cpu: bool=season.career!=null and Live.CommonStart.detached_earned_doubleheader(season.career.start_basis) and 0 not in seeds
	if seeds.size()!=4 or not Live.same(season.playoff_seeds,seeds) or (not earned_cpu and 0 not in [seeds[0],seeds[3]]): return {}
	var semis: Array=[]
	for pair: int in range(2): semis.append({"id":30+pair,"round":10,"home":seeds[pair],"away":seeds[3-pair]})
	if not Live.same(season.semifinals,semis) or season.results[30].get("id")!=30 or season.results[31].get("id")!=31 or season.results[32].get("id")!=32: return {}
	var winners: Array[int]=[SeasonState._winner(season.results[30]),SeasonState._winner(season.results[31])]
	if winners.has(0): return {}
	winners.sort_custom(func(a: int,b: int):return seeds.find(a)<seeds.find(b))
	var final: Dictionary={"id":32,"round":11,"home":winners[0],"away":winners[1],"neutral":true}
	if not Live.same(final,season.final_fixture) or season.champion!=SeasonState._winner(season.results[32]): return {}
	return {"semis":semis,"final":final}

static func _inspect_semifinal(source: SeasonState, pending: Dictionary, fixture: Dictionary, preview_job: bool, produce_settlement: bool) -> Dictionary:
	if fixture.id!=30 or (preview_job and pending.reports.size()==2) or Live._capture_structural(source).is_empty() or not Live._paid_history_valid(source): return {}
	var bracket: Dictionary=_semifinal_fixtures(source)
	if bracket.is_empty() or not Live.same(bracket.human,fixture): return {}
	var candidate: SeasonState=_detached(source,PrivateSemifinalFork)
	if candidate==null or Live.scope(candidate) or not Live.capture(candidate).is_empty(): return {}
	detached_calls+=1
	projection_calls+=1
	if produce_settlement:
		return _inspect_postseason_settlement(source,candidate,pending) if pending.reports.size()==2 and _unpaid_human_eliminated(pending.human,fixture) else {}
	if pending.reports.size()==2 or (preview_job and pending.reports.size()==1):
		return _inspect_npc_final(source,candidate,pending,preview_job) if _unpaid_human_eliminated(pending.human,fixture) else {}
	var shape: Dictionary={"human":pending.human,"reports":[]}
	if not GeneralPending._doubleheader_shape(shape,Live.Catalog.game_signature()): return {}
	var projected: Dictionary=GeneralPending._project_candidate(candidate,shape)
	if projected.has("error") or projected.get("candidate")!=candidate or not Live.same(projected.get("fixture"),bracket.other) or not Live.same(candidate.physical.needed,bracket.other): return {}
	if candidate.phase!=SeasonState.Phase.SEMIFINAL or candidate.round_index!=10 or not candidate.final_fixture.is_empty() or candidate.champion!=-1 or not candidate.physical.projecting or not candidate.physical.error.is_empty() or not candidate.physical.pending.is_empty(): return {}
	if candidate.results.size()!=31 or candidate.player_results.size()!=11 or not Live.same(candidate.results.slice(0,30),source.results) or not Live.same(candidate.player_results.slice(0,10),source.player_results): return {}
	if candidate.physical.cursor!=20 or not Live.same(candidate.physical.reports,source.physical.reports): return {}
	var human: Dictionary=candidate.player_results[-1]
	if human.get("id")!=30 or not Live.same(human,candidate.results[-1]) or human.away_runs!=pending.human[1] or human.home_runs!=pending.human[2]: return {}
	var expected: Dictionary=source.build._bank.view().rewards.duplicate(true)
	expected["30"]=(human.home_runs>human.away_runs)==(fixture.home==0)
	if not Live.same(candidate.build._bank.view().rewards,expected): return {}
	for club: String in source.opponents.clubs:
		if not Live.same(candidate.opponents.clubs[club].build._bank.view().rewards,source.opponents.clubs[club].build._bank.view().rewards): return {}
	if preview_job: return _job(candidate,bracket.other)
	if not pending.reports.is_empty():
		# Frozen General stopped at the missing #31 before any semifinal
		# settlement. Authenticate only that physical report on this fork;
		# never call the terminal continuation (it could auto-play #32).
		candidate.physical.reports.append(pending.reports[0].duplicate(true))
		var other: Dictionary=candidate.physical.resolve(candidate,bracket.other)
		if other.is_empty() or other.get("id")!=31 or candidate.physical.cursor!=21 or not candidate.physical.needed.is_empty() or not candidate.physical.error.is_empty(): return {}
		if candidate.phase!=SeasonState.Phase.SEMIFINAL or candidate.results.size()!=31 or not candidate.final_fixture.is_empty() or candidate.champion!=-1: return {}
	return {"accepted":true} # Neither the projected reward nor bracket escapes.


static func followup_projection(season: SeasonState) -> bool:
	return season!=null and season.get_script()==PrivateFollowupFork and season.round_index>=2 and human_enabled(season.round_index) and season.physical!=null and season.physical.projecting and season.physical.pending.is_empty()

# Called only after the settled prefix authenticates. Without a guaranteed
# top-four seat, frozen General may immediately simulate both semifinals.
static func _clinched_regular_end(source: SeasonState, fixtures: Dictionary) -> bool:
	var remaining: Dictionary={}
	for row: Dictionary in [fixtures.human]+fixtures.npc:
		for club: int in [row.away,row.home]:
			remaining[club]=int(remaining.get(club,0))+1
	if remaining.size()!=6 or remaining.values().any(func(games):return games!=1): return false
	var standings: Array[Dictionary]=source.standings()
	var human_wins: int=-1
	for row: Dictionary in standings:
		if row.team==0: human_wins=int(row.wins)
	if human_wins<0: return false
	var possible_rivals: int=0
	for row: Dictionary in standings:
		if row.team!=0 and int(row.wins)+int(remaining.get(row.team,0))>=human_wins: possible_rivals+=1
	return possible_rivals<4

static func _unplayed_semifinals(candidate: SeasonState) -> bool:
	return candidate.physical.pending.is_empty() and _unplayed_bracket(candidate)

static func _unplayed_bracket(candidate: SeasonState) -> bool:
	if candidate.phase!=SeasonState.Phase.SEMIFINAL or candidate.results.size()!=30 or candidate.player_results.size()!=10 or candidate.physical.cursor!=20 or not candidate.final_fixture.is_empty() or candidate.champion!=-1: return false
	var seeds: Array=[]
	for row: Dictionary in candidate.standings().slice(0,4): seeds.append(int(row.team))
	var earned: bool=candidate.career!=null and Live.CommonStart.detached_earned_doubleheader(candidate.career.start_basis)
	if seeds.size()!=4 or (not earned and not seeds.has(0)) or not Live.same(candidate.playoff_seeds,seeds): return false
	var games: Array=[]
	for pair: int in range(2):
		games.append({"id":30+pair,"round":10,"home":seeds[pair],"away":seeds[3-pair]})
	return Live.same(candidate.semifinals,games)

# The original resolver authenticates exact ordered NPC data. Fixed owned
# producers may materialize a paid owner only after exact terminal replay.
static func _followup_input(source: SeasonState, pending: Dictionary, preview_job: bool, produce_settlement: bool) -> bool:
	if source.round_index<2 or not human_enabled(source.round_index): return false
	if produce_settlement and not settlement_enabled(source.round_index): return false
	var limit: int=report_limit(source.round_index)
	if pending.reports.size()>limit or (preview_job and pending.reports.size()>=limit): return false
	return not Live._capture_structural(source).is_empty()

static func _inspect_followup(source: SeasonState, pending: Dictionary, fixture: Dictionary, preview_job: bool, produce_settlement: bool) -> Dictionary:
	if not _followup_input(source,pending,preview_job,produce_settlement) or not Live._paid_history_valid(source): return {}
	return _inspect_followup_physical(source,pending,fixture,preview_job,produce_settlement)

# Inspect the physical suffix; the fixed entry paths own the history check.
static func _inspect_followup_physical(source: SeasonState, pending: Dictionary, fixture: Dictionary, preview_job: bool, produce_settlement: bool) -> Dictionary:
	var fixtures: Dictionary=_round_fixtures(source.round_index)
	var complete: bool=pending.reports.size()==fixtures.npc.size()
	var regular_end: bool=complete and source.round_index==9
	var earned: bool=Live.CommonStart.detached_earned_doubleheader(source.career.start_basis)
	if regular_end and not earned and not _clinched_regular_end(source,fixtures): return {}
	var candidate: SeasonState=_detached(source,PrivateFollowupFork)
	if candidate==null or Live.scope(candidate) or not Live.capture(candidate).is_empty(): return {}
	detached_calls+=1
	projection_calls+=1
	var reports: Array=pending.reports
	var shape: Dictionary={"human":pending.human,"reports":reports}
	if not GeneralPending._doubleheader_shape(shape,Live.Catalog.game_signature()): return {}
	var projected: Dictionary=GeneralPending._project_candidate(candidate,shape)
	if projected.has("error") or projected.get("candidate")!=candidate or Live.scope(candidate): return {}
	var prior: int=source.results.size()
	if candidate.results.size()!=prior+1+reports.size() or candidate.player_results.size()!=source.player_results.size()+1: return {}
	if not Live.same(candidate.results.slice(0,prior),source.results) or not Live.same(candidate.player_results.slice(0,-1),source.player_results): return {}
	var result: Dictionary=candidate.player_results[-1]
	if result.get("id")!=fixture.id or not Live.same(candidate.results[prior],result): return {}
	for index: int in range(reports.size()):
		if candidate.results[prior+index+1].get("id")!=fixtures.npc[index].id: return {}
	if candidate.physical.cursor!=source.physical.cursor+reports.size() or not Live.same(candidate.physical.reports,source.physical.reports+reports): return {}
	var expected: Dictionary=source.build._bank.view().rewards.duplicate(true)
	expected[str(fixture.id)]=(pending.human[2]>pending.human[1])==(fixture.home==0)
	if not Live.same(candidate.build._bank.view().rewards,expected): return {}
	if complete:
		if preview_job or projected.has("fixture") or candidate.round_index!=source.round_index+1 or candidate.physical.projecting or not candidate.physical.needed.is_empty() or not candidate.physical.error.is_empty() or not GeneralPending._receipts(candidate): return {}
		if regular_end:
			if not _unplayed_semifinals(candidate): return {}
			return _concrete_settlement(candidate) if produce_settlement else {"accepted":true}
		if candidate.phase!=SeasonState.Phase.REGULAR: return {}
		return _concrete_settlement(candidate) if produce_settlement else {"accepted":true}
	if produce_settlement: return {}
	var needed: Dictionary=fixtures.npc[reports.size()]
	if candidate.round_index!=source.round_index or not Live.same(projected.get("fixture"),needed): return {}
	for club: String in source.opponents.clubs:
		if not Live.same(candidate.opponents.clubs[club].build._bank.view().rewards,source.opponents.clubs[club].build._bank.view().rewards): return {}
	return _job(candidate,needed) if preview_job else {"accepted":true}

static func return_projection(season: SeasonState) -> bool:
	if season==null or season.get_script()!=PrivateReturnFork or season.physical==null or not season.physical.projecting or not season.physical.pending.is_empty() or season.physical.cursor!=2 or season.physical.reports.size() not in [2,3,4]: return false
	for index: int in range(2,season.physical.reports.size()):
		var report: Dictionary=season.physical.reports[index]
		if not Live.keys(report,["fixture","request","report"]) or not Live.whole(report.fixture,index+2,index+2): return false
	return true

# Authenticated first-leg prefix plus ordered return reports. Only the fixed
# settlement operation requests concrete construction after complete replay.
static func _inspect_return(source: SeasonState, human: Array, reports: Array, preview_job: bool, produce_settlement: bool) -> Dictionary:
	if not Live.scope(source) or source.round_index!=1 or not Live.return_session_ready(source) \
		or not source.physical.pending.is_empty() or reports.size()>2: return {}
	var shape: Dictionary={"human":human.duplicate(true),"reports":reports.duplicate(true)}
	if not GeneralPending._doubleheader_shape(shape,Live.Catalog.game_signature()) or human[0]!=3: return {}
	if Live._capture_structural(source).is_empty() or not Live._first_human_score(source) or not PairedHistory.validate_settled(source): return {}
	var candidate: SeasonState=_detached(source,PrivateReturnFork)
	if candidate==null or Live.scope(candidate) or not Live.capture(candidate).is_empty(): return {}
	detached_calls+=1
	projection_calls+=1
	var projected: Dictionary=GeneralPending._project_candidate(candidate,shape)
	if projected.has("error") or projected.get("candidate")!=candidate or Live.scope(candidate): return {}
	if candidate.results.size()!=4+reports.size() or candidate.player_results.size()!=2: return {}
	for index: int in range(candidate.results.size()):
		if candidate.results[index].get("id")!=index: return {}
	if not Live.same(candidate.results.slice(0,3),source.results) or not Live.same(candidate.player_results[0],source.player_results[0]) or candidate.player_results[1].get("id")!=3: return {}
	if candidate.physical.cursor!=2+reports.size() or not Live.same(candidate.physical.reports,source.physical.reports+reports): return {}
	var expected: Dictionary=source.build._bank.view().rewards.duplicate(true)
	var result: Dictionary=candidate.player_results[1]
	expected["3"]=(human[2]>human[1])==(result.home==0)
	if not Live.same(candidate.build._bank.view().rewards,expected): return {}
	if reports.size()==2:
		if projected.has("fixture") or candidate.round_index!=2 or candidate.phase!=SeasonState.Phase.REGULAR or candidate.physical.projecting or not candidate.physical.needed.is_empty() or not candidate.physical.error.is_empty() or not GeneralPending._receipts(candidate): return {}
		return _concrete_settlement(candidate) if produce_settlement else {"accepted":true}
	if produce_settlement: return {}
	var needed: Dictionary={}
	for row: Dictionary in SeasonDoubleheaderRules.schedule():
		if row.id==4+reports.size(): needed=row
	if candidate.round_index!=1 or not Live.same(projected.get("fixture"),needed): return {}
	for club: String in source.opponents.clubs:
		if not Live.same(candidate.opponents.clubs[club].build._bank.view().rewards,source.opponents.clubs[club].build._bank.view().rewards): return {}
	return _job(candidate,needed) if preview_job else {"accepted":true}

static func _detached(source: SeasonState, type: Script) -> SeasonState:
	var old: SeasonState=source.fork_live()
	if old==null: return null
	var candidate: SeasonState=type.new()
	if not Live.install_fields(candidate,Live.FIELDS,Live.data_fields(old,Live.FIELDS)): return null
	for key: String in Live.OBJECTS:
		candidate.set(key,old.get(key))
	if candidate.build==source.build or candidate.ownership==source.ownership or candidate.physical==source.physical or candidate.career==source.career or candidate.opponents==source.opponents: return null
	for club: String in source.opponents.clubs:
		if candidate.opponents.clubs[club].build==source.opponents.clubs[club].build: return null
	return candidate

static func verify(source: SeasonState, human: Array) -> bool:
	return not _inspect(source,human,[]).is_empty()

static func _inspect(source: SeasonState, human: Array, reports: Array, produce_settlement: bool = false) -> Dictionary:
	if reports.size()>2: return {}
	if not Live.scope(source) or not Live.same(source.league_rules,LeagueRules.capture("doubleheader")) or source.phase!=SeasonState.Phase.REGULAR or source.round_index!=0 or source.picks.size()!=4 or not source.physical.pending.is_empty(): return {}
	var shape: Dictionary={"human":human.duplicate(true),"reports":reports.duplicate(true)}
	if not GeneralPending._doubleheader_shape(shape,Live.Catalog.game_signature()): return {}
	var fixture: Dictionary=source.pending_fixture()
	if fixture.is_empty() or int(fixture.round)!=0 or fixture.id!=human[0] or SeasonDoubleheaderRules.first(fixture.id)!=-1: return {}
	var needed: Dictionary={}
	for row: Dictionary in source.schedule:
		if row.round==0 and row.id==1+reports.size():
			needed=row.duplicate(true)
			break
	if reports.size()<2 and (needed.is_empty() or needed.away==0 or needed.home==0): return {}
	# The opening is pristine here. Earned starts reconstruct from their
	# exact authored basis and picks; old Genesis starts use original replay.
	var earned: bool=Live.CommonStart.detached_earned_doubleheader(source.career.start_basis)
	if Live._capture_structural(source).is_empty() or not (Live._history_valid(source) if earned else PairedHistory.validate_settled(source)): return {}
	var candidate: SeasonState=_detached(source,PrivateFirstFork)
	if candidate==null or not candidate.league_context_consistent() or Live.scope(candidate) or not Live.capture(candidate).is_empty(): return {}
	detached_calls+=1
	projection_calls+=1
	var projected: Dictionary=GeneralPending._project_candidate(candidate,shape)
	if projected.has("error") or projected.get("candidate")!=candidate or Live.scope(candidate): return {}
	if candidate.player_results.size()!=1 or candidate.physical.cursor!=reports.size() or not Live.same(candidate.physical.reports,reports): return {}
	for index: int in range(candidate.results.size()):
		if candidate.results[index].get("id")!=index: return {}
	if candidate.results.size()!=1+reports.size(): return {}
	var expected: Dictionary={}
	expected[str(int(fixture.id))]=(human[2]>human[1])==(fixture.home==0)
	if not Live.same(candidate.build._bank.view().rewards,expected): return {}
	if reports.size()==2:
		# Frozen Pending validates each NPC's exact game receipts on terminal
		# success. The nonpersistable projection never leaves this method.
		if projected.has("fixture") or candidate.round_index!=1 or candidate.phase!=SeasonState.Phase.REGULAR or candidate.physical.projecting or not candidate.physical.needed.is_empty() or not candidate.physical.error.is_empty() or not GeneralPending._receipts(candidate): return {}
		return _concrete_settlement(candidate) if produce_settlement else {"accepted":true}
	if candidate.round_index!=0 or not Live.same(projected.get("fixture"),needed): return {}
	for club: Dictionary in candidate.opponents.clubs.values():
		if not club.build._bank.view().rewards.is_empty(): return {}
	return _job(candidate,needed)

static func _concrete_settlement(candidate: SeasonState) -> Dictionary:
	var concrete: SeasonState=Live.Runtime.new()
	if not Live.install_fields(concrete,Live.FIELDS,Live.data_fields(candidate,Live.FIELDS)): return {}
	for key: String in Live.OBJECTS:
		concrete.set(key,candidate.get(key)) # Objects belong only to the detached projection.
	if not concrete.career.sync(concrete) or not Live.scope(concrete): return {}
	return {"settlement":concrete}

static func _job(candidate: SeasonState, needed: Dictionary) -> Dictionary:
	var state: MatchState=Physical.match_for(candidate,needed)
	var field: FieldDefinition=SeasonState.field_for_fixture(needed,candidate)
	var request: String=Physical.request_key(candidate,needed)
	if state==null or field==null or request.is_empty(): return {}
	return {"accepted":true,"job":{"state":state,"field":field,"seed":Physical.seed_for(candidate,needed),"binding":{"fixture":int(needed.id),"request":request}}}
