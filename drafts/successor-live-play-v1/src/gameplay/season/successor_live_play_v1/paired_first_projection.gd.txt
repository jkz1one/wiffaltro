## Pair boundary verification and detached concrete settlement construction.
const Live = preload("codec.gd")
const GeneralPending = preload("res://src/gameplay/season/successor_league_general_v2/pending.gd")
const GeneralOps = preload("res://src/gameplay/season/successor_league_general_v2/runtime_operations.gd")
const Physical = preload("physical_fixtures.gd")
const Request = preload("res://src/gameplay/season/successor_request_composition_v1/request_key.gd")
const PairedHistory = preload("res://src/gameplay/season/successor_settled_workload_v1/validator.gd")

# Geometry is reusable; only these exact private phases are qualified.
const HUMAN_PHASES: Array[int]=[0,1,2,3,4]
const REPORT_PHASES: Array[int]=[0,1,2,3,4]
const SETTLEMENT_PHASES: Array[int]=[0,1,2,3]

static func _round_fixtures(round_index: int) -> Dictionary:
	return Request.authored_phase(round_index) if round_index in HUMAN_PHASES else {}

static func bound_shape(value: Variant) -> bool:
	if not Live.keys(value,["human","reports","binding"]) or not value.human is Array or value.human.size()!=14 or not value.reports is Array or value.reports.size()>2: return false
	var fixtures: Dictionary={}
	for round_index: int in HUMAN_PHASES:
		var authored: Dictionary=_round_fixtures(round_index)
		if not authored.is_empty() and Live.whole(value.human[0],authored.human.id,authored.human.id):
			if round_index not in REPORT_PHASES and not value.reports.is_empty(): return false
			fixtures=authored
			break
	if fixtures.is_empty(): return false
	for index: int in range(value.reports.size()):
		var row: Variant=value.reports[index]
		var id: int=fixtures.npc[index].id
		if not Live.keys(row,["fixture","request","report"]) or not Live.whole(row.fixture,id,id) or not row.request is String or row.request.length()!=64 or not row.request.is_valid_hex_number(false) or not row.report is Dictionary: return false
	var binding: Variant=value.binding
	return Live.keys(binding,["version","request"]) and Live.whole(binding.version,1,1) and binding.request is String and binding.request.length()==64 and binding.request.is_valid_hex_number(false)

static func start(source: SeasonState) -> Dictionary:
	if not Live.scope(source) or not Live.same(source.league_rules,LeagueRules.capture("doubleheader")) or source.phase!=SeasonState.Phase.REGULAR or source.round_index not in HUMAN_PHASES or source.picks.size()!=4 or not source.physical.pending.is_empty(): return {}
	if source.round_index==1 and source.build.revision()!=5: return {}
	# Saved JSON hydrates whole IDs as floats. Bind to the semantically equal
	# frozen-authored row so warm and cold hash exactly the same primitives.
	var authored: Dictionary=_round_fixtures(source.round_index)
	if authored.is_empty(): return {}
	var fixture: Dictionary=authored.human
	if not Live.same(fixture,source.pending_fixture()): return {}
	var field: FieldDefinition=SeasonState.field_for_fixture(fixture,source)
	var state: MatchState=GeneralOps._source_public_make_match(source)
	if field==null or state==null or not state.stadium.is_empty() or not state.sure_shot.enable_current_evidence(): return {}
	var hash: String=Request.for_live_regular(source,fixture,state,field)
	if hash.is_empty(): return {}
	var club: TeamMatchState=state.home_team if fixture.home==0 else state.away_team
	var format: String=["private-first-human-start-v1","private-return-human-start-v1","private-next-first-human-start-v1","private-next-return-human-start-v1","private-pair3-first-human-start-v1"][source.round_index]
	var live: Dictionary={"format":format,"request":hash,"equipped":state.gear_usage.equipped,"held":club.tactics.held,"field_supply":[club.field_supply.receipt,club.field_supply.retired,club.field_supply.capacity],"scout":club.scouted_recipe,"copy":club.copy_source,"insurance":club.tactics.insured_receipt,"track_walks":club.tactics.track_walks}
	return {"state":state,"field":field,"fixture":fixture.duplicate(true),"binding":{"version":1,"request":JSON.stringify(live,"",true,true).sha256_text()}}

static func verify_bound(source: SeasonState) -> bool:
	return not _inspect_bound(source).is_empty()

# Read only the exact next slot. Job and Store admission independently replay it.
static func next_npc_fixture(source: SeasonState) -> Dictionary:
	if not Live.scope(source) or source.round_index not in REPORT_PHASES or not bound_shape(source.physical.pending): return {}
	var authored: Dictionary=_round_fixtures(source.round_index)
	var index: int=source.physical.pending.reports.size()
	if authored.is_empty() or index>=authored.npc.size() or source.physical.pending.human[0]!=authored.human.id: return {}
	return authored.npc[index]

# Only a detached match start is returned, never the projected season/payout.
static func npc_job(source: SeasonState, fixture_id: int) -> Dictionary:
	var next: Dictionary=next_npc_fixture(source)
	if next.is_empty() or fixture_id!=next.id: return {}
	return _inspect_bound(source,false,true).get("job",{})

# There is no reusable projection token or caller-supplied candidate.
static func materialize_settlement(source: SeasonState, leg: int=0) -> SeasonState:
	if leg not in SETTLEMENT_PHASES: return null
	var authored: Dictionary=_round_fixtures(leg)
	if authored.is_empty() or not Live.scope(source) or source.round_index!=leg or not bound_shape(source.physical.pending) or source.physical.pending.reports.size()!=authored.npc.size(): return null
	return _inspect_bound(source,true).get("settlement")

static func _inspect_bound(source: SeasonState, produce_settlement: bool = false, preview_job: bool = false) -> Dictionary:
	if not Live.scope(source) or not bound_shape(source.physical.pending): return {}
	var pending: Dictionary=source.physical.pending.duplicate(true)
	var pristine: SeasonState=source.fork_live()
	if pristine==null: return {}
	pristine.physical.pending.clear()
	var opening: Dictionary=start(pristine)
	if opening.is_empty() or not Live.same(opening.binding,pending.binding) or opening.fixture.id!=pending.human[0]: return {}
	if source.round_index in [2,3,4]:
		return _inspect_followup(pristine,pending,opening.fixture,preview_job,produce_settlement)
	if source.round_index==1:
		return _inspect_return(pristine,pending.human,pending.reports,preview_job,produce_settlement)
	return _inspect(pristine,pending.human,pending.reports,produce_settlement)

# Frozen Pending queues its already shape-checked archive before recording
# the human result. Only this exact private projection can carry that queue.
static func npc_queue(season: SeasonState) -> bool:
	if season==null or season.get_script()!=PrivateFirstFork or season.physical==null or not season.physical.projecting or season.physical.cursor!=0: return false
	if season.physical.reports.size() not in [1,2]: return false
	for index: int in range(season.physical.reports.size()):
		var row: Dictionary=season.physical.reports[index]
		if not Live.keys(row,["fixture","request","report"]) or not Live.whole(row.fixture,index+1,index+1): return false
	return true

# Exact non-persistable type and state after the original human-only projection.
# The mutable projecting field alone never admits any materialized owner.
static func projected_npc_fixture(season: SeasonState, fixture: Dictionary) -> Dictionary:
	if season==null or not Live.same(season.league_rules,LeagueRules.capture("doubleheader")) or season.phase!=SeasonState.Phase.REGULAR or season.picks.size()!=4: return {}
	var leg: int=season.round_index
	if not ((leg==0 and season.get_script()==PrivateFirstFork) or (leg==1 and season.get_script()==PrivateReturnFork) or (leg>=2 and leg in REPORT_PHASES and season.get_script()==PrivateFollowupFork)): return {}
	var fixtures: Dictionary=_round_fixtures(leg)
	var human: int=fixtures.human.id
	var archive: int=fixtures.prior_npcs.size()
	if season.physical==null or not season.physical.projecting or not season.physical.pending.is_empty() or season.player_results.size()!=fixtures.prior_humans.size()+1 or not season.build._bank.view().rewards.has(str(human)): return {}
	var progress: int=season.physical.cursor-archive
	if progress<0 or progress>=fixtures.npc.size() or season.physical.reports.size()<archive+progress: return {}
	var played: Array=fixtures.prior_results+[human]
	for index: int in range(progress): played.append(fixtures.npc[index].id)
	if not Live.same(season.results.map(func(row):return row.get("id")),played): return {}
	var next: Dictionary=fixtures.npc[progress]
	return next if Live.same(next,fixture) else {}

# Diagnostic counters only; never admit a source or release a candidate.
static var detached_calls: int = 0
static var projection_calls: int = 0

class PrivateFirstFork extends "res://src/gameplay/season/successor_live_play_v1/runtime.gd":
	func league_context_consistent() -> bool:
		return load("res://src/gameplay/season/successor_live_play_v1/codec.gd")._runtime_context(self)

	func record_doubleheader_result(human: Array) -> bool:
		return load("res://src/gameplay/season/successor_league_general_v2/runtime_operations.gd")._source_public_record_doubleheader_result(self,human)

class PrivateReturnFork extends PrivateFirstFork:
	pass

class PrivateFollowupFork extends PrivateFirstFork:
	pass

static func followup_projection(season: SeasonState) -> bool:
	return season!=null and season.get_script()==PrivateFollowupFork and season.round_index in [2,3,4] and season.physical!=null and season.physical.projecting and season.physical.pending.is_empty()

# The original resolver authenticates exact ordered NPC data. Fixed owned
# producers may materialize a paid owner only after exact terminal replay.
static func _inspect_followup(source: SeasonState, pending: Dictionary, fixture: Dictionary, preview_job: bool, produce_settlement: bool) -> Dictionary:
	if source.round_index not in [2,3,4]: return {}
	if produce_settlement and source.round_index not in SETTLEMENT_PHASES: return {}
	if (preview_job or not pending.reports.is_empty()) and source.round_index not in REPORT_PHASES: return {}
	if Live._capture_structural(source).is_empty(): return {}
	for paid: Dictionary in source.player_results:
		if not Live._human_score(source,paid): return {}
	if not PairedHistory.validate_settled(source): return {}
	var candidate: SeasonState=_detached(source,PrivateFollowupFork)
	if candidate==null or Live.scope(candidate) or not Live.capture(candidate).is_empty(): return {}
	detached_calls+=1
	projection_calls+=1
	var reports: Array=pending.reports
	var shape: Dictionary={"human":pending.human,"reports":reports}
	if not GeneralPending._doubleheader_shape(shape,Live.Catalog.game_signature()): return {}
	var projected: Dictionary=GeneralPending._project_candidate(candidate,shape)
	if projected.has("error") or projected.get("candidate")!=candidate or Live.scope(candidate): return {}
	var prior: int=source.results.size()
	if candidate.results.size()!=prior+1+reports.size() or candidate.player_results.size()!=source.player_results.size()+1: return {}
	if not Live.same(candidate.results.slice(0,prior),source.results) or not Live.same(candidate.player_results.slice(0,-1),source.player_results): return {}
	var result: Dictionary=candidate.player_results[-1]
	if result.get("id")!=fixture.id or not Live.same(candidate.results[prior],result): return {}
	var fixtures: Dictionary=_round_fixtures(source.round_index)
	for index: int in range(reports.size()):
		if candidate.results[prior+index+1].get("id")!=fixtures.npc[index].id: return {}
	if candidate.physical.cursor!=source.physical.cursor+reports.size() or not Live.same(candidate.physical.reports,source.physical.reports+reports): return {}
	var expected: Dictionary=source.build._bank.view().rewards.duplicate(true)
	expected[str(fixture.id)]=(pending.human[2]>pending.human[1])==(fixture.home==0)
	if not Live.same(candidate.build._bank.view().rewards,expected): return {}
	if reports.size()==fixtures.npc.size():
		if preview_job or projected.has("fixture") or candidate.round_index!=source.round_index+1 or candidate.phase!=SeasonState.Phase.REGULAR or candidate.physical.projecting or not candidate.physical.needed.is_empty() or not candidate.physical.error.is_empty() or not GeneralPending._receipts(candidate): return {}
		return _concrete_settlement(candidate) if produce_settlement else {"accepted":true}
	if produce_settlement: return {}
	var needed: Dictionary=fixtures.npc[reports.size()]
	if candidate.round_index!=source.round_index or not Live.same(projected.get("fixture"),needed): return {}
	for club: String in source.opponents.clubs:
		if not Live.same(candidate.opponents.clubs[club].build._bank.view().rewards,source.opponents.clubs[club].build._bank.view().rewards): return {}
	return _job(candidate,needed) if preview_job else {"accepted":true}

static func return_projection(season: SeasonState) -> bool:
	if season==null or season.get_script()!=PrivateReturnFork or season.physical==null or not season.physical.projecting or not season.physical.pending.is_empty() or season.physical.cursor!=2 or season.physical.reports.size() not in [2,3,4]: return false
	for index: int in range(2,season.physical.reports.size()):
		var report: Dictionary=season.physical.reports[index]
		if not Live.keys(report,["fixture","request","report"]) or not Live.whole(report.fixture,index+2,index+2): return false
	return true

# Authenticated first-leg prefix plus ordered return reports. Only the fixed
# settlement operation requests concrete construction after complete replay.
static func _inspect_return(source: SeasonState, human: Array, reports: Array, preview_job: bool, produce_settlement: bool) -> Dictionary:
	if not Live.scope(source) or source.round_index!=1 or source.build.revision()!=5 or not source.physical.pending.is_empty() or reports.size()>2: return {}
	var shape: Dictionary={"human":human.duplicate(true),"reports":reports.duplicate(true)}
	if not GeneralPending._doubleheader_shape(shape,Live.Catalog.game_signature()) or human[0]!=3: return {}
	if Live._capture_structural(source).is_empty() or not Live._first_human_score(source) or not PairedHistory.validate_settled(source): return {}
	var candidate: SeasonState=_detached(source,PrivateReturnFork)
	if candidate==null or Live.scope(candidate) or not Live.capture(candidate).is_empty(): return {}
	detached_calls+=1
	projection_calls+=1
	var projected: Dictionary=GeneralPending._project_candidate(candidate,shape)
	if projected.has("error") or projected.get("candidate")!=candidate or Live.scope(candidate): return {}
	if candidate.results.size()!=4+reports.size() or candidate.player_results.size()!=2: return {}
	for index: int in range(candidate.results.size()):
		if candidate.results[index].get("id")!=index: return {}
	if not Live.same(candidate.results.slice(0,3),source.results) or not Live.same(candidate.player_results[0],source.player_results[0]) or candidate.player_results[1].get("id")!=3: return {}
	if candidate.physical.cursor!=2+reports.size() or not Live.same(candidate.physical.reports,source.physical.reports+reports): return {}
	var expected: Dictionary=source.build._bank.view().rewards.duplicate(true)
	var result: Dictionary=candidate.player_results[1]
	expected["3"]=(human[2]>human[1])==(result.home==0)
	if not Live.same(candidate.build._bank.view().rewards,expected): return {}
	if reports.size()==2:
		if projected.has("fixture") or candidate.round_index!=2 or candidate.phase!=SeasonState.Phase.REGULAR or candidate.physical.projecting or not candidate.physical.needed.is_empty() or not candidate.physical.error.is_empty() or not GeneralPending._receipts(candidate): return {}
		return _concrete_settlement(candidate) if produce_settlement else {"accepted":true}
	if produce_settlement: return {}
	var needed: Dictionary={}
	for row: Dictionary in SeasonDoubleheaderRules.schedule():
		if row.id==4+reports.size(): needed=row
	if candidate.round_index!=1 or not Live.same(projected.get("fixture"),needed): return {}
	for club: String in source.opponents.clubs:
		if not Live.same(candidate.opponents.clubs[club].build._bank.view().rewards,source.opponents.clubs[club].build._bank.view().rewards): return {}
	return _job(candidate,needed) if preview_job else {"accepted":true}

static func _detached(source: SeasonState, type: Script) -> SeasonState:
	var old: SeasonState=source.fork_live()
	if old==null: return null
	var candidate: SeasonState=type.new()
	if not Live.install_fields(candidate,Live.FIELDS,Live.data_fields(old,Live.FIELDS)): return null
	for key: String in Live.OBJECTS:
		candidate.set(key,old.get(key))
	if candidate.build==source.build or candidate.ownership==source.ownership or candidate.physical==source.physical or candidate.career==source.career or candidate.opponents==source.opponents: return null
	for club: String in source.opponents.clubs:
		if candidate.opponents.clubs[club].build==source.opponents.clubs[club].build: return null
	return candidate

static func verify(source: SeasonState, human: Array) -> bool:
	return not _inspect(source,human,[]).is_empty()

static func _inspect(source: SeasonState, human: Array, reports: Array, produce_settlement: bool = false) -> Dictionary:
	if reports.size()>2: return {}
	if not Live.scope(source) or not Live.same(source.league_rules,LeagueRules.capture("doubleheader")) or source.phase!=SeasonState.Phase.REGULAR or source.round_index!=0 or source.picks.size()!=4 or not source.physical.pending.is_empty(): return {}
	var shape: Dictionary={"human":human.duplicate(true),"reports":reports.duplicate(true)}
	if not GeneralPending._doubleheader_shape(shape,Live.Catalog.game_signature()): return {}
	var fixture: Dictionary=source.pending_fixture()
	if fixture.is_empty() or int(fixture.round)!=0 or fixture.id!=human[0] or SeasonDoubleheaderRules.first(fixture.id)!=-1: return {}
	var needed: Dictionary={}
	for row: Dictionary in source.schedule:
		if row.round==0 and row.id==1+reports.size():
			needed=row.duplicate(true)
			break
	if reports.size()<2 and (needed.is_empty() or needed.away==0 or needed.home==0): return {}
	# Full structural+history authentication in this call, not a caller flag.
	if Live._capture_structural(source).is_empty() or not PairedHistory.validate_settled(source): return {}
	var candidate: SeasonState=_detached(source,PrivateFirstFork)
	if candidate==null or not candidate.league_context_consistent() or Live.scope(candidate) or not Live.capture(candidate).is_empty(): return {}
	detached_calls+=1
	projection_calls+=1
	var projected: Dictionary=GeneralPending._project_candidate(candidate,shape)
	if projected.has("error") or projected.get("candidate")!=candidate or Live.scope(candidate): return {}
	if candidate.player_results.size()!=1 or candidate.physical.cursor!=reports.size() or not Live.same(candidate.physical.reports,reports): return {}
	for index: int in range(candidate.results.size()):
		if candidate.results[index].get("id")!=index: return {}
	if candidate.results.size()!=1+reports.size(): return {}
	var expected: Dictionary={}
	expected[str(int(fixture.id))]=(human[2]>human[1])==(fixture.home==0)
	if not Live.same(candidate.build._bank.view().rewards,expected): return {}
	if reports.size()==2:
		# Frozen Pending validates each NPC's exact game receipts on terminal
		# success. The nonpersistable projection never leaves this method.
		if projected.has("fixture") or candidate.round_index!=1 or candidate.phase!=SeasonState.Phase.REGULAR or candidate.physical.projecting or not candidate.physical.needed.is_empty() or not candidate.physical.error.is_empty() or not GeneralPending._receipts(candidate): return {}
		return _concrete_settlement(candidate) if produce_settlement else {"accepted":true}
	if candidate.round_index!=0 or not Live.same(projected.get("fixture"),needed): return {}
	for club: Dictionary in candidate.opponents.clubs.values():
		if not club.build._bank.view().rewards.is_empty(): return {}
	return _job(candidate,needed)

static func _concrete_settlement(candidate: SeasonState) -> Dictionary:
	var concrete: SeasonState=Live.Runtime.new()
	if not Live.install_fields(concrete,Live.FIELDS,Live.data_fields(candidate,Live.FIELDS)): return {}
	for key: String in Live.OBJECTS:
		concrete.set(key,candidate.get(key)) # Objects belong only to the detached projection.
	if not concrete.career.sync(concrete) or not Live.scope(concrete): return {}
	return {"settlement":concrete}

static func _job(candidate: SeasonState, needed: Dictionary) -> Dictionary:
	var state: MatchState=Physical.match_for(candidate,needed)
	var field: FieldDefinition=SeasonState.field_for_fixture(needed,candidate)
	var request: String=Physical.request_key(candidate,needed)
	if state==null or field==null or request.is_empty(): return {}
	return {"accepted":true,"job":{"state":state,"field":field,"seed":Physical.seed_for(candidate,needed),"binding":{"fixture":int(needed.id),"request":request}}}
