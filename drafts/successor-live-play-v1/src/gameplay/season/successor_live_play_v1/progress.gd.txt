## Current materialized named ordinary summary, not history admission.
## Pure game/legend/row transforms retain the frozen general-career schemas.
## Development uses monotone current contracted profiles, never command replay.
## Not valid for Dynasty decay. The fixed Scramble start permutes stats only;
## these within-season repertoire, mastery and lesson aggregates are unchanged.
const Admissions = preload("ordinary_admission.gd")
const Legacy = preload("res://src/gameplay/season/successor_career_general_v1/career_progress.gd")
const Catalog = preload("res://src/gameplay/season/successor_league_general_v2/catalog_context.gd")
const Eligibility = preload("res://src/gameplay/season/successor_league_general_v2/exotic_eligibility.gd")
const VERSION = ClubPlayerProgress.VERSION
const HITS_VERSION = ClubPlayerProgress.HITS_VERSION
const PROGRESS_VERSION = Legacy.PROGRESS_VERSION
const FORMAT = Legacy.FORMAT

static func _capture_games(season: SeasonState) -> Dictionary:
	if season.build == null or season.build._market != 0:
		return {}
	var sponsors: Dictionary = _sponsor_counts(season.build)
	if sponsors.has("invalid"): return {}
	var basis: Dictionary=season.career.start_basis if season.career!=null else {}
	var development: Dictionary = _current_development(season.build,basis)
	if development.is_empty():
		return {}
	var games: Array = []
	for game: Dictionary in season.player_results:
		var own: Array = game.get("club_roster", season.teams[0].roster)
		var stats: Variant = null
		if game.has("performance"):
			if not game.performance is Dictionary:
				return {}
			var source: Dictionary = game.performance
			if (
				source.size() != 8
				or own.size() != 4
				or not SeasonPerformance.valid(source, source.keys())
			):
				return {}
			stats = {"homers": 0, "hitters": 0, "strikeouts": 0, "three_k": 0}
			for id: String in own:
				if not source.has(id):
					return {}
				stats.homers += int(source[id].hr)
				stats.hitters += 1 if source[id].h > 0 else 0
				stats.strikeouts += int(source[id].p_k)
				stats.three_k += 1 if source[id].p_k >= 3 else 0
		# Pre-physical saves accepted score/stat combinations without this cross-check.
		# Preserve their original game journal, but do not infer any new stat unlocks.
		# Owned-player seasons and every physical season remain strictly validated.
		var own_runs: int = int(game.home_runs if game.home == 0 else game.away_runs)
		if (
			stats != null
			and stats.homers > own_runs
			and season.physical == null
			and season.player_access.is_empty()
		):
			stats = null
		var row: Dictionary = {
			"game": int(game.id),
			"stats": stats,
			"sponsors": int(sponsors.get(int(game.id), 0)),
			"end_half": -1,
			"deficit": 0
		}
		var proofs: Dictionary = game.get("ai_tactics", {})
		for proof: Dictionary in proofs.values():
			if not proof.has("base"):
				continue
			var flow: Dictionary = proof.base
			row.end_half = int(flow.final.half)
			for point: Dictionary in flow.ready:
				var deficit: int = int(
					point.away - point.home if game.home == 0 else point.home - point.away
				)
				row.deficit = maxi(row.deficit, deficit)
		games.append(row)
	var result: Dictionary = {"version": VERSION, "games": games, "development": development}
	return result if load("res://src/gameplay/season/successor_career_general_v1/career_progress.gd").valid_capture(result, ClubSeasonRecord.capture(season).scores) else {}


static func _capture_hits(season: SeasonState) -> Dictionary:
	# New runs opt into a new evidence schema; the original capture remains exact.
	# Historic missing/null game stats are still unknown and never get reconstructed.
	var result: Dictionary = _capture_games(season)
	if result.is_empty():
		return {}
	for index in range(result.games.size()):
		var row: Dictionary = result.games[index]
		if row.stats == null:
			continue
		var game: Dictionary = season.player_results[index]
		var own: Array = game.get("club_roster", season.teams[0].roster)
		var source: Variant = game.get("performance")
		if (
			not source is Dictionary
			or source.size() != 8
			or own.size() != 4
			or not SeasonPerformance.valid(source, source.keys())
		):
			return {}
		var hits: int = 0
		var used: Dictionary = {}
		for id: String in own:
			if not source.has(id) or used.has(id):
				return {}
			used[id] = true
			hits += int(source[id].h)
		row.stats["hits"] = hits
	result.version = HITS_VERSION
	return result if load("res://src/gameplay/season/successor_career_general_v1/career_progress.gd").valid_capture(result, ClubSeasonRecord.capture(season).scores) else {}


static func _capture_legend(season: SeasonState) -> Dictionary:
	var result: Dictionary = _capture_hits(season)
	if result.is_empty():
		return {}
	for index in range(result.games.size()):
		var row: Dictionary = result.games[index]
		var game: Dictionary = season.player_results[index]
		row["players"] = null
		# Missing historic provenance is unknown, not the current roster or club totals.
		if row.stats == null or not game.has("club_roster"):
			continue
		var own: Variant = game.club_roster
		var source: Variant = game.get("performance")
		if (
			not ClubPlayerLegend._roster(own)
			or not source is Dictionary
			or not SeasonPerformance.valid(source, source.keys())
		):
			return {}
		var fielding: Variant = game.get("fielding")
		var known_fielding: bool = fielding != null
		if (
			known_fielding
			and (
				not fielding is Array
				or not SeasonLeftRight.own_halves(fielding, game.home != 0)
				or not SeasonJumpstart.valid_roster(own, fielding, source)
			)
		):
			return {}
		var players: Dictionary = {}
		for id: String in own:
			if not source.has(id):
				return {}
			var line: Dictionary = source[id]
			var clean: Variant = 0 if known_fielding else null
			var fielded: Variant = line.get("fielding_v1")
			if known_fielding:
				for play: Dictionary in fielding:
					if play.player == id:
						if fielded == false:
							return {}
						fielded = true
						clean += 1 if play.clean else 0
			players[id] = {
				"pa": int(line.pa),
				"pitches": int(line.pitches),
				"hits": int(line.h),
				"strikeouts": int(line.p_k),
				"outs": int(line.outs),
				"fielded": fielded,
				"clean_outs": clean
			}
		row.players = players
	result.version = ClubPlayerLegend.VERSION
	return result if load("res://src/gameplay/season/successor_career_general_v1/career_progress.gd").valid_capture(result, ClubSeasonRecord.capture(season).scores) else {}


static func capture(season: SeasonState) -> Dictionary:
	if season == null or season.build == null or not season.build.call("pitch_catalog_successor"):
		return {}
	var result: Dictionary = _capture_legend(season)
	if result.is_empty():
		return {}
	result.version = PROGRESS_VERSION
	result["format"] = FORMAT
	result["pitch_catalog"] = Catalog.signature()
	result["exotic"] = Eligibility.capture(season)
	return result if Legacy.valid(result, ClubSeasonRecord.capture(season).scores) else {}


static func _current_development(build: SeasonBuild,basis: Dictionary={}) -> Dictionary:
	if build == null or not build.call("pitch_catalog_successor") or build._book == null \
		or not build._book._dynasty_start.is_empty() \
		or (not build._league_rules.is_empty() and not ClubCareer.same(build._league_rules, Admissions.expected(build._league_rules,int(build._league_rules.get("season_seed",basis.get("seed",-1))),basis))):
		return {}
	# Contracts retain released players. Active count never shrinks; replacements
	# preserve count, mastery never decreases and forgotten recipes retain mastery.
	# Recruit catch-up changes levels, never recipe identity. Exchanges introduce
	# only recipes already learned by another contracted player, so the GLOBAL
	# lesson union is unchanged. Stat retraining does not affect these aggregates.
	var result: Dictionary = {"mastery": 0, "known": 0, "lessons": []}
	for key: Variant in build._contracts:
		if not key is String or not build._book._players.get(key) is Dictionary: return {}
		var id: String = key
		var player: Dictionary = build._book.player(id)
		var authored: Dictionary = SeasonPlayerCatalog.profile(id)
		if player.is_empty() or authored.is_empty() or not player.get("active") is Array or not player.get("mastery") is Dictionary: return {}
		for recipe: Variant in player.mastery:
			if not recipe is String or not SeasonOwnership._whole(player.mastery[recipe], 1, 5): return {}
		ClubPlayerProgress._profile(result, player)
		for recipe: String in player.mastery:
			if not authored.mastery.has(recipe) and not result.lessons.has(recipe):
				result.lessons.append(recipe)
	result.lessons.sort()
	return result

			"sides_earned": season.build._sides_earned,

	return result



static func _sponsor_counts(build: SeasonBuild) -> Dictionary:
	# Read-only bounded receipt projection. No wallet reconstruction, prices,
	# rewards, or ownership commands execute here. Their admitted journal remains
	# unchanged. Dugout D3 changes refunds, not active counts or expiry.
	# Keep all other terms contexts (including D4 expiry) outside this reader.
	var bank: SeasonOwnership = build._bank
	if bank == null or bank._events.size() > SeasonOwnership.MAX_EVENTS:
		return {"invalid": true}
	if not bank._sponsor_rule.is_empty() and not (
		bank._sponsor_rule == SeasonSponsorCatalog.D3_RULE
		and ClubCareer.same(build._league_rules, LeagueRules.capture_tier("dugout",3))
		and build.d3_sponsors_valid()
	):
		return {"invalid": true}
	var stock: Dictionary = {}
	var active: Dictionary = {}
	var result: Dictionary = {}
	for event: Dictionary in bank._events:
		if not event.get("op") is String: return {"invalid": true}
		match event.op:
			"stock":
				if not event.get("offers") is Dictionary: return {"invalid": true}
				for key: Variant in event.offers:
					if not key is String or not event.offers[key] is String: return {"invalid": true}
				stock = event.offers.duplicate()
			"buy":
				if not event.get("replace") is String or not event.get("offer") is String or not event.get("id") is String: return {"invalid": true}
				active.erase(event.replace)
				var item: Variant = bank._catalog.get(stock.get(event.offer, ""), {})
				if not item is Dictionary: return {"invalid": true}
				if item.get("kind") == "sponsor":
					active[event.id] = true
				stock.erase(event.offer)
			"sell", "inventory_sale_v1":
				if not event.get("receipt") is String: return {"invalid": true}
				active.erase(event.receipt)
			"sponsor_set":
				if not event.get("sales") is Array or not event.get("purchases") is Array or not event.get("id") is String: return {"invalid": true}
				for receipt: Variant in event.sales:
					if not receipt is String: return {"invalid": true}
				for purchase: Variant in event.purchases:
					if not purchase is Dictionary or not purchase.get("offer") is String: return {"invalid": true}
				for receipt: String in event.sales: active.erase(receipt)
				for index in range(event.purchases.size()):
					active[SeasonSponsorSet.receipt_id(event.id, index)] = true
					stock.erase(event.purchases[index].offer)
			"reward":
				if not SeasonOwnership._whole(event.get("game"), 0, 32): return {"invalid": true}
				result[int(event.game)] = active.size()
	return result
