static func create_fresh(seed_value: int, league: String = "standard") -> SeasonState:
	return _create_named(seed_value,league,Genesis.derive(seed_value)) if league in COMMON_ORDINARY_LEAGUES else null

static func _create_paired_draft(seed_value: int) -> SeasonState:
	return _create_named(seed_value,"doubleheader",Genesis.derive(seed_value))

static func create_common(basis: Dictionary, league: String = "standard") -> SeasonState:
	return _create_named(int(basis.get("seed",-1)),league,basis) if league in COMMON_ORDINARY_LEAGUES and CommonStart.valid(basis) else null

# Private draft construction only. Store will later derive the envelope from
# the exact admitted championship; callers may not publish a draft payload.
static func create_d1_draft(transition: Dictionary) -> SeasonState:
	var contract=load("res://src/gameplay/season/successor_dynasty_owner_v1/standard_d1_transition.gd")
	if not contract.valid(transition): return null
	var result: SeasonState=create_common(transition.basis,"standard")
	if result==null or not result.build.enable_d1_pricing(): return null
	result.career.start_basis=transition.duplicate(true)
	return result if result.career.sync(result) else null

func _d1_draft() -> bool:
	return career!=null and career.start_basis.get("kind")=="common-standard-d1-transition-v1"

static func _create_named(seed_value: int, league: String, basis: Dictionary) -> SeasonState:
	var earned: bool=basis.get("kind")==CommonStart.KIND
	if (earned and (not CommonStart.valid(basis) or basis.seed!=seed_value)) or (not earned and (not Genesis.valid_basis(basis) or basis.genesis.seed!=seed_value)): return null
	if not SeasonOwnership._whole(seed_value,0,2147483647) or league not in Admissions.BACKEND_LEAGUES or (earned and league=="doubleheader"): return null
	var result: SeasonState = load("res://src/gameplay/season/successor_live_play_v1/runtime.gd").new()
	result.season_seed = seed_value
	result.league_rules = Admissions.expected({"league":league},seed_value)
	result.league_live = league != "standard"
	result.build = Build.new(seed_value)
	if not result.build.enable_successor_pitch_catalog() or (result.league_live and not result.build.enable_league_rules(result.league_rules)): return null
	result.opponents = Opponents.new(); result.opponents._format = 13
	result.physical = Physical.new()
	var rng: RandomNumberGenerator = RandomNumberGenerator.new(); rng.seed = seed_value
	result.draft_pool = ClubPlayerCatalog.draft_pool(rng,PitchBatLab.DEBUG_PLAYER_ID,false,SeasonState.LEGACY_IDS)
	for index in range(6): result.teams.append({"name":SeasonState.TEAM_NAMES[index],"roster":[],"draw":rng.randf()})
	result._build_schedule()
	result.schedule = ScheduleSelector.select(LeagueRules.doubleheader(result.league_rules), result.schedule)
	if not (CommonStart.install(result.build,basis) if earned else Starts._install(result.build,basis)) or not ClubPlayerDraft.apply(result,basis.player_access): return null
	var ledger: ClubCareer = LiveCareer.new()
	ledger.current = 1; ledger.runs.append({})
	ledger.start_basis = basis.duplicate(true)
	ledger._assets = basis.assets.duplicate(true)
	ledger._pennants = basis.pennants.duplicate(true) if earned else preload("res://src/gameplay/season/successor_owner_career_v1/settlement.gd").birth_pennants(basis)
	ledger._plaques = basis.plaques.duplicate(true)
	ledger._pitch_catalog = preload("catalog.gd").game_signature()
	result.career = ledger; result.league_career = true
	return result if ledger.sync(result) else null

func league_context_consistent() -> bool:
	var codec=load("res://src/gameplay/season/successor_live_play_v1/codec.gd")
	return codec.d1_draft_scope(self) if _d1_draft() else codec.scope(self)
func _league_inventory_context_consistent() -> bool:
	return league_context_consistent()
func choose_player(id: String) -> bool:
	# Store owns a detached materialized candidate. This invokes114's local
	# gameplay operation, not117's historical clone/choose/capture path.
	if not league_context_consistent() or not super.choose_player(id): return false
	if picks.size() == 4:
		if _d1_draft():
			if not CommonStart.install(build,career.start_basis.basis): return false
		elif career.start_basis.get("kind")==CommonStart.KIND:
			if not CommonStart.install(build,career.start_basis): return false
		elif not Starts._install(build,career.start_basis): return false
	return career.sync(self)
func adopt_build(next: SeasonBuild) -> void:
	if _d1_draft(): return
	if next == null or not league_context_consistent(): return
	super.adopt_build(next)
	career.sync(self)

func fork_live() -> SeasonState:
	# Explicit detached transaction fork, not reconstruction of past commands.
	var codec = load("res://src/gameplay/season/successor_live_play_v1/codec.gd")
	if not codec.scope(self) or not codec.known_variables(self,codec.FIELDS+codec.OBJECTS): return null
	if _heat_replay_team != null or _doubleheader_replay_build != null or _pool_replay_ledger != null: return null
	var result: SeasonState = get_script().new()
	if not codec.install_fields(result,codec.FIELDS,codec.data_fields(self,codec.FIELDS)): return null
	var detach = preload("res://src/gameplay/season/successor_owner_career_v1/pending.gd")
	result.ownership = ownership.fork()
	result.build = detach._fork_build(build)
	result.career = career.fork()
	result.opponents = Opponents.new()
	result.opponents._format = opponents._format
	result.opponents.draft_rule = opponents.draft_rule
	result.opponents.draft_order.assign(opponents.draft_order)
	for id: String in opponents.clubs:
		var club: Dictionary = opponents.clubs[id].duplicate(true)
		club.build = detach._fork_build(opponents.clubs[id].build)
		result.opponents.clubs[id] = club
	result.physical = Physical.new()
	if not codec.install_fields(result.physical,codec.PHYSICAL_FIELDS,codec.data_fields(physical,codec.PHYSICAL_FIELDS)): return null
	return result

func make_match() -> MatchState:
	if career!=null and (CommonStart.detached_mixed_flat(career.start_basis) \
			or CommonStart.private_mixed_append_basis(career.start_basis)): return null
	return null if _d1_draft() or Admissions.paired_draft_only(self) else super.make_match()

# Called only by the bound compact native App route; the detached
# generic match entry above still rejects this versioned origin.
func _make_mixed_round() -> MatchState:
	if career==null or not CommonStart.valid(career.start_basis) \
			or phase not in [Phase.REGULAR,Phase.SEMIFINAL,Phase.FINAL] \
			or not physical.pending.is_empty() or build.pack_pending(): return null
	var basis: Dictionary=career.start_basis
	if CommonStart.detached_mixed_rollover(basis):
		if phase!=Phase.REGULAR or round_index<0 or round_index>=MIXED_ROLLOVER_REGULAR_STOP \
				or results.size()!=round_index*3 or player_results.size()!=round_index or picks.size()!=4: return null
	elif not CommonStart.detached_mixed_flat(basis) and not CommonStart.detached_mixed_append(basis):
		return null
	var fixture:=pending_fixture()
	if fixture.is_empty() or (fixture.home!=0 and fixture.away!=0): return null
	return super.make_match()

func shop_available() -> bool:

func record_player_result(
	fixture_id: int,
	away_runs: int,
	home_runs: int,
	performance: Dictionary = {},
	used_gear: Array = [],
	tactics: Array = [],
	batting: Dictionary = {},
	stances: Array = [],
	fielding: Array = [],
	pitching: Dictionary = {},
	field_supply: Dictionary = {},
	frozen: Array = [],
	ai_tactics: Dictionary = {},
	replayed_tactics: Array = [],
	property_key: String = "",
	pool_evidence: Variant = null,
	yard_evidence: Variant = null
) -> bool:
	if _d1_draft() or Admissions.paired_draft_only(self) or phase not in [Phase.REGULAR,Phase.SEMIFINAL,Phase.FINAL] or not league_context_consistent(): return false
	var ok: bool = super.record_player_result(fixture_id,away_runs,home_runs,performance,used_gear,tactics,batting,stances,fielding,pitching,field_supply,frozen,ai_tactics,replayed_tactics,property_key,pool_evidence,yard_evidence)
	# The native commit adds an empty key to an away/neutral human row
	# when a home snapshot exists. Keep those Ordinary wires unchanged.
	if ok and not home_property.is_empty() and property_key.is_empty():
		for rows: Array in [results,player_results]:
			for row: Dictionary in rows:
				if row.id==fixture_id and row.get("property")=="": row.erase("property")
	return ok and career.sync(self)
