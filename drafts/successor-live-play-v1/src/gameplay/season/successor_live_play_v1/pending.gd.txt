extends RefCounted
## Live current-round projection only. Never used by cold materialized hydration.
const Evidence = preload("evidence_boundary.gd")
const Codec = preload("codec.gd")
const Fixtures = preload("physical_fixtures.gd")
const Original = preload("res://src/gameplay/season/successor_league_general_v2/pending.gd")
static var project_calls: int = 0
static func shape(season: SeasonState) -> bool:
	if season==null or not Codec.scope(season) or season.physical.pending.is_empty(): return false
	var value: Dictionary=season.physical.pending
	if value.has("property"):
		if season.home_property.is_empty() or not Codec.StarterHome.pending_shape(season): return false
		value={"human":value.get("human"),"reports":value.get("reports")}
	elif not season.home_property.is_empty() and not Codec.StarterHome.pending_shape(season): return false
	return Evidence.pending_shape(value,season.league_rules,Codec.Catalog.game_signature())
static func project(owner: SeasonState) -> Dictionary:
	return _project(owner,false)

# Used only by mixed cold replay. No validation outcome can be supplied here.
static func _project_for_mixed_restore(owner: SeasonState) -> Dictionary:
	return _project(owner,true)

static func _project(owner: SeasonState, observe_mismatch: bool) -> Dictionary:
	project_calls += 1
	if not shape(owner): return {"error":"The saved current round is inconsistent."}
	var candidate: SeasonState = owner.fork_live()
	if candidate == null: return {"error":"The current owner could not be detached."}
	var pending: Dictionary = candidate.physical.pending.duplicate(true)
	candidate.physical.pending.clear()
	# Starts at the saved consumed-prefix cursor. Only this round's newly
	# appended reports are resolved. Earlier physical games are not rerun.
	if pending.has("property"):
		var human: Array = pending.human
		human.append_array([[],pending.property])
		pending.erase("property")
	if not observe_mismatch: return Original._project_candidate(candidate,pending)
	# Metadata is transient on this one fork; the native save schema is exact.
	var mismatch: Dictionary={}
	candidate.physical.set_meta("_mixed_request_observation",mismatch)
	var projected: Dictionary=Original._project_candidate(candidate,pending)
	candidate.physical.remove_meta("_mixed_request_observation")
	if projected.has("error") and mismatch.get("cursor")==candidate.physical.cursor:
		projected["_mixed_request_mismatch"]=mismatch
	return projected
static func stage_report(owner: SeasonState, raw: Dictionary) -> Dictionary:
	var result: Dictionary = project(owner)
	if result.has("error") or not result.get("fixture") is Dictionary: return {"error":"No matching physical job remains in this round."}
	var request: String = Fixtures.request_key(result.candidate,result.fixture)
	if request.is_empty(): return {"error":"The physical job binding is missing."}
	var candidate: SeasonState = owner.fork_live()
	if candidate == null: return {"error":"The physical checkpoint could not be detached."}
	candidate.physical.pending.reports.append({"fixture":int(result.fixture.id),"request":request,"report":raw.duplicate(true)})
	# Actual report acceptance is checked now against the current real fixture,
	# not deferred to cold load and not accepted as a caller validation token.
	var validated: Dictionary = project(candidate)
	if validated.has("error"): return validated
	# Expose only the already validated terminal result to the fixed Store producer.
	# This return value is not an admission token; Store never accepts one from callers.
	if not validated.has("fixture"): return {"candidate":candidate,"settled":validated.candidate}
	return {"candidate":candidate}
