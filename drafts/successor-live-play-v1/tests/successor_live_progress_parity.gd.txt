func check(ok: bool,label: String) -> bool:
	print(("CHECK: " if ok else "FAIL: ")+label)
	if not ok: failed=true
	return ok
func parity(build: SeasonBuild,label: String) -> bool:
	var trace: Dictionary = {"book":null,"count":0,"contracts":build._contracts.duplicate(true),"result":{"mastery":0,"known":0,"lessons":[]}}
	if not check(Reference._observe_development(trace,build),label+" legacy event observer accepts actual book commands"): return false
	trace.result.lessons.sort()
	return check(ClubCareer.same(Direct._current_development(build),trace.result),label+" direct aggregates equal legacy observer")
func command(build: SeasonBuild,op: String,player: String,target: String = "",extra: Dictionary = {}) -> bool:
	var event: Dictionary = {"id":"synthetic-progress:%d"%build._book.revision(),"rev":build._book.revision(),"op":op,"player":player}
	if op not in ["exchange","retrain"]: event.target=target
	event.merge(extra)
	var result: Dictionary = build._book.commit(event)
	return check(result.ok and not result.replayed,"actual book "+op+" "+player+" "+target)
func bank_command(build: SeasonBuild,op: String,extra: Dictionary = {}) -> String:
	var id: String = "synthetic-bank:%d"%build._bank.revision()
	var event: Dictionary = {"id":id,"rev":build._bank.revision(),"op":op}
	event.merge(extra)
	var result: Dictionary = build._bank.commit(event)
	if not check(result.ok and not result.replayed,"actual synthetic bank "+op): return ""
	return id

	if bank_command(build,"stock",{"offers":{"s0":"D01","s1":"A08"}}).is_empty(): return false
	var first: String = bank_command(build,"buy",{"offer":"s0","replace":"","discard":[]})
	if first.is_empty() or bank_command(build,"reward",{"game":1,"win":true}).is_empty(): return false
	var second: String = bank_command(build,"buy",{"offer":"s1","replace":"","discard":[]})
	if second.is_empty() or bank_command(build,"reward",{"game":2,"win":true}).is_empty(): return false
	if bank_command(build,"stock",{"offers":{"s2":"A09"}}).is_empty(): return false
	var third: String = bank_command(build,"buy",{"offer":"s2","replace":first,"discard":[]})
	if third.is_empty() or bank_command(build,"reward",{"game":3,"win":true}).is_empty(): return false
	if bank_command(build,"sell",{"receipt":second,"discard":[]}).is_empty(): return false
	if bank_command(build,"reward",{"game":4,"win":true}).is_empty(): return false
