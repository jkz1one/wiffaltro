func choose_player(id: String) -> bool:
	if not league_context_consistent() or not offers().has(id):
		return false
	# Rebuild a local candidate from the real predecessor; never publish a partial
	# fourth-pick reconstruction if inherited-context repair/validation fails.
	var candidate: SeasonState = create_next(career.predecessor,season_seed)
	if candidate == null:
		return false
	for previous: String in picks:
		if not candidate._choose_local(previous):
			return false
	var codec = load("res://src/gameplay/season/successor_owner_next_v1/codec.gd")
	if not ClubCareer.same(codec._snapshot(candidate),codec._snapshot(self)) or not candidate._choose_local(id):
		return false
	var boundary = preload("res://src/gameplay/season/successor_owner_terminal_v1/codec.gd")
	for field: String in boundary.FIELDS:
		var value: Variant = candidate.get(field)
		set(field,value.duplicate(true) if value is Array or value is Dictionary else value)
	for field: String in boundary.OBJECTS:
		set(field,candidate.get(field))
	return true

func _choose_local(id: String) -> bool:
	if not league_context_consistent() or not offers().has(id):
		return false
	if not super.choose_player(id):
		return false
	# General standalone fourth-pick reconstruction does not reinstall successor
	# history/full owned recruiting pool. Restore exact owner birth inputs here.
	if picks.size() == 4 and not load("res://src/gameplay/season/successor_owner_next_v1/start.gd")._install(build,career.start_basis):
		return false
	return career.sync(self)
