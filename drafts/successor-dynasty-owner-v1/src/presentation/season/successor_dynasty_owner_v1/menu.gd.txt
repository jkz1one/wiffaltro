var _round_resume: Button
var _round_progress: Label
var _next_round_refresh: int = 0

func _process(_delta: float) -> void:
	if not is_instance_valid(_round_progress) or not is_visible_in_tree(): return
	var now: int=Time.get_ticks_msec()
	if now<_next_round_refresh: return
	_next_round_refresh=now+250
	_refresh_round_progress()

func _refresh_round_progress() -> void:
	# Presentation only: the bound native runner owns its clock, evidence,
	# result validation and save. Never mutate the runner or projected fixture.
	var saved: int=app.season.physical.pending.get("reports",[]).size()
	var runner: Node=app.ai_runner
	var running: bool=is_instance_valid(runner)
	_round_resume.disabled=app._busy or running
	var message: String
	if running:
		var fixture: Dictionary=app.projection.get("fixture",{})
		var match_label: String=_matchup(fixture) if fixture.has("away") and fixture.has("home") else "NPC fixture"
		if app.starter_npc_running():
			_round_progress.text="Checking %s in the background. %d NPC results saved." % [match_label,saved]
			return
		var live: PitchBatLab=runner._lab
		var state: MatchState=live._match_state if is_instance_valid(live) else null
		if state==null:
			message="Preparing %s. %d NPC results saved."%[match_label,saved]
		else:
			message="Simulating %s · %s %d · %d plate appearances complete. %d NPC results saved."%[
				match_label,"Top" if state.top_half else "Bottom",state.inning,maxi(0,state.plate_appearance_number-1),saved]
	elif not app.retained_report.is_empty():
		message="NPC game complete, but its result is not saved yet. Resume to retry the checkpoint."
	elif app._busy:
		message="Checking the saved round. %d NPC results saved."%saved
	else:
		message="%d NPC results saved. %s to continue."%[saved,
			_round_resume.text.capitalize() if _round_resume.text.begins_with("RESUME ") else "Resume Saved Round"]
	if _round_progress.text!=message: _round_progress.text=message

func _show_standard_workshop(placement: bool = false) -> void:
	if not app.starter_player() or not app._available() or not app._owned(): return
	var workshop = Store.Owner.Common.StandardWorkshop
	var owner: RefCounted = app.career_owner
	if not workshop.visible(owner): return
	var record: Dictionary = owner.ordinary_state.get(workshop.FIELD,{})
	var saved: bool = not record.is_empty()
	_screen("standard_workshop","STADIUM WORKSHOP","STANDARD BASE · BACKYARD DRAFT")
	_bound_button(_footer,"BACK" if saved else "CANCEL",render)
	if not saved and not placement:
		_label(_body,"STARTER BACKSTOP CHAIR",24)
		_note("Place your free chair at Primary Receiver to preview the backyard draft. Nothing is saved until you confirm.")
		_bound_button(_body,"PREVIEW STARTER CHAIR",_show_standard_workshop.bind(true))
		return
	if not saved: record = workshop.record(owner.ordinary_state.lineage)
	if record.is_empty():
		_note("This layout could not be previewed. Nothing was changed.")
		return
	var opening: Dictionary=owner.ordinary_state.get(workshop.OPENING,{})
	if opening.get("locked",false):
		_label(_body,"OPENING DAY · HOME CHAIR LOCKED" if opening.choice=="chair" else "OPENING DAY · DEFAULT FIELD LOCKED",21)
		if opening.choice=="chair":
			_note("The chair is installed for your home games this season. Away games are unchanged.")
		elif saved:
			_note("The chair below is a saved draft preview only. This season uses the default field.")
		else:
			_note("This season uses the default field. No draft was saved.")
			return
	else:
		_label(_body,"SAVED DRAFT · PRIMARY RECEIVER" if saved else "UNSAVED PREVIEW · PRIMARY RECEIVER",21)
		_note("Preview only. This draft does not change the playing field and costs no Club Bucks.")
	var preview: StadiumDraftPreview = StadiumDraftPreview.new()
	_body.add_child(preview)
	var ledger: ClubAssetLedger = workshop.ledger(owner.ordinary_state.lineage)
	preview.build(record.layout,ledger.construction(),false)
	if not preview.positions.has(workshop.COPY):
		_note("The native preview could not build this layout. Nothing was changed.")
		return
	preview.focus_copy(workshop.COPY)
	_bound_button(_footer,"FIELD VIEW",preview.overview)
	if not saved and workshop.editable(owner):
		_bound_button(_footer,"SAVE DRAFT",Callable(app,"save_standard_workshop_draft"))
	elif workshop.player_opening_window(owner):
		_bound_button(_footer,"OPENING DAY LAYOUT",show_standard_opening_review)

	if not owner.workshop_draft.is_empty():
		_note("Stadium Workshop: %s backstop chair draft saved. This does not change the field."%("purchased" if paid else "starter"))

		_note("%s chair applied for subsequent Dynasty home games. This season's record stays unchanged."%("Purchased" if paid else "Starter"))

		and owner.get_script().workshop_window(owner):

		_bound_button(_body,"SAVE PURCHASED CHAIR DRAFT",Callable(app,"save_workshop_draft").bind(true))

	var balance: int=purchase.available_balance(owner)

	elif owner.get_script().workshop_window(owner) and balance>=purchase.PRICE:
		_bound_button(_body,"BUY EXTRA BACKSTOP CHAIR · %d CLUB BUCKS"%purchase.PRICE,Callable(app,"buy_workshop_chair"))

		_note("An extra chair costs %d Club Bucks. Only recorded Dynasty completion awards can fund it."%purchase.PRICE)

func _player_summary(player: PlayerDefinition) -> void:
	var repertoire: Array[String] = []
	var pitch_ids: Array[String] = []
	for pitch: PitchDefinition in player.starting_pitches:
		repertoire.append(pitch.display_name)
		pitch_ids.append(String(pitch.id))
	var pitch_label: Label = _label(_body,"Pitches: "+", ".join(repertoire),16)
	pitch_label.set_meta("pitch_ids",pitch_ids)
	# Authored attribute labels, no retuning and no raw data dump.
	var stats: Dictionary = app.season.build.player(String(player.id)).get("stats",{})
	var labels: Array[String] = []
	for stat: String in SeasonPlayerCatalog.STATS:
		if stats.has(stat): labels.append(stat.capitalize()+": "+str(stats[stat]))
	if not labels.is_empty():
		var stat_label: Label = _label(_body," · ".join(labels),16)
		stat_label.set_meta("stat_keys",SeasonPlayerCatalog.STATS.duplicate())

func _display_standard_record(history: Dictionary) -> void:
	var receipt: Dictionary = history.receipt
	var abandoned: bool = receipt.is_empty()
	var finish: String = {"missed":"Missed playoffs","semifinal":"Eliminated in semifinals",
		"runner_up":"Runner-up","champion":"Champion"}.get(history.summary.finish,"Abandoned" if abandoned else "")
	_label(_body,"%d–%d regular season · %s"%[history.summary.wins,
		history.summary.games-history.summary.wins,finish],22)
	if abandoned:
		_note("No completion payout. Standings reflect games played before the season closed.")
	else:
		_note("%d Club Bucks earned · %d finish + %d record + %d first title"%[
			receipt.total,receipt.finish_award,receipt.record_bonus,receipt.first_bonus])
	_label(_body,"STANDINGS AT CLOSE" if abandoned else "FINAL REGULAR-SEASON STANDINGS",19)
	var rank: int = 0
	for row: Dictionary in history.standings:
		rank += 1
		var line: Label = _label(_body,"%d. %s  ·  %d–%d  ·  %d scored / %d allowed"%[
			rank,history.teams[row.team],row.wins,row.losses,row.rf,row.ra],16)
		if row.team==0: line.add_theme_color_override("font_color",ClubhouseTheme.GOLD)
	_label(_body,"SEASON SCORES",19)
	for row: Dictionary in history.results:
		var round_label: String = "Round %d"%(int(row.id/3.0)+1) if row.id<30 else ("Semifinal" if row.id<32 else "Final")
		var line: Label = _label(_body,"%s  ·  %s  %d–%d  %s"%[round_label,
			history.teams[row.away],row.away_runs,row.home_runs,history.teams[row.home]],16)
		if row.away==0 or row.home==0: line.add_theme_color_override("font_color",ClubhouseTheme.GOLD)

func _installed_dugout_d1() -> void:
	var draft: Dictionary=app.installed_dugout_d1_draft
	var lane: String=draft.get("state","")
	if lane in ["doubleheader_cpu_playoffs","doubleheader_cpu_complete","doubleheader_cpu_closed"]:
		var closed: bool=lane=="doubleheader_cpu_closed"
		var completed: bool=lane!="doubleheader_cpu_playoffs"
		var label: String="SEASON CLOSED" if closed else ("PLAYOFFS COMPLETE" if completed else "CPU PLAYOFFS")
		_label(_body,"DOUBLEHEADER BASE · "+label,24)

			var winner: int=int(draft.cpu_champion)
			if winner in [1,2,3,4,5]: _note("League champion: %s. Playoff results are saved."%SeasonState.TEAM_NAMES[winner])
			for game: Dictionary in draft.results.slice(30):
				_note("%s · %d–%d · %s at %s"%["Final" if game.id==32 else "Semifinal",int(game.away_runs),int(game.home_runs),SeasonState.TEAM_NAMES[int(game.away)],SeasonState.TEAM_NAMES[int(game.home)]])

			var host: String=SeasonState.TEAM_NAMES[int(other.home)]

			else:
				_note("Other semifinal: %s at %s"%[visitor,host])

	if lane in ["doubleheader_draft","doubleheader_held","doubleheader_paid"]:
		var round_index: int=int(draft.round)
		var shown: int=round_index-1 if lane=="doubleheader_paid" else round_index

		if lane=="doubleheader_draft" and draft.picks.size()<4:
			label="DOUBLEHEADER BASE · DRAFT %d / 4"%draft.picks.size()
		_label(_body,label,24)
		_note("Dugout Base and D1 titles kept · %d Club Credits · %d Season Cash"%[draft.credits,draft.cash])
		_note("Each pair has two games. Pitching workload carries by player into the second game.")

			_note("Regular season complete. Qualified for the playoffs as seed %d."%seed if seed>0 else "Regular season complete. Yard Club did not qualify for the playoffs.")
		for id: String in draft.picks:
			var player: PlayerDefinition=ContentDB.get_player(StringName(id))
			_label(_body,player.display_name if player!=null else id,20)
		for id: String in draft.offers:
			var player: PlayerDefinition=ContentDB.get_player(StringName(id))
			_bound_button(_body,player.display_name if player!=null else id,Callable(app,"pick_installed_named_base").bind(id,draft.bytes.duplicate()))
		if lane=="doubleheader_held":
			_note("The club game is saved. Results have not been paid.")
			if app.installed_dugout_d1_cpu_running():
				_note("Playing CPU game %d of 2…"%int(draft.cpu_reports+1))

				_bound_button(_body,"SETTLE GAME %d"%(round_index%2+1),Callable(app,"settle_installed_dugout_d1").bind(draft.bytes.duplicate()))

			"scramble_draft","scramble_held","scramble_paid"]:
		var earnings: bool=lane.begins_with("earnings_")

	_label(_body,"SEASON CLOSED · "+owner.lifecycle_reason().to_upper(),24)

		_note("Your Dugout Base championship is saved. Starting D1 keeps that history and your Club Credits.")

		var payout:int=int(owner.ordinary_state.last_close.receipt.total)
		if app.mixed_append_next_visible():
			_note("Completion payout: %d Club Credits. Season results and previous season history are preserved."%payout)
			_bound_button(_body,"START NEXT DUGOUT BASE SEASON",Callable(app,"request_lifecycle").bind("next"))

			_note("Completion payout: %d Club Credits. Season results and previous season history are preserved. Next season is not available here."%payout)
		return

		var credits: int=int(owner.ordinary_state.last_close.receipt.total)
		_note("Completion payout: %d Club Credits. This season and its prior history remain saved."%credits)
		if app.mixed_append_next_visible():
			_bound_button(_body,"START NEXT DUGOUT BASE SEASON",Callable(app,"request_lifecycle").bind("next"))
		return
	if Store.Owner.Mixed.closed_complete_scope(owner):
		var payout:int=int(owner.ordinary_state.last_close.receipt.total)
		if app.mixed_append_next_visible():
			_note("Completion payout: %d Club Credits. Season results and previous season history are preserved."%payout)
			_bound_button(_body,"START NEXT DUGOUT BASE SEASON",Callable(app,"request_lifecycle").bind("next"))
		else:
			_note("Completion payout: %d Club Credits. Season results and previous season history are preserved. Next season is not available here."%payout)
		return
	if app.cross_review_visible():
		_note("Your saved season and earned progress stay intact. Activate Dugout Base below when you are ready to draft your new club.")
		return
	if Store.Owner.Common.StandardChain.marked(app.season):
		_note("Season 3 earned progress is banked. Both earlier seasons remain in Season History. No completion payout. Season 4 is not available.")
		if Store.standard_closed_scores_available(owner,app.saved_bytes):
			_bound_button(_body,"SEASON 3 SCORES",_show_standard_closed_scores)
		return
	if app.standard_repeat_archive():
		if not app.standard_chain_next_visible():
			_note("Earned progress is banked. The previous season is retained. This close is not eligible for archived rollover yet.")
			return
		_note("Season 2 was abandoned without a completion payout. Starting Season 3 keeps both prior seasons and banked progress.")
	else:
		_note("Earned progress is banked. The previous season archive is preserved. No completion payout or next season." if owner.ordinary_state.has("archive") else
			"Earned progress is banked. Return to the other saved session without starting Next. Abandonment earns no completion payout.")
	var proposal: Dictionary=owner.proposal if owner.active=="dynasty" else owner.ordinary_state.proposal
	if proposal.get("kind")=="common-ordinary-league-proposal-v1":
		_label(_body,"DUGOUT BASE · SEASON %d · PROPOSAL SAVED"%int(proposal.basis.ordinal),20)
		_note("Your completed season and banked rights remain intact. Starting Dugout from this proposal is not available yet.")
		return
	var access_block: String=app.next_league_block()
	if not access_block.is_empty():
		_note(access_block+" Return or Continue preserves this saved season.")
		return
	if not owner.next_supported():
		_note(owner.next_block_reason())
		return
	if proposal.is_empty():
		_bound_button(_body,"PREPARE NEXT SEASON",Callable(app,"propose_next"))
		return
	var next: Dictionary=owner.next_preview()
	if owner.active=="ordinary":
		if next.get("kind")!="ordinary":
			_note("The saved next-season proposal is not ready. No season has been replaced.")
			return
		_label(_body,"NEXT ORDINARY SEASON · "+str(next.league).to_upper()+" · seed %d"%int(next.seed),20)
		_note("Fresh four-player draft. Shared earned player access and progression starts apply; ordinary player development does not carry over.")
		_note("Seasonal Cash, items and sponsors reset only on confirmation. The other saved session stays unchanged.")
		_bound_button(_body,"START NEXT SEASON…",Callable(app,"request_lifecycle").bind("next"))
		_note("Cancel keeps this exact saved proposal and seed. Return does not start Next.")
		return
	_label(_body,"NEXT SEASON PREVIEW · seed %d"%owner.proposal.seed,20)
	_note("Provisional balance: up to 2 above-authored points per stat; 1 mastery per known pitch.")
	_note("Paid growth is lost first. Authored floors remain. All six clubs carry forward; no new draft.")
	_note("Seasonal Cash, items and sponsors reset only when you confirm Next.")
	var needs: Dictionary=Life.overflow(owner.last_close.profiles["club:0"])
	for player: String in needs:
		var definition: PlayerDefinition=ContentDB.player_by_id.get(StringName(player))
		_note("Choose an ability to forget for "+(definition.display_name if definition!=null else player)+". This is permanent on Next, not unequipped inventory.")
		for receipt: Dictionary in needs[player]:
			var selected: bool=owner.proposal.choices["club:0"].get(player,{}).get("forget","")==receipt.id
			_bound_button(_body,("SELECTED: " if selected else "FORGET ")+Profile.ABILITY_ITEMS[receipt.item].name,Callable(app,"choose_overflow").bind(player,receipt.id))
	var preview: Dictionary=owner.next_preview()
	if preview.size()!=6:
		_note("Resolve every ability overflow before starting. The saved seed will not reroll.")
		return
	for player: String in preview["club:0"].profile.roster:
		var before: Dictionary=owner.last_close.profiles["club:0"].players[player]
		var after: Dictionary=preview["club:0"].profile.players[player]
		var changes: Array[String]=[]
		for stat: String in SeasonPlayerCatalog.STATS:
			if before.stats[stat]!=after.stats[stat]: changes.append("%s %d → %d"%[stat.capitalize(),before.stats[stat],after.stats[stat]])
		for recipe: String in before.mastery:
			if before.mastery[recipe]!=after.mastery[recipe]: changes.append("%s mastery %d → %d"%[recipe,before.mastery[recipe],after.mastery[recipe]])
		var definition: PlayerDefinition=ContentDB.player_by_id.get(StringName(player))
		_note((definition.display_name if definition!=null else player)+": "+(", ".join(changes) if not changes.is_empty() else "No stat or mastery loss"))
	_bound_button(_body,"START NEXT SEASON…",Callable(app,"request_lifecycle").bind("next"))
	_note("Cancel the confirmation to keep this exact saved preview. Loading or closing the app never applies decay.")

	label.size_flags_horizontal=Control.SIZE_EXPAND_FILL
	return label

var _render_token: RefCounted
func _screen(key: String,title: String,subtitle: String) -> void:
	_render_token=RefCounted.new()
	_round_resume=null
	_round_progress=null
	_next_round_refresh=0
	super._screen(key,title,subtitle)
func _bound_button(parent: Node,text: String,action: Callable) -> Button:
	var token: RefCounted=_render_token
	var envelope: RefCounted=app.get("career_owner")
	var state: SeasonState=app.season
	var bytes: PackedByteArray=app.get("saved_bytes").duplicate()
	return SeasonMenu._button(parent,text,func() -> void:
		if is_queued_for_deletion() or not visible or token!=_render_token: return
		if app.get("career_owner")!=envelope or app.season!=state or app.get("saved_bytes")!=bytes: return
		action.call())
const EarnedUnlocks = preload("res://src/gameplay/season/league_unlocks.gd")
func _earned_leagues(selectable: bool = true) -> void:
	_label(_body,"LEAGUES" if app.starter_player() else "LEAGUE ACCESS PREVIEW",20)
	for league: String in EarnedUnlocks.IDS:
		var state: String=app.league_entry_state(league)
		if state=="playable" and selectable:
			_bound_button(_body,("SELECTED · " if app.selected_league==league else "SELECT · ")+league.to_upper()+" BASE",Callable(app,"select_league").bind(league))
		elif state=="playable":
			_note(league.to_upper()+" BASE · EARNED" if league not in ["standard","dugout"] else league.to_upper()+" BASE · AVAILABLE")
		else:
			_note("%s BASE · %s · %s"%[league.to_upper(),"EARNED, NOT PLAYABLE" if state=="unwired" else "LOCKED",app.league_entry_reason(league)])
		var tiers: Array[String]=[]
		for tier: int in range(1,5):
			tiers.append("D%d %s"%[tier,"earned, not playable" if app.league_entry_state(league,tier)=="unwired" else "locked"])
		_note(" · ".join(tiers)+" (win the preceding difficulty in this league)")
