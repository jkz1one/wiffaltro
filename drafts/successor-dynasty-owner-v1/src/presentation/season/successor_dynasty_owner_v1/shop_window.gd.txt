## Keep the full original shop; do not imply ordinary recruiting works in Dynasty.

	return owner!=null and (Mixed.marked_state(owner.ordinary_state) or MixedAppend.marked_state(owner.ordinary_state))

func _recruit(_shop: Dictionary) -> void:
	if _installed_d1_marked() or (app.get("career_owner")!=null and app.career_owner.active=="ordinary"):
		super._recruit(_shop);return
	_label("DYNASTY ROSTER")
	_label("In-season recruiting is unavailable. Between-season replacement remains planned; its cadence and price are not enabled in this engineering build.")


const IsolatedQuote = preload("shop_quote.gd")
var _quote_active: bool = false
var _quote_generation: int = 0
var _quote_owner: RefCounted
var _quote_session: SeasonState
var _quote_source: SeasonBuild
var _quote_revision: int = -1
var _quote_build: SeasonBuild


func _clear_quote_scope() -> void:
	_quote_active = false
	_quote_owner = null
	_quote_session = null
	_quote_source = null
	_quote_revision = -1
	_quote_build = null


func _refresh() -> void:
	if not _origin_matches(): return
	# No await and no cache across event callbacks. A nested refresh invalidates
	# the outer scope rather than allowing either render to reuse its scratch.
	var nested: bool = _quote_active
	_quote_generation += 1
	var generation: int = _quote_generation
	_clear_quote_scope()
	if not nested and app != null and app.season != null:
		_quote_owner = app.get("career_owner")
		_quote_session = app.season
		_quote_source = app.season.build
		_quote_revision = _quote_source.revision()
		_quote_active = true
	super._refresh()
	if generation == _quote_generation and _origin_matches() and app.get("retained")==null:
		if _mixed_career():
			var preview: Label=_label("Development packs are available. Other purchases and roster changes are not enabled for this career yet.")
			_body.move_child(preview,0)
		if _base_archive_marked():
			var scope: Label=_label("Season Cash and Club Credits are separate." if _installed_d1_marked() else
				"The separate Loadout overlay is not part of this private Shop check. Use this shop’s own reviewed controls.")
			_body.move_child(scope,0)
		var rule: String=app.call("earnings_income_rule")
		if not rule.is_empty():
			var explanation: Label=_label(rule)
			_body.move_child(explanation,0)
	if generation == _quote_generation:
		_clear_quote_scope()


func _quote_scope_matches() -> bool:
	return (
		_quote_active and app != null
		and app.get("career_owner") == _quote_owner
		and (_installed_d1_marked() if _quote_owner==null else _quote_owner.get("session")==_quote_session)
		and app.season == _quote_session and app.season.build == _quote_source
		and _quote_source.revision() == _quote_revision
	)


func _purchase(
	text: String, command: Dictionary, description: String, context: String = ""
) -> Button:
	var quote_build: SeasonBuild
	if _quote_scope_matches():
		if _quote_build == null:
			_quote_build = _quote_source._fork()
		quote_build = _quote_build
	else:
		# Recipient dialogs and any reentrant/replaced render get a fresh scratch.
		# Quotes are presentation only: the button still invokes current _preview,
		# and confirmation still uses App.commit_shop and the strict Store path.
		_clear_quote_scope()
		quote_build = app.season.build._fork()
	var button: Button=IsolatedQuote.button(self, quote_build, text, command, description, context)
	if _mixed_career() and not Mixed.SHOP_OPS.has(str(command.get("op",""))): button.disabled=true
	return button


const ManagedLoadout = preload("loadout_ui.gd")
func _ready() -> void:
	_bind_origin()
	super._ready()
	if _base_archive_marked():
		if _confirm.confirmed.is_connected(_commit): _confirm.confirmed.disconnect(_commit)
		_confirm.confirmed.connect(_base_archive_confirm_signal)
		_confirm.canceled.connect(_base_archive_cancel_quote)
		_confirm.close_requested.connect(_base_archive_cancel_quote)
		if _back.pressed.is_connected(_close): _back.pressed.disconnect(_close)
		_back.pressed.connect(_base_archive_close_signal)
		if close_requested.is_connected(_close): close_requested.disconnect(_close)
		close_requested.connect(_base_archive_close_signal)
	# The inherited shop creates its own utility. Replace it synchronously before
	# any input dispatch, not just the separate App entry.
	for child: Node in get_children():
		if child is SeasonLoadoutUI:
			remove_child(child)
			child.queue_free()
	var utility: SeasonLoadoutUI = ManagedLoadout.new()
	utility.app = app
	utility.shop = self
	add_child(utility)


var _origin_owner: RefCounted
var _origin_session: SeasonState
var _origin_bytes: PackedByteArray=PackedByteArray()
var _view_generation: int=0
func _bind_origin() -> void:
	_origin_owner=app.get("career_owner");_origin_session=app.season
	_origin_bytes=app.get("saved_bytes").duplicate()
	_view_generation+=1
func _origin_matches() -> bool:
	return not is_queued_for_deletion() and app!=null and _origin_session!=null \
		and app.get("career_owner")==_origin_owner and app.season==_origin_session \
		and (_installed_d1_marked() if _origin_owner==null else _origin_owner.get("session")==_origin_session) \
		and app.get("saved_bytes")==_origin_bytes
func _clear() -> void:
	_view_generation+=1
	_base_archive_cancel_quote()
	super._clear()
func _button(text: String,action: Callable) -> Button:
	var generation: int=_view_generation
	var control: Array[Button]=[]
	var button: Button=super._button(text,func() -> void:
		if not visible or generation!=_view_generation or not _origin_matches(): return
		if _base_archive_marked():
			if control.is_empty() or not is_instance_valid(control[0]) or control[0].disabled \
					or not control[0].is_visible_in_tree() or _confirm.visible: return
			_base_archive_cancel_quote()
			_base_archive_action=RefCounted.new()
		action.call()
		if _base_archive_action!=null and not _confirm.visible and not _base_archive_nested_visible():
			_base_archive_cancel_quote())
	control.append(button)
	return button
func _preview(command: Dictionary,description: String) -> void:
	if not visible or not _origin_matches(): return
	if _mixed_career() and not Mixed.SHOP_OPS.has(str(command.get("op",""))): return
	if _base_archive_marked() and _base_archive_action==null: return
	super._preview(command,description)
	if _base_archive_visible() and _base_archive_action!=null and _confirm.visible and not _pending.is_empty():
		_base_archive_quote=_pending.duplicate(true)
func _commit() -> void:
	if not visible or not _origin_matches(): return
	var departing: bool=_pending.get("op")=="reserve_offer"
	var ok: bool
	var marked: bool=_base_archive_marked()
	if marked:
		if not _base_archive_confirming or not ClubCareer.same(_base_archive_quote,_pending): return
		var command: Dictionary=_base_archive_quote.duplicate(true)
		_base_archive_cancel_quote()
		ok=_base_archive_dispatch(command)
	else:
		ok=app.commit_shop(_pending)
	_pending.clear()
	if ok: _bind_origin()
	if ok and departing:
		if marked: _base_archive_close_signal()
		else: _close()
		return
	_notice="Saved. Your build is ready for the next game." if ok else app.notice
	if ok and LeagueRules.academy(app.season.league_rules) and app.season.build.pack_pending():
		_notice="Saved. Choose or skip the remaining Academy pack choices before the next game."
	_refresh()
func _retry_checkpoint() -> void:
	if not _origin_matches(): return
	var candidate=app.get("retained_envelope")
	app.call("retry_save")
	if candidate!=null and app.get("career_owner")==candidate and app.get("retained")==null: _bind_origin()
	_notice=app.notice if app.get("retained")!=null else "Saved. Your build is ready for the next game."
	_refresh()
func _close() -> void:
	if not _origin_matches(): hide();queue_free();return
	if _base_archive_marked():
		if not _base_archive_closing: return
		_base_archive_cancel_quote()
		if SeasonSchoolSponsors.has_credit(app.season.build.view().shop) \
				and not _base_archive_dispatch(_request("leave_shop")):
			_notice=app.notice
			_refresh()
			return
		if _installed_d1_marked(): app.call("close_installed_dugout_d1_shop",self)
		else:
			app.show_season()
			queue_free()
		return
	super._close()
func _close_retained() -> void:
	if not _origin_matches(): hide();queue_free();return
	super._close_retained()


func _held(receipts: Array) -> void:
	var build: SeasonBuild=app.season.build
	if build._league_rules.get("league")=="dugout":
		_dugout_capacity(build)
	super._held(receipts)


# Discovery only. Current quotes and the original sponsor/overflow controls
# remain the sole mutation path, with their exact receipt/nomination proof.
func _dugout_capacity(build: SeasonBuild) -> void:
	var base: Dictionary=LeagueRules.capacities(build._league_rules)
	var catalog: Dictionary=SeasonSponsorCatalog.ownership_catalog()
	var small: Dictionary=catalog["G02"]
	var major: Dictionary=catalog["F09"]
	var held: int=int(base.held)
	var raised: int=held+int(small.held_delta)
	var lowered: int=held+int(major.held_delta)
	var combined: int=raised+int(major.held_delta)
	_label("DUGOUT SHARED BAG • %d base places for development cards and tactical supplies" % held)
	_label("Small Batch: %d held (up to %d sponsors). Double Major: %d held. Both: %d held." %
		[raised,int(small.sponsor_cap),lowered,combined])
	var shop: Dictionary=build.view().shop
	var wallet: Dictionary=build._bank.view()
	var pool: Dictionary=SeasonEarnedSponsors.eligible(build)
	var used: Array=[] if build._batch_start==null else SeasonSmallBatch.combine(build._batch_start,build._batch_used)
	_label("Small Batch Supply • %s · %d / %d required types" %
		[_dugout_modifier_state("G02",wallet,shop,pool),used.size(),SeasonSmallBatch.TYPES.size()])
	_label("Use Grip Tape, Recovery Pack and Swing Plan in completed games.")
	_label("Double Major Academy • %s" % _dugout_modifier_state("F09",wallet,shop,pool))
	_label("One Primary Fielder must make a clean ground out and a clean air catch in the same completed game.")
	_label("Earned access adds future shop eligibility, not a free copy or guaranteed offer. Resolve any capacity decrease before confirming.")


func _dugout_modifier_state(id: String, wallet: Dictionary, shop: Dictionary, pool: Dictionary) -> String:
	if wallet.sponsors.any(func(receipt: Dictionary)->bool: return receipt.item==id): return "ACTIVE"
	if id in shop.offers.values(): return "OFFERED THIS VISIT"
	return "ELIGIBLE FOR FUTURE OFFERS" if pool.has(id) else "LOCKED"

# Only an actual visible stock control can begin a bound marked preview.
# Nested native recipient/offer resolution may complete the same preview.
var _base_archive_action: RefCounted
var _base_archive_quote: Dictionary={}
var _base_archive_confirming: bool=false
var _base_archive_closing: bool=false
var _base_archive_ticket: RefCounted
var _base_archive_command: Dictionary={}

func _installed_d1_marked() -> bool:
	return app!=null and bool(app.call("installed_dugout_d1_shop_window",self))

func _base_archive_marked() -> bool:
	return app!=null and (app.base_archive_qualification_view() or _installed_d1_marked())

func _base_archive_visible() -> bool:
	return _base_archive_marked() and visible and _origin_matches() and get_parent()==app.menu

func _base_archive_nested_visible() -> bool:
	for child: Node in find_children("*","Window",true,false):
		if child!=_confirm and child.visible and not child.is_queued_for_deletion(): return true
	return false

func _base_archive_cancel_quote() -> void:
	_base_archive_action=null
	_base_archive_quote={}

func _base_archive_confirm_signal() -> void:
	if not _base_archive_visible() or _base_archive_action==null or _base_archive_quote.is_empty(): return
	_base_archive_confirming=true
	_commit()
	_base_archive_confirming=false

func _base_archive_close_signal() -> void:
	if not _base_archive_visible(): return
	_base_archive_closing=true
	_close()
	_base_archive_closing=false

func _base_archive_dispatch(command: Dictionary) -> bool:
	if not _base_archive_visible() or (not _base_archive_confirming and not _base_archive_closing): return false
	_base_archive_ticket=RefCounted.new()
	_base_archive_command=command.duplicate(true)
	var saved: bool=app.commit_base_archive_window(self,_base_archive_ticket)
	_base_archive_ticket=null
	_base_archive_command={}
	return saved

func take_base_archive_shop(ticket: RefCounted) -> Dictionary:
	if ticket==null or ticket!=_base_archive_ticket or not _base_archive_visible() \
			or (not _base_archive_confirming and not _base_archive_closing): return {}
	_base_archive_ticket=null
	var result: Dictionary=_base_archive_command
	_base_archive_command={}
	return result
