## Keep the full original shop; do not imply ordinary recruiting works in Dynasty.
func _recruit(_shop: Dictionary) -> void:
	if app.get("career_owner")!=null and app.career_owner.active=="ordinary":
		super._recruit(_shop);return
	_label("DYNASTY ROSTER")
	_label("In-season recruiting is unavailable. Between-season replacement remains planned; its cadence and price are not enabled in this engineering build.")


const IsolatedQuote = preload("shop_quote.gd")
var _quote_active: bool = false
var _quote_generation: int = 0
var _quote_owner: RefCounted
var _quote_session: SeasonState
var _quote_source: SeasonBuild
var _quote_revision: int = -1
var _quote_build: SeasonBuild


func _clear_quote_scope() -> void:
	_quote_active = false
	_quote_owner = null
	_quote_session = null
	_quote_source = null
	_quote_revision = -1
	_quote_build = null


func _refresh() -> void:
	if not _origin_matches(): return
	# No await and no cache across event callbacks. A nested refresh invalidates
	# the outer scope rather than allowing either render to reuse its scratch.
	var nested: bool = _quote_active
	_quote_generation += 1
	var generation: int = _quote_generation
	_clear_quote_scope()
	if not nested and app != null and app.season != null:
		_quote_owner = app.get("career_owner")
		_quote_session = app.season
		_quote_source = app.season.build
		_quote_revision = _quote_source.revision()
		_quote_active = true
	super._refresh()
	if generation == _quote_generation:
		_clear_quote_scope()


func _quote_scope_matches() -> bool:
	return (
		_quote_active and app != null and _quote_owner != null
		and app.get("career_owner") == _quote_owner
		and _quote_owner.get("session") == _quote_session
		and app.season == _quote_session and app.season.build == _quote_source
		and _quote_source.revision() == _quote_revision
	)


func _purchase(
	text: String, command: Dictionary, description: String, context: String = ""
) -> Button:
	var quote_build: SeasonBuild
	if _quote_scope_matches():
		if _quote_build == null:
			_quote_build = _quote_source._fork()
		quote_build = _quote_build
	else:
		# Recipient dialogs and any reentrant/replaced render get a fresh scratch.
		# Quotes are presentation only: the button still invokes current _preview,
		# and confirmation still uses App.commit_shop and the strict Store path.
		_clear_quote_scope()
		quote_build = app.season.build._fork()
	return IsolatedQuote.button(self, quote_build, text, command, description, context)


const ManagedLoadout = preload("loadout_ui.gd")
func _ready() -> void:
	_bind_origin()
	super._ready()
	# The inherited shop creates its own utility. Replace it synchronously before
	# any input dispatch, not just the separate App entry.
	for child: Node in get_children():
		if child is SeasonLoadoutUI:
			remove_child(child)
			child.queue_free()
	var utility: SeasonLoadoutUI = ManagedLoadout.new()
	utility.app = app
	utility.shop = self
	add_child(utility)


var _origin_owner: RefCounted
var _origin_session: SeasonState
var _origin_bytes: PackedByteArray=PackedByteArray()
var _view_generation: int=0
func _bind_origin() -> void:
	_origin_owner=app.get("career_owner");_origin_session=app.season
	_origin_bytes=app.get("saved_bytes").duplicate()
	_view_generation+=1
func _origin_matches() -> bool:
	return not is_queued_for_deletion() and app!=null and _origin_owner!=null \
		and app.get("career_owner")==_origin_owner and app.season==_origin_session \
		and _origin_owner.get("session")==_origin_session and app.get("saved_bytes")==_origin_bytes
func _clear() -> void:
	_view_generation+=1
	super._clear()
func _button(text: String,action: Callable) -> Button:
	var generation: int=_view_generation
	return super._button(text,func() -> void:
		if not visible or generation!=_view_generation or not _origin_matches(): return
		action.call())
func _preview(command: Dictionary,description: String) -> void:
	if not visible or not _origin_matches(): return
	super._preview(command,description)
func _commit() -> void:
	if not visible or not _origin_matches(): return
	var departing: bool=_pending.get("op")=="reserve_offer"
	var ok: bool=app.commit_shop(_pending)
	_pending.clear()
	if ok: _bind_origin()
	if ok and departing:
		_close();return
	_notice="Saved. Your build is ready for the next game." if ok else app.notice
	if ok and LeagueRules.academy(app.season.league_rules) and app.season.build.pack_pending():
		_notice="Saved. Choose or skip the remaining Academy pack choices before the next game."
	_refresh()
func _retry_checkpoint() -> void:
	if not _origin_matches(): return
	var candidate=app.get("retained_envelope")
	app.call("retry_save")
	if candidate!=null and app.get("career_owner")==candidate and app.get("retained")==null: _bind_origin()
	_notice=app.notice if app.get("retained")!=null else "Saved. Your build is ready for the next game."
	_refresh()
func _close() -> void:
	if not _origin_matches(): hide();queue_free();return
	super._close()
func _close_retained() -> void:
	if not _origin_matches(): hide();queue_free();return
	super._close_retained()


func _held(receipts: Array) -> void:
	var build: SeasonBuild=app.season.build
	if build._league_rules.get("league")=="dugout":
		_dugout_capacity(build)
	super._held(receipts)


# Discovery only. Current quotes and the original sponsor/overflow controls
# remain the sole mutation path, with their exact receipt/nomination proof.
func _dugout_capacity(build: SeasonBuild) -> void:
	var base: Dictionary=LeagueRules.capacities(build._league_rules)
	var catalog: Dictionary=SeasonSponsorCatalog.ownership_catalog()
	var small: Dictionary=catalog["G02"]
	var major: Dictionary=catalog["F09"]
	var held: int=int(base.held)
	var raised: int=held+int(small.held_delta)
	var lowered: int=held+int(major.held_delta)
	var combined: int=raised+int(major.held_delta)
	_label("DUGOUT SHARED BAG • %d base places for development cards and tactical supplies" % held)
	_label("Small Batch: %d held (up to %d sponsors). Double Major: %d held. Both: %d held." %
		[raised,int(small.sponsor_cap),lowered,combined])
	var shop: Dictionary=build.view().shop
	var wallet: Dictionary=build._bank.view()
	var pool: Dictionary=SeasonEarnedSponsors.eligible(build)
	var used: Array=[] if build._batch_start==null else SeasonSmallBatch.combine(build._batch_start,build._batch_used)
	_label("Small Batch Supply • %s · %d / %d required types" %
		[_dugout_modifier_state("G02",wallet,shop,pool),used.size(),SeasonSmallBatch.TYPES.size()])
	_label("Use Grip Tape, Recovery Pack and Swing Plan in completed games.")
	_label("Double Major Academy • %s" % _dugout_modifier_state("F09",wallet,shop,pool))
	_label("One Primary Fielder must make a clean ground out and a clean air catch in the same completed game.")
	_label("Earned access adds future shop eligibility, not a free copy or guaranteed offer. Resolve any capacity decrease before confirming.")


func _dugout_modifier_state(id: String, wallet: Dictionary, shop: Dictionary, pool: Dictionary) -> String:
	if wallet.sponsors.any(func(receipt: Dictionary)->bool: return receipt.item==id): return "ACTIVE"
	if id in shop.offers.values(): return "OFFERED THIS VISIT"
	return "ELIGIBLE FOR FUTURE OFFERS" if pool.has(id) else "LOCKED"
