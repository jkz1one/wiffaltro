const StarterDugoutReview = preload("res://src/gameplay/season/successor_dynasty_owner_v1/starter_dugout_review.gd")
var _starter_dugout_review: Node

func starter_dugout_review_running() -> bool:
	return is_instance_valid(_starter_dugout_review)

func starter_dugout_review_visible() -> bool:
	return StarterDugoutReview.supported() and _available() and career_owner!=null \
		and career_owner.session==season and not saved_bytes.is_empty() \
		and not career_owner.ordinary_state.has("archive") \
		and Owner.Common.cross_target_open(career_owner,"dugout")

func begin_starter_dugout_review() -> void:
	if not starter_dugout_review_visible() or not _owned(): return
	var child:=StarterDugoutReview.new()
	_starter_dugout_review=child
	add_child(child)
	child.completed.connect(_finish_starter_dugout_review.bind(child,career_owner,saved_bytes),CONNECT_ONE_SHOT)
	if not child.start(saved_bytes):
		_finish_starter_dugout_review("refused",child,career_owner,saved_bytes)
		return
	notice="Checking the saved Standard season and preparing a Dugout Base preview. Nothing will be saved."
	show_season()

func _finish_starter_dugout_review(outcome: String, child: Node, owner: RefCounted, revision: PackedByteArray) -> void:
	if _starter_dugout_review!=child: return
	_drop_starter_dugout_review()
	if career_owner!=owner or saved_bytes!=revision or outcome=="stale":
		notice="The saved career changed. Continue again before reviewing. Nothing was saved."
	elif outcome=="verified":
		notice="Dugout Base preview verified: Season 2, a fresh draft and season Cash, with banked career progress retained. Starting it is not available here. Nothing was saved."
	else:
		notice="The Dugout Base preview could not be verified. Nothing was saved."
	show_season()

func _drop_starter_dugout_review() -> void:
	var child: Node=_starter_dugout_review
	_starter_dugout_review=null
	if not is_instance_valid(child): return
	child.cancel()
	child.queue_free()

func cancel_starter_dugout_review() -> void:
	if not starter_dugout_review_running(): return
	_drop_starter_dugout_review()
	notice="Dugout Base review canceled. Nothing was saved."
	show_season()

func _exit_tree() -> void:
	Store.cancel_base_archive_cpu(self)
	_drop_mixed_continue()
	_drop_starter_dugout_review()
	Store.cancel_cross_activation(career_owner)
	_expire_d1_proof()
	var checking: Node=_d1_check
	_d1_check=null
	if is_instance_valid(checking): checking.cancel()
	_cancel_exit()
	_cancel_lifecycle()
	_loading_generation += 1
	if is_inside_tree(): get_tree().auto_accept_quit = _prior_auto_quit
	_content_roots.clear()

func _begin_private_pair_settlement() -> void:
	if not _available() or not Store.eligible_private_pair_settlement(career_owner): return
	var kind: String=D1Check.pair_settlement_kind(season.round_index)
	if not D1Check.supported(kind):
		notice="Background settlement is not supported by this engineering build. Nothing was written."
		show_season()
		return
	_expire_d1_proof()
	var check:=D1Check.new()
	_d1_check=check
	_pair_settling=true
	add_child(check)
	check.completed.connect(_complete_private_pair_settlement.bind(check,_d1_epoch,saved_bytes),CONNECT_ONE_SHOT)
	if not check.start(D1Check.T._source_digest(saved_bytes),0,kind):
		_complete_private_pair_settlement("invalid",{},check,_d1_epoch,saved_bytes)
		return
	notice="Settling the exact Doubleheader leg in the background…"
	show_season()

func _complete_private_pair_settlement(outcome: String,_proof: Dictionary,check: Node,epoch: int,expected: PackedByteArray) -> void:
	if _d1_check!=check or epoch!=_d1_epoch: return
	_d1_check=null
	_pair_settling=false
	check.queue_free()
	if _d1_close_after_check:
		_d1_close_after_check=false
		_quit_now()
		return
	if outcome=="qualified":
		var started:=Time.get_ticks_msec()
		outcome=Store.accept_private_pair_settlement(career_owner,expected,check) if expected==saved_bytes else "stale"
		_pair_settle_write_ms=Time.get_ticks_msec()-started
	if outcome in ["saved","renamed_unverified","unverified_io","stale"]:
		career_owner=null; season=null; saved_bytes=PackedByteArray()
	notice={
		"saved":"Doubleheader leg settled. Use Continue Saved Career to verify and reopen the save.",
		"renamed_unverified":"The save was replaced, but its contents could not be verified. Use Continue to check it before playing.",
		"unverified_io":"Settlement could not be verified after a save error. Use Continue to check the save before retrying.",
		"obstructed":"Save path blocked before writing. Resolve the obstruction, then retry settlement.",
		"cancelled":"Settlement cancelled before writing.",
		"stale":"The saved career changed. No settlement was written by this action.",
		"refused":"The pending Doubleheader leg did not qualify. Nothing was written."
	}.get(outcome,"The pending Doubleheader leg could not be verified. Nothing was written.")
	show_season()

	if pin.owner!=career_owner or pin.season!=season or pin.bytes!=saved_bytes or pin.lab!=lab or pin.state!=lab._match_state or not Codec.same(pin.fixture,season.pending_fixture()) or not season.physical.pending.is_empty(): return

	if human.is_empty(): return
	var pending: Dictionary={"human":human.duplicate(true),"reports":[],"binding":pin.binding.duplicate(true)}
	var factory=load("res://src/gameplay/season/successor_live_play_v1/paired_first_projection.gd")
	if round_index>0 and factory.human_enabled(round_index):
		var expected: PackedByteArray=saved_bytes.duplicate()

		match round_index:
			1: ticket=Store._prepare_paired_return_human(career_owner,expected,pending)
			2: ticket=Store._prepare_paired_next_first_human(career_owner,expected,pending)
			3: ticket=Store._prepare_paired_next_return_human(career_owner,expected,pending)
			4: ticket=Store._prepare_paired_third_first_human(career_owner,expected,pending)
			5: ticket=Store._prepare_paired_third_return_human(career_owner,expected,pending)

			_: ticket=Store._prepare_paired_human_checkpoint(career_owner,expected,pending,season.round_index)
		if ticket!=null:
			_retain_prepared(ticket,Store.prepared_candidate(ticket),expected)
			_paired_first_pin.clear()

	_expire_d1_proof()
	var check:=D1Check.new()
	_d1_check=check
	_cross_reviewing=true
	add_child(check)
	check.completed.connect(_complete_cross_review.bind(check,_d1_epoch),CONNECT_ONE_SHOT)
	_cross_third_write=_third_activation()
	var kind: String=_cross_activation_kind()
	if not check.start(D1Check.T._source_digest(saved_bytes),0,kind):
		_complete_cross_review("invalid",{},check,_d1_epoch)
		return
	notice="Preparing the exact Dugout Base draft in the background…"
	show_season()

func _complete_cross_review(outcome: String, _proof: Dictionary, check: Node, epoch: int) -> void:
	if _d1_check!=check or epoch!=_d1_epoch: return
	_d1_check=null
	_cross_reviewing=false
	check.queue_free()
	if _d1_close_after_check:
		_d1_close_after_check=false
		_quit_now()
		return
	if outcome=="qualified":
		var started:=Time.get_ticks_msec()
		var result:=Store.accept_cross_activation(career_owner,saved_bytes,check)
		_cross_write_ms=Time.get_ticks_msec()-started
		_finish_cross_activation(result)
		return
	notice={
		"cancelled":"League transition check cancelled. Your career is unchanged.",
		"stale":"The saved career changed during preparation. No transition was written.",
		"refused":"The saved league transition did not qualify. Your career is unchanged."
	}.get(outcome,"The saved league transition could not be verified. Your career is unchanged.")
	show_season()

func retry_cross_activation() -> void:
	if not _cross_retry: return
	var started:=Time.get_ticks_msec()
	var result:=Store.retry_cross_activation(career_owner,saved_bytes)
	_cross_write_ms=Time.get_ticks_msec()-started
	_finish_cross_activation(result)

func cancel_cross_retry() -> void:
	if not _cross_retry: return
	Store.cancel_cross_activation(career_owner)
	_cross_retry=false
	notice="Dugout activation cancelled. No transition was written."
	show_season()

func _finish_cross_activation(outcome: String) -> void:
	_cross_retry=outcome=="obstructed"
	if outcome in ["saved","renamed_unverified"]:
		career_owner=null; season=null; saved_bytes=PackedByteArray()
		var use_private_reader: bool=not _cross_third_write and D1Check.supported(D1Check.CROSS_READ_KIND)
		notice=("Dugout Base activated. Use Private Continue Archived Dugout Draft to draft your club." if use_private_reader else "Dugout Base activated. Use Continue to draft your club.") if outcome=="saved" else (
			"The save was replaced, but its contents could not be verified. Use Continue to check the saved career before playing.")
	elif _cross_retry:
		notice="The exact Dugout draft could not be saved. Resolve the save obstruction, then retry."
	elif outcome=="stale":
		notice="The saved career changed. No transition was written."
	else:
		notice="The prepared Dugout draft could not be verified. No transition was written."
	show_season()

	check.completed.connect(_complete_cross_read.bind(check,_d1_epoch,source),CONNECT_ONE_SHOT)
	if source.is_empty() or not check.start(Owner._hash(source),0,D1Check.CROSS_READ_KIND):
		_complete_cross_read("invalid",{},check,_d1_epoch,source)
		return
	notice="Checking the saved archived Dugout draft in the background…"
	show_season()

func _complete_cross_read(outcome: String,_proof: Dictionary,check: Node,epoch: int,source: PackedByteArray) -> void:
	if _d1_check!=check or epoch!=_d1_epoch: return
	_d1_check=null
	_cross_reading=false
	check.queue_free()
	if _d1_close_after_check:
		_d1_close_after_check=false
		_quit_now()
		return
	if outcome=="qualified":
		var start:=Time.get_ticks_msec()
		var owner:=Store.accept_cross_draft_read(source,check)
		_cross_read_ms=Time.get_ticks_msec()-start
		if owner!=null:
			career_owner=owner;season=owner.session;saved_bytes=source
			notice="Archived Dugout draft ready."
			show_season()
			return
		outcome="invalid"
	notice={
		"refused":"This slot is outside the private Dugout draft route. Use normal Continue for other careers.",
		"cancelled":"Dugout draft check cancelled. Your save is unchanged.",
		"stale":"The saved career changed during verification. Try Continue again."
	}.get(outcome,"The archived Dugout draft could not be verified. No save was changed.")
	show_season()

func _complete_mixed_continue(outcome: String,process_node: Node,epoch: int,expected: PackedByteArray) -> void:
	if process_node!=_mixed_continue or epoch!=_mixed_continue_epoch: return
	var candidate: RefCounted
	if outcome=="verified": candidate=await Store.accept_mixed_continue(expected,process_node)
	if process_node!=_mixed_continue or epoch!=_mixed_continue_epoch: return
	_drop_mixed_continue()
	if candidate!=null:
		career_owner=candidate; season=candidate.session; saved_bytes=expected; notice=""
	elif outcome=="stale" or not Store._write_revision_matches(expected,true):
		notice="The saved career changed during verification. Try Continue again."
	else:
		notice="The saved career could not be verified. Nothing was saved."
	show_season()

func reload_save() -> void:
	if not _available(): return
	_expire_d1_proof()
	var kind:=Store.standard_d1_slot_kind()
	if not kind.is_empty() and not Store.LeagueAdmission.starter_feature():
		_continue_standard_d1(kind)
		return
	_busy = true
	var before: PackedByteArray = Store.raw()
	if Store._base_archive_qa() and Store._base_archive_family(before):
		if not _adopt_base_archive_view(Store.read_base_archive_app_source(),before):
			notice="The marked career could not be verified. No files were changed."
		_busy=false; show_season()
		return
	var candidate: RefCounted = Store.read()
	if candidate != null and Store.raw() == before:
		career_owner = candidate; season = career_owner.session; saved_bytes = before; notice = ""
		_base_archive_view=false
	else: notice = Store.last_error
	_busy = false; show_season()

func draft(id: String) -> void:
	if career_owner!=null and not career_owner.playable(): return
	if not _available() or season == null or season.phase != SeasonState.Phase.DRAFT or not season.offers().has(id): return
	if base_archive_qualification_view():
		_busy=true
		var committed: bool=Store.write_base_archive_pick(career_owner,saved_bytes,id)
		var written: Dictionary=Store.base_archive_app_result(saved_bytes) if committed else {}
		if written.is_empty() or not _adopt_base_archive_view(written.owner,written.bytes):
			notice="This choice could not be saved. Use Continue to verify the marked career."
		_busy=false; show_season()
		return
	if not _owned():
		notice = "The save changed outside this view. Reload before drafting."
		show_season(); return
	_busy = true
	# One explicit detached operation fork, no historical reconstruction.
	var candidate: SeasonState = season.fork_live()
	if candidate != null and not candidate.choose_player(id): candidate = null
	_busy = false
	if candidate == null:
		notice = "This choice could not be applied. Your current saved roster is unchanged."
		show_season(); return
	_retain(candidate,saved_bytes)
func _retain(candidate: SeasonState, expected: PackedByteArray) -> void:
	if retained != null or candidate == null: return
	if career_owner != null and career_owner.session != season: return
	var envelope: RefCounted = Owner.new()
	envelope.session = candidate
	if career_owner != null:
		if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(career_owner,Owner.DATA_FIELDS)): return
	else: return
	_retain_envelope(envelope,expected)

func _retain_envelope(envelope: RefCounted, expected: PackedByteArray) -> void:
	if _busy or Store.saving or retained!=null or envelope==null or envelope.session==null: return
	if expected != saved_bytes: return
	if career_owner!=null and career_owner.session!=season: return
	var ticket: RefCounted = Store.prepare_and_write(career_owner,envelope,expected)
	if ticket == null:
		notice = "The candidate could not be represented consistently. Saved progress is unchanged."
		show_season(); return
	_retain_prepared(ticket,envelope,expected)

func _retain_prepared(ticket: RefCounted, envelope: RefCounted, expected: PackedByteArray) -> void:
	# Same synchronous producer handoff; Store retry/promote remain authoritative.
	if _busy or Store.saving or retained != null or envelope == null or envelope.session == null: return
	if expected != saved_bytes or (career_owner != null and career_owner.session != season): return
	if not Store.prepared_binding(ticket,career_owner,envelope,expected): return
	var candidate: SeasonState = envelope.session
	retained = candidate; retained_owner = season; retained_expected = expected.duplicate()
	retained_envelope = envelope; retained_owner_envelope = career_owner
	_save_ticket = ticket
	retained_before = {}
	retained_lab = lab; retained_state = lab._match_state if is_instance_valid(lab) else null
	retained_saved_bytes = Store.written_bytes(ticket)
	retained_saved = not retained_saved_bytes.is_empty()
	_busy = true
	_finish_save()

func retry_save() -> void:
	Store.forget_projection() # Retry never transfers a warm projection lease.
	if _busy or retained == null or season != retained_owner: return
	if career_owner != retained_owner_envelope or retained_envelope == null or retained_envelope.session != retained: return
	if career_owner != null and career_owner.session != season: return
	if lab != retained_lab or (is_instance_valid(lab) and lab._match_state != retained_state): return
	if retained_saved and not Store.prepared_matches(_save_ticket,retained_owner_envelope,retained_envelope,retained_expected): return
	_busy = true
	if not retained_saved:
		retained_saved = Store.write_prepared(_save_ticket)
		if retained_saved: retained_saved_bytes = Store.written_bytes(_save_ticket)
	_finish_save()

func rollover_close_recovery_pending() -> bool:
	return not _rollover_close_recovery.is_empty()

func recover_rollover_close() -> void:
	if not rollover_close_recovery_pending() or _busy or Store.saving or is_instance_valid(lab) \
		or is_instance_valid(ai_runner) or career_owner!=null or season!=null: return
	_busy = true
	var candidate: RefCounted=Store.read_rollover_close_recovery(_rollover_close_recovery)
	if candidate==null:
		notice=Store.last_error
	else:
		career_owner=candidate; season=candidate.session; saved_bytes=_rollover_close_recovery
		_rollover_close_recovery=PackedByteArray()
		notice=""
	_busy = false
	show_season()

func _finish_save() -> void:
	if not retained_saved:
		var replacement:=Store.take_rollover_close_recovery(_save_ticket,retained_owner_envelope,retained_envelope,retained_expected)
		if not replacement.is_empty():
			_rollover_close_recovery=replacement
			_save_ticket=null
			career_owner=null; season=null; saved_bytes=PackedByteArray()
			retained_envelope=null; retained_owner_envelope=null
			retained=null; retained_owner=null; retained_lab=null; retained_state=null
			retained_expected=PackedByteArray(); retained_before={}
			retained_saved_bytes=PackedByteArray()
			notice="The season save was replaced, but its contents could not be verified. Use Continue to check it."
			_busy=false; show_season(); return
		Store.forget_projection()
		notice = Store.last_error; _busy = false; show_season(); return
	if Store.raw() != retained_saved_bytes:
		Store.forget_projection()
		notice = "Saved candidate changed before promotion. Exact progress remains retained."
		_busy = false; return
	if is_instance_valid(lab) and not _close_match():
		Store.forget_projection()
		notice = "Saved. Original match cleanup must finish before continuing."
		_busy = false; return
	# A successful close consumed the original pinned lab. Record that boundary
	# even if a synchronous callback attached another lab: never promote or
	# close the replacement. Retry can continue only after it is removed.
	retained_lab = null; retained_state = null
	if is_instance_valid(lab):
		Store.forget_projection()
		notice = "Saved candidate retained. A replacement match blocks promotion."
		_busy = false; return
	if season != retained_owner or career_owner != retained_owner_envelope or (career_owner != null and career_owner.session != season) or retained_envelope.session != retained or Store.raw() != retained_saved_bytes:
		Store.forget_projection()
		notice = "Saved candidate retained; ownership changed during cleanup."
		_busy = false; return
	if not Store.promote(_save_ticket,retained_owner_envelope,retained_envelope,retained_expected):
		Store.forget_projection()
		notice = "Saved candidate remains retained; its private admission changed."
		_busy = false; return
	_save_ticket = null
	career_owner = retained_envelope; season = career_owner.session; saved_bytes = retained_saved_bytes.duplicate()
	retained_envelope = null; retained_owner_envelope = null
	retained = null; retained_owner = null; retained_lab = null; retained_state = null
	retained_expected = PackedByteArray(); retained_before = {}
	retained_saved = false; retained_saved_bytes = PackedByteArray()
	retained_human.clear(); retained_report.clear(); _clear_npc_pins()
	_result_recorded = false; _result_saved = false
	notice = ""; _busy = false; show_season()

func _request_exit(mode: String) -> void:
	if mode not in ["leave", "quit"] or _exit_blocked():
		return
	if not is_instance_valid(lab):
		if mode == "quit":
			_cancel_exit()
			_quit_now()
		return
	if lab._match_state == null or not is_instance_valid(_exit_dialog):
		return
	if not _exit_mode.is_empty():
		return # Repeated/different requests cannot retarget an open confirmation.
	_exit_mode = mode
	_exit_season = season
	_exit_envelope = career_owner
	_exit_lab = lab
	_exit_state = lab._match_state
	_exit_dialog.title = "Quit unfinished game" if mode == "quit" else "Leave unfinished game"
	_exit_dialog.dialog_text = "Quit? This unfinished game will restart. Saved progress stays." if mode == "quit" else "Leave this unfinished game? It will restart. Saved progress stays."
	_exit_dialog.popup_centered(Vector2i(560, 180))
	_exit_dialog.get_cancel_button().grab_focus()


func _cancel_exit() -> void:
	_exit_mode = ""
	_exit_season = null
	_exit_envelope = null
	_exit_lab = null
	_exit_state = null
	if is_instance_valid(_exit_dialog):
		_exit_dialog.hide()


func _confirm_exit() -> void:
	var mode: String = _exit_mode
	var exiting_season: SeasonState = _exit_season
	var envelope: RefCounted = _exit_envelope
	var d1: bool=not installed_dugout_d1_draft.is_empty() and Store.installed_dugout_d1_match_bound(self)
	var matches: bool = mode in ["leave", "quit"] and season == _exit_season \
		and career_owner == envelope and (d1 or (career_owner != null and career_owner.session == season)) \
		and is_instance_valid(lab) and lab == _exit_lab and lab._match_state == _exit_state
	_cancel_exit() # Consume once before cleanup or any potential callback.
	if not matches or _exit_blocked():
		return
	if not _close_match(false):
		notice = "The original match could not close safely. It remains available."
		return
	if season != exiting_season or career_owner != envelope or (not d1 and (career_owner == null or career_owner.session != season)) or is_instance_valid(lab) or _exit_blocked():
		return # Cleanup may have invoked callbacks; never quit a replacement owner.
	if d1: season=null
	if mode == "quit":
		_quit_now()
	else:
		show_season()


func _quit_now() -> void:
	# Sole final OS action; tests may spy on it without bypassing any exit guard.
	get_tree().quit()


class StarterLocalRecords extends PlayRecordExport:
	var _lab: WeakRef
	var _state: MatchState

	static func ordinary_live(opened: PitchBatLab, state: MatchState) -> bool:
		return is_instance_valid(opened) and opened.get_script()==PitchBatLab \
			and opened._managed_match and opened._match_mode and state!=null \
			and opened._match_state==state and opened._automation==null \
			and state.stadium.is_empty() and opened.get_meta("stadium_snapshot",{})=={} \
			and not opened.ac02_startup_pending() and not opened._ac02_issued_history \
			and opened._ac02_issued_owner==null

	func bind(opened: PitchBatLab, state: MatchState) -> void:
		_lab = weakref(opened)
		_state = state

	func save(records: Array[PlayRecord], _field: FieldDefinition, stadium: Dictionary = {}) -> Error:
		var opened: PitchBatLab = _lab.get_ref() as PitchBatLab if _lab!=null else null
		if not ordinary_live(opened,_state) or opened._record_export!=self \
			or not is_same(records,opened._play_records) or not stadium.is_empty():
			return ERR_UNAUTHORIZED
		# Native finish_record already committed these records in memory. The
		# non-issued path has no QC journal or disk receipt to acknowledge.
		return OK

func earnings_income_rule() -> String:
	if career_owner==null or career_owner.active!="ordinary" or career_owner.session!=season \
		or season==null or not LeagueRules.earnings(season.league_rules): return ""
	return "Earnings game Cash replaces normal win/loss Cash: 15 plus your final run margin, bounded to 10–24. Sponsor effects are separate."

func earnings_paid_result(result: Dictionary) -> String:
	if earnings_income_rule().is_empty(): return ""
	if result.get("away")!=0 and result.get("home")!=0: return ""
	var home: bool=result.get("home")==0
	var own: Variant=result.get("home_runs" if home else "away_runs")
	var opponent: Variant=result.get("away_runs" if home else "home_runs")
	for event: Dictionary in season.build._bank._events:
		if event.get("op")!="reward" or event.get("game")!=result.get("id") \
			or event.get("own_runs")!=own or event.get("opponent_runs")!=opponent \
			or not LeagueRules.final_scores_valid(own,opponent,event.get("win")): continue
		var game: String=str(int(event.game))
		if season.build._bank.view().rewards.get(game)!=event.win: return ""
		var amount: int=LeagueRules.game_income(season.league_rules,event.win,own,opponent)
		var margin: int=int(own)-int(opponent)
		return "%d base game Cash paid • %s%d run margin" % [amount,"+" if margin>0 else "",margin]
	return ""

func play_next() -> void:
	if career_owner!=null and not career_owner.playable(): return
	var mixed: bool=career_owner!=null and (Owner.Mixed.marked_state(career_owner.ordinary_state) \
		or Owner.MixedAppend.marked_state(career_owner.ordinary_state))
	if mixed and not Store.mixed_round_entry(career_owner): return
	var workshop=Owner.Common.StandardWorkshop
	if workshop.opening_unlocked(career_owner):
		if standard_opening_enabled() and _available() and _owned() and workshop.player_opening_window(career_owner):
			menu.call("show_standard_opening_review")
		return
	var marked: bool=base_archive_qualification_view()
	if not _available() or season==null or not season.physical.pending.is_empty(): return
	if marked:
		if not Store.base_archive_app_binding(career_owner,saved_bytes) or not Store._base_archive_managed_ready(career_owner):
			return
		for window: Node in find_children("*","Window",true,false):
			if window.visible and not window.is_queued_for_deletion(): return
	elif not _owned() or not _round_scope():
		return
	if academy_pack_choice_pending():
		notice="Choose or explicitly skip the remaining paid Academy upgrade before playing."
		show_season()
		return
	var fixture: Dictionary = season.pending_fixture()
	var state: MatchState = season._make_mixed_round() if mixed else season.make_match()
	var field: FieldDefinition = Adapter.runtime(_active_side()).field_for_fixture(fixture,season)
	if state==null or field==null or not ClubCareer.same(state.stadium,_expected_stadium(fixture)) or not state.sure_shot.enable_current_evidence():
		notice = "This fixture could not be prepared."
		show_season()
		return
	_fixture_id = int(fixture.id)
	_result_recorded = false
	_result_saved = false
	_open_match(state,int(fixture.home)==0,true,field.id)
	if not state.stadium.is_empty() and not _played_field_matches():
		_close_match()
		notice="The saved home layout could not be installed. Your career is unchanged."
		show_season()


func _expected_stadium(fixture: Dictionary) -> Dictionary:
	if _active_side()=="dynasty": return Owner.WorkshopApply.fixture(season,fixture)
	# Installed Ordinary owners are admitted only in the private lane.
	return Owner.Common.StandardWorkshop.Home.fixture(season,fixture)

func _played_field_matches() -> bool:
	if Store.installed_earned_dynasty_match_bound(self): return true
	if season==null or (not _round_scope() and not base_archive_qualification_view()) \
			or not is_instance_valid(lab) or lab._match_state==null:
		return false
	var fixture: Dictionary = season.pending_fixture()
	var expected: FieldDefinition = Adapter.runtime(_active_side()).field_for_fixture(fixture,season)
	var stadium:=_expected_stadium(fixture)
	if not ClubCareer.same(lab._match_state.stadium,stadium): return false
	if not stadium.is_empty():
		if not ClubCareer.same(lab.get_meta("stadium_snapshot",{}),stadium) or not lab._player_home: return false
		var property: Variant=lab.get_meta("stadium_property",null)
		if not is_instance_valid(property) or not property is StadiumPropertyGeometry or not property.objects.has(stadium.layout.receiver): return false
		var receiver: StadiumPitchReceiver=property.get_node_or_null("PitchReceiverPresentation")
		if receiver==null or receiver.chair!=property.objects[stadium.layout.receiver] or lab._pitch_actor.receiver_router!=receiver.route: return false
	return expected!=null and lab._field_definition==expected and lab._field_id==expected.id and ClubCareer.same(lab._match_state.league_rules, season.league_rules) and _fixture_id==int(fixture.get("id",-1)) and lab._match_state.pitch_catalog==season.build.pitch_catalog_signature()


func finish_game() -> void:
	if not installed_dugout_d1_draft.is_empty():
		_finish_installed_dugout_d1_game()
		return
	if base_archive_qualification_view():
		_finish_base_archive_game()
		return
	if Store.Flat.is_flat(career_owner):
		if not Store.private_standard_chain_opening() or _busy or is_instance_valid(_loadout_modal): return
		if retained!=null:
			retry_save()
			return
		var ticket: RefCounted=Store.prepare_private_standard_flat_human(self)
		if ticket!=null:
			_retain_prepared(ticket,Store.prepared_candidate(ticket),saved_bytes)
		else:
			notice="The private opening game could not be saved. It remains open."
		return
	if is_instance_valid(_paired_menu_lab) and lab==_paired_menu_lab:
		if not (private_paired_opening_visible() or private_unpurchased_return_visible() \
			or ((private_unpurchased_next_pair() or private_unpurchased_next_reverse()) and season.physical.pending.is_empty())): return
		if retained!=null: retry_save()
		elif season.round_index==0: _finish_paired_first()
		elif season.round_index==1: _finish_paired_return()
		elif season.round_index==2: _finish_paired_next_first()
		else: _finish_paired_next_return()
		return
	if not _round_scope() or not career_owner.playable(): return
	if _busy or is_instance_valid(_loadout_modal):
		return
	if retained!=null:
		retry_save()
		return

func _resume_round_native(slow: bool) -> void:

		_retain_prepared(ticket,Store.prepared_candidate(ticket),saved_bytes)
		return

		var job: Dictionary=StarterNpc.request(career_owner,saved_bytes,projection)

func _checkpoint_report() -> void:
	if _busy or retained != null or retained_report.is_empty() or not _npc_owner_matches(): return
	if career_owner.paired_draft_only():
		var projection=load("res://src/gameplay/season/successor_live_play_v1/paired_first_projection.gd")
		var expected_fixture: Dictionary=projection.next_npc_fixture(season)
		if expected_fixture.is_empty() or _paired_npc_binding.get("fixture")!=expected_fixture.id: return
		var row: Dictionary=_paired_npc_binding.duplicate(true)
		row.report=retained_report.duplicate(true)
		if season.round_index>0 and projection.report_limit(season.round_index)>0:
			var expected: PackedByteArray=_npc_expected.duplicate()
			var paired_ticket: RefCounted
			match season.round_index:
				1: paired_ticket=Store._prepare_paired_return_report(career_owner,expected,row)
				2: paired_ticket=Store._prepare_paired_next_first_report(career_owner,expected,row)
				3: paired_ticket=Store._prepare_paired_next_return_report(career_owner,expected,row)
				4: paired_ticket=Store._prepare_paired_third_first_report(career_owner,expected,row)
				5: paired_ticket=Store._prepare_paired_third_return_report(career_owner,expected,row)
				6: paired_ticket=Store._prepare_paired_fourth_first_report(career_owner,expected,row)
				_: paired_ticket=Store._prepare_paired_report(career_owner,expected,row,season.round_index)
			if paired_ticket!=null: _retain_prepared(paired_ticket,Store.prepared_candidate(paired_ticket),expected)
			return
		var candidate: SeasonState=_clone()
		if candidate==null: return
		candidate.physical.pending.reports.append(row)
		_retain(candidate,saved_bytes)
		return
	_busy = true
	var ticket: RefCounted = Store.prepare_report_and_write(career_owner,_npc_expected,retained_report)
	_busy = false
	if ticket == null:
		notice = "The completed physical report could not be admitted. It remains retained."
		show_season(); return
	_retain_prepared(ticket,Store.prepared_candidate(ticket),_npc_expected)

func commit_shop(command: Dictionary) -> bool:
	if not Codec.primitive(command): return false
	if base_archive_qualification_view(): return false # Actual widget owns this marked write.
	var appended: bool=_append_shop_scope()
	if career_owner==null or (not career_owner.playable() and not appended): return false
	if not _available() or season == null or (not _scope() and not appended) or not season.physical.pending.is_empty() or not season.shop_available(): return false
	if not SeasonBuild.SHOP_OPS.has(str(command.get("op",""))) \
			or (appended and not Store.SHOP_PRODUCER_OPS.has(str(command.get("op","")))): return false
	var request_id: Variant = command.get("id")
	if request_id is String and not request_id.is_empty() and season.build._requests.has(request_id): return false
	# Choose the fixed producer before any transaction. Never fall back after it
	# has attempted a write: an obstructed candidate must retain its exact ticket.
	if Store.SHOP_PRODUCER_OPS.has(str(command.get("op",""))):
		if career_owner.session != season: return false
		var ticket: RefCounted = Store.prepare_shop_and_write(career_owner,saved_bytes,command)
		if ticket == null:
			if not Store.last_error.is_empty(): notice = Store.last_error
			return false
		var envelope: RefCounted = Store.prepared_candidate(ticket)
		var produced: SeasonState = envelope.session
		_retain_prepared(ticket,envelope,saved_bytes)
		return season == produced and retained == null
	var candidate: SeasonState = _clone()
	if candidate == null: return false
	var next: SeasonBuild = candidate.build.candidate(command)
	if next == null:
		notice = candidate.build.last_error; return false
	candidate.adopt_build(next)
	if candidate.build != next: return false
	_retain(candidate,saved_bytes)
	return season == candidate and retained == null

		return
	var private_flat: bool=action=="private_flat_abandon"
	var valid: bool=not action.is_empty() and career_owner==_lifecycle_owner and season==_lifecycle_season and career_owner!=null and career_owner.session==season and saved_bytes==_lifecycle_bytes and (Store.private_standard_flat_close_available(self) if private_flat else _owned())
	_cancel_lifecycle()
	if not valid: return
	if private_flat:
		var ticket: RefCounted=Store.prepare_private_standard_flat_abandon(self)
		if ticket!=null: _retain_prepared(ticket,Store.prepared_candidate(ticket),saved_bytes)
	elif action=="mixed_append_next":
		if not _mixed_append_next_ready(): return
		var expected:=saved_bytes.duplicate()
		var ticket: RefCounted=Store.prepare_mixed_append_next_and_write(career_owner,expected)
		if ticket!=null: _retain_prepared(ticket,Store.prepared_candidate(ticket),expected)
		else:
			notice="The next season was not saved. Your closed career is unchanged."
			show_season()
	elif action=="next": commit_next()
	else: close_current(action)

func _clear_lineup_ticket() -> void:
	_lineup_ticket=null;_lineup_owner=null;_lineup_session=null
	_lineup_bytes=PackedByteArray()

func lineup_allowed() -> bool:
	if not _available() or career_owner==null or career_owner.active!="dynasty" or career_owner.status!="active" or season==null: return false
	if career_owner.session!=season or not _scope(): return false
	if season.phase in [SeasonState.Phase.DRAFT,SeasonState.Phase.COMPLETE] or season.picks.size()!=4 or not season.physical.pending.is_empty(): return false
	var fixture: Dictionary=season.pending_fixture()
	if fixture.is_empty() or (fixture.get("home",-1)!=0 and fixture.get("away",-1)!=0) or season.build.pack_pending(): return false
	for node: Node in find_children("*","Window",true,false):
		if is_instance_valid(node) and not node.is_queued_for_deletion() and node.visible: return false
	# Display eligibility is not authority. Actions authenticate hidden state/disk.
	return season.teams.size()==6 \
		and Codec.human_lineup_matches(season.teams[0].get("roster"),season.build.roster()) \
		and Codec.whole(season.starter_index,0,3) and Codec.whole(season.fielder_index,0,3) \
		and season.starter_index!=season.fielder_index

func lineup_ticket() -> RefCounted:
	_clear_lineup_ticket()
	if not lineup_allowed(): return null
	_lineup_ticket=RefCounted.new();_lineup_owner=career_owner;_lineup_session=season
	_lineup_bytes=saved_bytes.duplicate()
	return _lineup_ticket

func _lineup_binding(ticket: RefCounted) -> bool:
	return ticket!=null and ticket==_lineup_ticket and career_owner==_lineup_owner \
		and season==_lineup_session and saved_bytes==_lineup_bytes and lineup_allowed()

func _lineup_current(ticket: RefCounted) -> bool:
	return _lineup_binding(ticket) and _owned()

func open_lineup() -> void:
	if lineup_allowed(): menu.call("show_lineup")

func _lineup_edit(kind: String, first: int, second: int, ticket: RefCounted) -> void:
	if not _lineup_binding(ticket) or first<0 or first>=4: return
	if kind=="swap":
		if second<0 or second>=4 or first==second: return
	elif kind=="starter":
		if first==season.starter_index: return
	elif kind=="fielder":
		if first==season.starter_index or first==season.fielder_index: return
	else: return
	if not _lineup_current(ticket):
		notice="Lineup change was not saved. Reload the saved career before editing."
		_clear_lineup_ticket()
		menu.call("show_lineup_blocked")
		return
	# No callback/await since authoritative admission; Store remains strict below.
	var candidate: SeasonState=season.fork_live()
	if candidate==null: return
	if kind=="swap": candidate.swap_batters(first,second)
	elif kind=="starter": candidate.select_starter(first)
	else: candidate.fielder_index=first
	_retain(candidate,saved_bytes)
	if season==candidate and retained==null: open_lineup()

func lineup_swap(first: int, second: int, ticket: RefCounted) -> void:
	_lineup_edit("swap",first,second,ticket)
func lineup_starter(index: int, ticket: RefCounted) -> void:
	_lineup_edit("starter",index,-1,ticket)
func lineup_fielder(index: int, ticket: RefCounted) -> void:
	_lineup_edit("fielder",index,-1,ticket)
func lineup_play(ticket: RefCounted) -> void:
	if not _lineup_binding(ticket): return
	# play_next performs the current authoritative _owned check synchronously.
	_clear_lineup_ticket();play_next()
func lineup_back(ticket: RefCounted) -> void:
	if not _lineup_binding(ticket): return
	show_season()

func _finish_base_archive_game() -> void:
	if _busy or Store.saving or retained!=null or is_instance_valid(_loadout_modal) or not is_instance_valid(lab): return
	var source: RefCounted=career_owner
	var prior: SeasonState=season
	if _base_archive_return.is_empty():
		_busy=true
		var committed: bool=Store.write_base_archive_human(self)
		_busy=false
		if not committed:
			notice="The completed game could not be saved. It remains open."
			return
		var written: Dictionary=Store.base_archive_app_result(saved_bytes)
		if written.is_empty():
			notice="The saved game could not be verified. The match remains open."
			return
		_base_archive_return=written.bytes
	var expected: PackedByteArray=_base_archive_return
	var next: Dictionary=Store.base_archive_app_result(saved_bytes)
	var candidate: RefCounted=next.get("owner")
	if next.is_empty() or next.bytes!=expected:
		notice="The saved game changed. The match has not been closed."
		return
	_busy=true
	var closed: bool=_close_match()
	_busy=false
	if not closed or is_instance_valid(lab):
		notice="The game is saved. Return again to close this match."
		return
	if career_owner!=source or season!=prior or not _adopt_base_archive_view(candidate,expected):
		notice="The game is saved. Use Continue to verify the checkpoint."
	show_season()

func _resume_base_archive_round() -> void:
	if not _available() or not Store.base_archive_app_binding(career_owner,saved_bytes): return
	if Store._base_archive_final_human(career_owner):
		_busy=true
		var final_settled: bool=Store.settle_base_archive_final(self)
		_busy=false
		_adopt_base_archive_progress(final_settled,"The final could not be verified for settlement.")
		return
	var count: int=Store._base_archive_report_count(career_owner)
	if count<0: return
	var last_regular: bool=Store._base_archive_last_regular(career_owner)
	var semifinal: bool=Store._base_archive_semifinal_count(career_owner)>=0
	if semifinal or last_regular or count<2:
		var decision: Dictionary
		if semifinal: decision=Store.resume_base_archive_semifinal(self)
		elif last_regular: decision=Store.resume_base_archive_last_regular(self)
		else: decision={"runner":Store.start_base_archive_cpu(self)}
		ai_runner=decision.get("runner")
		if is_instance_valid(ai_runner):
			_busy=true
			notice="Simulating the next CPU game. No rewards have been paid."
			show_season()
			return
		if decision.get("settle")!=true:
			notice="The saved round could not verify or start its next CPU game. No rewards were paid."
			show_season()
			return
	_busy=true
	var settled: bool=Store.settle_base_archive_opening(self)
	_busy=false
	_adopt_base_archive_progress(settled,"The round could not be verified for settlement.")
