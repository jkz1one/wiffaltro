extends Node
## A fixed read-only child proves the immutable title without stalling the menu.
## Only revision-bound journal number types may accompany a verified read.
## No path, save body, draft choice, or write authority crosses this protocol.
signal completed(outcome: String)

const Store=preload("owner_store.gd")
const Earned=preload("earned_dugout_d1.gd")
const ARG: String="--dugout-d1-read-v1="
const FRAME: String="WIFFALTRO_DUGOUT_D1_READ_V1 "
const MAX_FRAME: int=12288
const MAX_MS: int=720000
# Exported replay repeatedly asks ResourceLoader for these authored inputs.
# Retain them only in the one-shot worker, with no report or verdict caching.
const REPLAY_ASSET_PATHS: Array[String]=[
	"res://src/content/manifest/content_manifest.tres",
	"res://src/content/manifest/working_pitch_manifest.tres",
	"res://src/content/manifest/working_pitch_manifest_v6.tres",
	"res://src/content/pitches/exotic/corkscrew.tres",
	"res://src/content/pitches/exotic/sidearm_corkscrew.tres",
	"res://src/content/pitches/exotic/sidearm_switchback.tres",
	"res://src/content/pitches/exotic/sidearm_trapdoor.tres",
	"res://src/content/pitches/exotic/switchback.tres",
	"res://src/content/pitches/exotic/trapdoor.tres",
	"res://src/content/pitches/successor_v7/forkball.tres",
	"res://src/content/pitches/successor_v7/gypsy_drop.tres",
	"res://src/content/pitches/successor_v7/knuckle_drop.tres",
	"res://src/content/pitches/successor_v7/super_curve.tres",
]

var _job: Dictionary={}
var _verified: Dictionary={}
var _verified_wallet_types: String=""
var _pid: int=-1
var _out: FileAccess
var _err: FileAccess
var _buffer: PackedByteArray
var _start_ms: int
var _next_poll: int

static func _wire(job: Dictionary) -> String:
	return JSON.stringify(job,"",true,true)

static func _valid(job: Variant) -> bool:
	return Store.Codec.keys(job,["revision","page","nonce"]) \
		and Store.BaseArchive._digest(job.revision) and Store.BaseArchive._digest(job.page) \
		and Store.BaseArchive._digest(job.nonce) and job.revision!=job.nonce and job.page!=job.nonce

static func parse(args: PackedStringArray) -> Dictionary:
	if args.size()!=1 or args[0].length()>480 or not args[0].begins_with(ARG): return {}
	var encoded: String=args[0].trim_prefix(ARG)
	if encoded.is_empty() or encoded.length()%4!=0: return {}
	for character: String in encoded:
		if not "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/=".contains(character): return {}
	var raw: String=Marshalls.base64_to_utf8(encoded)
	if raw.is_empty() or Marshalls.utf8_to_base64(raw)!=encoded: return {}
	var job: Variant=JSON.parse_string(raw)
	return job if _valid(job) and raw==_wire(job) else {}

static func _package() -> bool:
	var exe: String=OS.get_executable_path()
	return exe.is_absolute_path() and FileAccess.file_exists(exe.get_basename()+".pck")

static func _matches(source: Dictionary, job: Dictionary) -> bool:
	return not source.is_empty() and Store.BaseArchive._hash(source.bytes)==job.revision \
		and source.pin==job.page

func start(expected: PackedByteArray) -> bool:
	if _pid>0 or not is_inside_tree() or not Store._installed_dugout_d1_lane() or not _package(): return false
	var source:=Store._installed_dugout_d1_source()
	if source.is_empty() or expected.is_empty() or source.bytes!=expected: return false
	var job: Dictionary={"revision":Store.BaseArchive._hash(expected),"page":source.pin,
		"nonce":Crypto.new().generate_random_bytes(32).hex_encode()}
	if not _valid(job): return false
	_verified.clear()
	_verified_wallet_types=""
	var path: String=OS.get_executable_path().get_base_dir().path_join("Wiffaltro")
	var launched:=OS.execute_with_pipe(path,[ARG+Marshalls.utf8_to_base64(_wire(job))],false)
	_pid=int(launched.get("pid",-1))
	_out=launched.get("stdio"); _err=launched.get("stderr")
	if _pid<=0 or _out==null or _err==null:
		cancel(); return false
	_job=job
	_start_ms=Time.get_ticks_msec()
	_next_poll=_start_ms
	process_mode=Node.PROCESS_MODE_ALWAYS
	set_process(true)
	return true

func cancel() -> void:
	_stop_child()
	_verified.clear()
	_verified_wallet_types=""

func _stop_child() -> void:
	if _pid>0 and OS.is_process_running(_pid): OS.kill(_pid)
	_pid=-1
	_out=null; _err=null
	_buffer.clear()
	_job.clear()
	set_process(false)

func _exit_tree() -> void: cancel()

func _drain() -> bool:
	if _out==null or _err==null or _err.get_length()!=0: return false
	var n: int=_out.get_length()
	if n<0 or n>MAX_FRAME-_buffer.size(): return false
	if n>0:
		var part: PackedByteArray=_out.get_buffer(n)
		if part.size()!=n: return false
		_buffer.append_array(part)
	return true

static func _decode(bytes: PackedByteArray, job: Dictionary, code: int) -> Dictionary:
	if code!=0 or not _valid(job) or bytes.is_empty() or bytes.size()>MAX_FRAME: return {}
	var text: String=bytes.get_string_from_utf8()
	var lines: PackedStringArray=text.split("\n",false)
	if not text.ends_with("\n") or lines.size()!=2 or not lines[0].begins_with("Godot Engine v"): return {}
	var prefix: String=FRAME+_wire(job)
	if lines[1]==prefix: return {"wallet_types":""}
	if not lines[1].begins_with(prefix+" "): return {}
	var wallet: String=lines[1].trim_prefix(prefix+" ")
	return {"wallet_types":wallet} if Earned.valid_wallet_types(wallet) else {}

func _process(_delta: float) -> void:
	if _pid<=0: return
	var now: int=Time.get_ticks_msec()
	if now<_next_poll: return
	_next_poll=now+80
	if now-_start_ms>MAX_MS or not _drain():
		cancel(); completed.emit("failed"); return
	if OS.is_process_running(_pid): return
	var exit_code: int=OS.get_process_exit_code(_pid)
	var reply: Dictionary=_decode(_buffer,_job,exit_code) if _drain() else {}
	var fresh: bool=not reply.is_empty() and _matches(Store._installed_dugout_d1_source(),_job)
	var witness: Dictionary=_job.duplicate() if fresh else {}
	_stop_child()
	_verified=witness
	_verified_wallet_types=reply.wallet_types if fresh else ""
	completed.emit("verified" if fresh else ("stale" if not reply.is_empty() else "refused"))

func verified_wallets() -> String:
	return _verified_wallet_types if _pid<0 and not _verified.is_empty() else ""

# The parent may consume only its completed matching, still-current job.
func verified_revision(expected: PackedByteArray) -> bool:
	if _pid>0 or _verified.is_empty() or not is_inside_tree() \
		or not Store._installed_dugout_d1_lane() or Store.BaseArchive._hash(expected)!=_verified.revision: return false
	var source:=Store._installed_dugout_d1_source()
	return _matches(source,_verified) and source.bytes==expected

func worker(job: Dictionary) -> void:
	set_process(false)
	if not _valid(job) or not _package() or not Store._installed_dugout_d1_read_worker():
		get_tree().quit(3); return
	var source:=Store._installed_dugout_d1_source(true)
	if not _matches(source,job):
		get_tree().quit(8); return
	# Hold strong references for the existing replay. The same catalog and
	# archive checks still run against the same bytes; no paths come from saves.
	var assets: Array[Resource]=[]
	for path: String in REPLAY_ASSET_PATHS:
		if ResourceLoader.exists(path):
			var asset: Resource=ResourceLoader.load(path)
			if asset!=null: assets.append(asset)
	var restored: SeasonState=Store._restore_installed_dugout_d1_source(source)
	if restored==null:
		get_tree().quit(8); return
	var wallets: String=""
	if Earned.needs_wallet_types(restored):
		wallets=Earned.wallet_types(restored)
		if wallets.is_empty():
			get_tree().quit(8); return
	var again:=Store._installed_dugout_d1_source(true)
	if not _matches(again,job) or again.bytes!=source.bytes or again.pages!=source.pages:
		get_tree().quit(8); return
	assets.clear()
	print(FRAME+_wire(job)+(" "+wallets if not wallets.is_empty() else ""))
	get_tree().quit()
