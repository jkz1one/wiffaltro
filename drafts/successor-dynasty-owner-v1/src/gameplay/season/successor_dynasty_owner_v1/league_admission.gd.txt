extends RefCounted
## Entry policy for the Common V4 launcher only. The starter flavor has a fixed profile.
const Unlocks = preload("res://src/gameplay/season/league_unlocks.gd")
const Ordinary = preload("res://src/gameplay/season/successor_live_play_v1/runtime.gd")
const Titles = preload("res://src/gameplay/season/club_league_career.gd")

const STARTER_FEATURE: String="wiffaltro_starter_beta"
const STARTER_MAIN: String="res://starter_bootstrap/entry.tscn"
const MIXED_APPEND_QA_FEATURE: String="wiffaltro_private_mixed_append_qa"
const MIXED_APPEND_QA_DIR: String="WiffaltroPrivateMixedAppendQA-v1"
const STARTER_USER_DIR: String="WiffaltroStarterBeta-v1"
const STARTER_MAC_USER_DIR: String="WiffaltroStarterBeta-Mac-v1"
const STARTER_LEAGUES: Array[String]=["standard","dugout"]

const CAREER_FEATURE: String="wiffaltro_linux_career_qualification"
const CAREER_MAIN: String="res://career_linux_bootstrap/entry.tscn"
const CAREER_USER_DIR: String="WiffaltroCareerQualification-v1"
const CAREER_FORWARD_FEATURE: String="wiffaltro_catalog_forward_v1"
const CAREER_FORWARD_USER_DIR: String="WiffaltroCareerForwardQualification-v1"

static func installed_career_directory() -> String:
	return CAREER_FORWARD_USER_DIR if OS.has_feature(CAREER_FORWARD_FEATURE) else CAREER_USER_DIR

static func installed_career_feature() -> bool:
	return OS.has_feature(CAREER_FEATURE)

static func _installed_career_profile() -> bool:
	if not installed_career_feature() or OS.get_name()!="Linux" or OS.has_feature("editor") \
		or OS.has_feature(STARTER_FEATURE) or OS.has_feature(MIXED_APPEND_QA_FEATURE) \
		or OS.has_feature("wiffaltro_engineering_export") or OS.has_feature("wiffaltro_base_archive_read_playtest") \
		or ProjectSettings.get_setting("application/config/custom_user_dir_name","")!=installed_career_directory() \
		or ProjectSettings.get_setting("application/run/main_scene","")!=CAREER_MAIN \
		or ProjectSettings.get_setting("application/config/use_custom_user_dir",false)!=true: return false
	return true

static func _installed_career_lock(pid: int, process_dir: String) -> bool:
	if pid<1: return false
	var root: String=OS.get_user_data_dir().simplify_path().trim_suffix("/")
	if root.get_file()!=installed_career_directory(): return false
	var fd:=DirAccess.open(process_dir+"/fd")
	if fd==null or fd.read_link("198")!=root: return false
	var info:=FileAccess.open(process_dir+"/fdinfo/198",FileAccess.READ)
	if info==null: return false
	# procfs fdinfo has a nominal size of zero: read the bounded line set.
	for i in 16:
		var fields:=info.get_line().replace("\t"," ").split(" ",false)
		if fields.size()==9 and fields[0]=="lock:" and fields[2]=="FLOCK" and fields[3]=="ADVISORY" \
			and fields[4]=="WRITE" and fields[5]==str(pid) and fields[7]=="0" and fields[8]=="EOF": return true
		if info.eof_reached(): break
	return false

static func installed_career_owned() -> bool:
	return _installed_career_profile() and OS.get_cmdline_args().is_empty() and OS.get_cmdline_user_args().is_empty() \
		and _installed_career_lock(OS.get_process_id(),"/proc/self")

# Read-only installed CPU worker, never a lock holder or save owner.
# The native shim closes its inherited copy before entering Godot.
static func installed_career_worker() -> bool:
	if not _installed_career_profile() or DisplayServer.get_name()!="headless" \
		or not OS.get_cmdline_args().is_empty() or OS.get_cmdline_user_args().size()!=1: return false
	var descriptors:=DirAccess.open("/proc/self/fd")
	if descriptors==null or not descriptors.read_link("198").is_empty(): return false
	var status:=FileAccess.open("/proc/self/status",FileAccess.READ)
	if status==null: return false
	for i in 24:
		var line:=status.get_line()
		if line.begins_with("PPid:"):
			var pid: String=line.trim_prefix("PPid:").strip_edges()
			return pid.is_valid_int() and int(pid)>1 and int(pid)!=OS.get_process_id() and _installed_career_lock(int(pid),"/proc/"+str(int(pid)))
		if status.eof_reached(): break
	return false

# The qualification inherits the same public controls and tier restrictions,
# not the Starter save profile or Starter-only worker processes.
static func starter_feature() -> bool:
	return OS.has_feature(STARTER_FEATURE) or installed_career_feature()

static func starter_player() -> bool:
	if installed_career_feature(): return installed_career_owned()
	var leaf: String=ProjectSettings.get_setting("application/config/custom_user_dir_name","")
	return OS.has_feature(STARTER_FEATURE) and (starter_platform_profile(OS.get_name(),leaf) or mixed_append_qa_profile()) and not OS.has_feature("editor") \
		and not OS.has_feature("wiffaltro_engineering_export") \
		and ProjectSettings.get_setting("application/run/main_scene")==STARTER_MAIN \
		and ProjectSettings.get_setting("application/config/use_custom_user_dir",false)==true

# This compiled feature and fixed Linux-only save leaf are NEVER set by
# a public package. An environment variable alone cannot select this profile.
static func mixed_append_qa_profile() -> bool:
	return OS.has_feature(MIXED_APPEND_QA_FEATURE) and OS.get_name()=="Linux" \
		and ProjectSettings.get_setting("application/config/custom_user_dir_name","")==MIXED_APPEND_QA_DIR

static func starter_platform_profile(platform: String, leaf: String) -> bool:
	return (platform=="Linux" and leaf==STARTER_USER_DIR) \
		or (platform=="macOS" and leaf==STARTER_MAC_USER_DIR)

static func starter_fast_worker() -> bool:
	return OS.get_name()=="Linux" and OS.has_feature(STARTER_FEATURE) and starter_player()

static func enabled() -> bool:
	return starter_feature() or OS.get_environment("WIFFALTRO_EARNED_LEAGUES_PREVIEW") == "1"

static func state(league: String, tier: int, titles: Dictionary) -> String:
	if tier not in [0,1,2,3,4] or league not in Unlocks.from_titles(titles):
		return "locked"
	if tier>0:
		return "unwired" if titles.get(Titles.title_key(league,tier-1),false)==true else "locked"
	if starter_feature():
		return "playable" if starter_player() and league in STARTER_LEAGUES else "unwired"
	return "playable" if league=="dynasty" or league in Ordinary.COMMON_ORDINARY_LEAGUES else "unwired"

static func reason(league: String, tier: int, titles: Dictionary) -> String:
	if league not in Unlocks.from_titles(titles): return Unlocks.requirement(league)
	if tier>0 and state(league,tier,titles)=="locked":
		return "Win "+league.capitalize()+" "+("Base" if tier==1 else "D%d"%(tier-1))+"."
	return "Earned. This league or difficulty is not playable in this starter build yet." if starter_feature() else \
		"Earned. This league or difficulty is not playable in this preview yet."

static func guard(league: String, tier: int, titles: Dictionary) -> bool:
	return not enabled() or state(league,tier,titles)=="playable"
