extends RefCounted
## Entry policy for the Common V4 launcher only. The starter flavor has a fixed profile.
const Unlocks = preload("res://src/gameplay/season/league_unlocks.gd")
const Ordinary = preload("res://src/gameplay/season/successor_live_play_v1/runtime.gd")
const Titles = preload("res://src/gameplay/season/club_league_career.gd")

const STARTER_FEATURE: String="wiffaltro_starter_beta"
const STARTER_MAIN: String="res://starter_bootstrap/entry.tscn"
const MIXED_APPEND_QA_FEATURE: String="wiffaltro_private_mixed_append_qa"
const MIXED_APPEND_QA_DIR: String="WiffaltroPrivateMixedAppendQA-v1"
const STARTER_USER_DIR: String="WiffaltroStarterBeta-v1"
const STARTER_MAC_USER_DIR: String="WiffaltroStarterBeta-Mac-v1"
const STARTER_LEAGUES: Array[String]=["standard","dugout"]

static func starter_feature() -> bool:
	return OS.has_feature(STARTER_FEATURE)

static func starter_player() -> bool:
	var leaf: String=ProjectSettings.get_setting("application/config/custom_user_dir_name","")
	return starter_feature() and (starter_platform_profile(OS.get_name(),leaf) or mixed_append_qa_profile()) and not OS.has_feature("editor") \
		and not OS.has_feature("wiffaltro_engineering_export") \
		and ProjectSettings.get_setting("application/run/main_scene")==STARTER_MAIN \
		and ProjectSettings.get_setting("application/config/use_custom_user_dir",false)==true

# This compiled feature and fixed Linux-only save leaf are NEVER set by
# a public package. An environment variable alone cannot select this profile.
static func mixed_append_qa_profile() -> bool:
	return OS.has_feature(MIXED_APPEND_QA_FEATURE) and OS.get_name()=="Linux" \
		and ProjectSettings.get_setting("application/config/custom_user_dir_name","")==MIXED_APPEND_QA_DIR

static func starter_platform_profile(platform: String, leaf: String) -> bool:
	return (platform=="Linux" and leaf==STARTER_USER_DIR) \
		or (platform=="macOS" and leaf==STARTER_MAC_USER_DIR)

static func starter_fast_worker() -> bool:
	return OS.get_name()=="Linux" and starter_player()

static func enabled() -> bool:
	return starter_feature() or OS.get_environment("WIFFALTRO_EARNED_LEAGUES_PREVIEW") == "1"

static func state(league: String, tier: int, titles: Dictionary) -> String:
	if tier not in [0,1,2,3,4] or league not in Unlocks.from_titles(titles):
		return "locked"
	if tier>0:
		return "unwired" if titles.get(Titles.title_key(league,tier-1),false)==true else "locked"
	if starter_feature():
		return "playable" if starter_player() and league in STARTER_LEAGUES else "unwired"
	return "playable" if league=="dynasty" or league in Ordinary.COMMON_ORDINARY_LEAGUES else "unwired"

static func reason(league: String, tier: int, titles: Dictionary) -> String:
	if league not in Unlocks.from_titles(titles): return Unlocks.requirement(league)
	if tier>0 and state(league,tier,titles)=="locked":
		return "Win "+league.capitalize()+" "+("Base" if tier==1 else "D%d"%(tier-1))+"."
	return "Earned. This league or difficulty is not playable in this starter build yet." if starter_feature() else \
		"Earned. This league or difficulty is not playable in this preview yet."

static func guard(league: String, tier: int, titles: Dictionary) -> bool:
	return not enabled() or state(league,tier,titles)=="playable"
