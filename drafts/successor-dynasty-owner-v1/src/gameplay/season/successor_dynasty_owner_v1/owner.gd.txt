static func _new(): return load("res://src/gameplay/season/successor_dynasty_owner_v1/owner.gd").new()
static func create_development(seed_value: int):
	var prior=Legacy.create_development(seed_value)
	if prior==null: return null
	var result=_new(); result.session=prior.session
	for name: String in DYNASTY_FIELDS: result.set(name,prior.get(name).duplicate(true) if prior.get(name) is Dictionary else prior.get(name))
	return result if Common.initialize(result) else null
static func create_ordinary(seed_value: int, league: String):
	var result=_new(); result.active="ordinary"
	result.session=Adapter.OrdinaryRuntime.create_fresh(seed_value,league)
	return result if result.session!=null and Common.initialize(result) else null
static func _create_paired_draft(seed_value: int):
	var result=_new(); result.active="ordinary"
	result.session=Adapter.OrdinaryRuntime._create_paired_draft(seed_value)
	return result if result.session!=null and Common.initialize(result) else null
func paired_draft_only() -> bool:
	return active=="ordinary" and session!=null and session.league_rules.get("league","")=="doubleheader"
func fork():
	var result=_new()
	result.session=session.fork_live() if session!=null else null
	if not Codec.install_fields(result,DATA_FIELDS,Codec.data_fields(self,DATA_FIELDS)): return null
	return result if result.session!=null else null
func playable() -> bool:
	if ordinary_state.has("archive"): return active=="ordinary" and ordinary_state.status=="active" and Common.archive_play_scope(session)
	if paired_draft_only(): return ordinary_state.get("status")=="active" and session.phase==SeasonState.Phase.DRAFT
	return (active=="ordinary" and ordinary_state.get("status")=="active") or (active=="dynasty" and status=="active")
func lifecycle_status() -> String: return str(ordinary_state.get("status","absent")) if active=="ordinary" else status
func current_ordinal() -> int: return int(ordinary_state.get("ordinal",0)) if active=="ordinary" else ordinal
func lifecycle_reason() -> String: return str(ordinary_state.get("last_close",{}).get("reason","")) if active=="ordinary" else str(last_close.get("reason",""))
func next_block_reason() -> String: return Common.next_block_reason(self)
func next_supported() -> bool: return next_block_reason().is_empty()
func close_current(reason: String) -> bool: return Common.close(self,reason)
func propose_next(seed_value: int) -> bool: return Common.propose(self,seed_value)
func choose_overflow(player: String, receipt: String) -> bool: return active=="dynasty" and Life.choose(self,player,receipt)
func next_preview() -> Dictionary: return Common.preview(self)
func commit_next() -> bool: return Common.advance(self)
func discovery_ids() -> Array:

	if playable():
		for id: String in ClubCollection.acquired(session.build):
			if not result.has(id): result.append(id)
	result.sort(); return result
func awards() -> Dictionary: return Common.awards(self)

func dynasty_metadata() -> Dictionary: return Codec.data_fields(self,DYNASTY_FIELDS)
static func _digest(value: Variant) -> bool: return Legacy._digest(value)
static func _hash(bytes: PackedByteArray) -> String:
	var h: HashingContext=HashingContext.new()
	return h.finish().hex_encode() if h.start(HashingContext.HASH_SHA256)==OK and h.update(bytes)==OK else ""
static func _bytes_value(bytes: PackedByteArray) -> Variant:
	if bytes.is_empty() or bytes.size()>64*1024*1024: return null
	var parser: JSON=JSON.new()
	return parser.data if parser.parse(bytes.get_string_from_utf8())==OK else null
static func _adopt_legacy(prior):
	var result=_new();result.session=prior.session
	for name: String in DYNASTY_FIELDS: result.set(name,prior.get(name).duplicate(true) if prior.get(name) is Dictionary else prior.get(name))
	result.paused=prior.alternate.duplicate(true);result.imported_sha256=prior.imported_sha256
	return result
static func import_v3(bytes: PackedByteArray):
	var input: Variant=_bytes_value(bytes)
	var prior=PreviousOwner.restore(input)
	if prior==null or not Codec.same(PreviousOwner.capture(prior),input): return null
	var result=_new();result.session=prior.session
	if not Codec.install_fields(result,PreviousOwner.DATA_FIELDS,Codec.data_fields(prior,PreviousOwner.DATA_FIELDS)): return null
	result.migration_v3={"format":PreviousOwner.FORMAT,"sha256":_hash(bytes),"bytes":bytes.size()}
	return result if Common.initialize(result) and not capture(result).is_empty() else null
static func import_v2(bytes: PackedByteArray):
	var input: Variant=_bytes_value(bytes)
	var prior=Legacy.restore(input)
	if prior==null or not Codec.same(Legacy.capture(prior),input): return null
	var result=_adopt_legacy(prior)
	result.migration_v2={"format":Legacy.FORMAT,"sha256":_hash(bytes),"bytes":bytes.size()}
	return result if Common.initialize(result) and not capture(result).is_empty() else null
static func import_75(bytes: PackedByteArray):
	var prior=Legacy.import_75(bytes)
	if prior==null: return null
	var result=_adopt_legacy(prior)
	return result if Common.initialize(result) and not capture(result).is_empty() else null
func import_119(bytes: PackedByteArray) -> bool:
	if active!="dynasty" or not paused.is_empty(): return false
	var value: Variant=_bytes_value(bytes)
	# This explicit import is legacy119 only. New earned sessions keep their
	# common lineage/ordinal through v4 restore, never re-enter as ordinal1.
	if not PreviousOwner._ordinary_v1(value): return false
	var checked: Dictionary=Adapter.checked_restore("ordinary",value)
	if checked.is_empty(): return false
	paused=checked.captured; imported_sha256=_hash(bytes)
	return not imported_sha256.is_empty() and Common.initialize_ordinary(self,checked.season)
func switch_existing() -> bool:
	if paired_draft_only() or ordinary_state.has("archive") or paused.is_empty(): return false
	var outgoing: Dictionary=Adapter.capture(active,session)
	var next: Dictionary=Adapter.checked_restore(Adapter.other(active),paused)
	if outgoing.is_empty() or next.is_empty(): return false
	session=next.season;paused=outgoing;active=Adapter.other(active)
	return true
func create_missing(seed_value: int, league: String) -> bool:
	if paired_draft_only() or ordinary_state.has("archive") or not paused.is_empty(): return false
	var outgoing: Dictionary=Adapter.capture(active,session)
	if outgoing.is_empty(): return false
	if active=="dynasty":
		var next: SeasonState=Common.ordinary_birth(self,seed_value,league)
		if next==null: return false
		session=next;active="ordinary"
	else:
		if league!="dynasty": return false
		if common.sequence==0 and common.aggregate.exotic_rights.is_empty():
			var next=Legacy.create_development(seed_value)
			if next==null: return false
			for name: String in DYNASTY_FIELDS: set(name,next.get(name).duplicate(true) if next.get(name) is Dictionary else next.get(name))
			session=next.session
		else:
			var basis: Dictionary=Common.dynasty_birth_basis(self,seed_value)
			if basis.is_empty(): return false
			var next: SeasonState=Session.create_common_birth(basis)
			if next==null: return false
			var fresh_ledger: Dictionary=Ledger.birth(next)
			if not Ledger.valid(fresh_ledger): return false
			session=next;birth_seed=seed_value;ordinal=1;status="active";ledger=fresh_ledger
		active="dynasty"
	paused=outgoing
	return Common.initialize_missing(self)

## First Dynasty workshop family: exactly one free native starter layout draft.
## This never writes the ordinary journal, spends, or sets the physical property.
static func starter_workshop_draft(owner) -> Dictionary:
	if owner.status not in ["active","closed"] or owner.ordinal<1: return {}
	var ledger_basis := ClubAssetLedger.new(owner.birth_seed,[0],[],{})
	var stock := ledger_basis.construction()
	var layout := StadiumStarterLayout.empty()
	for copy: String in stock.copies:
		if stock.copies[copy].item=="ST-079":
			if not layout.receiver.is_empty(): return {}
			layout.receiver=copy
	if layout.receiver.is_empty(): return {}
	var before: Dictionary=ledger_basis.view()
	var result: Dictionary=ledger_basis.commit({"id":"stadium-draft:0","rev":ledger_basis.revision(),"op":"stadium_draft","layout":layout})
	var after: Dictionary=ledger_basis.view()
	if not result.get("ok",false) or not Codec.same(after.stadium_draft,layout) or not Codec.same(after.stadium_active,before.stadium_active) or ledger_basis.balance()!=0: return {}
	return {"version":1,"lineage":Life.lineage(owner),"layout":layout}

static func workshop_window(owner) -> bool:
	if owner==null or owner.active!="dynasty" or owner.session==null or not Adapter.scope("dynasty",owner.session): return false
	var current: SeasonState=owner.session
	if not WorkshopApply.matches_season(owner,current) or not current.physical.pending.is_empty() or not current.physical.needed.is_empty() or current.physical.projecting or current.build.pack_pending(): return false
	return (owner.status=="active" and owner.ordinal==1 and current.phase==SeasonState.Phase.DRAFT and current.picks.is_empty()) or (owner.status=="closed" and owner.proposal.is_empty())

static func _base_valid(owner) -> bool:
	if owner==null or owner.get_script()!=load("res://src/gameplay/season/successor_dynasty_owner_v1/owner.gd") or not Codec.known_variables(owner,DATA_FIELDS+["session"]): return false
	if not Adapter.scope(owner.active,owner.session) or not Codec.primitive(Codec.data_fields(owner,DATA_FIELDS)): return false
	if not owner.imported_sha256.is_empty() and not _digest(owner.imported_sha256): return false
	if owner.active=="dynasty" and owner.paused.is_empty() and not owner.imported_sha256.is_empty(): return false
	if not owner.migration_v2.is_empty():
		if not Codec.keys(owner.migration_v2,["format","sha256","bytes"]) or owner.migration_v2.format!=Legacy.FORMAT or not _digest(owner.migration_v2.sha256) or not Codec.whole(owner.migration_v2.bytes,1,64*1024*1024): return false
	if not owner.migration_v3.is_empty():
		if not Codec.keys(owner.migration_v3,["format","sha256","bytes"]) or owner.migration_v3.format!=PreviousOwner.FORMAT or not _digest(owner.migration_v3.sha256) or not Codec.whole(owner.migration_v3.bytes,1,64*1024*1024): return false
	if not owner.workshop_draft.is_empty():
		var authored: Dictionary=WorkshopApply.paid_draft(owner) if owner.workshop_draft.get("version")==2 else starter_workshop_draft(owner)
		if authored.is_empty() or not Codec.same(owner.workshop_draft,authored): return false
	if not Common.valid(owner): return false
	return (owner.workshop_close_history.is_empty() or WorkshopHistory.valid_history(owner.workshop_close_history,owner.ledger,Life.lineage(owner),owner.last_close)) and WorkshopPurchase.valid(owner) and WorkshopApply.valid(owner)
static func _missing_dynasty(owner) -> bool:
	return owner.imported_sha256.is_empty() and owner.birth_seed==0 and owner.ordinal==0 and owner.status=="absent" and owner.ledger.is_empty() and owner.last_close.is_empty() and owner.proposal.is_empty() and owner.last_transition.is_empty() and owner.migration.is_empty() and owner.migration_v2.is_empty()
static func _dynasty_valid(owner, season: SeasonState, current: Dictionary) -> bool:
	if not WorkshopApply.matches_season(owner,season) or not WorkshopHistory.matches_current(owner.workshop_close_history,season,owner.last_close): return false
	var view=Legacy.new();view.session=season
	for name: String in DYNASTY_FIELDS: view.set(name,owner.get(name))
	# Ordinary import provenance is common-owner metadata, not a second copy in
	# this Dynasty-only validation view. Original v2 inputs use Legacy.restore.
	var basis: Dictionary=season.career.start_basis
	if basis.get("kind")==Codec.CommonBirth.KIND:
		return Common.dynasty_birth_bound(owner,basis) and Legacy.envelope_valid(view,current,true)
	return not Legacy.pack_current(view,current).is_empty()
static func pack_current(owner, current: Dictionary) -> Dictionary:
	# Pure assembly, no writer authority. For ordinary actions Store separately
	# pins every dormant Dynasty byte AND metadata to its strict admission.
	if not _base_valid(owner) or current.is_empty(): return {}
	if owner.active=="dynasty" and not _dynasty_valid(owner,owner.session,current): return {}
	if owner.active=="ordinary" and not Common.ordinary_current_valid(owner,owner.session,current): return {}
	if owner.active=="ordinary" and owner.paused.is_empty() and not _missing_dynasty(owner): return {}
	var result: Dictionary=Codec.data_fields(owner,META_FIELDS)
	if owner.workshop_draft.is_empty(): result.erase("workshop_draft")
	if owner.workshop_close_history.is_empty(): result.erase("workshop_close_history")
	if owner.workshop_chair_purchase.is_empty(): result.erase("workshop_chair_purchase")
	if owner.workshop_applied.is_empty(): result.erase("workshop_applied")
	result.merge({"version":4,"format":FORMAT,"active":owner.active,"dynasty":current if owner.active=="dynasty" else owner.paused.duplicate(true),"ordinary":current if owner.active=="ordinary" else owner.paused.duplicate(true)})
	return result
static func capture(owner) -> Dictionary:
	if not _base_valid(owner): return {}
	var current: Dictionary=Adapter.capture(owner.active,owner.session)
	if current.is_empty(): return {}
	if not owner.paused.is_empty():
		var dormant: Dictionary=Adapter.checked_restore(Adapter.other(owner.active),owner.paused)
		if dormant.is_empty(): return {}
		if owner.active=="ordinary" and not _dynasty_valid(owner,dormant.season,dormant.captured): return {}
		if owner.active=="dynasty" and not Common.ordinary_current_valid(owner,dormant.season,dormant.captured): return {}
	return pack_current(owner,current)
# Cold-only strict predecessor replay. Refuse recursion before calling
# the original Common reader; never use the rotating backup as authority.
static func _archive_valid(owner, current: Dictionary) -> bool:
	if not owner.ordinary_state.has("archive"): return true
	if not Common._archive_shape(owner): return false
	var archive: Dictionary=owner.ordinary_state.archive
	if archive.get("kind")==Common.StandardArchive.KIND: return Common.StandardArchive.validate(owner,current)
	if archive.get("kind")==Common.StandardChain.KIND: return Common.StandardChain.validate(owner,current)
	if archive.get("kind")==Common.ARCHIVE_CHAIN: return _archive_chain_valid(owner,current)
	var bytes: PackedByteArray=archive.source.to_utf8_buffer()
	if bytes.size()>Common.ARCHIVE_MAX or _hash(bytes)!=archive.sha256: return false
	var old: Variant=_bytes_value(bytes)
	if not old is Dictionary or not old.get("ordinary_state") is Dictionary \
		or old.ordinary_state.has("archive"): return false
	var predecessor: RefCounted=load("res://src/gameplay/season/successor_dynasty_owner_v1/owner.gd").restore(old)
	if not Common.cross_commit_ready(predecessor): return false
	var p: Dictionary=predecessor.ordinary_state.proposal
	var state: Dictionary=owner.ordinary_state
	if archive.kind==Common.STARTER_DUGOUT_ARCHIVE and (not Common.starter_dugout_standard_scope(predecessor) \
			or p.league!="dugout" or p.basis.seed!=Common.starter_dugout_seed(state.lineage,int(state.ordinal))): return false
	var basis: Dictionary=p.basis if not state.last_transition.has("kind") else Common.Start.bind_archive(p.basis,archive.sha256,p.league)
	return state.lineage==predecessor.ordinary_state.lineage and state.last_transition.source==p.source and state.last_transition.league==p.league \
		and Codec.same(state.last_transition.basis,basis) \
		and _archive_fold_matches(owner,predecessor,current) and Codec.same(owner.paused,predecessor.paused) \
		and Codec.same(owner.dynasty_metadata(),predecessor.dynasty_metadata()) \
		and Codec.same(owner.workshop_draft,predecessor.workshop_draft) \
		and Codec.same(owner.workshop_close_history,predecessor.workshop_close_history) \
		and Codec.same(owner.workshop_chair_purchase,predecessor.workshop_chair_purchase) \
		and Codec.same(owner.workshop_applied,predecessor.workshop_applied) \
		and owner.imported_sha256==predecessor.imported_sha256 \
		and Codec.same(owner.migration_v2,predecessor.migration_v2) and Codec.same(owner.migration_v3,predecessor.migration_v3)

static func _archive_chain_valid(owner, current: Dictionary) -> bool:
	var state: Dictionary=owner.ordinary_state
	var a: Dictionary=state.archive
	var second: PackedByteArray=Common._chain_second_bytes(a)
	if second.is_empty(): return false
	var value: Variant=_bytes_value(second)
	if not value is Dictionary or not value.get("ordinary_state") is Dictionary: return false
	var o: Dictionary=value.ordinary_state
	if not o.get("archive") is Dictionary or o.archive.get("kind")!=Common.ARCHIVE_KIND \
			or not Codec.same(o.archive,a.entries[0]) or o.get("ordinal")!=2 \
			or o.get("proposal",{}).get("kind")!=Common.ARCHIVED_NEXT_KIND \
			or not Common._flat_chain_encoding(a.entries[1],value): return false
	# Strict ordinal-two replay authenticates the original Standard archive
	# and independently re-folds the complete abandoned Dugout source.
	var prev: RefCounted=load("res://src/gameplay/season/successor_dynasty_owner_v1/owner.gd").restore(value)
	if prev==null or not Common._ordinary_proposal_valid(prev) \
			or not Codec.same(Common._make_archive_chain(prev,second),a): return false
	var p: Dictionary=prev.ordinary_state.proposal
	var basis: Dictionary=Common.Start.bind_archive_chain(p.basis,a.entries[0].sha256,a.entries[1].sha256)
	if state.lineage!=prev.ordinary_state.lineage or state.last_transition.source!=p.source \
			or not Codec.same(state.last_transition.basis,basis) or not Codec.same(owner.paused,prev.paused): return false
	var saved: Dictionary=Codec.data_fields(owner,META_FIELDS)
	var previous: Dictionary=Codec.data_fields(prev,META_FIELDS)
	for name: String in ["ordinary_state","common"]:
		saved.erase(name);previous.erase(name)
	return Codec.same(saved,previous) and _archive_fold_matches(owner,prev,current)

# Reconstruct the native close from the exact earlier fold. Legacy kinds
# remain Abandon-only; only the strict Starter Dugout custody can complete.
# The current snapshot already passed full ordinary restore; reuse it.
static func _archive_fold_matches(owner, predecessor, current: Dictionary) -> bool:
	if owner.ordinary_state.status=="active": return Codec.same(owner.common,predecessor.common)
	if owner.ordinary_state.status!="closed" or current.is_empty(): return false
	var expected=_new()
	expected.session=owner.session
	if not Codec.install_fields(expected,DATA_FIELDS,Codec.data_fields(owner,DATA_FIELDS)): return false
	expected.common=predecessor.common.duplicate(true)
	expected.ordinary_state.status="active"
	expected.ordinary_state.last_close={}
	expected.ordinary_state.proposal={}
	var reason: String="complete" if owner.ordinary_state.last_close.get("reason")=="complete" \
		and Common.starter_dugout_complete_scope(owner,owner.session) else "abandon"
	return Common._apply_close(expected,reason,current) \
		and Codec.same(expected.common,owner.common) and Codec.same(expected.ordinary_state.last_close,owner.ordinary_state.last_close)

# No current-session object or write authority is restored here.
static func _restore_metadata(value: Variant) -> RefCounted:
	var names: Array=META_FIELDS+["version","format","active","dynasty","ordinary"]
	if value is Dictionary:
		for field: String in ["workshop_draft","workshop_close_history","workshop_chair_purchase","workshop_applied"]:
			if not value.has(field): names.erase(field)
	if not Codec.primitive(value) or not Codec.keys(value,names) or value.version!=4 or value.format!=FORMAT or not value.active is String or not Adapter.side(value.active): return null
	if not value.dynasty is Dictionary or not value.ordinary is Dictionary: return null
	var result=_new(); var fields: Dictionary={}
	for name: String in META_FIELDS: fields[name]=value.get(name,{})
	if not Codec.install_fields(result,META_FIELDS,fields): return null
	result.active=value.active;result.paused=value[Adapter.other(value.active)].duplicate(true)
	return result

class CanonicalRestore extends RefCounted:
	var owner: RefCounted
	var current: Dictionary

# No archive authority on its own. Only native restore and the fixed ordered
# Standard archive reader (which verifies both predecessors) consume this.
static func _restore_canonical(value: Variant) -> CanonicalRestore:
	var result: RefCounted=_restore_metadata(value)
	if result==null: return null
	var current: Dictionary=Adapter.checked_restore(result.active,value[result.active])
	if current.is_empty(): return null
	result.session=current.season
	if not result.paused.is_empty():
		var dormant: Dictionary=Adapter.checked_restore(Adapter.other(result.active),result.paused)
		if dormant.is_empty(): return null
		if result.active=="ordinary" and not _dynasty_valid(result,dormant.season,dormant.captured): return null
		if result.active=="dynasty" and not Common.ordinary_current_valid(result,dormant.season,dormant.captured): return null
	var checked: Dictionary=pack_current(result,current.captured)
	if checked.is_empty() or not Codec.same(checked,value): return null
	var native := CanonicalRestore.new()
	native.owner=result
	native.current=current.captured
	return native

static func restore(value: Variant):
	var native := _restore_canonical(value)
	return native.owner if native!=null and _archive_valid(native.owner,native.current) else null
