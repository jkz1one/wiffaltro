extends Node
## Same-binary, read-only strict cold validation of the exact current slot.
const Work=preload("standard_d1_witness_process.gd")
const T=preload("standard_d1_transition.gd")

func _ready() -> void:
	call_deferred("_run")

func _run() -> void:
	var args:=OS.get_cmdline_user_args()
	var prefix: String="--d1-witness="
	if not Work.supported() or args.size()!=1 or not args[0].begins_with(prefix) or args[0].length()>1024:
		get_tree().quit(2)
		return
	var json:=Marshalls.base64_to_utf8(args[0].trim_prefix(prefix))
	var request: Variant=JSON.parse_string(json)
	if not Work.request_valid(request) or JSON.stringify(request,"",true,true)!=json:
		get_tree().quit(2)
		return
	var bytes:=Work._bounded_slot(request.kind)
	if bytes.is_empty() or T._source_digest(bytes)!=request.source_sha256:
		get_tree().quit(3)
		return
	if request.kind in [Work.CROSS_KIND,Work.ACTIVATE_KIND]:
		_review_cross(request,bytes)
		return
	if request.kind==Work.CONTINUE_KIND:
		_continue_exact(request,bytes)
		return
	var witness: Dictionary=T._pack_closed_source(bytes)
	var owner: RefCounted=T._replay_closed_source(witness)
	if owner==null or Work._revision(request.kind)!=request.source_sha256:
		get_tree().quit(3)
		return
	var qualified: Dictionary=T.derive(owner,int(request.seed))
	var result: Dictionary={"request":request,"outcome":"refused" if qualified.is_empty() else "qualified","transition":qualified}
	_emit(result,not qualified.is_empty())

# Review the actual owner and candidate. Only the native draft fingerprint
# crosses the private pipe; this process cannot write or mint a Store ticket.
func _review_cross(request: Dictionary, bytes: PackedByteArray) -> void:
	if not Work.Store.cross_league_preview():
		get_tree().quit(2)
		return
	var owner: RefCounted=Work.Store.Owner.restore(Work.Store.Owner._bytes_value(bytes))
	var current: Dictionary={}
	var candidate: Dictionary={}
	if owner!=null and Work.Store.Owner.Common._activate_cross(owner,bytes):
		current=Work.Store.Adapter.capture("ordinary",owner.session)
		candidate=Work.Store.Owner.pack_current(owner,current)
	if Work._revision(request.kind)!=request.source_sha256:
		get_tree().quit(3)
		return
	if request.kind==Work.ACTIVATE_KIND:
		_stage_cross(request,candidate)
		return
	var digest: String="" if candidate.is_empty() else Work.Store.Owner.Common.Life.digest(current)
	_emit({"request":request,"outcome":"refused" if digest.is_empty() else "qualified",
		"draft_sha256":digest},not digest.is_empty())

func _stage_cross(request: Dictionary,candidate: Dictionary) -> void:
	var result: Dictionary={"request":request,"outcome":"refused","size":0,"candidate_sha256":""}
	if candidate.is_empty():
		_emit(result,false)
		return
	var bytes:=JSON.stringify(candidate,"",true,true).to_utf8_buffer()
	if bytes.is_empty() or bytes.size()>Work.CROSS_MAX \
		or not Work.Store.Codec.same(JSON.parse_string(bytes.get_string_from_utf8()),candidate):
		get_tree().quit(3)
		return
	var path:=Work._activation_dir(request.token)
	var root:=DirAccess.open("user://")
	var target:=path.path_join(Work.CROSS_FILE)
	if root==null or root.is_link(path.get_file()) or not DirAccess.dir_exists_absolute(path) \
		or FileAccess.file_exists(target) or DirAccess.dir_exists_absolute(target):
		get_tree().quit(3)
		return
	var dir:=DirAccess.open(path)
	if dir==null or dir.is_link(Work.CROSS_FILE):
		get_tree().quit(3)
		return
	var f:=FileAccess.open(target,FileAccess.WRITE)
	if f==null:
		get_tree().quit(3)
		return
	f.store_buffer(bytes)
	f.flush()
	var ok: bool=f.get_error()==OK
	f.close()
	if not ok or Work._revision(request.kind)!=request.source_sha256:
		get_tree().quit(3)
		return
	result.merge({"outcome":"qualified","size":bytes.size(),"candidate_sha256":T._source_digest(bytes)},true)
	_emit(result,true)

func _continue_exact(request: Dictionary, bytes: PackedByteArray) -> void:
	var data:=T._durable_shape(bytes)
	var verified:=T._cold_durable_checked(data)
	if Work._revision(request.kind)!=request.source_sha256:
		get_tree().quit(3)
		return
	var digests: Dictionary={}
	if not verified.is_empty():
		digests={"stamp":data.stamp,"witness":data.witness.source_sha256,
			"draft":T._slot_text(data.draft).sha256_text()}
	_emit({"request":request,"outcome":"refused" if digests.is_empty() else "qualified",
		"digests":digests},not digests.is_empty())

func _emit(result: Dictionary, qualified: bool) -> void:
	var frame: String=Work.FRAME+Work.canonical(result)
	if frame.to_utf8_buffer().size()>Work.MAX_FRAME-1024:
		get_tree().quit(3)
		return
	print(frame)
	get_tree().quit(0 if qualified else Work.REFUSED_EXIT)
