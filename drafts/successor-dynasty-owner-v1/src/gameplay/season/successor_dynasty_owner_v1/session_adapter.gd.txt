extends RefCounted
## Fixed named current-session dispatch. No caller-selected script or save writer.
const Dynasty = preload("codec.gd")
const Ordinary = preload("res://src/gameplay/season/successor_live_play_v1/codec.gd")
const DynastyRuntime = preload("runtime.gd")
const OrdinaryRuntime = preload("res://src/gameplay/season/successor_live_play_v1/runtime.gd")
const DynastyPending = preload("pending.gd")
const OrdinaryPending = preload("res://src/gameplay/season/successor_live_play_v1/pending.gd")
const DynastyFixtures = preload("physical_fixtures.gd")
const OrdinaryFixtures = preload("res://src/gameplay/season/successor_live_play_v1/physical_fixtures.gd")
const DynastyRunner = preload("physical_runner.gd")
const OrdinaryRunner = preload("res://src/gameplay/season/successor_league_general_v2/physical_runner.gd")
const DynastyCapture = preload("res://src/presentation/season/successor_dynasty_owner_v1/capture_human.gd")
const OrdinaryCapture = preload("res://src/presentation/season/successor_live_play_v1/capture_human.gd")
static func side(value: String) -> bool: return value in ["dynasty","ordinary"]
static func other(value: String) -> String: return "ordinary" if value=="dynasty" else "dynasty"
static func codec(value: String) -> Script: return Dynasty if value=="dynasty" else (Ordinary if value=="ordinary" else null)
static func runtime(value: String) -> Script: return DynastyRuntime if value=="dynasty" else (OrdinaryRuntime if value=="ordinary" else null)
static func pending(value: String) -> Script: return DynastyPending if value=="dynasty" else (OrdinaryPending if value=="ordinary" else null)
static func fixtures(value: String) -> Script: return DynastyFixtures if value=="dynasty" else (OrdinaryFixtures if value=="ordinary" else null)
static func runner(value: String) -> Script: return DynastyRunner if value=="dynasty" else (OrdinaryRunner if value=="ordinary" else null)
static func human_capture(value: String) -> Script: return DynastyCapture if value=="dynasty" else (OrdinaryCapture if value=="ordinary" else null)
static func scope(value: String, session: SeasonState) -> bool:
	return side(value) and session!=null and session.get_script()==runtime(value) and codec(value).scope(session) \
		and not (value=="ordinary" and (OrdinaryRuntime.CommonStart.detached_standard_flat(session.career.start_basis) \
			or OrdinaryRuntime.CommonStart.detached_mixed_flat(session.career.start_basis) \
			or OrdinaryRuntime.CommonStart.private_mixed_append_basis(session.career.start_basis) \
			or OrdinaryRuntime.CommonStart.detached_earned_dugout_d1(session.career.start_basis) \
			or OrdinaryRuntime.CommonStart.detached_earned_dugout_d2(session.career.start_basis)
			or OrdinaryRuntime.CommonStart.detached_earned_dugout_d3(session.career.start_basis)))
static func capture(value: String, session: SeasonState) -> Dictionary:
	return codec(value).capture(session) if scope(value,session) else {}
static func checked_restore(value: String, data: Variant) -> Dictionary:
	if value=="dynasty": return Dynasty._hydrate_current_checked(data)
	if value!="ordinary": return {}
	# Local native draft bytes are not a persisted Common-owner dialect.
	if data is Dictionary and data.get("career") is Dictionary and data.career.get("start_basis") is Dictionary \
			and (OrdinaryRuntime.CommonStart.detached_standard_flat(data.career.start_basis) \
			or OrdinaryRuntime.CommonStart.detached_earned_dugout_d1(data.career.start_basis) \
			or OrdinaryRuntime.CommonStart.detached_earned_dugout_d2(data.career.start_basis)
			or OrdinaryRuntime.CommonStart.detached_earned_dugout_d3(data.career.start_basis)): return {}
	var session: SeasonState=Ordinary.restore_common(data)
	# Ordinary.restore itself captures and compares the complete named schema.
	# Return a detached copy of that checked input, never a caller admission pair.
	return {} if session==null else {"season":session,"captured":data.duplicate(true)}
static func restore(value: String, data: Variant) -> SeasonState:
	return checked_restore(value,data).get("season")
