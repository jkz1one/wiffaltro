extends RefCounted
## Authenticated detached next draft. This creates neither a Common owner,
## a Store ticket, a launch option, nor a persistent archive reader.
const Custody=preload("starter_mixed_custody.gd")

class Draft:
	extends RefCounted
	var season: SeasonState
	var _basis: Dictionary
	var _common: Dictionary
	var _manifest: PackedByteArray
	var _source: Dictionary
	var _native_usec: int
	var _transport_usec: int
	func basis() -> Dictionary: return _basis.duplicate(true)
	func common() -> Dictionary: return _common.duplicate(true)
	func manifest() -> PackedByteArray: return _manifest
	func source() -> Dictionary: return _source.duplicate(true)

static func _api() -> Script:
	return load("res://src/gameplay/season/successor_dynasty_owner_v1/owner.gd")

static func draft_from_closed(raw: PackedByteArray, expected_sha256: String) -> Draft:
	var api:=_api()
	if raw.is_empty() or raw.size()>Custody.MAX_MEMBER \
			or not Custody._sha(expected_sha256) or api._hash(raw)!=expected_sha256: return null
	var parsed: Variant=api._bytes_value(raw)
	if not parsed is Dictionary or not parsed.get("ordinary_state") is Dictionary: return null
	var state: Dictionary=parsed.ordinary_state
	if state.get("ordinal")!=3 or state.get("status")!="closed" \
			or not state.get("last_close") is Dictionary or state.last_close.get("reason")!="complete" \
			or not Custody._chain_shape(state.get("archive")): return null
	# The existing cold reader authenticates each predecessor and the native
	# S3 completion fold recursively, once. Hash, shape, caller pins and the
	# structural custody transport alone have no authority to derive starts.
	var tick:=Time.get_ticks_usec()
	var owner: RefCounted=api.restore(parsed)
	var native_usec:=Time.get_ticks_usec()-tick
	if not closed_scope(owner): return null
	var tick_wire:=Time.get_ticks_usec()
	var wire:=Custody.migrate_closed(raw,owner.ordinary_state.lineage)
	var draft:=_draft(owner,wire,expected_sha256)
	if draft!=null:
		draft._native_usec=native_usec
		draft._transport_usec=Time.get_ticks_usec()-tick_wire
	return draft

static func closed_scope(owner: RefCounted) -> bool:
	var api:=_api()
	return owner!=null and owner.get_script()==api and api.Common.starter_dugout_chain_scope(owner) \
		and owner.ordinary_state.get("status")=="closed" and owner.common.sequence==3 \
		and api.Codec.same(owner.common.cursors,{"ordinary":3,"dynasty":0})

# Store passes only its concurrently revision/RawGuard-checked owner.
# The cold caller obtains its predecessor with verified compact transport
# and the existing strict native owner reader, never from a metadata pin.
static func _draft(owner: RefCounted, wire: PackedByteArray, expected_sha256: String) -> Draft:
	if not closed_scope(owner): return null
	var api:=_api()
	var state: Dictionary=owner.ordinary_state
	var lineage: String=state.lineage
	var manifest: Dictionary=Custody.manifest_fields(wire)
	if manifest.is_empty() or manifest.lineage!=lineage or manifest.rows[2].raw_sha256!=expected_sha256: return null
	var ordered: Array[String]=[]
	for row: Dictionary in manifest.rows: ordered.append(row.raw_sha256)
	var ordinal: int=ordered.size()+1
	var seed: int=api.Common.starter_dugout_seed(lineage,ordinal)
	var native: Dictionary=api.Common.Start.make(seed,lineage,ordinal,int(owner.common.sequence),
		api.Common.Life.digest(owner.common.aggregate),api.Common.projection(owner))
	var basis: Dictionary=api.Common.Start.bind_mixed_flat(native,ordered,api._hash(wire))
	var next: SeasonState=api.Adapter.OrdinaryRuntime.create_common(basis,"dugout")
	if next==null or next.phase!=SeasonState.Phase.DRAFT or next.round_index!=0 or not next.picks.is_empty() \
			or not next.results.is_empty() or not next.player_results.is_empty() or next.season_seed!=seed \
			or not next.league_live or next.league_rules.get("league")!="dugout" or next.league_rules.get("tier")!=0 \
			or not next.home_property.is_empty() or next.cash()!=0 or not api.Codec.same(next.career.start_basis,basis): return null
	var result:=Draft.new()
	result.season=next
	result._basis=basis
	result._manifest=wire
	result._common=owner.common.duplicate(true)
	result._source={"sha256":expected_sha256,"lineage":lineage,"ordinal":int(state.ordinal),
		"close":state.last_close.duplicate(true)}
	return result

# Comparison only; this accepts no candidate as authority to write.
static func matches_detached(draft: Draft, basis: Variant, wire: PackedByteArray) -> bool:
	return draft!=null and draft.season!=null and draft.season.phase==SeasonState.Phase.DRAFT \
		and _api().Codec.same(draft.season.career.start_basis,draft._basis) \
		and _api().Codec.same(basis,draft._basis) and wire==draft._manifest

const TRANSITION := "common-starter-dugout-compact-draft-v1"

static func marked_state(state: Variant) -> bool:
	return state is Dictionary and state.get("archive") is Dictionary and state.archive.get("kind")==Custody.KIND

# A shape check is never native custody or write authority. This dialect is
# deliberately limited to the exact generated draft and one native pick.
static func draft_owner_scope(owner: RefCounted) -> bool:
	var api:=_api()
	if owner==null or owner.get_script()!=api or owner.active!="ordinary" or not owner.paused.is_empty() \
			or not api._missing_dynasty(owner) or not owner.migration_v3.is_empty() \
			or not owner.workshop_draft.is_empty() or not owner.workshop_close_history.is_empty() \
			or not owner.workshop_chair_purchase.is_empty() or not owner.workshop_applied.is_empty(): return false
	var s: SeasonState=owner.session
	var state: Dictionary=owner.ordinary_state
	if s==null or s.get_script()!=api.Adapter.OrdinaryRuntime or s.career==null or s.build==null \
			or s.phase!=SeasonState.Phase.DRAFT or s.round_index!=0 or s.picks.size()>1 \
			or not s.results.is_empty() or not s.player_results.is_empty() or s.physical==null \
			or not s.physical.pending.is_empty() or not s.physical.reports.is_empty() or s.physical.cursor!=0 \
			or not s.home_property.is_empty() or not s.league_live or s.league_rules.get("league")!="dugout" \
			or s.league_rules.get("tier")!=0 or s.cash()!=0: return false
	if not api.Codec.keys(state,api.Common.ORDINARY_FIELDS+["archive"]) or not marked_state(state) \
			or state.status!="active" or state.ordinal!=4 or state.proposal!={} or state.last_close!={} \
			or not api.Codec.keys(state.archive,["kind","wire"]) or not state.archive.wire is String \
			or state.archive.wire.is_empty() or state.archive.wire.length()>Custody.MAX_WIRE: return false
	var t: Variant=state.last_transition
	if not api.Codec.keys(t,["kind","id","source","from","to","league","basis"]) \
			or t.kind!=TRANSITION or t.league!="dugout" or t.from!=3 or t.to!=4 \
			or t.id!="%s:3:next"%state.lineage or not Custody._sha(t.source) \
			or not api.Common.Start.valid(t.basis) or not api.Common.Start.detached_mixed_flat(t.basis): return false
	return t.basis.lineage==state.lineage and s.season_seed==t.basis.seed \
		and state.start_basis_sha256==api.Common.Life.digest(t.basis) \
		and api.Codec.same(t.basis,s.career.start_basis) \
		and owner.common.get("sequence")==3 and api.Codec.same(owner.common.get("cursors"),{"ordinary":3,"dynasty":0})

static func _owned_draft(prior: RefCounted, draft: Draft) -> RefCounted:
	if draft==null: return null
	var api:=_api()
	var next: RefCounted=api._new()
	if not api.Codec.install_fields(next,api.DATA_FIELDS,api.Codec.data_fields(prior,api.DATA_FIELDS)): return null
	next.session=draft.season
	var lineage: String=draft._basis.lineage
	next.ordinary_state={"status":"active","ordinal":4,"lineage":lineage,"last_close":{},"proposal":{},
		"start_basis_sha256":api.Common.Life.digest(draft._basis),
		"last_transition":{"kind":TRANSITION,"id":"%s:3:next"%lineage,"source":draft._source.close.session_sha256,
			"from":3,"to":4,"league":"dugout","basis":draft.basis()},
		"archive":{"kind":Custody.KIND,"wire":draft._manifest.get_string_from_utf8()}}
	return next if draft_owner_scope(next) else null

static func _owned_from_pinned(prior: RefCounted, raw: PackedByteArray) -> RefCounted:
	if not closed_scope(prior): return null
	var wire:=Custody.migrate_closed(raw,prior.ordinary_state.lineage)
	return _owned_draft(prior,_draft(prior,wire,Custody._hash(raw)))

# This explicit pure envelope assembly is not a general v5 session codec.
# All writes still originate in fixed Store producers.
static func compact_payload(owner: RefCounted) -> Dictionary:
	if not draft_owner_scope(owner): return {}
	var api:=_api()
	var native: Script=api.Adapter.Ordinary
	if not native._structural_valid(owner.session): return {}
	var current: Dictionary=native._payload(owner.session)
	return api._pack_fields(owner,current) if not current.is_empty() else {}

static func first_pick_payload(owner: RefCounted, candidate: RefCounted) -> Dictionary:
	if not draft_owner_scope(owner) or not draft_owner_scope(candidate) \
			or not owner.session.picks.is_empty() or candidate.session.picks.size()!=1: return {}
	var api:=_api()
	# Confirm the App fork against one independently derived native choice
	# from the exact admitted active session without rereading predecessors.
	var replay: SeasonState=owner.session.fork_live()
	var id: String=candidate.session.picks[0]
	if replay==null or not replay.offers().has(id) or not replay.choose_player(id): return {}
	var native: Script=api.Adapter.Ordinary
	var data:=compact_payload(candidate)
	return data if not data.is_empty() and api.Codec.same(native._payload(replay),data.ordinary) else {}

static func restore_owned(value: Variant) -> RefCounted:
	var api:=_api()
	var meta: RefCounted=api._restore_metadata(value)
	if meta==null or meta.active!="ordinary" or not meta.paused.is_empty() or not api._missing_dynasty(meta) \
			or not marked_state(meta.ordinary_state): return null
	var archive: Dictionary=meta.ordinary_state.archive
	var transition: Variant=meta.ordinary_state.get("last_transition")
	var current: Variant=value.get("ordinary")
	if not api.Codec.keys(archive,["kind","wire"]) or not archive.wire is String \
			or archive.wire.length()>Custody.MAX_WIRE or not transition is Dictionary \
			or not api.Common.Start.valid(transition.get("basis")) or not api.Common.Start.detached_mixed_flat(transition.basis) \
			or not current is Dictionary or not current.get("state") is Dictionary \
			or current.state.get("phase")!=SeasonState.Phase.DRAFT or not current.state.get("picks") is Array \
			or current.state.picks.size()>1: return null
	var wire: PackedByteArray=archive.wire.to_utf8_buffer()
	var manifest:=Custody.manifest_fields(wire)
	if manifest.is_empty() or Custody._hash(wire)!=transition.basis.archive_origin.manifest_sha256 \
			or manifest.lineage!=transition.basis.lineage \
			or not api.Codec.same(manifest.rows.map(func(row): return row.raw_sha256),transition.basis.archive_origin.ordered_sha256): return null
	var last: Array[PackedByteArray]=[]
	var retain_terminal:=func(i: int, raw: PackedByteArray) -> bool:
		if i==3: last.append(raw)
		return true
	if not Custody._walk(wire,retain_terminal) or last.size()!=1: return null
	# The strict stock reader replays the exact three original owners and
	# authentic folds once; never use the rotating .bak or rows alone.
	var prior: RefCounted=api.restore(api._bytes_value(last[0]))
	var candidate:=_owned_draft(prior,_draft(prior,wire,manifest.rows[2].raw_sha256))
	if candidate==null: return null
	if not current.state.picks.is_empty():
		var id: Variant=current.state.picks[0]
		if not id is String or not candidate.session.offers().has(id) or not candidate.session.choose_player(id): return null
	var rebuilt:=compact_payload(candidate)
	return candidate if not rebuilt.is_empty() and api.Codec.same(rebuilt,value) else null

# Display only; Store admits and pins the exact slot before calling this.
static func display(archive: Dictionary) -> Array:
	var api:=_api()
	var rows: Array=[]
	var collect:=func(i: int, raw: PackedByteArray) -> bool:
		var row: Dictionary=api.Common.StandardArchive._display_scores(api._bytes_value(raw),true)
		if row.get("ordinal")!=i or row.results.size()!=33: return false
		row["league"]="standard" if i==1 else "dugout"
		rows.append(row)
		return true
	return rows if Custody._walk(archive.wire.to_utf8_buffer(),collect) and rows.size()==3 else []
