extends RefCounted
## Fixed compact owner, reconstructed from the strict historical reader and
## the current season’s original chronological native evidence.
const Custody=preload("starter_mixed_custody.gd")

class Draft:
	extends RefCounted
	var season: SeasonState
	var _basis: Dictionary
	var _common: Dictionary
	var _manifest: PackedByteArray
	var _source: Dictionary
	var _native_usec: int
	var _transport_usec: int
	func basis() -> Dictionary: return _basis.duplicate(true)
	func common() -> Dictionary: return _common.duplicate(true)
	func manifest() -> PackedByteArray: return _manifest
	func source() -> Dictionary: return _source.duplicate(true)

static func _api() -> Script:
	return load("res://src/gameplay/season/successor_dynasty_owner_v1/owner.gd")

static func draft_from_closed(raw: PackedByteArray, expected_sha256: String) -> Draft:
	var api:=_api()
	if raw.is_empty() or raw.size()>Custody.MAX_MEMBER \
			or not Custody._sha(expected_sha256) or api._hash(raw)!=expected_sha256: return null
	var parsed: Variant=api._bytes_value(raw)
	if not parsed is Dictionary or not parsed.get("ordinary_state") is Dictionary: return null
	var state: Dictionary=parsed.ordinary_state
	if state.get("ordinal")!=3 or state.get("status")!="closed" \
			or not state.get("last_close") is Dictionary or state.last_close.get("reason")!="complete" \
			or not Custody._chain_shape(state.get("archive")): return null
	# The existing cold reader authenticates each predecessor and the native
	# S3 completion fold recursively, once. Hash, shape, caller pins and the
	# structural custody transport alone have no authority to derive starts.
	var tick:=Time.get_ticks_usec()
	var owner: RefCounted=api.restore(parsed)
	var native_usec:=Time.get_ticks_usec()-tick
	if not closed_scope(owner): return null
	var tick_wire:=Time.get_ticks_usec()
	var wire:=Custody.migrate_closed(raw,owner.ordinary_state.lineage)
	var draft:=_draft(owner,wire,expected_sha256)
	if draft!=null:
		draft._native_usec=native_usec
		draft._transport_usec=Time.get_ticks_usec()-tick_wire
	return draft

static func closed_scope(owner: RefCounted) -> bool:
	var api:=_api()
	return owner!=null and owner.get_script()==api and api.Common.starter_dugout_chain_scope(owner) \
		and owner.ordinary_state.get("status")=="closed" and owner.common.sequence==3 \
		and api.Codec.same(owner.common.cursors,{"ordinary":3,"dynasty":0})

# Store passes only its concurrently revision/RawGuard-checked owner.
# The cold caller obtains its predecessor with verified compact transport
# and the existing strict native owner reader, never from a metadata pin.
static func _draft(owner: RefCounted, wire: PackedByteArray, expected_sha256: String) -> Draft:
	if not closed_scope(owner): return null
	var api:=_api()
	var state: Dictionary=owner.ordinary_state
	var lineage: String=state.lineage
	var manifest: Dictionary=Custody.manifest_fields(wire)
	if manifest.is_empty() or manifest.lineage!=lineage or manifest.rows[2].raw_sha256!=expected_sha256: return null
	var ordered: Array[String]=[]
	for row: Dictionary in manifest.rows: ordered.append(row.raw_sha256)
	var ordinal: int=ordered.size()+1
	var seed: int=api.Common.starter_dugout_seed(lineage,ordinal)
	var native: Dictionary=api.Common.Start.make(seed,lineage,ordinal,int(owner.common.sequence),
		api.Common.Life.digest(owner.common.aggregate),api.Common.projection(owner))
	var basis: Dictionary=api.Common.Start.bind_mixed_flat(native,ordered,api._hash(wire))
	var next: SeasonState=api.Adapter.OrdinaryRuntime.create_common(basis,"dugout")
	if next==null or next.phase!=SeasonState.Phase.DRAFT or next.round_index!=0 or not next.picks.is_empty() \
			or not next.results.is_empty() or not next.player_results.is_empty() or next.season_seed!=seed \
			or not next.league_live or next.league_rules.get("league")!="dugout" or next.league_rules.get("tier")!=0 \
			or not next.home_property.is_empty() or next.cash()!=0 or not api.Codec.same(next.career.start_basis,basis): return null
	var result:=Draft.new()
	result.season=next
	result._basis=basis
	result._manifest=wire
	result._common=owner.common.duplicate(true)
	result._source={"sha256":expected_sha256,"lineage":lineage,"ordinal":int(state.ordinal),
		"close":state.last_close.duplicate(true)}
	return result

# Comparison only; this accepts no candidate as authority to write.
static func matches_detached(draft: Draft, basis: Variant, wire: PackedByteArray) -> bool:
	return draft!=null and draft.season!=null and draft.season.phase==SeasonState.Phase.DRAFT \
		and _api().Codec.same(draft.season.career.start_basis,draft._basis) \
		and _api().Codec.same(basis,draft._basis) and wire==draft._manifest

const TRANSITION := "common-starter-dugout-compact-draft-v1"

static func marked_state(state: Variant) -> bool:
	return state is Dictionary and state.get("archive") is Dictionary and state.archive.get("kind")==Custody.KIND

# A scope condition is never native custody or write authority. The active
# epoch covers Stock Dugout through play; close/rollover remain separate.
static func season_owner_scope(owner: RefCounted) -> bool:
	var api:=_api()
	if owner==null or owner.get_script()!=api or owner.active!="ordinary" or not owner.paused.is_empty() \
			or not api._missing_dynasty(owner) or not owner.migration_v3.is_empty() \
			or not owner.workshop_draft.is_empty() or not owner.workshop_close_history.is_empty() \
			or not owner.workshop_chair_purchase.is_empty() or not owner.workshop_applied.is_empty(): return false
	var s: SeasonState=owner.session
	var state: Dictionary=owner.ordinary_state
	if s==null or s.get_script()!=api.Adapter.OrdinaryRuntime or s.career==null or s.build==null \
			or s.physical==null or not s.physical.needed.is_empty() or not s.physical.error.is_empty() or s.physical.projecting \
			or not s.home_property.is_empty() or not s.league_live or s.league_rules.get("league")!="dugout" \
			or s.league_rules.get("tier")!=0 or not _season_phase(s): return false
	if not api.Codec.keys(state,api.Common.ORDINARY_FIELDS+["archive"]) or not marked_state(state) \
			or state.status!="active" or state.ordinal!=4 or state.proposal!={} or state.last_close!={} \
			or not api.Codec.keys(state.archive,["kind","wire"]) or not state.archive.wire is String \
			or state.archive.wire.is_empty() or state.archive.wire.length()>Custody.MAX_WIRE: return false
	var t: Variant=state.last_transition
	if not api.Codec.keys(t,["kind","id","source","from","to","league","basis"]) \
			or t.kind!=TRANSITION or t.league!="dugout" or t.from!=3 or t.to!=4 \
			or t.id!="%s:3:next"%state.lineage or not Custody._sha(t.source) \
			or not api.Common.Start.valid(t.basis) or not api.Common.Start.detached_mixed_flat(t.basis): return false
	return t.basis.lineage==state.lineage and s.season_seed==t.basis.seed \
		and state.start_basis_sha256==api.Common.Life.digest(t.basis) \
		and api.Codec.same(t.basis,s.career.start_basis) \
		and owner.common.get("sequence")==3 and api.Codec.same(owner.common.get("cursors"),{"ordinary":3,"dynasty":0})

static func draft_owner_scope(owner: RefCounted) -> bool:
	return season_owner_scope(owner) and owner.session.phase==SeasonState.Phase.DRAFT

static func _season_phase(s: SeasonState) -> bool:
	var api:=_api()
	if s.phase==SeasonState.Phase.DRAFT:
		return s.picks.size()<4 and s.cash()==0 and s.round_index==0 and s.results.is_empty() \
			and s.player_results.is_empty() and s.physical.pending.is_empty() and s.physical.reports.is_empty() \
			and s.physical.cursor==0
	if s.picks.size()!=4 or s.physical.cursor!=s.physical.reports.size() \
			or s.results.size()!=s.player_results.size()+s.physical.cursor: return false
	var played:=s.results.size()
	match s.phase:
		SeasonState.Phase.REGULAR:
			if s.round_index<0 or s.round_index>=10 or played!=s.round_index*3 \
					or s.player_results.size()!=s.round_index: return false
		SeasonState.Phase.SEMIFINAL:
			if s.round_index!=10 or played!=30 or s.player_results.size()!=10: return false
		SeasonState.Phase.FINAL:
			if s.round_index!=10 or played!=32 or s.player_results.size()!=11: return false
		SeasonState.Phase.COMPLETE:
			return s.round_index==10 and played==33 and s.physical.pending.is_empty() \
				and not s.build.pack_pending() and api.Common._starter_dugout_paid_terminal(s)
		_: return false
	if not api.Common._starter_dugout_rewards_match(s): return false
	var fixture:=s.pending_fixture()
	if fixture.is_empty() or (fixture.home!=0 and fixture.away!=0): return false
	if s.physical.pending.is_empty(): return true
	var pending:=s.physical.pending
	var human: Variant=pending.get("human")
	var reports: Variant=pending.get("reports")
	return not s.build.pack_pending() and api.Codec.keys(pending,["human","reports"]) \
		and human is Array and human.size()==13 and api.Codec.whole(human[0],fixture.id,fixture.id) \
		and reports is Array and reports.size()<=5

static func game_entry_scope(owner: RefCounted) -> bool:
	return season_owner_scope(owner) and owner.session.phase!=SeasonState.Phase.DRAFT \
		and owner.session.phase!=SeasonState.Phase.COMPLETE and owner.session.physical.pending.is_empty() \
		and not owner.session.build.pack_pending()

# Fixed native Shop actions only. The same bound applies at hot admission,
# UI quoting and chronological replay; prices/targets are still native.
const SHOP_OPS: Array[String]=["open","pack_open","pack_pick","pack_skip"]

static func shop_command_allowed(s: SeasonState, command: Dictionary) -> bool:
	if s==null or not s.shop_available() or not s.physical.pending.is_empty() \
			or not SHOP_OPS.has(str(command.get("op",""))) or not command.get("id") is String \
			or s.build._requests.has(command.id): return false
	if command.op=="open":
		return not s.build._visit.get("open") and _api().Codec.same(command,
			{"id":"shop-open:%d"%s.build.revision(),"rev":s.build.revision(),"op":"open"})
	return true

static func _owned_draft(prior: RefCounted, draft: Draft) -> RefCounted:
	if draft==null: return null
	var api:=_api()
	var next: RefCounted=api._new()
	if not api.Codec.install_fields(next,api.DATA_FIELDS,api.Codec.data_fields(prior,api.DATA_FIELDS)): return null
	next.session=draft.season
	var lineage: String=draft._basis.lineage
	next.ordinary_state={"status":"active","ordinal":4,"lineage":lineage,"last_close":{},"proposal":{},
		"start_basis_sha256":api.Common.Life.digest(draft._basis),
		"last_transition":{"kind":TRANSITION,"id":"%s:3:next"%lineage,"source":draft._source.close.session_sha256,
			"from":3,"to":4,"league":"dugout","basis":draft.basis()},
		"archive":{"kind":Custody.KIND,"wire":draft._manifest.get_string_from_utf8()}}
	return next if season_owner_scope(next) else null

static func _owned_from_pinned(prior: RefCounted, raw: PackedByteArray) -> RefCounted:
	if not closed_scope(prior): return null
	var wire:=Custody.migrate_closed(raw,prior.ordinary_state.lineage)
	return _owned_draft(prior,_draft(prior,wire,Custody._hash(raw)))

# This explicit pure envelope assembly is not a general v5 session codec.
# All writes still originate in fixed Store producers.
static func compact_payload(owner: RefCounted) -> Dictionary:
	if not season_owner_scope(owner): return {}
	var api:=_api()
	var native: Script=api.Adapter.Ordinary
	if not native._structural_valid(owner.session): return {}
	var current: Dictionary=native._payload(owner.session)
	return api._pack_fields(owner,current) if not current.is_empty() else {}

static func draft_pick_payload(owner: RefCounted, candidate: RefCounted) -> Dictionary:
	if not draft_owner_scope(owner) or not season_owner_scope(candidate) \
			or candidate.session.picks.size()!=owner.session.picks.size()+1: return {}
	var api:=_api()
	# Confirm the App fork against one independently derived native choice
	# from the exact admitted active session without rereading predecessors.
	var replay: SeasonState=owner.session.fork_live()
	var id: String=candidate.session.picks[-1]
	if replay==null or not replay.offers().has(id) or not replay.choose_player(id): return {}
	var native: Script=api.Adapter.Ordinary
	var data:=compact_payload(candidate)
	return data if not data.is_empty() and api.Codec.same(native._payload(replay),data.ordinary) else {}

static func restore_owned(value: Variant) -> RefCounted:
	var api:=_api()
	var meta: RefCounted=api._restore_metadata(value)
	if meta==null or meta.active!="ordinary" or not meta.paused.is_empty() or not api._missing_dynasty(meta) \
			or not marked_state(meta.ordinary_state): return null
	var archive: Dictionary=meta.ordinary_state.archive
	var transition: Variant=meta.ordinary_state.get("last_transition")
	var current: Variant=value.get("ordinary")
	if not api.Codec.keys(archive,["kind","wire"]) or not archive.wire is String \
			or archive.wire.length()>Custody.MAX_WIRE or not transition is Dictionary \
			or not api.Common.Start.valid(transition.get("basis")) or not api.Common.Start.detached_mixed_flat(transition.basis) \
			or not current is Dictionary or not current.get("state") is Dictionary \
			or not current.state.get("picks") is Array: return null
	var phase: Variant=current.state.get("phase")
	if phase==SeasonState.Phase.DRAFT:
		if current.state.picks.size()>=4: return null
	elif not api.Codec.whole(phase,SeasonState.Phase.REGULAR,SeasonState.Phase.COMPLETE) \
			or current.state.picks.size()!=4 or not api.Codec.whole(current.state.get("round_index"),0,10): return null
	var wire: PackedByteArray=archive.wire.to_utf8_buffer()
	var manifest:=Custody.manifest_fields(wire)
	if manifest.is_empty() or Custody._hash(wire)!=transition.basis.archive_origin.manifest_sha256 \
			or manifest.lineage!=transition.basis.lineage \
			or not api.Codec.same(manifest.rows.map(func(row): return row.raw_sha256),transition.basis.archive_origin.ordered_sha256): return null
	var last: Array[PackedByteArray]=[]
	var retain_terminal:=func(i: int, raw: PackedByteArray) -> bool:
		if i==3: last.append(raw)
		return true
	if not Custody._walk(wire,retain_terminal) or last.size()!=1: return null
	# The strict stock reader replays the exact three original owners and
	# authentic folds once; never use the rotating .bak or rows alone.
	var prior: RefCounted=api.restore(api._bytes_value(last[0]))
	var candidate:=_owned_draft(prior,_draft(prior,wire,manifest.rows[2].raw_sha256))
	if candidate==null: return null
	for id: Variant in current.state.picks:
		if not id is String or not candidate.session.offers().has(id) or not candidate.session.choose_player(id): return null
	if phase!=SeasonState.Phase.DRAFT:
		candidate.session=_replay_current(candidate.session,current)
		if candidate.session==null or not season_owner_scope(candidate): return null
	var rebuilt:=compact_payload(candidate)
	return candidate if not rebuilt.is_empty() and api.Codec.same(rebuilt,value) else null

# Dugout stored its original native H13 arguments in the player row,
# including the exact per-opponent physical evidence. No new data or source
# compression schema is required to reissue it on the authored draft.
static func _paid_human(raw: Variant, fixture: Dictionary) -> Array:
	var api:=_api()
	if not raw is Dictionary or fixture.is_empty(): return []
	var row: Dictionary=raw
	if not api.Codec.whole(row.get("id"),fixture.id,fixture.id) or not api.Codec.whole(row.get("away_runs"),0,9999) \
			or not api.Codec.whole(row.get("home_runs"),0,9999) or not row.get("pitching_current") is Dictionary: return []
	var opponent: String=str(int(fixture.away if fixture.home==0 else fixture.home))
	var ai: Variant=row.get("ai_tactics")
	if not api.Codec.keys(ai,[opponent]) or not ai[opponent] is Dictionary: return []
	return [row.id,row.away_runs,row.home_runs,row.get("performance",{}),row.get("used_gear",[]),
		row.get("tactics",[]),row.get("batting",{}),row.get("stances",[]),row.get("fielding",[]),
		row.pitching_current,row.get("field_supply",{}),row.get("frozen",[]),ai[opponent]]

# A normal regular transaction consumes two CPU jobs. If the native bracket
# eliminates the managed club, missing semis/final are resolved in that same
# transaction; the receipt list only locates candidates, never defines a winner.
static func _paid_report_count(s: SeasonState, more_human: bool, phase: int) -> int:
	match s.phase:
		SeasonState.Phase.REGULAR:
			return 5 if s.round_index==9 and not more_human and phase==SeasonState.Phase.COMPLETE else 2
		SeasonState.Phase.SEMIFINAL:
			return 2 if not more_human and phase==SeasonState.Phase.COMPLETE else 1
		SeasonState.Phase.FINAL:
			return 0
	return -1

# Compare each emitted authored Build event to the supplied prefix. Consume
# only a contiguous visit’s allowed native Shop events; every other event must
# be emitted anew by play from the reconstructed state and actual evidence.
static func _journal_prefix(s: SeasonState, journal: Array, first: int) -> bool:
	var events: Array=s.build._events
	return first>=0 and first<=events.size() and events.size()<=journal.size() \
		and _api().Codec.same(events.slice(first),journal.slice(first,events.size()))

static func _replay_shop(s: SeasonState, journal: Array) -> bool:
	var first: int=s.build.revision()
	var index:=first
	while index<journal.size():
		var command: Variant=journal[index]
		if not command is Dictionary: return false
		if not SHOP_OPS.has(str(command.get("op",""))): break
		if not shop_command_allowed(s,command): return false
		var bought: SeasonBuild=s.build.candidate(command)
		if bought==null: return false
		s.adopt_build(bought)
		index+=1
		if s.build!=bought or s.build.revision()!=index: return false
	return _journal_prefix(s,journal,first)

# The marked mixed reader has already reconstructed this pre-round
# source. Older responsive Continue can give otherwise equal fixture and
# ownership journal numbers their JSON types. Use only the exact, detached
# native image, never a supplied digest or a relaxed report validator.
static func _project_mixed_round(source: SeasonState) -> Dictionary:
	var api:=_api()
	var pending: Dictionary=source.physical.pending
	var eligible: bool=source.phase==SeasonState.Phase.REGULAR and pending.get("reports") is Array \
			and not pending.reports.is_empty() and pending.reports.size()<=2
	var original: Dictionary=api.Adapter.OrdinaryPending._project_for_mixed_restore(source) \
			if eligible else api.Adapter.OrdinaryPending.project(source)
	var mismatch: Dictionary=original.get("_mixed_request_mismatch",{})
	original.erase("_mixed_request_mismatch")
	var stock: Dictionary={"projection":original,"source":source}
	if not eligible or mismatch.is_empty() \
			or original.get("error")!="The pending round contains an unexpected physical fixture.": return stock
	var index: int=mismatch.cursor-source.physical.cursor
	if index<0 or index>=pending.reports.size(): return stock
	var row: Variant=pending.reports[index]
	if not row is Dictionary or row.get("fixture")!=mismatch.fixture \
			or not row.get("request") is String or row.request.length()!=64 \
			or not row.request.is_valid_hex_number(false) or row.request!=mismatch.saved \
			or row.request==mismatch.expected: return stock
	# This records only why the stock candidate failed. The exact typed image
	# and the complete stock projector below still decide admission.
	var core: SeasonState=source.fork_live()
	if core==null: return stock
	core.physical.pending.clear()
	var native: Script=api.Adapter.Ordinary
	var earned: Dictionary=native._payload(core)
	if earned.is_empty(): return stock
	var typed: SeasonState=native._hydrate_current_shape(earned,false)
	if typed==null or not native._canonical_named_context(typed) \
			or not api.Codec.same(native._payload(typed),earned): return stock
	typed.physical.pending=pending.duplicate(true)
	var checked: Dictionary=api.Adapter.OrdinaryPending.project(typed)
	return stock if checked.has("error") else {"projection":checked,"source":typed}

# The input session only comes from the genuine authored seed and picks when
# used for admission. Each settled boundary is projected once by the original
# native path. Current pending is validated but not adopted as a paid result.
static func _replay_current(source: SeasonState, current: Variant) -> SeasonState:
	var api:=_api()
	if source==null or source.round_index!=0 or source.picks.size()!=4 or not source.physical.pending.is_empty() \
			or not source.results.is_empty() or source.pending_fixture().get("id")!=0 \
			or not current is Dictionary or not current.get("state") is Dictionary \
			or not current.get("physical") is Dictionary or not current.physical.get("pending") is Dictionary \
			or not current.physical.get("reports") is Array or not current.get("build") is Dictionary \
			or not current.build.get("build") is Dictionary or not current.build.build.get("_events") is Array:
		return null
	var target: Variant=current.state.get("phase")
	var people: Variant=current.state.get("player_results")
	var reports: Array=current.physical.reports
	var journal: Array=current.build.build._events
	if not api.Codec.whole(target,SeasonState.Phase.REGULAR,SeasonState.Phase.COMPLETE) \
			or not people is Array or people.size()>12 or reports.size()>23 or journal.size()>SeasonBuild.MAX_EVENTS \
			or not _journal_prefix(source,journal,0): return null
	var next: SeasonState=source
	var cursor:=0
	for i: int in people.size():
		var count:=_paid_report_count(next,i+1<people.size(),int(target))
		if count<0 or cursor+count>reports.size() or next.build.pack_pending(): return null
		var human:=_paid_human(people[i],next.pending_fixture())
		if human.is_empty(): return null
		var revision: int=next.build.revision()
		next.physical.pending={"human":human,"reports":reports.slice(cursor,cursor+count)}
		var projected: Dictionary=_project_mixed_round(next).projection
		next=projected.get("candidate")
		cursor+=count
		if projected.has("error") or projected.has("fixture") or next==null \
				or not _season_phase(next) or next.player_results.size()!=i+1 or next.physical.cursor!=cursor \
				or not _journal_prefix(next,journal,revision) or not _replay_shop(next,journal): return null
	if cursor!=reports.size() or next.build.revision()!=journal.size(): return null
	next.physical.pending=current.physical.pending.duplicate(true)
	if not _season_phase(next): return null
	if not next.physical.pending.is_empty():
		var pending_read:=_project_mixed_round(next)
		if pending_read.projection.has("error"): return null
		next=pending_read.source
	var native: Script=api.Adapter.Ordinary
	return next if native._structural_valid(next) and api.Codec.same(native._payload(next),current) else null

# Display only; Store admits and pins the exact slot before calling this.
static func display(archive: Dictionary) -> Array:
	var api:=_api()
	var rows: Array=[]
	var collect:=func(i: int, raw: PackedByteArray) -> bool:
		var row: Dictionary=api.Common.StandardArchive._display_scores(api._bytes_value(raw),true)
		if row.get("ordinal")!=i or row.results.size()!=33: return false
		row["league"]="standard" if i==1 else "dugout"
		rows.append(row)
		return true
	return rows if Custody._walk(archive.wire.to_utf8_buffer(),collect) and rows.size()==3 else []
