extends RefCounted
## Ordered native archive, close, and derived admission for an earned named
## Base league. Pure projection only: Store owns the revision and byte pin.
const Archive = preload("starter_base_archive.gd")
const Equipment = preload("earned_equipment_base.gd")
const Life = preload("lifecycle.gd")
const Progress = preload("common_progress.gd")
const Unlocks = preload("res://src/gameplay/season/league_unlocks.gd")
const KIND := "installed-earned-named-base-v1"
const EQUIPMENT_TO_EARNINGS := "equipment:0>earnings:0"
const DUGOUT_D2_TO_SCRAMBLE := "dugout:2>scramble:0"
const DUGOUT_D1_TO_DOUBLEHEADER := "dugout:1>doubleheader:0"
const FITTED_DUGOUT_KIND := "installed-forward-dugout-d1-draft-v1"
const DugoutD2 = preload("earned_dugout_d2.gd")
const Paired = preload("res://src/gameplay/season/successor_live_play_v1/paired_first_projection.gd")

static func target(value: Dictionary) -> String:
	match value.get("offer",{}).get("route","") if value.get("offer") is Dictionary else "":
		EQUIPMENT_TO_EARNINGS: return "earnings"
		DUGOUT_D2_TO_SCRAMBLE: return "scramble"
		DUGOUT_D1_TO_DOUBLEHEADER: return "doubleheader"
		_: return ""

static func page_ids(value: Dictionary) -> Array:
	var goal:=target(value)
	if goal.is_empty(): return []
	var offer: Dictionary=value.offer
	if goal in ["earnings","doubleheader"]: return [offer.get("origin_page_sha256"),offer.get("close_page_sha256")]
	return [offer.get("origin_page_sha256"),offer.get("prior_title_page_sha256"),offer.get("close_page_sha256")]

# Only this authored pair is installed. Other named legs must supply their
# own authenticated source proof, not a caller-selected league or badge.
static func stage(source: Dictionary, season: SeasonState, origin_page: PackedByteArray) -> Dictionary:
	if season==null or season.career==null or Equipment.finish(season)!="final_champion" \
			or season.champion!=0 or not season.physical.pending.is_empty() or not season.physical.needed.is_empty() \
			or season.physical.projecting or season.build.pack_pending() \
			or not Archive.C.same(season.league_rules,LeagueRules.capture("equipment")): return {}
	if source.get("kind")!=Equipment.DRAFT_KIND or not source.get("head") is Dictionary \
			or not source.get("current") is Dictionary or not source.get("aggregate") is Dictionary \
			or not source.get("offer") is Dictionary: return {}
	var api:=Archive._owner()
	var image: Dictionary=api.Adapter.capture("ordinary",season)
	var basis: Dictionary=season.career.start_basis
	var previous: Dictionary=source.aggregate
	if image.is_empty() or not Archive.C.same(source.current,image) or not Progress.valid(previous) \
			or previous.closed!=1 or previous.titles!={"dugout:0":true} \
			or basis.get("ordinal")!=2 or basis.get("common_sequence")!=1 \
			or basis.get("common_sha256")!=Life.digest(previous) or source.offer.get("basis_sha256")!=Life.digest(basis): return {}
	var old_head_bytes:=Archive._wire(source.head)
	var old_head:=Archive._head(old_head_bytes)
	if old_head.is_empty() or old_head.ordinal!=2 or Archive._hash(origin_page)!=old_head.pages[0] \
			or old_head.pages[0]!=source.offer.get("page_sha256"): return {}
	var lineage: String=basis.get("lineage","")
	var id: String="%s:2:close"%lineage
	var folded:=Progress.fold(previous,"ordinary",season,id,2,"complete")
	if folded.is_empty() or folded.receipt.get("finish")!="champion" or not Progress.valid(folded.aggregate) \
			or folded.aggregate.closed!=2 or folded.aggregate.titles!={"dugout:0":true,"equipment:0":true} \
			or not "earnings" in Unlocks.from_titles(folded.aggregate.titles): return {}
	var close: Dictionary={"reason":"complete","ordinal":2,"id":id,"league":"equipment","tier":0,
		"session_sha256":Life.digest(source),"native_session_sha256":Life.digest(image),
		"aggregate_sha256":Life.digest(folded.aggregate),"receipt":folded.receipt}
	# Page 2 carries the exact Equipment reducer journal and its native
	# terminal image. Its predecessor is the immutable Dugout title page.
	var chain:=Archive.make_page_and_head(old_head_bytes,source,close)
	if chain.is_empty(): return {}
	var head:=Archive._head(chain.head)
	if head.is_empty() or head.ordinal!=3 or head.pages.size()!=2 or head.pages[0]!=old_head.pages[0] \
			or head.pages[1]!=Archive._hash(chain.page): return {}
	var seed: int=("ordinary:%s:3:nextseed"%lineage).hash() & 0x7fffffff
	# Projection only consumes its common aggregate and native source basis.
	var detached: Dictionary={"common":{"aggregate":folded.aggregate},"session":season}
	var start: Dictionary=api.Adapter.OrdinaryRuntime.CommonStart.make(seed,lineage,3,2,
		close.aggregate_sha256,api.Common.projection(detached,season))
	var draft: SeasonState=null if start.is_empty() else api.Adapter.OrdinaryRuntime.create_common(start,"earnings")
	if draft==null or not Archive.C.same(draft.league_rules,LeagueRules.capture("earnings")) \
			or not Archive.C.same(draft.career.start_basis,start) \
			or draft.phase!=SeasonState.Phase.DRAFT or not draft.picks.is_empty() or draft.build.cash()!=0: return {}
	var current: Dictionary=api.Adapter.capture("ordinary",draft)
	if current.is_empty(): return {}
	var offer: Dictionary={"route":EQUIPMENT_TO_EARNINGS,"source_close":id,
		"source_session_sha256":close.native_session_sha256,"previous_head_sha256":Archive._hash(old_head_bytes),
		"origin_page_sha256":head.pages[0],"close_page_sha256":head.pages[1],
		"head_sha256":Archive._hash(chain.head),"basis_sha256":Life.digest(start)}
	var value: Dictionary={"kind":KIND,"version":1,"head":head,"offer":offer,
		"current":current,"aggregate":folded.aggregate,"receipt":folded.receipt}
	return {"value":value,"draft":draft,"page":chain.page}

# D2's authenticated championship is already banked. Re-use the native
# source verification from the same-league D3 route, without folding it again.
static func stage_scramble(source: Dictionary, season: SeasonState, previous_pages: Array) -> Dictionary:
	if not Archive.C.keys(source,["kind","version","head","current"]) or source.kind!=FITTED_DUGOUT_KIND \
			or not Archive.C.whole(source.version,4,4) or not source.head is Dictionary \
			or not source.current is Dictionary or previous_pages.size()!=2 \
			or not previous_pages[0] is PackedByteArray or not previous_pages[1] is PackedByteArray: return {}
	var verified:=DugoutD2._champion_source(source.current,source.head,season,2)
	if verified.is_empty() or verified.pages!=[Archive._hash(previous_pages[0]),Archive._hash(previous_pages[1])]: return {}
	var common: Dictionary=source.current.common
	var native_close: Dictionary=source.current.ordinary_state.last_close
	if not Progress.valid(common.aggregate) or common.aggregate.closed!=3 \
			or common.aggregate.titles!={"dugout:0":true,"dugout:1":true,"dugout:2":true} \
			or not Unlocks.from_titles(common.aggregate.titles).has("scramble"): return {}
	var old_head:=Archive._wire(verified)
	var close: Dictionary={"reason":"complete","ordinal":3,"id":native_close.id,"league":"dugout","tier":2,
		"session_sha256":Life.digest(source),"native_session_sha256":native_close.session_sha256,
		"aggregate_sha256":Life.digest(common.aggregate),"receipt":native_close.receipt}
	var linked:=Archive.make_page_and_head(old_head,source,close)
	if linked.is_empty(): return {}
	var head:=Archive._head(linked.head)
	if head.is_empty() or head.ordinal!=4 or head.pages.slice(0,2)!=verified.pages \
			or head.pages[2]!=Archive._hash(linked.page): return {}
	var api:=Archive._owner()
	var lineage: String=source.current.ordinary_state.lineage
	var seed: int=("ordinary:%s:4:nextseed"%lineage).hash() & 0x7fffffff
	var live: Script=api.Adapter.OrdinaryRuntime
	var detached: Dictionary={"common":common,"session":season}
	var basis: Dictionary=live.CommonStart.make(seed,lineage,4,3,close.aggregate_sha256,api.Common.projection(detached,season))
	var draft: SeasonState=null if basis.is_empty() else live.create_common(basis,"scramble")
	if draft==null or draft.career==null or not Archive.C.same(draft.career.start_basis,basis) \
			or not Equipment.named_runtime(draft) or draft.phase!=SeasonState.Phase.DRAFT \
			or not draft.picks.is_empty() or draft.build.cash()!=0: return {}
	var image: Dictionary=api.Adapter.capture("ordinary",draft)
	if image.is_empty(): return {}
	var offer: Dictionary={"route":DUGOUT_D2_TO_SCRAMBLE,"source_close":close.id,
		"source_session_sha256":close.native_session_sha256,"previous_head_sha256":Archive._hash(old_head),
		"origin_page_sha256":head.pages[0],"prior_title_page_sha256":head.pages[1],"close_page_sha256":head.pages[2],
		"head_sha256":Archive._hash(linked.head),"basis_sha256":Life.digest(basis)}
	var value: Dictionary={"kind":KIND,"version":1,"head":head,"offer":offer,"current":image,
		"aggregate":common.aggregate.duplicate(true),"receipt":close.receipt.duplicate(true)}
	return {"value":value,"draft":draft,"page":linked.page}

# The existing D1 validator proves the close. The earned paired descriptor
# is made only after we pin the exact origin and append the locked D1 page.
static func stage_doubleheader(source: Dictionary,season: SeasonState,previous: Array) -> Dictionary:
	if not Archive.C.keys(source,["kind","version","head","current"]) or source.kind!=FITTED_DUGOUT_KIND \
			or not Archive.C.whole(source.version,2,2) or not source.head is Dictionary \
			or not source.current is Dictionary or previous.size()!=1 or not previous[0] is PackedByteArray: return {}
	var verified:=DugoutD2._champion_source(source.current,source.head,season,1)
	if verified.is_empty() or verified.pages!=[Archive._hash(previous[0])]: return {}
	var common: Dictionary=source.current.common
	var native_close: Dictionary=source.current.ordinary_state.last_close
	if not Progress.valid(common.aggregate) or common.aggregate.closed!=2 \
			or common.aggregate.titles!={"dugout:0":true,"dugout:1":true} \
			or not Unlocks.from_titles(common.aggregate.titles).has("doubleheader"): return {}
	var old_head:=Archive._wire(verified)
	var close: Dictionary={"reason":"complete","ordinal":2,"id":native_close.id,"league":"dugout","tier":1,
		"session_sha256":Life.digest(source),"native_session_sha256":native_close.session_sha256,
		"aggregate_sha256":Life.digest(common.aggregate),"receipt":native_close.receipt}
	var linked:=Archive.make_page_and_head(old_head,source,close)
	if linked.is_empty(): return {}
	var head:=Archive._head(linked.head)
	var next_head_sha:=Archive._hash(linked.head)
	if head.is_empty() or head.ordinal!=3 or head.pages.slice(0,1)!=verified.pages \
			or head.pages[1]!=Archive._hash(linked.page): return {}
	var api:=Archive._owner()
	var lineage: String=source.current.ordinary_state.lineage
	var seed: int=("ordinary:%s:3:nextseed"%lineage).hash() & 0x7fffffff
	var runtime: Script=api.Adapter.OrdinaryRuntime
	var detached: Dictionary={"common":common,"session":season}
	var base: Dictionary=runtime.CommonStart.make(seed,lineage,3,2,close.aggregate_sha256,api.Common.projection(detached,season))
	var basis: Dictionary=runtime.CommonStart.bind_earned_doubleheader(base,head.pages,close.id,next_head_sha)
	var draft: SeasonState=null if basis.is_empty() else runtime.create_earned_doubleheader_draft(basis)
	if not _paired_opening(draft) or draft.phase!=SeasonState.Phase.DRAFT or not draft.picks.is_empty() \
			or not draft.physical.pending.is_empty() or not Archive.C.same(draft.career.start_basis,basis): return {}
	var current: Dictionary=api.Adapter.Ordinary.capture(draft)
	if current.is_empty(): return {}
	var offer: Dictionary={"route":DUGOUT_D1_TO_DOUBLEHEADER,"source_close":close.id,
		"source_session_sha256":close.native_session_sha256,"previous_head_sha256":Archive._hash(old_head),
		"origin_page_sha256":head.pages[0],"close_page_sha256":head.pages[1],
		"head_sha256":next_head_sha,"basis_sha256":Life.digest(basis)}
	var value: Dictionary={"kind":KIND,"version":1,"head":head,"offer":offer,
		"current":current,"aggregate":common.aggregate.duplicate(true),"receipt":close.receipt.duplicate(true)}
	return {"value":value,"draft":draft,"page":linked.page}

static func _paired_opening(session: SeasonState) -> bool:
	if session==null or session.career==null or session.round_index!=0 or not session.results.is_empty() \
			or session.build.cash()!=0: return false
	return Archive.C.same(session.league_rules,LeagueRules.capture("doubleheader")) \
		and Archive.C.same(session.schedule,SeasonDoubleheaderRules.schedule()) \
		and Archive._owner().Adapter.OrdinaryRuntime.CommonStart.detached_earned_doubleheader(session.career.start_basis)

static func _paired_current(session: SeasonState) -> bool:
	if _paired_opening(session): return true
	if session!=null and session.round_index in [2,3]:
		return session.career!=null and Paired.Live.CommonStart.detached_earned_doubleheader(session.career.start_basis) \
			and Paired.Live.scope(session)
	if session==null or session.career==null or session.phase!=SeasonState.Phase.REGULAR \
			or session.round_index!=1 or session.results.size()!=3 or session.player_results.size()!=1 \
			or session.physical.reports.size()!=2: return false
	var pending: Dictionary=session.physical.pending
	if not pending.is_empty() and (not Paired.Live.return_session_ready(session) \
			or not Paired.bound_shape(pending) or pending.human[0]!=3): return false
	return Archive.C.same(session.league_rules,LeagueRules.capture("doubleheader")) \
		and Archive.C.same(session.schedule,SeasonDoubleheaderRules.schedule()) \
		and Paired.Live.CommonStart.detached_earned_doubleheader(session.career.start_basis) \
		and Paired.Live._earned_pair_first_shop_prefix(session.build)

static func paired_shop_window(session: SeasonState) -> bool:
	return session!=null and session.round_index in [1,3] and _paired_current(session) \
		and session.physical.pending.is_empty() and session.pending_fixture().get("id")==3*session.round_index

static func paired_return_ready(session: SeasonState) -> bool:
	return paired_shop_window(session) and session.round_index==1 and Paired.Live.return_session_ready(session)

static func paired_next_opening_ready(session: SeasonState) -> bool:
	return session!=null and session.round_index==2 and _paired_current(session) \
		and session.physical.pending.is_empty() and Paired.Live.next_first_session_ready(session) \
		and session.pending_fixture().get("id")==6

static func paired_next_return_ready(session: SeasonState) -> bool:
	return paired_shop_window(session) and session.round_index==3 and Paired.Live.earned_next_return_ready(session)

static func paired_shop_step(previous: SeasonState,command: Dictionary) -> SeasonState:
	if not paired_shop_window(previous) or not Archive.C.primitive(command) \
			or not SeasonBuild.SHOP_OPS.has(str(command.get("op",""))): return null
	var op: String=command.op
	var id: Variant=command.get("id")
	var rev: int=previous.build.revision()
	if not id is String or id.is_empty() or previous.build._requests.has(id) \
			or not Archive.C.whole(command.get("rev"),rev,rev) or previous.build._visit.open==(op=="open"): return null
	if op=="open" and not Archive.C.same(command,{"id":"shop-open:%d"%rev,"rev":rev,"op":"open"}): return null
	var candidate: SeasonState=previous.fork_live()
	if candidate==null: return null
	var build: SeasonBuild=candidate.build.candidate(command.duplicate(true))
	if build==null: return null
	candidate.adopt_build(build)
	var before: Array=previous.build._events
	if candidate.build!=build or not paired_shop_window(candidate) or not build._visit.open \
			or build.revision()!=rev+1 or not Archive.C.same(build._events.slice(0,-1),before) \
			or not Archive.C.same(build._events.back(),command): return null
	return candidate

static func capture_current(value: Dictionary,session: SeasonState) -> Dictionary:
	var adapter: Script=Archive._owner().Adapter
	if target(value)=="doubleheader":
		return adapter.Ordinary.capture(session) if _paired_current(session) else {}
	return adapter.capture("ordinary",session)

static func _route_value(value: Variant) -> bool:
	if not value is Dictionary: return false
	if target(value)=="doubleheader" and (value.has("playoffs") \
			or (value.has("rounds") and (not value.rounds is Array or value.rounds.size() not in [1,2,3]))): return false
	var fields: Array=["kind","version","head","offer","current","aggregate","receipt"]
	if value.has("rounds"):
		if not value.rounds is Array or value.rounds.is_empty() or value.rounds.size()>Equipment.REGULAR_LIMIT: return false
		fields.append("rounds")
	if value.has("playoffs"):
		if not value.has("rounds") or value.rounds.size()!=Equipment.REGULAR_LIMIT \
				or not value.playoffs is Array or value.playoffs.is_empty() or value.playoffs.size()>2: return false
		fields.append("playoffs")
	return Archive.C.keys(value,fields) \
		and value.kind==KIND and Archive.C.whole(value.version,1,1) and value.head is Dictionary \
		and value.offer is Dictionary and not target(value).is_empty() \
		and value.current is Dictionary and value.aggregate is Dictionary and value.receipt is Dictionary

# Cheap rehydration of bytes already proven and pinned by the fixed reader.
# Never an alternative way to award a championship.
static func local_draft(value: Variant, live: bool=false) -> SeasonState:
	if not _route_value(value) or not Progress.valid(value.aggregate) or value.receipt.get("finish")!="champion": return null
	var goal:=target(value)
	var closed: int=3 if goal=="scramble" else 2
	var titles: Dictionary={"dugout:0":true}
	if goal=="earnings": titles["equipment:0"]=true
	else: titles["dugout:1"]=true
	if goal=="scramble": titles["dugout:2"]=true
	if value.aggregate.closed!=closed or value.aggregate.titles!=titles: return null
	var adapter: Script=Archive._owner().Adapter
	var paired: bool=goal=="doubleheader"
	var session: SeasonState=adapter.Ordinary.restore_common(value.current) if paired else adapter.restore("ordinary",value.current)
	if session==null or not (_paired_current(session) if paired else Equipment.named_runtime(session)) \
			or session.picks.size()>4 or session.league_rules.get("league")!=goal: return null
	if paired:
		var origin: Dictionary=session.career.start_basis.archive_origin
		if origin.ordered_sha256!=page_ids(value) or origin.target_head_sha256!=value.offer.get("head_sha256") \
				or origin.source_close!=value.offer.get("source_close"): return null
	var rounds: Array=value.get("rounds",[])
	var playoffs: Array=value.get("playoffs",[])
	if rounds.size()!=session.round_index or (not live and (not rounds.is_empty() or not session.physical.pending.is_empty())): return null
	if rounds.is_empty():
		if not session.results.is_empty() or session.build.cash()!=0: return null
	elif playoffs.is_empty():
		if session.phase==SeasonState.Phase.SEMIFINAL:
			if Equipment.DugoutTitle.semifinal_pair(session).is_empty(): return null
		elif not paired and not Equipment._regular_boundary(session,rounds.size()): return null
	else:
		var destination:=Equipment.finish(session)
		if (playoffs.size()==1 and destination not in ["final_ready","semifinal_exit"]) \
				or (playoffs.size()==2 and destination not in ["final_champion","final_runner_up"]): return null
	if not session.physical.pending.is_empty() and not paired and Equipment.pending_plan(session).is_empty(): return null
	var basis: Dictionary=session.career.start_basis
	if basis.get("ordinal")!=closed+1 or basis.get("common_sequence")!=closed \
			or basis.get("common_sha256")!=Life.digest(value.aggregate) \
			or Life.digest(basis)!=value.offer.get("basis_sha256"): return null
	return session

# The next fixture signs the persisted paid state. Only the independently
# replayed prefix may cross the same native JSON boundary as live play.
static func _paired_native_prefix(draft: SeasonState) -> SeasonState:
	if not paired_return_ready(draft) and not paired_next_opening_ready(draft) and not paired_next_return_ready(draft): return null
	var api:=Archive._owner()
	var prefix: Dictionary=api.Adapter.Ordinary.capture(draft)
	var native: Variant=api._bytes_value(JSON.stringify(prefix,"",true,true).to_utf8_buffer())
	if prefix.is_empty() or not Archive.C.same(prefix,native): return null
	var ready: SeasonState=api.Adapter.Ordinary.restore_common(native)
	return ready if paired_return_ready(ready) or paired_next_opening_ready(ready) or paired_next_return_ready(ready) else null

# Re-derive both proven seasons and the exact close, then the draft and its
# original picks. The supplied title, aggregate and page are never authority.
static func replay(value: Variant, pages: Array) -> Dictionary:
	if not OS.has_feature("wiffaltro_catalog_forward_v1") or not _route_value(value): return {}
	var goal:=target(value)
	var paired: bool=goal=="doubleheader"
	var expected: int=3 if goal=="scramble" else 2
	if pages.size()!=expected: return {}
	var digests: Array=[]
	for page: Variant in pages:
		if not page is PackedByteArray: return {}
		digests.append(Archive._hash(page))
	var head:=Archive._head(Archive._wire(value.head))
	if head.is_empty() or head.ordinal!=expected+1 or head.pages!=digests or digests!=page_ids(value): return {}
	var closing:=Archive._page(pages.back(),head,expected)
	if closing.is_empty() or closing.close.get("reason")!="complete": return {}
	var computed: Dictionary={}
	if goal=="earnings":
		if closing.close.get("league")!="equipment": return {}
		var recognized:=Equipment.replay_draft(closing.image,pages[0])
		if recognized.is_empty(): return {}
		computed=stage(closing.image,recognized.draft,pages[0])
	elif goal=="doubleheader":
		if closing.close.get("league")!="dugout" or closing.close.get("tier")!=1 \
				or not closing.image.get("head") is Dictionary or not closing.image.get("current") is Dictionary: return {}
		var prefix: Array[PackedByteArray]=[pages[0]]
		var proved:=DugoutD2.D1.restore_completed_from_pages(Archive._wire(closing.image.head),prefix,closing.image.current)
		if proved==null: return {}
		computed=stage_doubleheader(closing.image,proved,prefix)
	else:
		if closing.close.get("league")!="dugout" or closing.close.get("tier")!=2 \
				or not closing.image.get("head") is Dictionary or not closing.image.get("current") is Dictionary: return {}
		var prefix: Array[PackedByteArray]=[pages[0],pages[1]]
		var proved:=DugoutD2.restore_completed_from_pages(Archive._wire(closing.image.head),prefix,closing.image.current)
		if proved==null: return {}
		computed=stage_scramble(closing.image,proved,prefix)
	if computed.is_empty() or computed.page!=pages.back(): return {}
	var picks: Variant=value.current.get("state",{}).get("picks") if value.current.get("state") is Dictionary else null
	if not picks is Array or picks.size()>4: return {}
	var draft: SeasonState=computed.draft
	for pick: Variant in picks:
		if not pick is String or not draft.choose_player(pick): return {}
	var journal: Array=value.get("rounds",[])
	for row: Variant in journal:
		if not Archive.C.keys(row,["pending","shop"]) or not row.pending is Dictionary \
				or not row.shop is Array or row.shop.size()>128: return {}
		var leg: int=draft.round_index
		if paired and leg in [1,2]:
			if leg==1 and not row.shop.is_empty(): return {}
			draft=_paired_native_prefix(draft)
			if draft==null: return {}
		draft.physical.pending=row.pending.duplicate(true)
		if paired:
			draft=Paired.materialize_settlement(draft,leg)
			if draft==null or not _paired_current(draft): return {}
		else:
			var plan:=Equipment.pending_plan(draft,true)
			if plan.is_empty() or plan.projection.has("fixture") or plan.get("semifinal",false) or plan.get("managed_final",false): return {}
			draft=plan.projection.candidate
		for command: Variant in row.shop:
			if not command is Dictionary: return {}
			draft=paired_shop_step(draft,command) if paired else Equipment.shop_step(draft,row.pending,command)
			if draft==null: return {}
	if value.has("rounds"): computed.value.rounds=journal.duplicate(true)
	var playoffs: Array=value.get("playoffs",[])
	for row: Variant in playoffs:
		if not row is Dictionary or Equipment.play_slot(draft) not in ["semifinal","final"]: return {}
		draft.physical.pending=row.duplicate(true)
		var plan:=Equipment.pending_plan(draft,true)
		if plan.is_empty() or plan.projection.has("fixture"): return {}
		draft=plan.projection.candidate
	if value.has("playoffs"): computed.value.playoffs=playoffs.duplicate(true)
	var physical: Variant=value.current.get("physical")
	if not physical is Dictionary or not physical.get("pending") is Dictionary: return {}
	if not physical.pending.is_empty():
		if paired and draft.round_index in [1,2,3]:
			draft=_paired_native_prefix(draft)
			if draft==null: return {}
		draft.physical.pending=physical.pending.duplicate(true)
		if not paired and Equipment.pending_plan(draft,true).is_empty(): return {}
	computed.value.current=capture_current(computed.value,draft)
	if computed.value.current.is_empty() or not Archive.C.same(computed.value,value): return {}
	return {"draft":draft,"aggregate":computed.value.aggregate.duplicate(true)}
