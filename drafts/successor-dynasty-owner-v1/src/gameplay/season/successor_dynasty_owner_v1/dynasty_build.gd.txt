## Current-catalog Dynasty build. No old schema or reader is relaxed.
const Profile = preload("dynasty_profile.gd")

func _init(seed_value: int = 0, roster: Array[String] = [], pool: Array[String] = [], blocked: Array[String] = [], fork_only: bool = false) -> void:
	super(seed_value,roster,pool,blocked,true)
	if not fork_only:
		_bank = SeasonOwnership.new(SeasonSponsorCatalog.ownership_catalog())
		_book = DynastyBook.new("season:%d" % _seed)

func pitch_catalog_successor() -> bool:
	return _book != null and _book.get_script() == DynastyBook and _book._pitch_catalog_version == V7Catalog.VERSION

func dynasty_context_valid() -> bool:
	if not LeagueRules.dynasty(_league_rules): return _dynasty_start.is_empty() and _book._dynasty_start.is_empty()
	if not Profile.same(_dynasty_start,_book._dynasty_start): return false
	if _dynasty_start.is_empty(): return true
	var profile: Dictionary = Profile.normalize(_dynasty_start)
	return not profile.is_empty() and profile.lineage_id.length() <= 152 and _market in [0,12] and profile.seed == _seed and profile.roster == _initial_roster

func enable_dynasty_start(value: Variant) -> bool:
	if not LeagueRules.dynasty(_league_rules) or _market not in [0,12] or revision() != 0 or not _requests.is_empty() or _bank.revision() != 0 or not _dynasty_start.is_empty(): return false
	var profile: Dictionary = Profile.normalize(value)
	if profile.is_empty() or profile.lineage_id.length() > 152 or profile.seed != _seed or profile.roster != _initial_roster: return false
	for id: String in profile.roster:
		if profile.players[id].temporary_fielding_slots != 1: return false
	var book: SeasonDevelopment = _book.fork()
	if not book.enable_dynasty(_league_rules,profile): return false
	var abilities: SeasonAbilities = _abilities.fork()
	for id: String in profile.roster:
		abilities.learned[id] = []
		for row: Dictionary in profile.players[id].abilities:
			var receipt: Dictionary = row.duplicate(true)
			receipt.erase("source_season")
			abilities.learned[id].append(receipt)
	_book = book
	_abilities = abilities
	_dynasty_start = profile
	return dynasty_context_valid()

func _apply(command: Dictionary) -> String:
	if str(command.get("op","")) == "exotic_admit":
		if pack_pending(): return "Choose or skip the open pack before leaving or doing other shopping."
		return preload("exotic_access.gd").settle(self,command)
	return super._apply(command)
