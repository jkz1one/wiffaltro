## Common close identity is independent of the Dynasty-local ordinal chain.
const C = preload("codec.gd")
const Adapter = preload("session_adapter.gd")
const Life = preload("lifecycle.gd")
const WorkshopHistory = preload("workshop_close_history.gd")
const Progress = preload("common_progress.gd")
const Start=preload("res://src/gameplay/season/successor_live_play_v1/common_start.gd")
const ARCHIVE_KIND: String="common-ordinary-predecessor-v1"
const ARCHIVE_MAX: int=16*1024*1024
const CROSS_KIND: String="common-ordinary-league-proposal-v1"
const ORDINARY_FIELDS: Array[String]=["lineage","ordinal","status","last_close","proposal","last_transition","start_basis_sha256"]

static func initialize(owner) -> bool:
	if not owner.common.is_empty() or not owner.ordinary_state.is_empty(): return false
	var ordinary_session: SeasonState=owner.session if owner.active=="ordinary" else null
	if ordinary_session==null and not owner.paused.is_empty():
		ordinary_session=Adapter.restore("ordinary",owner.paused)
		if ordinary_session==null: return false
	var dynasty_session: SeasonState=owner.session if owner.active=="dynasty" else null
	if dynasty_session==null and not owner.paused.is_empty():
		dynasty_session=Adapter.restore("dynasty",owner.paused)
		if dynasty_session==null: return false
	var access: Array=[]
	for season: SeasonState in [ordinary_session,dynasty_session]:
		if season==null: continue
		for id: String in season.player_access:
			if not access.has(id): access.append(id)
	access.sort()
	var aggregate: Dictionary=Progress.birth(owner.ledger,ordinary_session,access)
	if aggregate.is_empty(): return false
	if dynasty_session!=null:
		var start: Dictionary=dynasty_session.build._dynasty_start
		var ids: Array=dynasty_session.build._book._exotic_ids if owner.status=="closed" else start.get("exotic_rights",[])
		var source: String=Life.digest(Adapter.capture("dynasty",dynasty_session)) if owner.status=="closed" else Life.digest(start)
		for id: String in ids: aggregate.exotic_rights[id]={"origin":"baseline:"+source,"source":source}
	if aggregate.is_empty(): return false
	var closed: int=int(owner.ledger.get("closed",0))
	owner.common={"version":1,"baseline":{"dynasty_closed":closed,"ledger_sha256":Life.digest(owner.ledger)},"sequence":closed,"cursors":{"dynasty":closed,"ordinary":0},"aggregate":aggregate,"last_fold":{}}
	if ordinary_session!=null and not initialize_ordinary(owner,ordinary_session): return false
	return valid(owner)

static func initialize_ordinary(owner, season: SeasonState) -> bool:
	if not owner.ordinary_state.is_empty() or not Adapter.scope("ordinary",season): return false
	var current: Dictionary=Adapter.capture("ordinary",season)
	if current.is_empty(): return false
	var basis: Dictionary=season.career.start_basis
	if basis.get("kind")==Start.KIND and basis.ordinal!=1: return false
	var lineage: String=basis.lineage if basis.get("kind")==Start.KIND else "ordinary:"+Life.digest(current)
	owner.ordinary_state={"lineage":lineage,"ordinal":1,"status":"active","last_close":{},"proposal":{},"last_transition":{},"start_basis_sha256":Life.digest(season.career.start_basis)}
	return true

static func initialize_missing(owner) -> bool:
	if owner.active=="ordinary": return initialize_ordinary(owner,owner.session)
	# A newly created Dynasty starts its own local ordinal; common accumulated
	# ordinary rights are not rewritten or counted as Dynasty-local history.
	return owner.common.cursors.dynasty==0 and owner.ordinal==1 and owner.ledger.closed==0

static func valid(owner) -> bool:
	var value: Dictionary=owner.common
	if not C.keys(value,["version","baseline","sequence","cursors","aggregate","last_fold"]) or value.version!=1: return false
	if not C.keys(value.baseline,["dynasty_closed","ledger_sha256"]) or not C.whole(value.baseline.dynasty_closed,0,ClubCareer.MAX_RUNS) or not owner._digest(value.baseline.ledger_sha256): return false
	if not C.keys(value.cursors,["dynasty","ordinary"]): return false
	for side: String in ["dynasty","ordinary"]:
		if not C.whole(value.cursors[side],0,ClubCareer.MAX_RUNS): return false
	if not C.whole(value.sequence,0,ClubCareer.MAX_RUNS*2) or value.sequence!=value.cursors.dynasty+value.cursors.ordinary or value.cursors.dynasty<value.baseline.dynasty_closed: return false
	if value.cursors.dynasty!=owner.ledger.get("closed",0) or not Progress.valid(value.aggregate): return false
	if value.aggregate.closed!=value.sequence or value.aggregate.legacy_closed!=value.baseline.dynasty_closed: return false
	if value.last_fold.is_empty():
		if value.sequence!=value.baseline.dynasty_closed or value.cursors.ordinary!=0: return false
	else:
		var row: Dictionary=value.last_fold
		if not C.keys(row,["id","side","ordinal","sequence","source","aggregate_sha256","receipt"]) or not Adapter.side(row.side) or not row.id is String or not C.whole(row.ordinal,1,ClubCareer.MAX_RUNS) or row.ordinal!=value.cursors[row.side] or row.sequence!=value.sequence or not owner._digest(row.source) or not row.receipt is Dictionary: return false
		if row.aggregate_sha256!=Life.digest(value.aggregate): return false
		var lineage: String=Life.lineage(owner) if row.side=="dynasty" else str(owner.ordinary_state.get("lineage",""))
		if row.id!="%s:%d:close"%[lineage,row.ordinal]: return false
	var has_ordinary: bool=owner.active=="ordinary" or not owner.paused.is_empty()
	if not has_ordinary: return owner.ordinary_state.is_empty() and value.cursors.ordinary==0
	var state: Dictionary=owner.ordinary_state
	if not C.keys(state,ORDINARY_FIELDS+(["archive"] if state.has("archive") else [])) or not state.lineage is String or not state.lineage.begins_with("ordinary:") or not owner._digest(state.lineage.trim_prefix("ordinary:")) or not C.whole(state.ordinal,1,ClubCareer.MAX_RUNS) or state.status not in ["active","closed"]: return false
	if not owner._digest(state.start_basis_sha256) or not state.last_close is Dictionary or not state.proposal is Dictionary or not state.last_transition is Dictionary: return false
	if value.cursors.ordinary!=int(state.ordinal)-int(state.status=="active"): return false
	if state.status=="active" and (not state.last_close.is_empty() or not state.proposal.is_empty()): return false
	if state.ordinal==1 and not state.last_transition.is_empty(): return false
	if state.ordinal>1:
		var t: Dictionary=state.last_transition
		if not C.keys(t,["id","source","from","to","league","basis"]) or t.to!=state.ordinal or t.from!=state.ordinal-1 or t.id!="%s:%d:next"%[state.lineage,t.from] or not owner._digest(t.source) or t.league not in Adapter.OrdinaryRuntime.COMMON_ORDINARY_LEAGUES or not Start.valid(t.basis): return false
		if t.basis.lineage!=state.lineage or t.basis.ordinal!=state.ordinal or Life.digest(t.basis)!=state.start_basis_sha256 or t.basis.common_sequence>value.sequence: return false
	return _archive_shape(owner) and _ordinary_proposal_valid(owner)

static func _archive_shape(owner) -> bool:
	var state: Dictionary=owner.ordinary_state
	if not state.has("archive"): return true
	var a: Variant=state.archive
	return C.keys(a,["kind","sha256","source"]) and a.kind==ARCHIVE_KIND \
		and a.source is String and not a.source.is_empty() and a.source.length()<=ARCHIVE_MAX \
		and owner._digest(a.sha256) and owner.active=="ordinary" and state.ordinal==2 \
		and state.proposal.is_empty() and state.last_transition.get("league")=="dugout" \
		and (state.status=="active" or (state.status=="closed" and state.last_close.get("reason")=="abandon"))

static func cross_commit_ready(owner) -> bool:
	if owner==null or owner.active!="ordinary" or owner.ordinary_state.has("archive"): return false
	var state: Dictionary=owner.ordinary_state
	var p: Dictionary=state.get("proposal",{})
	if p.get("kind")!=CROSS_KIND or not _cross_proposal_valid(owner,p): return false
	var fold: Dictionary=owner.common.last_fold
	return p.basis.common_sequence==owner.common.sequence and fold.get("side")=="ordinary" \
		and fold.get("sequence")==owner.common.sequence and fold.get("id")==p.close_id \
		and fold.get("source")==p.source and C.same(fold.get("receipt"),state.last_close.get("receipt"))

static func _activate_cross(owner, raw: PackedByteArray) -> bool:
	if not cross_commit_ready(owner) or raw.is_empty() or raw.size()>ARCHIVE_MAX: return false
	var text:=raw.get_string_from_utf8()
	if text.to_utf8_buffer()!=raw: return false
	var p: Dictionary=owner.ordinary_state.proposal
	var next: SeasonState=Adapter.OrdinaryRuntime.create_common(p.basis,p.league)
	if next==null or next.phase!=SeasonState.Phase.DRAFT or not next.picks.is_empty(): return false
	var state: Dictionary=owner.ordinary_state
	state.last_transition={"id":"%s:1:next"%state.lineage,"source":p.source,"from":1,"to":2,
		"league":p.league,"basis":p.basis.duplicate(true)}
	state.ordinal=2;state.status="active";state.last_close={};state.proposal={}
	state.start_basis_sha256=Life.digest(p.basis)
	state.archive={"kind":ARCHIVE_KIND,"sha256":owner._hash(raw),"source":text}
	owner.session=next
	return valid(owner)

# The named archive lane admits only replayed stock opening proof, not a
# second fixture launch, second archive, mode changes or unrelated modes.
static func archive_opening_scope(season: SeasonState) -> bool:
	if season==null or season.league_rules.get("league")!="dugout" \
		or season.league_rules.get("tier")!=0 or season.phase not in [SeasonState.Phase.DRAFT,SeasonState.Phase.REGULAR]:
		return false
	if season.round_index==0:
		return season.results.is_empty() and season.player_results.is_empty() and season.physical.reports.is_empty() \
			and (season.phase==SeasonState.Phase.REGULAR or season.physical.pending.is_empty())
	return season.round_index==1 and season.phase==SeasonState.Phase.REGULAR \
		and season.results.size()==3 and season.player_results.size()==1 and season.physical.reports.size()==2 \
		and season.physical.pending.is_empty() and season.physical.needed.is_empty()

static func ordinary_current_valid(owner, season: SeasonState, current: Dictionary) -> bool:
	if not Adapter.scope("ordinary",season) or current.is_empty() or owner.ordinary_state.is_empty(): return false
	var state: Dictionary=owner.ordinary_state
	if season.league_rules.get("league","")=="doubleheader":
		if owner.active!="ordinary" or not owner.paused.is_empty() or not owner.migration_v3.is_empty() or not owner._missing_dynasty(owner): return false
		if state.ordinal!=1: return false
		var birth: Dictionary=Progress.birth({},season,season.player_access)
		if state.status=="closed": return _paired_closed_valid(owner,season,current,birth)
		if state.status!="active" or owner.common.sequence!=0 or not C.same(owner.common.aggregate,birth): return false
	if state.has("archive") and (not archive_opening_scope(season) \
		or (state.status=="closed" and season.round_index!=1)): return false
	if state.start_basis_sha256!=Life.digest(season.career.start_basis): return false
	var basis: Dictionary=season.career.start_basis
	if basis.get("kind")==Start.KIND and (basis.lineage!=state.lineage or basis.ordinal!=state.ordinal): return false
	if state.ordinal>1 and not C.same(season.career.start_basis,state.last_transition.basis): return false
	if state.status=="active": return state.last_close.is_empty()
	var row: Dictionary=state.last_close
	if not C.keys(row,["id","ordinal","reason","session_sha256","receipt","first_title","league","tier"]) or row.id!="%s:%d:close"%[state.lineage,state.ordinal] or row.ordinal!=state.ordinal or row.reason not in ["complete","abandon"] or not row.first_title is bool: return false
	if row.session_sha256!=Life.digest(current): return false
	var league: String=str(season.league_rules.get("league","standard"))
	var tier: int=int(season.league_rules.get("tier",0))
	if row.league!=league or row.tier!=tier: return false
	var proof: Dictionary=ClubSeasonRecord.capture(season)
	var summary: Dictionary=ClubSeasonRecord.analyze(proof)
	if summary.is_empty() or (row.reason=="complete")!=(summary.finish!="unfinished"): return false
	if (row.reason=="complete")!=(season.phase==SeasonState.Phase.COMPLETE): return false
	var expected: Dictionary=ClubSeasonRecord.receipt(proof,row.first_title,tier) if row.reason=="complete" else {}
	return C.same(row.receipt,expected)

# Only the already-qualified private eliminated semifinal COMPLETE can close.
static func _paired_complete_scope(owner, season: SeasonState) -> bool:
	return owner.paired_draft_only() and owner.active=="ordinary" and owner.paused.is_empty() \
		and owner.migration_v3.is_empty() and owner._missing_dynasty(owner) and Adapter.scope("ordinary",season) \
		and Adapter.Ordinary._completed_postseason_shape(season) and season.champion!=0 \
		and ClubSeasonRecord.analyze(ClubSeasonRecord.capture(season)).get("finish")=="semifinal" \
		and not season.build.pack_pending()

static func _paired_closed_valid(owner, season: SeasonState, current: Dictionary, birth: Dictionary) -> bool:
	if birth.is_empty() or not _paired_complete_scope(owner,season): return false
	var state: Dictionary=owner.ordinary_state
	if not state.last_transition.is_empty() or not _ordinary_proposal_valid(owner): return false
	if owner.common.sequence!=1 or not C.same(owner.common.cursors,{"ordinary":1,"dynasty":0}) \
		or not C.same(owner.common.baseline,{"dynasty_closed":0,"ledger_sha256":Life.digest({})}): return false
	# This first ordinal may only originate from the deterministic private draft,
	# not an arbitrary lineage whose two close envelopes agree with each other.
	var draft: SeasonState=Adapter.OrdinaryRuntime._create_paired_draft(season.season_seed)
	if draft==null or not C.same(draft.career.start_basis,season.career.start_basis): return false
	var origin: Dictionary=Adapter.capture("ordinary",draft)
	if origin.is_empty() or state.lineage!="ordinary:"+Life.digest(origin) \
		or state.start_basis_sha256!=Life.digest(season.career.start_basis): return false
	var id: String="%s:1:close"%state.lineage
	var source: String=Life.digest(current)
	var folded: Dictionary=Progress.fold(birth,"ordinary",season,id,1,"complete")
	if folded.is_empty() or not C.same(owner.common.aggregate,folded.aggregate): return false
	return C.same(state.last_close,{"id":id,"ordinal":1,"reason":"complete","session_sha256":source,"receipt":folded.receipt,"first_title":true,"league":"doubleheader","tier":0}) \
		and C.same(owner.common.last_fold,{"id":id,"side":"ordinary","ordinal":1,"sequence":1,"source":source,"aggregate_sha256":Life.digest(folded.aggregate),"receipt":folded.receipt})

# Pure metadata fold on a detached owner; only Store can bind existing
# admitted session bytes, enforce exact raw equality, and write the envelope.
static func _close_paired_complete(owner, current: Dictionary) -> bool:
	if not _paired_complete_scope(owner,owner.session) or not valid(owner) or owner.ordinary_state.status!="active": return false
	return ordinary_current_valid(owner,owner.session,current) and _apply_close(owner,"complete",current)

static func close(owner, reason: String) -> bool:
	if owner.ordinary_state.has("archive") and (reason!="abandon" or owner.active!="ordinary" \
		or not archive_opening_scope(owner.session) or owner.session.round_index!=1): return false
	if not valid(owner) or not owner.playable() or reason not in ["complete","abandon"]: return false
	var season: SeasonState=owner.session
	if season.picks.size()!=4 or not season.physical.pending.is_empty() or not season.physical.needed.is_empty() or season.physical.projecting or season.build.pack_pending(): return false
	if (reason=="complete")!=(season.phase==SeasonState.Phase.COMPLETE): return false
	var source: Dictionary=Adapter.capture(owner.active,season)
	return not source.is_empty() and _apply_close(owner,reason,source)

static func _apply_close(owner, reason: String, source: Dictionary) -> bool:
	var season: SeasonState=owner.session
	var ordinal: int=owner.ordinal if owner.active=="dynasty" else int(owner.ordinary_state.ordinal)
	var lineage: String=Life.lineage(owner) if owner.active=="dynasty" else owner.ordinary_state.lineage
	var id: String="%s:%d:close"%[lineage,ordinal]
	if owner.common.cursors[owner.active]!=ordinal-1: return false
	var sequence: int=int(owner.common.sequence)+1
	var folded: Dictionary=Progress.fold(owner.common.aggregate,owner.active,season,id,sequence,reason)
	if folded.is_empty(): return false
	if owner.active=="dynasty":
		if not Life.close(owner,reason) or not WorkshopHistory.append_close(owner): return false
	else:
		var league: String=str(season.league_rules.get("league","standard"))
		var tier: int=int(season.league_rules.get("tier",0))
		var first_title: bool=not owner.common.aggregate.titles.get(ClubLeagueCareer.title_key(league,tier),false)
		owner.ordinary_state.last_close={"id":id,"ordinal":ordinal,"reason":reason,"session_sha256":Life.digest(source),"receipt":folded.receipt.duplicate(true),"first_title":first_title,"league":league,"tier":tier}
		owner.ordinary_state.status="closed";owner.ordinary_state.proposal={}
	owner.common.aggregate=folded.aggregate
	owner.common.cursors[owner.active]=ordinal;owner.common.sequence=sequence
	owner.common.last_fold={"id":id,"side":owner.active,"ordinal":ordinal,"sequence":sequence,"source":Life.digest(source),"aggregate_sha256":Life.digest(folded.aggregate),"receipt":folded.receipt.duplicate(true)}
	return valid(owner)

static func projection(owner, source: SeasonState=null) -> Dictionary:
	var aggregate: Dictionary=owner.common.aggregate
	var ids: Array=aggregate.exotic_rights.keys();ids.sort()
	# Retain actual issued legacy records without converting compact awards into
	# fictitious asset journals. New awards have their own named projection.
	var basis: Dictionary=(owner.session if source==null else source).career.start_basis
	var inherited: Dictionary=basis.rights if basis.get("kind")=="dynasty-carried-start-v2" else basis
	return {"player_access":aggregate.player_access.duplicate(),"starts":aggregate.starts.duplicate(true),"exotic_rights":ids,
		"assets":inherited.get("assets",{}).duplicate(true),"plaques":inherited.get("plaques",{}).duplicate(true),
		"pennants":{"version":StadiumPennantRewards.MASTERY_VERSION,"from_season":1,"eras":{"1":aggregate.era.duplicate(true)},"mastery_from":1,"mastery_contract":StadiumPennantRewards.MASTERY_CONTRACT},
		"common_pennants":aggregate.pennants.duplicate(true),"common_plaques":awards(owner)}

static func ordinary_birth(owner, seed_value: int, league: String) -> SeasonState:
	if not valid(owner) or not owner.ordinary_state.is_empty() or not C.whole(seed_value,0,2147483647) or league not in Adapter.OrdinaryRuntime.COMMON_ORDINARY_LEAGUES: return null
	var lineage: String="ordinary:"+Life.digest({"common":owner.common,"seed":seed_value,"league":league})
	var basis: Dictionary=Start.make(seed_value,lineage,1,int(owner.common.sequence),Life.digest(owner.common.aggregate),projection(owner))
	return Adapter.OrdinaryRuntime.create_common(basis,league) if not basis.is_empty() else null

static func dynasty_birth_basis(owner, seed_value: int) -> Dictionary:
	if owner.active!="ordinary" or not owner.paused.is_empty() or not valid(owner) or not C.whole(seed_value,0,2147483647) or owner.common.sequence<1: return {}
	var projection_basis: Dictionary=Start.make(seed_value,owner.ordinary_state.lineage,1,int(owner.common.sequence),Life.digest(owner.common.aggregate),projection(owner))
	return C.CommonBirth.make(seed_value,projection_basis)

static func dynasty_birth_bound(owner, basis: Dictionary) -> bool:
	var rights: Dictionary=basis.projection
	if rights.common_sequence>owner.common.sequence or rights.lineage!=owner.ordinary_state.lineage: return false
	for id: String in rights.exotic_rights:
		if not owner.common.aggregate.exotic_rights.has(id): return false
	if rights.common_sequence==owner.common.sequence:
		var ordinary: SeasonState=owner.session if owner.active=="ordinary" else Adapter.restore("ordinary",owner.paused)
		if ordinary==null: return false
		var expected: Dictionary=Start.make(int(rights.seed),rights.lineage,1,int(owner.common.sequence),Life.digest(owner.common.aggregate),projection(owner,ordinary))
		if not C.same(rights,expected): return false
	return true

# Only the active private first closed season may freeze a proposal for
# ordinal two. Its current ordinary payload remains the complete source archive
# until an independently qualified flat archive rollover can consume it.
static func _paired_proposal_valid(owner, row: Dictionary) -> bool:
	var state: Dictionary=owner.ordinary_state
	if not owner.paired_draft_only() or state.ordinal!=1 or state.status!="closed" or owner.common.sequence!=1: return false
	if not _paired_complete_scope(owner,owner.session) or row.league!="doubleheader" or row.tier!=0: return false
	if row.id!="%s:1:next"%state.lineage or not owner._digest(row.source) or row.source!=state.last_close.get("session_sha256"): return false
	if not Start.valid(row.basis): return false
	var basis: Dictionary=Start.make(int(row.basis.seed),state.lineage,2,1,Life.digest(owner.common.aggregate),projection(owner))
	return not basis.is_empty() and C.same(row.basis,basis)

static func _propose_paired_next(owner, current: Dictionary, seed_value: int) -> bool:
	if not C.whole(seed_value,0,2147483647) or not valid(owner) or owner.ordinary_state.status!="closed" or not owner.ordinary_state.proposal.is_empty(): return false
	if not _paired_complete_scope(owner,owner.session) or current.is_empty(): return false
	var state: Dictionary=owner.ordinary_state
	var source: String=Life.digest(current)
	if source!=state.last_close.get("session_sha256"): return false
	var basis: Dictionary=Start.make(seed_value,state.lineage,2,1,Life.digest(owner.common.aggregate),projection(owner))
	if basis.is_empty(): return false
	state.proposal={"id":"%s:1:next"%state.lineage,"source":source,"league":"doubleheader","tier":0,"basis":basis}
	return valid(owner)

# First live cross-league route is scoped to the two authored starting Bases.
# Locked modes stay closed until an exact closed transition qualifies them.
static func cross_target_open(owner, league: String) -> bool:
	if owner==null or owner.active!="ordinary" or not owner.ordinary_state.proposal.is_empty(): return false
	var state: Dictionary=owner.ordinary_state
	return state.status=="closed" and state.ordinal==1 and state.last_close.get("reason")=="complete" \
		and state.last_close.get("league")=="standard" and state.last_close.get("tier")==0 \
		and owner.session!=null and owner.session.phase==SeasonState.Phase.COMPLETE \
		and owner.session.league_rules.is_empty() and league=="dugout" \
		and league in preload("res://src/gameplay/season/league_unlocks.gd").from_titles(owner.common.aggregate.titles)

static func _prepare_cross_proposal(owner, current: Dictionary, league: String, seed_value: int) -> bool:
	if not cross_target_open(owner,league) or not C.whole(seed_value,0,2147483647) \
		or not valid(owner) or not ordinary_current_valid(owner,owner.session,current): return false
	var state: Dictionary=owner.ordinary_state
	var close: Dictionary=state.last_close
	var basis: Dictionary=Start.make(seed_value,state.lineage,2,int(owner.common.sequence),
		Life.digest(owner.common.aggregate),projection(owner))
	if basis.is_empty(): return false
	state.proposal={"kind":CROSS_KIND,"id":"%s:1:next-league"%state.lineage,"source":close.session_sha256,
		"close_id":close.id,"source_league":"standard","source_tier":0,"league":league,"tier":0,"basis":basis}
	return valid(owner)

static func _cross_proposal_valid(owner, row: Dictionary) -> bool:
	var state: Dictionary=owner.ordinary_state
	var close: Dictionary=state.last_close
	if not C.keys(row,["kind","id","source","close_id","source_league","source_tier","league","tier","basis"]) \
		or state.status!="closed" or state.ordinal!=1 or close.get("reason")!="complete" \
		or row.id!="%s:1:next-league"%state.lineage or row.close_id!=close.get("id") \
		or not owner._digest(row.source) or row.source!=close.get("session_sha256") \
		or row.source_league!="standard" or close.get("league")!=row.source_league \
		or not C.whole(row.source_tier,0,0) or close.get("tier")!=row.source_tier \
		or row.league!="dugout" or not C.whole(row.tier,0,0) or not Start.valid(row.basis): return false
	if row.league not in preload("res://src/gameplay/season/league_unlocks.gd").from_titles(owner.common.aggregate.titles) \
		or row.basis.lineage!=state.lineage or row.basis.ordinal!=2 or row.basis.common_sequence>owner.common.sequence: return false
	if row.basis.common_sequence==owner.common.sequence:
		var season: SeasonState=owner.session if owner.active=="ordinary" else Adapter.restore("ordinary",owner.paused)
		if season==null: return false
		var exact:=Start.make(int(row.basis.seed),state.lineage,2,int(owner.common.sequence),
			Life.digest(owner.common.aggregate),projection(owner,season))
		if not C.same(row.basis,exact): return false
	return true

static func _ordinary_proposal_valid(owner) -> bool:
	var state: Dictionary=owner.ordinary_state
	var row: Dictionary=state.proposal
	if row.is_empty(): return true
	if row.get("kind")==CROSS_KIND: return _cross_proposal_valid(owner,row)
	if state.status!="closed" or state.ordinal>=ClubCareer.MAX_RUNS or not C.keys(row,["id","source","league","tier","basis"]): return false
	if owner.paired_draft_only(): return _paired_proposal_valid(owner,row)
	if row.id!="%s:%d:next"%[state.lineage,state.ordinal] or row.source!=state.last_close.get("session_sha256") or row.league not in Adapter.OrdinaryRuntime.COMMON_ORDINARY_LEAGUES or row.league!=state.last_close.get("league") or row.tier!=0: return false
	if not Start.valid(row.basis) or row.basis.lineage!=state.lineage or row.basis.ordinal!=state.ordinal+1 or row.basis.common_sequence>owner.common.sequence: return false
	# A proposal is frozen at creation. While that aggregate is still current,
	# require exact derivation. Older persisted proposals remain bounded named
	# materialized checkpoints, never permission to rewrite paused starts.
	if row.basis.common_sequence==owner.common.sequence:
		if row.basis.common_sha256!=Life.digest(owner.common.aggregate): return false
		# The ordinary session may be dormant; projection uses its own legacy
		# rights, not whichever side happens to be selected now.
		var season: SeasonState=owner.session if owner.active=="ordinary" else Adapter.restore("ordinary",owner.paused)
		if season==null: return false
		var expected: Dictionary=projection(owner,season)
		for key: String in expected:
			if not C.same(row.basis[key],expected[key]): return false
	return true

static func next_block_reason(owner) -> String:
	if owner.ordinary_state.has("archive"): return "Further seasons from this archived rollover are not available."
	if owner.active=="dynasty":
		return "" if dynasty_next_supported(owner) else "Shared earned rights are retained. Applying differing common rights to Dynasty Next is not yet available; Return keeps both saved seasons intact."
	if owner.ordinary_state.get("proposal",{}).get("kind")==CROSS_KIND:
		return "A different Base league is prepared. Activation awaits safe archive rollover; your completed season stays intact."
	var basis: Dictionary=owner.session.career.start_basis
	if not Start.assets_valid(basis.get("assets",{})) or not Start.legacy_plaques_valid(basis.get("plaques",{})):
		return "This saved asset history is retained. Its earned next-season integration is not yet available; Return preserves both saved seasons."
	return ""

static func dynasty_next_supported(owner) -> bool:
	if owner.active!="dynasty" or owner.status!="closed": return false
	var local: Dictionary=Life.rights(owner)
	var aggregate: Dictionary=owner.common.aggregate
	if not C.same(local.starts,aggregate.starts) or not C.same(local.player_access,aggregate.player_access): return false
	var profile: Dictionary=owner.last_close.profiles.get("club:0",{})
	for id: String in aggregate.exotic_rights:
		if id not in profile.get("exotic_rights",[]): return false
	# Compact common awards outside the local Dynasty bank require the explicit
	# future carried descriptor. Do not silently omit them at an apparent Next.
	return aggregate.pennants.is_empty() and C.same(awards(owner),owner.ledger.plaques)

static func propose(owner, seed_value: int) -> bool:
	if owner.ordinary_state.has("archive"): return false
	if owner.paired_draft_only(): return false
	if owner.active=="dynasty": return dynasty_next_supported(owner) and Life.propose(owner,seed_value)
	if not valid(owner) or owner.ordinary_state.status!="closed" or owner.ordinary_state.ordinal>=ClubCareer.MAX_RUNS or not C.whole(seed_value,0,2147483647): return false
	var state: Dictionary=owner.ordinary_state
	if not state.proposal.is_empty(): return state.proposal.get("kind")!=CROSS_KIND and state.proposal.basis.seed==seed_value
	var source: Dictionary=Adapter.capture("ordinary",owner.session)
	if not ordinary_current_valid(owner,owner.session,source): return false
	var league: String=str(owner.session.league_rules.get("league","standard"))
	if league not in Adapter.OrdinaryRuntime.COMMON_ORDINARY_LEAGUES or int(owner.session.league_rules.get("tier",0))!=0: return false
	var basis: Dictionary=Start.make(seed_value,state.lineage,int(state.ordinal)+1,int(owner.common.sequence),Life.digest(owner.common.aggregate),projection(owner))
	if basis.is_empty(): return false
	state.proposal={"id":"%s:%d:next"%[state.lineage,state.ordinal],"source":Life.digest(source),"league":league,"tier":0,"basis":basis}
	return _ordinary_proposal_valid(owner)

static func preview(owner) -> Dictionary:
	if owner.active=="dynasty": return Life.preview(owner) if dynasty_next_supported(owner) else {}
	if not valid(owner) or owner.ordinary_state.proposal.is_empty(): return {}
	var p: Dictionary=owner.ordinary_state.proposal
	var b: Dictionary=p.basis
	return {"kind":p.get("kind","ordinary"),"league":p.league,"tier":p.tier,"seed":b.seed,"ordinal":b.ordinal,"player_access":b.player_access.duplicate(),"starts":b.starts.duplicate(true),"exotic_rights":b.exotic_rights.duplicate()}

static func advance(owner) -> bool:
	if owner.ordinary_state.has("archive"): return false
	if owner.paired_draft_only(): return false
	if owner.active=="dynasty": return dynasty_next_supported(owner) and Life.advance(owner)
	if not valid(owner) or owner.ordinary_state.status!="closed" or owner.ordinary_state.proposal.is_empty(): return false
	var state: Dictionary=owner.ordinary_state
	var source: Dictionary=Adapter.capture("ordinary",owner.session)
	if not ordinary_current_valid(owner,owner.session,source): return false
	var p: Dictionary=state.proposal.duplicate(true)
	if p.get("kind")==CROSS_KIND: return false
	var next: SeasonState=Adapter.OrdinaryRuntime.create_common(p.basis,p.league)
	if next==null: return false
	state.last_transition={"id":p.id,"source":p.source,"from":state.ordinal,"to":p.basis.ordinal,"league":p.league,"basis":p.basis.duplicate(true)}
	state.ordinal=p.basis.ordinal;state.status="active";state.last_close={};state.proposal={};state.start_basis_sha256=Life.digest(p.basis)
	owner.session=next
	return valid(owner) and ordinary_current_valid(owner,next,Adapter.capture("ordinary",next))
static func awards(owner) -> Dictionary:
	var aggregate: Dictionary=owner.common.get("aggregate",{})
	var result: Dictionary=aggregate.get("legacy_plaques",{}).duplicate(true)
	result.merge(aggregate.get("plaques",{}))
	return result
