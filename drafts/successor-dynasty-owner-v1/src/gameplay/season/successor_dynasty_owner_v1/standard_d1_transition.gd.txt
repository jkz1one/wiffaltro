## Versioned Common V4 start envelope. Shape alone confers no rights or writes.
const Start=preload("res://src/gameplay/season/successor_live_play_v1/common_start.gd")
const Common=preload("common_lifecycle.gd")
const Intent=preload("standard_d1_intent.gd")
const Life=preload("lifecycle.gd")
const KIND: String="common-standard-d1-transition-v1"

static func derive(owner: RefCounted, seed_value: int) -> Dictionary:
	var title: Dictionary=Intent.qualify(owner,seed_value)
	if title.is_empty(): return {}
	var basis: Dictionary=Start.make(seed_value,title.lineage,title.to,int(owner.common.sequence),
		Life.digest(owner.common.aggregate),Common.projection(owner))
	var row: Dictionary={"version":1,"kind":KIND,"basis":basis,"entitlement":title}
	return row if valid(row) else {}

static func valid(value: Variant) -> bool:
	if not value is Dictionary or not SeasonOwnership._keys(value,["version","kind","basis","entitlement"]) \
			or not SeasonOwnership._whole(value.version,1,1) or value.kind!=KIND or not Start.valid(value.basis): return false
	var basis: Dictionary=value.basis
	var title: Variant=value.entitlement
	if not title is Dictionary or not SeasonOwnership._keys(title,["kind","profile","lineage","from","to","seed","close","fold","proof","title_key"]) \
			or title.kind!=Intent.KIND or not ClubCareer.same(title.profile,Intent.Replay.profile(1)) \
			or title.title_key!=Intent.ClubLeagueCareer.title_key("standard",0) \
			or not SeasonOwnership._whole(title.from,1,ClubCareer.MAX_RUNS-1) \
			or not SeasonOwnership._whole(title.to,int(title.from)+1,int(title.from)+1) or title.to!=basis.ordinal \
			or title.lineage!=basis.lineage or not SeasonOwnership._whole(title.seed,int(basis.seed),int(basis.seed)): return false
	var close: Variant=title.close
	var fold: Variant=title.fold
	if not close is Dictionary or not fold is Dictionary or not SeasonOwnership._keys(close,["id","ordinal","reason","session_sha256","receipt","first_title","league","tier"]) \
			or not SeasonOwnership._keys(fold,["id","side","ordinal","sequence","source","aggregate_sha256","receipt"]): return false
	var id: String="%s:%d:close"%[title.lineage,int(title.from)]
	if close.id!=id or not SeasonOwnership._whole(close.ordinal,int(title.from),int(title.from)) or close.reason!="complete" or close.league!="standard" \
			or not SeasonOwnership._whole(close.tier,0,0) or not close.first_title is bool or not Start.sha(close.session_sha256) \
			or fold.id!=id or fold.side!="ordinary" \
			or not SeasonOwnership._whole(fold.ordinal,int(title.from),int(title.from)) or fold.sequence!=basis.common_sequence \
			or not SeasonOwnership._whole(fold.sequence,int(title.from),ClubCareer.MAX_RUNS*2) \
			or fold.aggregate_sha256!=basis.common_sha256 or fold.source!=close.session_sha256: return false
	var proof: Variant=title.proof
	if not proof is Dictionary or not SeasonOwnership._keys(proof,["draws","scores"]) or not proof.scores is Array or proof.scores.size()!=33 \
			or ClubSeasonRecord.analyze(proof).get("finish")!="champion": return false
	var receipt: Dictionary=ClubSeasonRecord.receipt(proof,close.first_title,0)
	return not receipt.is_empty() and ClubCareer.same(receipt,close.receipt) and ClubCareer.same(receipt,fold.receipt)

# Retain the exact admitted Common V4 bytes, not the rotating backup or only
# the compact score projection. The digest is an integrity check, not authority.
const WITNESS_KIND: String="common-standard-d1-closed-source-v1"
const MAX_SOURCE: int=64*1024*1024
const MAX_ENCODED: int=8*1024*1024

static func _pack_closed_source(bytes: PackedByteArray) -> Dictionary:
	if bytes.is_empty() or bytes.size()>MAX_SOURCE: return {}
	var encoded: String=Marshalls.raw_to_base64(bytes.compress(FileAccess.COMPRESSION_DEFLATE))
	if encoded.is_empty() or encoded.length()>MAX_ENCODED: return {}
	return {"version":1,"kind":WITNESS_KIND,"source_bytes":bytes.size(),
		"source_sha256":_source_digest(bytes),"deflate_base64":encoded}

static func _source_digest(bytes: PackedByteArray) -> String:
	var digest:=HashingContext.new()
	if digest.start(HashingContext.HASH_SHA256)!=OK or digest.update(bytes)!=OK: return ""
	return digest.finish().hex_encode()

static func _unpack_closed_source(value: Variant) -> PackedByteArray:
	if not value is Dictionary or not SeasonOwnership._keys(value,["version","kind","source_bytes","source_sha256","deflate_base64"]) \
			or not SeasonOwnership._whole(value.version,1,1) or value.kind!=WITNESS_KIND \
			or not SeasonOwnership._whole(value.source_bytes,1,MAX_SOURCE) or not Start.sha(value.source_sha256) \
			or not value.deflate_base64 is String or value.deflate_base64.is_empty() \
			or value.deflate_base64.length()>MAX_ENCODED: return PackedByteArray()
	var compressed:=Marshalls.base64_to_raw(value.deflate_base64)
	if compressed.is_empty() or Marshalls.raw_to_base64(compressed)!=value.deflate_base64: return PackedByteArray()
	var bytes:=compressed.decompress(int(value.source_bytes),FileAccess.COMPRESSION_DEFLATE)
	return bytes if bytes.size()==int(value.source_bytes) and _source_digest(bytes)==value.source_sha256 else PackedByteArray()

static func _replay_closed_source(value: Variant) -> RefCounted:
	var bytes:=_unpack_closed_source(value)
	if bytes.is_empty(): return null
	var parser:=JSON.new()
	if parser.parse(bytes.get_string_from_utf8())!=OK: return null
	# Same strict Common V4 production restore as Store.read. Normal restore
	# does not recognize this witness or the D1 draft dialect.
	var restore=load("res://src/gameplay/season/successor_dynasty_owner_v1/owner.gd")
	return restore.restore(parser.data)

static func rederive_from_witness(value: Variant, seed_value: int) -> Dictionary:
	if not SeasonOwnership._whole(seed_value,0,2147483647): return {}
	var owner:=_replay_closed_source(value)
	return derive(owner,seed_value) if owner!=null else {}

static func restore_witnessed_d1_draft(witness: Variant, draft: Variant) -> SeasonState:
	var codec=load("res://src/gameplay/season/successor_live_play_v1/codec.gd")
	var season: SeasonState=codec.restore_d1_draft(draft)
	if season==null: return null
	var stored: Dictionary=season.career.start_basis
	var derived:=rederive_from_witness(witness,int(stored.basis.seed))
	return season if not derived.is_empty() and ClubCareer.same(stored,derived) else null

# A separate candidate dialect until the owner has a strict async Continue route.
# The entire former Common career (including dormant Dynasty/metadata) sits in
# the witness; duplicating a caller's parked fields here would add another truth.
const SLOT_FORMAT: String="wiffaltro-common-career-standard-d1-v1"
const MAX_SLOT: int=16*1024*1024

static func _durable_candidate(witness: Dictionary, draft: Dictionary) -> PackedByteArray:
	var body: Dictionary={"version":1,"format":SLOT_FORMAT,"witness":witness,"draft":draft}
	var canonical:=_slot_text(body)
	if canonical.is_empty(): return PackedByteArray()
	body.stamp=canonical.sha256_text()
	var bytes:=_slot_text(body).to_utf8_buffer()
	return bytes if bytes.size()<=MAX_SLOT else PackedByteArray()

static func _slot_text(value: Variant) -> String:
	# Normalize integral JSON values consistently with the strict materialized codec.
	var engine:=JSON.stringify(value,"",true,true)
	if engine.length()>MAX_SLOT: return ""
	return JSON.stringify(JSON.parse_string(engine),"",true,true)

static func _durable_shape(bytes: PackedByteArray) -> Dictionary:
	if bytes.is_empty() or bytes.size()>MAX_SLOT: return {}
	var text:=bytes.get_string_from_utf8()
	var data: Variant=JSON.parse_string(text)
	if not data is Dictionary or not SeasonOwnership._keys(data,["version","format","witness","draft","stamp"]) \
			or not SeasonOwnership._whole(data.version,1,1) or data.format!=SLOT_FORMAT \
			or not data.witness is Dictionary or not data.draft is Dictionary or not Start.sha(data.stamp) \
			or _slot_text(data)!=text: return {}
	var body: Dictionary=data.duplicate()
	body.erase("stamp")
	return data if _slot_text(body).sha256_text()==data.stamp else {}

# This is the strict background replay contract. It grants no Store binding.
# On success the caller must separately match exact slot bytes/epoch and adopt
# the verified owner+draft with its own new raw guard.
static func _cold_durable_candidate(bytes: PackedByteArray) -> Dictionary:
	var data:=_durable_shape(bytes)
	if data.is_empty(): return {}
	var codec=load("res://src/gameplay/season/successor_live_play_v1/codec.gd")
	var draft: SeasonState=codec.restore_d1_draft(data.draft)
	if draft==null or draft.phase!=SeasonState.Phase.DRAFT or not draft.picks.is_empty(): return {}
	var owner:=_replay_closed_source(data.witness)
	if owner==null: return {}
	var stored: Dictionary=draft.career.start_basis
	var derived:=derive(owner,int(stored.basis.seed))
	return {"previous_owner":owner,"draft":draft} if not derived.is_empty() and ClubCareer.same(stored,derived) else {}
