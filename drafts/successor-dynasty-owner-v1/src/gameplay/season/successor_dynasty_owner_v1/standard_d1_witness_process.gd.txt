extends Node
## Private same-binary proof transport. Only activation stages disposable bytes.
const Store=preload("owner_store.gd")
const T=preload("standard_d1_transition.gd")
const View=preload("standard_d1_view.gd")
const KIND: String="common-standard-d1-witness-worker-v2"
const CROSS_READ_KIND: String="common-cross-league-draft-read-worker-v1"
const PAIR_SETTLE_KIND: String="common-doubleheader-no-purchase-return-settle-v1"
const NEXT_OPENER_SETTLE_KIND: String="common-doubleheader-no-purchase-next-opener-settle-v1"
const NEXT_REVERSE_SETTLE_KIND: String="common-doubleheader-no-purchase-next-return-settle-v1"
const ACTIVATE_KIND: String="common-cross-league-activate-worker-v1"
const ARCHIVE_ACTIVATE_KIND: String="common-archived-dugout-ordinal-three-activate-v1"
const CROSS_FILE: String="candidate.json"
const CROSS_MAX: int=24*1024*1024
const CROSS_KIND: String="common-cross-league-review-worker-v2"
const CONTINUE_KIND: String="common-standard-d1-continue-worker-v2"
const PACK_FEATURE: String="wiffaltro_engineering_export"
const PACK_BOOTSTRAP: String="res://engineering_bootstrap/entry.tscn"
const PACK_WITNESS_FLAG: String="WIFFALTRO_PACKAGED_CROSS_DRAFT"
const PACK_ARCHIVED_DUGOUT_FLAG: String="WIFFALTRO_PACKAGED_ARCHIVED_DUGOUT_ACTIVATION"
const ARG: String="--d1-witness="
const FRAME: String="WIFFALTRO_D1_WITNESS_V2 "
const MAX_FRAME: int=16384
const MAX_MS: int=300000
const REFUSED_EXIT: int=23
signal completed(outcome: String, proof: Dictionary)
var running: bool=false
var _pid: int=-1
var _stdout: FileAccess
var _stderr: FileAccess
var _output: PackedByteArray
var _request: Dictionary={}
var _started: int=0
var _last_poll: int=0
var _stopping: String=""
var _completed_view: RefCounted
var _completed_cross: Dictionary={}
var _completed_cross_read: Dictionary={}
var _cross_dir: String=""

static func supported(kind: String=KIND) -> bool:
	if OS.get_name()!="Linux" or OS.get_environment("WIFFALTRO_DYNASTY_ENGINEERING")!="1": return false
	if OS.has_feature("editor"): return true
	if not Store.cross_league_preview() or not OS.has_feature(PACK_FEATURE) \
			or ProjectSettings.get_setting("application/run/main_scene")!=PACK_BOOTSTRAP: return false
	return (kind==CROSS_READ_KIND and OS.get_environment(PACK_WITNESS_FLAG)=="1") \
		or (kind==ARCHIVE_ACTIVATE_KIND and OS.get_environment(PACK_ARCHIVED_DUGOUT_FLAG)=="1")

# Both fixed routers decode the same canonical bounded capsule. It chooses a
# contract and exact revision, never a resource, executable, output or slot.
static func parse_args(args: PackedStringArray) -> Dictionary:
	if args.size()!=1 or not args[0].begins_with(ARG) or args[0].length()>1024: return {}
	var json:=Marshalls.base64_to_utf8(args[0].trim_prefix(ARG))
	var request: Variant=JSON.parse_string(json)
	return request if request_valid(request) and JSON.stringify(request,"",true,true)==json else {}

# A directly loaded scene cannot arm itself. Only the compiled packaging
# bootstrap installs this fixed, one-shot in-process boot lease.
static var _packed_admission: Dictionary={}

static func arm_packaged_worker(request: Dictionary) -> bool:
	if not _packed_admission.is_empty() or not request_valid(request) \
		or OS.has_feature("editor") or not supported(request.kind) \
		or DisplayServer.get_name()!="headless": return false
	_packed_admission=request.duplicate(true)
	return true

static func take_packaged_worker(request: Dictionary) -> bool:
	var admitted: Dictionary=_packed_admission
	_packed_admission={}
	return not admitted.is_empty() and ClubCareer.same(admitted,request)

static func _bounded_slot(kind: String) -> PackedByteArray:
	var file:=FileAccess.open(Store.PATH,FileAccess.READ)
	if file==null: return PackedByteArray()
	var length:=file.get_length()
	if length<1 or length>(T.MAX_SLOT if kind==CONTINUE_KIND else (CROSS_MAX if kind in [CROSS_READ_KIND,ARCHIVE_ACTIVATE_KIND] else Store.MAX_BYTES)): return PackedByteArray()
	var bytes:=file.get_buffer(length)
	var valid: bool=bytes.size()==length and file.get_length()==length
	file.close()
	return bytes if valid else PackedByteArray()

static func _revision(kind: String=KIND) -> String:
	var bytes:=_bounded_slot(kind)
	return "" if bytes.is_empty() else T._source_digest(bytes)

static func request_valid(value: Variant) -> bool:
	return value is Dictionary and SeasonOwnership._keys(value,["version","kind","token","source_sha256","seed"]) \
		and SeasonOwnership._whole(value.version,2,2) and value.kind in [KIND,CONTINUE_KIND,CROSS_KIND,ACTIVATE_KIND,ARCHIVE_ACTIVATE_KIND,CROSS_READ_KIND,PAIR_SETTLE_KIND,NEXT_OPENER_SETTLE_KIND,NEXT_REVERSE_SETTLE_KIND] \
		and value.token is String and value.token.length()==32 and value.token.is_valid_hex_number(false) \
		and T.Start.sha(value.source_sha256) and SeasonOwnership._whole(value.seed,0,2147483647) \
		and (value.kind==KIND or value.seed==0)

# Godot's JSON parser materializes numbers as floats; normalize once before
# writing so the receiver can check the exact canonical wire representation.
static func canonical(value: Variant) -> String:
	return JSON.stringify(JSON.parse_string(JSON.stringify(value,"",true,true)),"",true,true)

func start(source_sha256: String, seed_value: int, kind: String=KIND) -> bool:
	if running or not is_inside_tree() or not supported(kind) or not T.Start.sha(source_sha256) \
			or not SeasonOwnership._whole(seed_value,0,2147483647) or _revision(kind)!=source_sha256 \
			or kind not in [KIND,CONTINUE_KIND,CROSS_KIND,ACTIVATE_KIND,ARCHIVE_ACTIVATE_KIND,CROSS_READ_KIND,PAIR_SETTLE_KIND,NEXT_OPENER_SETTLE_KIND,NEXT_REVERSE_SETTLE_KIND] or (kind!=KIND and seed_value!=0) \
			or (kind in [CROSS_KIND,ACTIVATE_KIND,ARCHIVE_ACTIVATE_KIND,CROSS_READ_KIND] and not Store.cross_league_preview()): return false
	_request={"version":2,"kind":kind,"token":Crypto.new().generate_random_bytes(16).hex_encode(),
		"source_sha256":source_sha256,"seed":seed_value}
	if kind in [PAIR_SETTLE_KIND,NEXT_OPENER_SETTLE_KIND,NEXT_REVERSE_SETTLE_KIND] and not Store.private_pair_settlement_preview():
		_clear()
		return false
	if kind in [ACTIVATE_KIND,ARCHIVE_ACTIVATE_KIND,PAIR_SETTLE_KIND,NEXT_OPENER_SETTLE_KIND,NEXT_REVERSE_SETTLE_KIND]:
		var folder:=_stage_dir(_request.token,kind)
		if folder.is_empty() or DirAccess.make_dir_absolute(folder)!=OK:
			_clear()
			return false
		_cross_dir=folder
	var packet: String=ARG+Marshalls.utf8_to_base64(canonical(_request))
	var args: PackedStringArray=["--headless","--",packet]
	if OS.has_feature("editor"):
		var worker: String=(get_script() as GDScript).resource_path.get_base_dir().path_join("standard_d1_witness_worker.tscn")
		args=["--headless","--path",ProjectSettings.globalize_path("res://"),worker,"--",packet]
	# Release child boots the same official executable and adjacent PCK. No
	# scene, project path or pack override is handed to the release process.
	var launched:=OS.execute_with_pipe(OS.get_executable_path(),args,false)
	_pid=int(launched.get("pid",-1))
	_stdout=launched.get("stdio")
	_stderr=launched.get("stderr")
	if _pid<=0 or _stdout==null or _stderr==null:
		if _pid>0: OS.kill(_pid)
		_clear()
		_clean_cross()
		return false
	process_mode=Node.PROCESS_MODE_ALWAYS
	_started=Time.get_ticks_msec()
	_last_poll=_started
	running=true
	return true

func cancel() -> void:
	if not running: return
	_stopping="cancelled"
	_stop()

func _exit_tree() -> void:
	if running:
		_stopping="cancelled"
		_stop(false)

func _stop(emit_result: bool=true) -> void:
	if OS.is_process_running(_pid) and OS.kill(_pid)!=OK: return
	_finish(_stopping,{},emit_result)

func _clear() -> void:
	running=false
	_stdout=null;_stderr=null;_output=PackedByteArray();_request={};_pid=-1;_stopping=""

func _finish(outcome: String, proof: Dictionary={}, emit_result: bool=true) -> void:
	_clear()
	if emit_result: completed.emit(outcome,proof)
	_completed_cross={}
	_completed_cross_read={}
	_clean_cross()

static func _stage_dir(token: String,kind: String) -> String:
	if token.length()!=32 or not token.is_valid_hex_number(false): return ""
	if kind in [PAIR_SETTLE_KIND,NEXT_OPENER_SETTLE_KIND,NEXT_REVERSE_SETTLE_KIND]: return "user://.pair-settle-"+token
	if kind==ARCHIVE_ACTIVATE_KIND: return "user://.archive-activate-"+token
	return "user://.cross-activate-"+token if kind==ACTIVATE_KIND else ""

func _clean_cross() -> void:
	if _cross_dir.is_empty(): return
	var root:=DirAccess.open("user://")
	if root!=null and not root.is_link(_cross_dir.get_file()):
		DirAccess.remove_absolute(_cross_dir.path_join(CROSS_FILE))
		DirAccess.remove_absolute(_cross_dir)
	_cross_dir=""

# The file is not authority: only the currently validated child pipe can
# unlock one read of its exact size and content in this synchronous callback.
func take_cross_bytes(source_sha256: String,kind: String=ACTIVATE_KIND) -> PackedByteArray:
	return _take_staged_bytes(source_sha256,kind) if kind in [ACTIVATE_KIND,ARCHIVE_ACTIVATE_KIND] else PackedByteArray()

static func pair_settlement_kind(leg: int) -> String:
	return {1:PAIR_SETTLE_KIND,2:NEXT_OPENER_SETTLE_KIND,3:NEXT_REVERSE_SETTLE_KIND}.get(leg,"")

func take_pair_settled_bytes(source_sha256: String,leg: int) -> PackedByteArray:
	return _take_staged_bytes(source_sha256,pair_settlement_kind(leg))

func _take_staged_bytes(source_sha256: String,kind: String) -> PackedByteArray:
	var proof: Dictionary=_completed_cross
	_completed_cross={}
	if proof.is_empty() or proof.request.source_sha256!=source_sha256 \
		or proof.request.kind!=kind or _cross_dir!=_stage_dir(proof.request.token,kind): return PackedByteArray()
	var root:=DirAccess.open("user://")
	var folder:=DirAccess.open(_cross_dir)
	if root==null or root.is_link(_cross_dir.get_file()) or folder==null or folder.is_link(CROSS_FILE):
		return PackedByteArray()
	var f:=FileAccess.open(_cross_dir.path_join(CROSS_FILE),FileAccess.READ)
	if f==null or f.get_length()!=int(proof.size): return PackedByteArray()
	var b:=f.get_buffer(int(proof.size))
	f.close()
	return b if b.size()==int(proof.size) and T._source_digest(b)==proof.candidate_sha256 else PackedByteArray()

func take_cross_read_seal(source_sha256: String) -> String:
	var proof: Dictionary=_completed_cross_read
	_completed_cross_read={}
	return proof.draft_sha256 if not proof.is_empty() and proof.request.source_sha256==source_sha256 else ""

# The view is obtainable once, synchronously while this exact private child
# emits completion. No caller-supplied payload can populate the handoff.
func take_completed_view() -> RefCounted:
	var view:=_completed_view
	_completed_view=null
	return view if view!=null and _revision(CONTINUE_KIND)==view._revision else null

func _drain() -> bool:
	if _stdout==null or _stderr==null or _stderr.get_length()>0: return false
	var available: int=_stdout.get_length()
	if available>MAX_FRAME-_output.size(): return false
	if available>0: _output.append_array(_stdout.get_buffer(available))
	return true

func _process(_delta: float) -> void:
	if not running or Time.get_ticks_msec()-_last_poll<100: return
	_last_poll=Time.get_ticks_msec()
	if not _stopping.is_empty():
		_stop()
		return
	if _last_poll-_started>MAX_MS or not _drain():
		_stopping="invalid"
		_stop()
		return
	if OS.is_process_running(_pid): return
	var exit_code:=OS.get_process_exit_code(_pid)
	if not _drain():
		_finish("invalid")
		return
	if _revision(_request.kind)!=_request.source_sha256:
		_finish("stale")
		return
	var packet:=_decode(_output,_request,exit_code)
	if packet.is_empty():
		_finish("invalid")
	elif _request.kind==CONTINUE_KIND and packet.outcome=="qualified":
		var view:=View._from_bytes(_bounded_slot(CONTINUE_KIND),_request.source_sha256,packet.digests)
		if view==null:
			_finish("stale" if _revision(CONTINUE_KIND)!=_request.source_sha256 else "invalid")
			return
		_completed_view=view
		_finish("qualified")
		_completed_view=null
	elif _request.kind==CROSS_READ_KIND and packet.outcome=="qualified":
		_completed_cross_read=packet
		_finish("qualified")
	elif _request.kind in [ACTIVATE_KIND,ARCHIVE_ACTIVATE_KIND,PAIR_SETTLE_KIND,NEXT_OPENER_SETTLE_KIND,NEXT_REVERSE_SETTLE_KIND] and packet.outcome=="qualified":
		_completed_cross=packet
		_finish("qualified")
	else:
		_finish(packet.outcome,packet if packet.outcome=="qualified" else {})

static func _decode(bytes: PackedByteArray, request: Dictionary, exit_code: int) -> Dictionary:
	if bytes.is_empty() or bytes.size()>MAX_FRAME or not request_valid(request): return {}
	var text:=bytes.get_string_from_utf8().replace("\r\n","\n")
	if not text.ends_with("\n"): return {}
	var lines:=text.split("\n",true)
	# The same executable prints an engine banner and blank line before the scene.
	if lines.size()!=4 or not lines[0].begins_with("Godot Engine v") \
			or not lines[0].ends_with(" - https://godotengine.org") or not lines[1].is_empty() \
			or not lines[2].begins_with(FRAME): return {}
	var json: String=lines[2].trim_prefix(FRAME)
	var result: Variant=JSON.parse_string(json)
	if not result is Dictionary or not ClubCareer.same(result.get("request"),request) \
			or JSON.stringify(result,"",true,true)!=json: return {}
	if request.kind in [ACTIVATE_KIND,ARCHIVE_ACTIVATE_KIND,PAIR_SETTLE_KIND,NEXT_OPENER_SETTLE_KIND,NEXT_REVERSE_SETTLE_KIND]:
		if not SeasonOwnership._keys(result,["request","outcome","size","candidate_sha256"]) or not result.candidate_sha256 is String: return {}
		if exit_code==REFUSED_EXIT and result.outcome=="refused" and result.size==0 and result.candidate_sha256.is_empty(): return result
		if exit_code==0 and result.outcome=="qualified" and SeasonOwnership._whole(result.size,1,CROSS_MAX) \
			and T.Start.sha(result.candidate_sha256): return result
		return {}
	if request.kind in [CROSS_KIND,CROSS_READ_KIND]:
		if not SeasonOwnership._keys(result,["request","outcome","draft_sha256"]) \
			or not result.draft_sha256 is String: return {}
		if exit_code==REFUSED_EXIT and result.outcome=="refused" and result.draft_sha256.is_empty(): return result
		if exit_code==0 and result.outcome=="qualified" and T.Start.sha(result.draft_sha256): return result
		return {}
	if request.kind==CONTINUE_KIND:
		if not SeasonOwnership._keys(result,["request","outcome","digests"]) \
				or not result.digests is Dictionary: return {}
		if exit_code==REFUSED_EXIT and result.outcome=="refused" and result.digests.is_empty(): return result
		if exit_code==0 and result.outcome=="qualified" \
				and SeasonOwnership._keys(result.digests,["stamp","witness","draft"]) \
				and T.Start.sha(result.digests.stamp) and T.Start.sha(result.digests.witness) \
				and T.Start.sha(result.digests.draft): return result
		return {}
	if not SeasonOwnership._keys(result,["request","outcome","transition"]): return {}
	if exit_code==REFUSED_EXIT and result.outcome=="refused" and result.transition is Dictionary and result.transition.is_empty(): return result
	if exit_code==0 and result.outcome=="qualified" and T.valid(result.transition) \
			and result.transition.basis.seed==request.seed: return result
	return {}
