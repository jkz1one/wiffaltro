extends "res://src/gameplay/season/successor_league_general_v2/physical_runner.gd"
## Same physical engine, bounded Dynasty-only request/report boundary.
## Inherited cleanup and finished/failed signals remain unchanged.
const DynastyRequest = preload("physical_request.gd")
const DynastyReport = preload("physical_report.gd")

func start(source: MatchState, seed: int, field: StringName = PitchBatLab.FIELD_ID,
	ledger: ClubAssetLedger = null, pool_request: Dictionary = {}, npc_season: SeasonState = null) -> bool:
	if ledger != null or not pool_request.is_empty() or npc_season != null \
		or field not in [PitchBatLab.FIELD_ID,SeasonState.AWAY_FIELD_ID] \
		or not is_inside_tree() or _running or _viewport != null or _lab != null \
		or seed < 0 or seed > 2147483647 or ContentDB.get_field(field) == null: return false
	var snapshot: MatchState = DynastyRequest.capture(source)
	if snapshot == null: return false
	var viewport: SubViewport = SubViewport.new()
	viewport.size = Vector2i(1280,720)
	viewport.own_world_3d = true
	viewport.gui_disable_input = true
	viewport.render_target_update_mode = SubViewport.UPDATE_DISABLED
	_viewport = viewport
	_running = true # A child callback cannot start a second job during installation.
	add_child(viewport)
	if not is_instance_valid(viewport) or _viewport != viewport or viewport.get_parent()!=self \
		or _lab != null or not _running:
		if _viewport == viewport and _lab == null: cancel()
		return false
	var lab: PitchBatLab = PitchBatLab.new()
	lab._configured_match = snapshot
	lab._field_id = field
	lab._sky_backdrop_enabled = false
	lab._automation = MatchAutomation.new()
	lab._automation.seed = seed
	lab._automation.presentation_enabled = presentation_enabled
	_lab = lab
	viewport.add_child(lab)
	if not is_instance_valid(lab) or not is_instance_valid(viewport) or _lab != lab \
		or _viewport != viewport or lab.get_parent()!=viewport or lab._match_state!=snapshot \
		or not _running:
		if _viewport == viewport and _lab == lab: cancel()
		return false
	_elapsed = 0.0
	_stalled = 0.0
	_progress = ""
	set_process(true)
	return true

func _process(delta: float) -> void:
	if not _running: return
	if not is_instance_valid(_lab):
		_fail("Physical match ownership was lost; no result was settled.")
		return
	_elapsed += delta
	var lab: PitchBatLab = _lab
	var state: MatchState = lab._match_state
	if state == null:
		_fail("Physical match initialization failed.")
		return
	if state.phase == MatchState.Phase.GAME_END:
		var report: Dictionary = DynastyReport.capture(lab)
		if _lab != lab or lab._match_state != state or not _running: return
		if not DynastyReport.valid(report,state.pitch_catalog):
			_fail("Physical match evidence did not balance.")
			return
		cancel()
		if _running or _lab != null or _viewport != null: return
		finished.emit(report)
		return
	var progress: String = "%d:%d:%d" % [state.phase,state.plate_appearance_number,lab._throw_number]
	_stalled = _stalled + delta if progress == _progress else 0.0
	_progress = progress
	if _elapsed >= MAX_SECONDS or _stalled >= STALL_SECONDS:
		_fail("Physical match exceeded its time or progress limit; no result was settled.")
