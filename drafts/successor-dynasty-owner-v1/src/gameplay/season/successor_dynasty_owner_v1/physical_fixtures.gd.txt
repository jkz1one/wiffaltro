extends SeasonPhysicalFixtures
const Report = preload("physical_report.gd")
## Same physical report algorithms, explicit materialized owner scope.
## No cold reader calls resolve; only new current-round settlement does.
static var resolve_calls: int = 0
## Versioned evidence archive. Historical score-only seasons never instantiate this.
# gdlint: disable=max-returns



func to_data() -> Dictionary:
	return {"version": 1, "reports": reports.duplicate(true), "pending": pending.duplicate(true)}


static func from_data(value: Variant, pitch_catalog: String = "") -> SeasonPhysicalFixtures:
	if not load("res://src/gameplay/season/successor_league_general_v2/pitch_evidence.gd").context_valid(pitch_catalog):
		return null
	if (
		not value is Dictionary
		or not SeasonOwnership._keys(value, ["version", "reports", "pending"])
	):
		return null
	if (
		value.version != 1
		or not value.reports is Array
		or value.reports.size() > 23
		or not value.pending is Dictionary
	):
		return null
	var result: SeasonPhysicalFixtures = load("res://src/gameplay/season/successor_dynasty_owner_v1/physical_fixtures.gd").new()
	var ids: Array = []
	for row: Variant in value.reports:
		if not envelope(row, pitch_catalog) or ids.has(row.fixture):
			return null
		ids.append(row.fixture)
		result.reports.append(row.duplicate(true))
	result.pending = value.pending.duplicate(true)
	result.projecting = true
	return result


static func envelope(value: Variant, pitch_catalog: String = "") -> bool:
	return (
		value is Dictionary
		and SeasonOwnership._keys(value, ["fixture", "request", "report"])
		and SeasonOwnership._whole(value.fixture, 0, 32)
		and value.request is String
		and value.request.length() == 64
		and Report.valid(value.report, pitch_catalog)
	)


static func seed_for(season: SeasonState, fixture: Dictionary) -> int:
	return (season.season_seed + int(fixture.id) * 104729) & 0x7fffffff


static func match_for(season: SeasonState, fixture: Dictionary) -> MatchState:
	if not load("res://src/gameplay/season/successor_dynasty_owner_v1/codec.gd").scope(season): return null
	for key: String in ["inventory_rules","home_property","owned_surface_night","npc_venues","final_venue","_league_home_context"]:
		if not season.get(key).is_empty(): return null
	if not SeasonOwnership._whole(fixture.get("away"),0,5) \
		or not SeasonOwnership._whole(fixture.get("home"),0,5) \
		or fixture.away==fixture.home or not SeasonOwnership._whole(fixture.get("id"),0,32) \
		or not SeasonOwnership._whole(fixture.get("round"),0,32): return null
	if (fixture.away==0 or fixture.home==0) and SeasonInventoryRules.closed(season.build): return null
	if not season.academy_fixture_ready(fixture): return null
	var away: TeamMatchState = season._make_team(int(fixture.away),int(fixture.id))
	var home: TeamMatchState = season._make_team(int(fixture.home),int(fixture.id))
	if away==null or home==null: return null
	var state: MatchState = MatchState.create(away,home)
	state.league_rules=season.league_rules.duplicate(true)
	state.pitch_catalog=catalog_for(season)
	away.pitch_catalog=state.pitch_catalog
	home.pitch_catalog=state.pitch_catalog
	state.ai_tactical_quality=clampf(0.15+season.difficulty*0.25+minf(int(fixture.round),9)*0.025,0.0,1.0)
	if away.dynasty_start.get("owner_id")!="club:%d"%int(fixture.away) \
		or home.dynasty_start.get("owner_id")!="club:%d"%int(fixture.home): return null
	return state if Report.context_valid(state) else null


static func request_key(season: SeasonState, fixture: Dictionary) -> String:
	var field: FieldDefinition = SeasonState.field_for_fixture(fixture, season)
	if field == null:
		return ""
	var state: MatchState = match_for(season, fixture)
	if state == null:
		return ""
	var rows: Array = []
	for team: TeamMatchState in [state.away_team, state.home_team]:
		var players: Array = []
		for player: PlayerMatchState in team.roster:
			players.append(
				{
					"definition": _resource(player.definition,
						season.build != null and (season.build.pitch_catalog_current() or season.build.call("pitch_catalog_successor"))),
					"capacity": player.stamina_max,
					"initial": player.stamina_remaining
				}
			)
		var row: Dictionary = {
			"name": team.display_name,
			"pitcher": team.pitcher_index,
			"fielder": team.fielder_index,
			"players": players
		}
		if team.ai_sponsor_choices:
			row["choice_policy"] = 1
		if not team.ai_tactical_hitter.is_empty():
			row["batting_supplies"] = {
				"featured": team.ai_tactical_hitter,
				"held": team.ai_tactical_initial.duplicate(true)
			}
		if team.ai_heat:
			row["heat_policy"] = 1
		if not team.ai_recovery_pitcher.is_empty():
			row["recovery_policy"] = {"version": 1, "pitcher": team.ai_recovery_pitcher}
		if team.ai_base:
			row["base_policy"] = 1
		if not team.ai_budget.is_empty():
			row["budget_policy"] = team.ai_budget.duplicate(true)
		if not state.league_rules.is_empty():
			row["league_ownership"] = team.league_ownership.duplicate(true)
			if LeagueRules.dynasty(state.league_rules):
				row["dynasty_start"] = team.dynasty_start.duplicate(true)
			if LeagueRules.doubleheader(state.league_rules):
				row["league_pair"] = team.league_pair.duplicate(true)
		rows.append(row)
	var request: Dictionary = {
		"resolver": Report.RESOLVER,
		"seed": seed_for(season, fixture),
		"field": String(field.id),
		"quality": state.ai_tactical_quality,
		"fixture": fixture,
		"teams": rows
	}
	request["league_rules"] = state.league_rules.duplicate(true)
	request["pitch_catalog"] = state.pitch_catalog
	return JSON.stringify(request,"",true,true).sha256_text()


func resolve(season: SeasonState, fixture: Dictionary) -> Dictionary:
	resolve_calls += 1
	return _resolve(season,fixture)

# The optional argument is retained for diagnostic callers, not a validation token.
func _resolve(season: SeasonState, fixture: Dictionary, _allow_reuse: bool = true) -> Dictionary:
	needed = {}
	if cursor < 0 or cursor > reports.size():
		error="The physical evidence cursor is invalid."
		return {}
	if cursor == reports.size():
		needed=fixture.duplicate(true)
		return {}
	var catalog: String = catalog_for(season)
	var row: Variant = reports[cursor]
	if not envelope(row,catalog):
		error="The carried-profile physical report envelope is invalid."
		return {}
	# Full report validation above happens inside this call. No persistent cache
	# or caller-provided validity is accepted before the pure matching seam.
	var report: Dictionary = row.report.physical
	var field: FieldDefinition = SeasonState.field_for_fixture(fixture,season)
	var state: MatchState = match_for(season,fixture)
	if field==null or state==null or row.fixture!=fixture.id \
		or row.request!=request_key(season,fixture) or report.seed!=seed_for(season,fixture) \
		or report.field!=String(field.id) or not Report._matches_validated_current(row.report,state):
		error="Saved physical fixture does not match its scheduled clubs and carried builds."
		return {}
	cursor += 1
	var result: Dictionary = fixture.duplicate(true)
	result["away_runs"]=int(report.away_runs)
	result["home_runs"]=int(report.home_runs)
	result["performance"]=report.performance.duplicate(true)
	result["ai_tactics"]={}
	for index in range(2):
		result.ai_tactics[str(int(fixture.away if index==0 else fixture.home))] = PhysicalBaseEvidence.enriched(
			report.tactics.teams[index],report.tactics.base).duplicate(true)
	return result


static func _resource(value: Variant, current_pitches: bool = false) -> Variant:
	if value is Resource:
		var data: Dictionary = {}
		for property: Dictionary in value.get_property_list():
			if not current_pitches and WorkingPitchProgression.neutral_pitch_property(value, property.name):
				continue
			if (
				int(property.usage) & PROPERTY_USAGE_SCRIPT_VARIABLE
				and int(property.usage) & PROPERTY_USAGE_STORAGE
			):
				data[property.name] = _resource(value.get(property.name), current_pitches)
		return data
	if value is Array:
		return value.map(func(item: Variant) -> Variant: return _resource(item, current_pitches))
	if value is Dictionary:
		var data: Dictionary = {}
		for key: Variant in value:
			data[key] = _resource(value[key], current_pitches)
		return data
	if current_pitches and value is Vector2:
		return [value.x, value.y]
	if value is Vector3:
		return [value.x, value.y, value.z]
	return value


# The caller supplies a season reconstructed by its exact versioned save boundary.
static func catalog_for(season: SeasonState) -> String:
	return season.build.pitch_catalog_signature() if season != null and season.build != null else ""
