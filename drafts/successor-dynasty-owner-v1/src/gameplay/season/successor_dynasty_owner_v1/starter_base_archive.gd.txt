extends RefCounted
## Inactive canonical page transport. This module grants no Store or filesystem authority.
const C = preload("codec.gd")
const Core = preload("starter_base_chain.gd")
const Life = preload("lifecycle.gd")
const HEAD := "starter-base-head-v1"
const PAGE := "starter-base-page-v1"
const VERSION := 1
const MAX_HEAD := 128 * 1024
const MAX_PAGE := 64 * 1024 * 1024

static func _owner() -> Script:
	return load("res://src/gameplay/season/successor_dynasty_owner_v1/owner.gd")

static func _wire(value: Dictionary) -> PackedByteArray:
	return JSON.stringify(Life._canonical_numbers(value),"",true,true).to_utf8_buffer()

static func _hash(raw: PackedByteArray) -> String:
	var context := HashingContext.new()
	if context.start(HashingContext.HASH_SHA256)!=OK or context.update(raw)!=OK: return ""
	return context.finish().hex_encode()

static func _digest(value: Variant) -> bool:
	if not value is String or value.length()!=64: return false
	for i: int in 64:
		var code: int=value.unicode_at(i)
		if not ((code>=48 and code<=57) or (code>=97 and code<=102)): return false
	return true

static func _decode(raw: PackedByteArray, cap: int) -> Dictionary:
	if raw.is_empty() or raw.size()>cap: return {}
	var decoded: Variant=_owner()._bytes_value(raw)
	return decoded if decoded is Dictionary and C.primitive(decoded) and _wire(decoded)==raw else {}

static func _root(seed: int, league: String) -> String:
	return _hash(_wire({"kind":HEAD,"version":VERSION,"seed":seed,"league":league}))

static func make_head(seed: int, league: String) -> PackedByteArray:
	if not C.whole(seed,0,2147483647) or league!="dugout": return PackedByteArray()
	return _wire({"kind":HEAD,"version":VERSION,"seed":seed,"league":league,
		"root":_root(seed,league),"ordinal":1,"pages":[]})

static func _head(raw: PackedByteArray) -> Dictionary:
	var h: Dictionary=_decode(raw,MAX_HEAD)
	if not C.keys(h,["kind","version","seed","league","root","ordinal","pages"]) \
			or h.kind!=HEAD or h.version!=VERSION or h.league!="dugout" \
			or not C.whole(h.seed,0,2147483647) or not C.whole(h.ordinal,1,ClubCareer.MAX_RUNS) \
			or not h.pages is Array or h.pages.size()!=int(h.ordinal)-1 \
			or not _digest(h.root) or h.root!=_root(int(h.seed),h.league): return {}
	var seen: Dictionary={}
	for id: Variant in h.pages:
		if not _digest(id) or id==h.root or seen.has(id): return {}
		seen[id]=true
	return h

static func _previous(h: Dictionary, ordinal: int) -> String:
	return h.root if ordinal==1 else h.pages[ordinal-2]

static func _page(raw: PackedByteArray, h: Dictionary, ordinal: int) -> Dictionary:
	var p: Dictionary=_decode(raw,MAX_PAGE)
	if not C.keys(p,["kind","version","root","ordinal","previous","image","close"]) \
			or p.kind!=PAGE or p.version!=VERSION or p.root!=h.root or p.ordinal!=ordinal \
			or p.previous!=_previous(h,ordinal) or not p.image is Dictionary or not p.close is Dictionary \
			or p.close.get("ordinal")!=ordinal or p.close.get("reason") not in ["complete","abandon"] \
			or not _digest(p.close.get("session_sha256")): return {}
	if p.close.session_sha256!=Life.digest(p.image): return {}
	return p

# Pure candidate construction, not authority. An exact but fabricated Close can
# never pass restore_pages: the normal native round/Close path is rederived there.
static func make_page(head: PackedByteArray, image: Dictionary, close: Dictionary) -> PackedByteArray:
	var h: Dictionary=_head(head)
	if h.is_empty() or int(h.ordinal)>=ClubCareer.MAX_RUNS: return PackedByteArray()
	if not C.primitive(image) or not C.primitive(close): return PackedByteArray()
	var ordinal: int=int(h.ordinal)
	var raw: PackedByteArray=_wire({"kind":PAGE,"version":VERSION,"root":h.root,
		"ordinal":ordinal,"previous":_previous(h,ordinal),"image":image,"close":close})
	return raw if not _page(raw,h,ordinal).is_empty() else PackedByteArray()

# Appending creates a new head value. It changes neither predecessor bytes nor
# the caller's filesystem, and cannot install an owner.
static func append_head(head: PackedByteArray, page: PackedByteArray) -> PackedByteArray:
	var h: Dictionary=_head(head)
	if h.is_empty() or int(h.ordinal)>=ClubCareer.MAX_RUNS or _page(page,h,int(h.ordinal)).is_empty():
		return PackedByteArray()
	var pin: String=_hash(page)
	if not _digest(pin) or pin==h.root or h.pages.has(pin): return PackedByteArray()
	h.pages.append(pin)
	h.ordinal=int(h.ordinal)+1
	var raw: PackedByteArray=_wire(h)
	return raw if raw.size()<=MAX_HEAD else PackedByteArray()

# Same ordinary no-argument seed as App.propose_next. No caller-supplied archive
# seed or transition is permitted to change authored progression.
static func _default_next_seed(native: RefCounted) -> int:
	return ("ordinary:%s:%d:nextseed"%[native.ordinary_state.lineage,native.current_ordinal()+1]).hash() & 0x7fffffff

static func _current_image(current: PackedByteArray, ordinal: int) -> Dictionary:
	if current.is_empty() or current.size()>MAX_PAGE: return {}
	var value: Variant=_owner()._bytes_value(current)
	if not value is Dictionary or not value.get("ordinary_state") is Dictionary \
			or not value.get("ordinary") is Dictionary or value.ordinary_state.get("ordinal")!=ordinal: return {}
	return value

static func _advance_closed_page(native: RefCounted, raw: PackedByteArray, h: Dictionary, i: int) -> bool:
	if _hash(raw)!=h.pages[i]: return false
	var page: Dictionary=_page(raw,h,i+1)
	return not page.is_empty() and native.current_ordinal()==i+1 \
		and Core._apply_closed(native,page.image,page.close) \
		and native.propose_next(_default_next_seed(native)) and native.commit_next()

static func _restore_current(native: RefCounted, value: Dictionary) -> RefCounted:
	var resumed: SeasonState=Core._replay(native.session,value.ordinary)
	if resumed==null: return null
	native.session=resumed
	return _complete_current(native,value)

static func _complete_current(native: RefCounted,value: Dictionary) -> RefCounted:
	var phase: Dictionary=value.ordinary_state
	if phase.get("status")=="closed":
		if not Core._apply_closed(native,value.ordinary,phase.get("last_close"),true): return null
		var proposal: Variant=phase.get("proposal")
		if not proposal is Dictionary or (not proposal.is_empty() and not native.propose_next(_default_next_seed(native))):
			return null
	var rebuilt: Dictionary=_owner().pack_current(native,value.ordinary)
	return native if not rebuilt.is_empty() and C.same(rebuilt,value) else null

# Caller provides an ordered vector of immutable page bytes, not paths/scripts/
# callbacks. Only one decoded history page is needed at a time. The individual
# native file size guard applies to every page, never to all 1024 as one JSON.
static func restore_pages(head: PackedByteArray, current: PackedByteArray, pages: Array[PackedByteArray]) -> RefCounted:
	var h: Dictionary=_head(head)
	if h.is_empty() or pages.size()!=h.pages.size() or current.is_empty() or current.size()>MAX_PAGE: return null
	for raw: PackedByteArray in pages:
		if raw.is_empty() or raw.size()>MAX_PAGE: return null
	var value: Dictionary=_current_image(current,int(h.ordinal))
	if value.is_empty(): return null
	var native: RefCounted=Core.root(int(h.seed),h.league)
	if native==null: return null
	for i: int in pages.size():
		if not _advance_closed_page(native,pages[i],h,i): return null
	return _restore_current(native,value)

static func _detached_birth(native: RefCounted) -> RefCounted:
	if native.session==null or native.session.phase!=SeasonState.Phase.DRAFT or not native.session.picks.is_empty():
		return null
	var branch: RefCounted=native.fork()
	if branch==null or branch==native or branch.session==native.session or is_same(branch.common,native.common) \
			or is_same(branch.ordinary_state,native.ordinary_state): return null
	var source: SeasonState=native.session
	var fork: SeasonState=branch.session
	if fork.ownership==source.ownership or fork.build==source.build or fork.build._bank==source.build._bank \
			or fork.build._book==source.build._book or fork.career==source.career or fork.opponents==source.opponents \
			or fork.physical==source.physical or not fork.opponents.clubs.is_empty() \
			or not source.opponents.clubs.is_empty(): return null
	return branch

# Used only after an exact full native replay (active or closed). Ordinary
# fork_live deep-copies primitives; ensure every native mutable build/ledger
# (including drafted CPU clubs) is actually distinct on this closed branch.
static func _detached_populated(native: RefCounted) -> RefCounted:
	var other: RefCounted=native.fork()
	if other==null or other==native or other.session==null or other.session==native.session \
			or is_same(other.common,native.common) or is_same(other.ordinary_state,native.ordinary_state): return null
	var source: SeasonState=native.session
	var fork: SeasonState=other.session
	if fork.ownership==source.ownership or fork.career==source.career or fork.opponents==source.opponents \
			or fork.physical==source.physical or not _distinct_build(fork.build,source.build) \
			or fork.opponents.clubs.size()!=source.opponents.clubs.size(): return null
	for id: String in source.opponents.clubs:
		var club: Dictionary=source.opponents.clubs[id]
		var copied: Dictionary=fork.opponents.clubs.get(id,{})
		if copied.is_empty() or is_same(club,copied) or not _distinct_build(copied.get("build"),club.get("build")):
			return null
	return other

static func _distinct_build(a: SeasonBuild,b: SeasonBuild) -> bool:
	return a!=null and b!=null and a!=b and a._bank!=b._bank and a._book!=b._book

# A differing-head prepared sibling can carry the identical final archived
# image and Close receipt. Only byte-equal canonical full images qualify;
# the verified native Close result is used once, not a saved result/token.
static func _exact_closed_sibling(page: Dictionary,value: Dictionary) -> bool:
	if page.is_empty(): return false
	var phase: Dictionary=value.ordinary_state
	var close: Variant=phase.get("last_close")
	var proposal: Variant=phase.get("proposal")
	return phase.get("status")=="closed" and close is Dictionary and proposal is Dictionary \
		and not proposal.is_empty() and _wire(value.ordinary)==_wire(page.image) and _wire(close)==_wire(page.close)

static func _restore_closed_sibling(native: RefCounted,value: Dictionary) -> RefCounted:
	if native.lifecycle_status()!="closed" or not native.propose_next(_default_next_seed(native)): return null
	var rebuilt: Dictionary=_owner().pack_current(native,value.ordinary)
	return native if not rebuilt.is_empty() and C.same(rebuilt,value) else null

# Same authored birth, same complete ordinary image; only the deterministic
# lifecycle tail differs. This information is merely a preflight. No wire
# status, Close receipt or proposed seed is accepted without the native tail.
static func _same_completed_ordinary(value: Dictionary,previous: Dictionary) -> bool:
	var primary: Dictionary=value.ordinary_state
	var sibling: Dictionary=previous.ordinary_state
	var close: Variant=primary.get("last_close")
	var proposal: Variant=primary.get("proposal")
	if primary.get("status")!="closed" or not close is Dictionary or close.get("reason")!="complete" \
			or not proposal is Dictionary or not C.same(sibling.get("proposal"),{}): return false
	if proposal.is_empty():
		if sibling.get("status")!="active" or not C.same(sibling.get("last_close"),{}): return false
	elif sibling.get("status")!="closed" or not sibling.get("last_close") is Dictionary \
			or _wire(close)!=_wire(sibling.last_close): return false
	if not value.ordinary.get("state") is Dictionary or value.ordinary.state.get("phase")!=SeasonState.Phase.COMPLETE:
		return false
	return _wire(value.ordinary)==_wire(previous.ordinary)

# Only Store's two exact same-head completed generation pairs use this
# entry. It performs the full native authored replay once, then each
# independently detached owner runs its original Close / Prepare / pack tail.
static func restore_completed_related_pages(head: PackedByteArray,current: PackedByteArray,pages: Array[PackedByteArray],
		sibling_current: PackedByteArray) -> Dictionary:
	var h: Dictionary=_head(head)
	if h.is_empty() or pages.size()!=h.pages.size() or current.is_empty() or current.size()>MAX_PAGE \
			or sibling_current.is_empty() or sibling_current.size()>MAX_PAGE: return {}
	for raw: PackedByteArray in pages:
		if raw.is_empty() or raw.size()>MAX_PAGE: return {}
	var value: Dictionary=_current_image(current,int(h.ordinal))
	var previous: Dictionary=_current_image(sibling_current,int(h.ordinal))
	if value.is_empty() or previous.is_empty() or not _same_completed_ordinary(value,previous): return {}
	var active: RefCounted=Core.root(int(h.seed),h.league)
	if active==null: return {}
	for i: int in pages.size():
		if not _advance_closed_page(active,pages[i],h,i): return {}
	var resumed: SeasonState=Core._replay(active.session,value.ordinary)
	if resumed==null or resumed.phase!=SeasonState.Phase.COMPLETE or active.lifecycle_status()!="active": return {}
	active.session=resumed
	var detached: RefCounted=_detached_populated(active)
	if detached==null: return {}
	var guard: Script=load("res://src/gameplay/season/successor_dynasty_owner_v1/raw_guard.gd")
	var active_snapshot: Dictionary=guard.capture(active)
	if active_snapshot.is_empty() or not guard.matches(detached,active_snapshot): return {}
	var primary: RefCounted=_complete_current(active,value)
	if primary==null or not guard.matches(detached,active_snapshot): return {}
	var primary_snapshot: Dictionary=guard.capture(primary)
	if primary_snapshot.is_empty(): return {}
	var sibling: RefCounted=_complete_current(detached,previous)
	if sibling==null or not guard.matches(primary,primary_snapshot): return {}
	return {"primary":primary,"sibling":sibling}

# One synchronous related proof, no token or cache. Authenticate the common
# physical history once, then replay each distinct current from its own birth.
# The only differing-head case is appending one exact next page.
static func restore_related_pages(head: PackedByteArray, current: PackedByteArray, pages: Array[PackedByteArray],
		sibling_head: PackedByteArray,sibling_current: PackedByteArray) -> Dictionary:
	var h: Dictionary=_head(head)
	var sibling: Dictionary=_head(sibling_head)
	if h.is_empty() or sibling.is_empty() or pages.size()!=h.pages.size() \
			or current.is_empty() or current.size()>MAX_PAGE or sibling_current.is_empty() \
			or sibling_current.size()>MAX_PAGE: return {}
	for raw: PackedByteArray in pages:
		if raw.is_empty() or raw.size()>MAX_PAGE: return {}
	var depth: int=sibling.pages.size()
	if head!=sibling_head and (depth+1!=pages.size() or append_head(sibling_head,pages.back())!=head): return {}
	var value: Dictionary=_current_image(current,int(h.ordinal))
	var sibling_value: Dictionary=_current_image(sibling_current,int(sibling.ordinal))
	if value.is_empty() or sibling_value.is_empty(): return {}
	var shared_close: Dictionary={}
	if head!=sibling_head and _hash(pages.back())==h.pages.back():
		var page: Dictionary=_page(pages.back(),h,pages.size())
		if _exact_closed_sibling(page,sibling_value): shared_close=page
	var native: RefCounted=Core.root(int(h.seed),h.league)
	if native==null: return {}
	var guard: Script=load("res://src/gameplay/season/successor_dynasty_owner_v1/raw_guard.gd")
	var branch: RefCounted
	var branch_before: Dictionary={}
	for i: int in range(pages.size()+1):
		if i==depth and not shared_close.is_empty():
			if native.current_ordinal()!=i+1 or not Core._apply_closed(native,shared_close.image,shared_close.close): return {}
			branch=_detached_populated(native)
			if branch==null: return {}
			branch_before=guard.capture(native)
			if branch_before.is_empty() or not guard.matches(branch,branch_before) \
					or not native.propose_next(_default_next_seed(native)) or not native.commit_next(): return {}
			continue
		if i==depth:
			branch=_detached_birth(native)
			if branch==null: return {}
			branch_before=guard.capture(native)
			if branch_before.is_empty() or not guard.matches(branch,branch_before): return {}
		if i<pages.size() and not _advance_closed_page(native,pages[i],h,i): return {}
	var primary: RefCounted=_restore_current(native,value)
	if primary==null or branch==null or not guard.matches(branch,branch_before): return {}
	var primary_before: Dictionary=guard.capture(primary)
	if primary_before.is_empty(): return {}
	var previous: RefCounted=_restore_current(branch,sibling_value) if shared_close.is_empty() \
		else _restore_closed_sibling(branch,sibling_value)
	if previous==null or not guard.matches(primary,primary_before): return {}
	return {"primary":primary,"sibling":previous}
