# A display-only memo for one admitted archive. Never consulted by a save,
# restore, league gate or the pending-result projection.
static var _previous_standard_pin: Dictionary = {}
static var _previous_standard_view: Dictionary = {}
static var previous_standard_projection_calls: int = 0

static func previous_standard_available(owner: RefCounted, expected: PackedByteArray) -> bool:
	if saving or _ticket!=null or owner==null or owner!=_owner or owner.session!=_owner_session \
		or expected.is_empty() or expected!=_owner_bytes:
		return false
	var pin: Variant = _metadata.get("ordinary_state",{}).get("archive")
	return pin is Dictionary and RawGuard.same(owner.ordinary_state.archive,pin) \
		and ((Owner.Common.StandardArchive.marked(owner.session) and Owner.Common.StandardArchive.shape(owner)) \
			or (LeagueAdmission.starter_player() and Owner.Common.starter_dugout_archive(owner)))

static func previous_standard_season(owner: RefCounted, expected: PackedByteArray) -> Dictionary:
	if not previous_standard_available(owner,expected) or _starter_slot_linked() \
		or not _write_revision_matches(expected,true): return {}
	var pin: Dictionary = _metadata.ordinary_state.archive
	if not RawGuard.same(pin,_previous_standard_pin):
		previous_standard_projection_calls += 1
		var view: Dictionary = Owner.Common.StandardArchive._display_scores(Owner._bytes_value(pin.source.to_utf8_buffer()),true) \
			if pin.get("kind")==Owner.Common.STARTER_DUGOUT_ARCHIVE else Owner.Common.StandardArchive.display(pin)
		if not previous_standard_available(owner,expected) or not _write_revision_matches(expected,true): return {}
		_previous_standard_pin = pin.duplicate(true)
		_previous_standard_view = view
	return _previous_standard_view.duplicate(true)

static func standard_history_available(owner: RefCounted, expected: PackedByteArray) -> bool:
	if saving or _ticket!=null or owner==null or owner!=_owner or owner.session!=_owner_session \
		or expected.is_empty() or expected!=_owner_bytes:
		return false
	var pin: Variant = _metadata.get("ordinary_state",{}).get("archive")

		and ((Owner.Common.StandardChain.marked(owner.session) and Owner.Common.StandardChain.shape(owner)) \

				or Owner.Mixed.season_owner_scope(owner) or Owner.Mixed.closed_complete_scope(owner) \
				or (private_mixed_append() and (Owner.MixedAppend.draft_scope(owner) or Owner.MixedAppend.season_scope(owner) \
					or Owner.MixedAppend.closed_complete_scope(owner) or Owner.MixedAppend.rollover_play_scope(owner) \
					or Owner.MixedAppend.rollover_closed_complete_scope(owner) \
					or Owner.MixedAppend.seven_opening_scope(owner))))))

static func standard_history(owner: RefCounted, expected: PackedByteArray) -> Array:
	if not standard_history_available(owner,expected) or _starter_slot_linked() \
		or not _write_revision_matches(expected,true): return []
	var pin: Dictionary = _metadata.ordinary_state.archive
	if not RawGuard.same(pin,_standard_history_pin):
		standard_history_projection_calls += 1
		var mixed: bool=Owner.Mixed.marked_state(owner.ordinary_state)
		var appended: bool=Owner.MixedAppend.marked_state(owner.ordinary_state)
		var view: Array = Owner.MixedAppend.display(owner) if appended else (Owner.Mixed.display(pin) if mixed else (
			_starter_dugout_history(pin) if pin.get("kind")==Owner.Common.STARTER_DUGOUT_CHAIN
			else Owner.Common.StandardChain.display(pin)))
		var expected_count: int=Owner.MixedAppend.history_count(owner) if appended else (3 if mixed else 2)
		if view.size()!=expected_count or not standard_history_available(owner,expected) \
			or not RawGuard.same(pin,_metadata.ordinary_state.archive) or not _write_revision_matches(expected,true):
			return []
		_standard_history_pin = pin.duplicate(true)
		_standard_history_view = view
	return _standard_history_view.duplicate(true)

# Display only: the two exact members were natively admitted on cold
# Continue or a fixed Store promotion. No write path consumes these views.
static func _starter_dugout_history(pin: Dictionary) -> Array:
	var second:=Owner.Common.starter_dugout_second_bytes(pin)
	if second.is_empty(): return []
	var first:=Owner.Common.StandardArchive._display_scores(Owner._bytes_value(pin.entries[0].source.to_utf8_buffer()),true)
	var latest:=Owner.Common.StandardArchive._display_scores(Owner._bytes_value(second),true)
	if first.get("ordinal")!=1 or latest.get("ordinal")!=2: return []
	first["league"]="standard"
	latest["league"]="dugout"
	return [first,latest]

static func slot_path() -> String:

# A fixed header and digest bind the native bytes to this profile. This
# detects corruption and rejects foreign dialects; it is not anti-cheat.
static func _installed_header() -> String:
	return INSTALLED_FORWARD_HEADER if OS.has_feature(LeagueAdmission.CAREER_FORWARD_FEATURE) else INSTALLED_HEADER

static func _installed_payload(wire: PackedByteArray) -> PackedByteArray:
	var header: String=_installed_header()
	var start: int=header.length()+65
	if wire.size()<=start or wire.size()>MAX_BYTES \
		or wire.slice(0,header.length()).get_string_from_ascii()!=header \
		or wire[start-1]!=10: return PackedByteArray()
	var body:=wire.slice(start)
	return body if wire.slice(header.length(),start-1).get_string_from_ascii()==Owner._hash(body) else PackedByteArray()

static func _installed_wire(body: PackedByteArray) -> PackedByteArray:
	var header: String=_installed_header()
	if body.is_empty() or body.size()+header.length()+65>MAX_BYTES: return PackedByteArray()
	var digest: String=Owner._hash(body)
	if digest.length()!=64: return PackedByteArray()
	return (header+digest+"\n").to_utf8_buffer()+body

static func _installed_paths_blocked() -> bool:
	var path: String=OS.get_user_data_dir().trim_suffix("/")
	while path!="" and path!="/":
		var parent: String=path.get_base_dir()
		var directory:=DirAccess.open(parent)
		if parent==path or directory==null or directory.is_link(path.get_file()): return true
		path=parent
	var dir:=DirAccess.open("user://")
	if dir==null: return true
	for suffix: String in ["", ".bak", ".tmp", INSTALLED_BACKUP_STAGE]:
		var member: String=INSTALLED_PATH+suffix
		if dir.is_link(member.get_file()) or DirAccess.dir_exists_absolute(member): return true
	return false

static func _starter_slot_linked() -> bool:
	if LeagueAdmission.installed_career_feature():
		return not LeagueAdmission.installed_career_owned() or _installed_paths_blocked()
	if not LeagueAdmission.starter_feature(): return false
	var root: String=OS.get_user_data_dir().trim_suffix("/")
	var parent:=DirAccess.open(root.get_base_dir())
	if parent==null or parent.is_link(root.get_file()): return true
	var dir:=DirAccess.open("user://")
	if dir==null: return true
	for suffix: String in ["", ".bak", ".tmp"]:
		if dir.is_link((slot_path()+suffix).get_file()): return true
	return false

static func raw() -> PackedByteArray:

# Bounded prefix probe; unsupported family versions remain protected.
static func standard_d1_slot_kind() -> String:
	if LeagueAdmission.installed_career_feature(): return ""
	if _starter_slot_linked(): return ""
	var contract=preload("standard_d1_transition.gd")
	var file:=FileAccess.open(slot_path(),FileAccess.READ)
	if file==null: return ""
	var prefix: String=file.get_buffer(contract.SLOT_HEADER.length()).get_string_from_ascii()
	var size:=file.get_length()
	file.close()
	if not prefix.begins_with(contract.SLOT_FAMILY): return ""
	return "supported" if prefix==contract.SLOT_HEADER and size<=contract.MAX_SLOT else "unsupported"

static func _starter_owner_allowed(owner: RefCounted) -> bool:
	if LeagueAdmission.installed_career_feature():
		return LeagueAdmission.installed_career_owned() and installed_owner_shape(owner)
	if owner!=null and Owner.MixedAppend.marked_state(owner.ordinary_state):
		return private_mixed_append() and (Owner.MixedAppend.draft_scope(owner) or Owner.MixedAppend.season_scope(owner) \
			or Owner.MixedAppend.closed_complete_scope(owner) or Owner.MixedAppend.rollover_play_scope(owner) \
			or Owner.MixedAppend.rollover_closed_complete_scope(owner) or Owner.MixedAppend.seven_opening_scope(owner))
	if Flat.is_flat(owner): return false
	if owner!=null and Owner.Mixed.marked_state(owner.ordinary_state):
		return LeagueAdmission.starter_player() and (Owner.Mixed.season_owner_scope(owner) or Owner.Mixed.closed_complete_scope(owner))
	if owner!=null and owner.ordinary_state.get("archive",{}).get("kind")==Owner.Common.STARTER_DUGOUT_ARCHIVE:
		return LeagueAdmission.starter_player() and Owner.Common.starter_dugout_archive(owner) and Owner.Common._archive_shape(owner)
	if owner!=null and owner.ordinary_state.get("archive",{}).get("kind")==Owner.Common.STARTER_DUGOUT_CHAIN:
		return LeagueAdmission.starter_player() and Owner.Common.starter_dugout_chain_scope(owner)
	if owner!=null and Owner.Common.StandardWorkshop.Home.opening_chain(owner.session):
		return (LeagueAdmission.starter_player() or private_standard_chain_opening()) and _starter_opening_owner(owner)
	if not LeagueAdmission.starter_feature(): return true
	if not LeagueAdmission.starter_player() or owner==null or owner.get_script()!=Owner \
		or owner.active!="ordinary" or owner.session==null or not owner.paused.is_empty() \
		or not Owner._missing_dynasty(owner) or not owner.migration_v3.is_empty() \
		or owner.ordinary_state.has(Owner.Common.StandardWorkshop.APPLIED) \
		or (owner.ordinary_state.has("archive") and not Owner.Common.StandardArchive.shape(owner) \
			and not Owner.Common.StandardChain.shape(owner)) \
		or Owner.Common.cross_proposal_kind(owner.ordinary_state.get("proposal",{}).get("kind")): return false
	var current: SeasonState=owner.session
	var workshop=Owner.Common.StandardWorkshop
	if owner.ordinary_state.has(workshop.OPENING) or not current.home_property.is_empty():
		return _starter_opening_owner(owner)
	# Standard Base's authored envelope has no named league rules. This is
	# not a fallback for missing fields; the strict native codec took priority.
	if current.league_rules.is_empty():
		return not current.league_live and current.career!=null \
			and current.career.start_basis.get("kind")!="common-standard-d1-transition-v1"
	return current.league_live and current.league_rules.get("league")=="dugout" \
		and current.league_rules.get("tier",-1)==0

static func _starter_opening_owner(owner: RefCounted) -> bool:
	if owner==null: return false
	var w = Owner.Common.StandardWorkshop
	var scoped: bool=Owner.Common.StandardChain.shape(owner) if w.Home.opening_chain(owner.session) \
		else Owner.Common.StandardArchive.shape(owner)
	return scoped and w.player_opening_allowed(owner)

# The strict worker authenticates these fields before the parent sees
# the seal. JSON has erased the distinction between integer and whole float;
# rebuild the same explicitly authored metadata numerics as the native birth
# so the first normal write matches a cold native owner byte for byte.
static func _mixed_continue_authored_metadata(owner: RefCounted) -> bool:
	var state: Dictionary=owner.ordinary_state
	var transition: Variant=state.get("last_transition")
	if not transition is Dictionary or not transition.get("basis") is Dictionary: return false
	var basis: Dictionary=transition.basis
	if not Owner.Common.Start.valid(basis) or not Owner.Common.Start.detached_mixed_flat(basis) \
			or not basis.get("pennants") is Dictionary: return false
	var pennants: Dictionary=basis.pennants
	for row: Array in [[state,["ordinal"]],[transition,["from","to"]],
			[basis,["common_sequence","ordinal","seed","version"]],
			[pennants,["from_season","mastery_from","version"]]]:
		var target: Dictionary=row[0]
		for field: String in row[1]:
			if not Codec.whole(target.get(field),0,2147483647): return false
			target[field]=int(target[field])
	return true

static func accept_mixed_continue(expected: PackedByteArray,child: Node) -> RefCounted:
	if _rollover_close_unverified or not _rollover_recovery_expected.is_empty(): return null
	var process_script: Script=load("res://src/gameplay/season/successor_dynasty_owner_v1/starter_mixed_continue.gd")
	if not LeagueAdmission.starter_fast_worker() or saving or _owner!=null or _ticket!=null \
			or child==null or child.get_script()!=process_script or expected.is_empty() \
			or expected.size()>MAX_BYTES or _starter_slot_linked() or not _write_revision_matches(expected,true): return null
	var revision:=Owner._hash(expected)
	var seal: Dictionary=child.take_current_seal(revision)
	if seal.is_empty(): return null
	var valid_binding: Callable = func() -> bool:
		return is_instance_valid(child) and child.binding_live(seal.nonce,revision) \
			and not saving and _owner==null and _ticket==null
	if not valid_binding.call(): return null
	var tree:=child.get_tree()
	child.binding_stage="revision"
	var payload: Variant=Owner._bytes_value(expected)
	if not _mixed_continue_shape(payload) or process_script.current_digest(payload.ordinary)!=seal.current: return null
	await tree.process_frame
	if not valid_binding.call(): return null
	child.binding_stage="metadata"
	var owner:=Owner._restore_metadata(payload)
	if owner==null or not _mixed_continue_authored_metadata(owner): return null
	await tree.process_frame
	if not valid_binding.call(): return null
	child.binding_stage="typed"
	owner.session=await Adapter.Ordinary._hydrate_mixed_continue(payload.ordinary,child,valid_binding)
	if owner.session==null or not valid_binding.call() \
			or not Owner.Mixed.season_owner_scope(owner): return null
	child.binding_stage="native bank representation"
	if not process_script.apply_bank_witness(owner.session,seal.banks) or not valid_binding.call(): return null
	child.binding_stage="native bracket representation"
	if not process_script.apply_bracket_witness(owner.session,seal.bracket) or not valid_binding.call(): return null
	await tree.process_frame
	if not valid_binding.call(): return null
	child.binding_stage="capture"
	var current: Dictionary=await Adapter.Ordinary._recapture_mixed_continue(owner.session,child,valid_binding)
	if current.is_empty() or not valid_binding.call(): return null
	child.binding_stage="compare"
	var rebuilt:=Owner._pack_fields(owner,current)
	if not Codec.keys(payload,rebuilt.keys()) or not Codec.keys(payload.ordinary,current.keys()): return null
	for field: String in rebuilt:
		if field!="ordinary" and not Codec.same(rebuilt[field],payload[field]): return null
	for field: String in current:
		await tree.process_frame
		if not valid_binding.call() \
				or not Codec.same(current[field],payload.ordinary[field]): return null
	await tree.process_frame
	if not valid_binding.call(): return null
	child.binding_stage="raw guard"
	var snapshot: Dictionary=await RawGuard.capture_mixed_continue(owner,child,valid_binding)
	if not valid_binding.call(): return null
	if snapshot.is_empty() or saving or _owner!=null or _ticket!=null or _starter_slot_linked() \
			or not _write_revision_matches(expected,true) or not child.binding_live(seal.nonce,revision,true): return null
	_invalidate()
	_owner=owner; _owner_session=owner.session; _owner_raw=snapshot; _owner_bytes=expected.duplicate()
	_paused=owner.paused.duplicate(true); _active=owner.active
	_metadata=Codec.data_fields(owner,Owner.META_FIELDS); _imported_sha256=owner.imported_sha256
	return owner

static func _read_common_slot(flat: bool, recovery_expected: PackedByteArray=PackedByteArray()) -> RefCounted:
	if (_rollover_close_unverified or not _rollover_recovery_expected.is_empty()) and recovery_expected.is_empty():
		last_error="Use Continue to verify the replacement season save."
		return null
	if not recovery_expected.is_empty() and (flat or _rollover_close_unverified \
		or recovery_expected!=_rollover_recovery_expected): return null
	if flat and (saving or _ticket!=null or not private_standard_chain_opening()): return null
	forget_projection()
	last_error = ""
	if _starter_slot_linked():
		last_error="This Starter Beta slot points outside its own files. No files were changed."
		return null
	if not standard_d1_slot_kind().is_empty():
		last_error="This save is outside the Starter Beta slot. No files were changed." if LeagueAdmission.starter_feature() else \
			"This Standard D1 slot requires private Continue. No files were changed."
		return null
	if not FileAccess.file_exists(slot_path()):
		last_error = "No common career is saved in this slot. Earlier saves are unchanged."
		return null
	var file: FileAccess = FileAccess.open(slot_path(),FileAccess.READ)
	if file == null or file.get_length() > MAX_BYTES:
		last_error = "The save could not be read safely. It has not been changed."
		return null
	var bytes: PackedByteArray = file.get_buffer(file.get_length())
	file.close()
	if LeagueAdmission.installed_career_feature():
		if FileAccess.file_exists(INSTALLED_PATH+".tmp") or FileAccess.file_exists(INSTALLED_PATH+INSTALLED_BACKUP_STAGE):
			last_error="This career has an unfinished save. No files were changed."
			return null
		bytes=_installed_payload(bytes)
		if bytes.is_empty():
			last_error="This file is not a verified installed career. No files were changed."
			return null
	if not recovery_expected.is_empty() and bytes!=recovery_expected:
		last_error="The replacement save does not match the pending Close. No files were changed."
		return null
	if _base_archive_family(bytes):
		last_error="This archive save requires its isolated qualification reader. No files were changed."
		return null
	var parser: JSON = JSON.new()
	if parser.parse(bytes.get_string_from_utf8()) != OK:
		last_error = "The save is unreadable. Original and backup are preserved."
		return null
	if parser.data is Dictionary and Owner.MixedAppend.marked_state(parser.data.get("ordinary_state")) \
			and not private_mixed_append():
		last_error="This test career requires its isolated qualification build. No files were changed."
		return null
	var season: RefCounted = Flat._restore_private(parser.data) if flat else Owner.restore(parser.data)
	if season == null:
		last_error = "This common career has an unsupported version or inconsistent state. No files were changed."
		return null
	if not flat and not _starter_owner_allowed(season):
		last_error="This save is outside the Starter Beta slot. No files were changed."
		return null
	if not recovery_expected.is_empty() and not Owner.MixedAppend.rollover_closed_complete_scope(season):
		last_error="The replacement does not verify as a closed Season 6. No files were changed."
		return null
	var snapshot: Dictionary = RawGuard.capture(season)
	if snapshot.is_empty() or raw() != bytes:
		last_error = "The save changed during read or could not be admitted. No files were changed."
		return null
	_invalidate()
	_owner = season; _owner_session = season.session; _owner_raw = snapshot; _owner_bytes = bytes.duplicate()
	_paused = season.paused.duplicate(true); _active = season.active
	_metadata = Codec.data_fields(season,Owner.META_FIELDS); _imported_sha256 = season.imported_sha256
	return season

static func save(season: RefCounted, expected: PackedByteArray, expected_snapshot: Dictionary = {}) -> bool:
	forget_projection()
	if saving: return false
	saving = true
	var result: bool = _write(season,expected,expected_snapshot)
	saving = false
	if result: _invalidate()
	return result
static func _matches(expected: PackedByteArray) -> bool:
	if _base_archive_family(expected) or _starter_slot_linked(): return false
	# Generic read/admission/write/retry paths never own the D1 dialect.
	if not standard_d1_slot_kind().is_empty(): return false
	if LeagueAdmission.installed_career_feature() and FileAccess.file_exists(INSTALLED_PATH+INSTALLED_BACKUP_STAGE): return false
	return not DirAccess.dir_exists_absolute(slot_path()) and ((expected.is_empty() and not FileAccess.file_exists(slot_path())) or (not expected.is_empty() and FileAccess.file_exists(slot_path()) and raw() == expected))
static func _write(season: RefCounted, expected: PackedByteArray, expected_snapshot: Dictionary) -> bool:
	# Legacy unbound writer never modifies the private paired slot.
	if season==null or season.get_script()!=Owner or _paired_owner(season) or _paired_settled(_owner): return false
	if not _starter_owner_allowed(season) or _rollover_close_unverified or not _rollover_recovery_expected.is_empty(): return false
	last_error = "The exact candidate is retained. Resolve the save obstruction, then retry."
	if not _matches(expected):
		last_error = "The saved file changed. It was not overwritten; your candidate remains retained."
		return false
	for suffix: String in ["", ".bak", ".tmp"]:
		if DirAccess.dir_exists_absolute(slot_path()+suffix): return false
	if FileAccess.file_exists(slot_path()+".tmp") or (expected.is_empty() and FileAccess.file_exists(slot_path()+".bak")): return false
	# Unbound legacy writes cannot create, replace, or erase workshop metadata.
	var previous: Variant=Owner._bytes_value(expected)
	# The unbound writer cannot erase or manufacture a typed career proposal.
	if season.ordinary_state.has("archive") or Owner.Common.cross_proposal_kind(season.ordinary_state.get("proposal",{}).get("kind")): return false
	if previous is Dictionary and previous.get("ordinary_state") is Dictionary:
		var prior_proposal: Variant=previous.ordinary_state.get("proposal")
		if previous.ordinary_state.has("archive") or (prior_proposal is Dictionary and Owner.Common.cross_proposal_kind(prior_proposal.get("kind"))): return false
	if not season.workshop_draft.is_empty() or not season.workshop_close_history.is_empty() or not season.workshop_chair_purchase.is_empty() or not season.workshop_applied.is_empty(): return false
	if previous is Dictionary and (previous.has("workshop_draft") or previous.has("workshop_close_history") or previous.has("workshop_chair_purchase") or previous.has("workshop_applied")): return false
	var data: Dictionary = Owner.capture(season)
	if data.is_empty() or (not expected_snapshot.is_empty() and not Codec.same(data,expected_snapshot)):
		last_error = "This candidate is inconsistent with the materialized save contract."
		return false
	return _write_data(data,expected)

static func _native_encoded(data: Dictionary) -> PackedByteArray:
	var bytes: PackedByteArray = JSON.stringify(data,"",true,true).to_utf8_buffer()
	if bytes.is_empty() or bytes.size() > MAX_BYTES: return PackedByteArray()
	# Validate the encoded representation against the consistent native value.
	var parser: JSON = JSON.new()
	if parser.parse(bytes.get_string_from_utf8()) != OK or not Codec.same(parser.data,data):
		return PackedByteArray()
	return bytes

static func _write_data(data: Dictionary, expected: PackedByteArray) -> bool:
	var bytes := _native_encoded(data)
	return not bytes.is_empty() and _write_exact_bytes(bytes,expected)

static func _write_exact_bytes(bytes: PackedByteArray, expected: PackedByteArray) -> bool:
	return _write_exact_result(bytes,expected,false)=="saved"

static func _write_exact_result(bytes: PackedByteArray, expected: PackedByteArray, bounded: bool) -> String:
	if OS.has_feature(BASE_ARCHIVE_READER_FEATURE): return "not_renamed"
	var installed: bool=LeagueAdmission.installed_career_feature()
	if installed and not _matches(expected): return "not_renamed"
	var wire:=_installed_wire(bytes) if installed else bytes
	if wire.is_empty(): return "not_renamed"
	var file: FileAccess = FileAccess.open(slot_path()+".tmp",FileAccess.WRITE)
	if file == null: return "not_renamed"
	file.store_buffer(wire); file.flush()
	var ok: bool = file.get_error() == OK
	file.close()
	if not ok or FileAccess.get_file_as_bytes(slot_path()+".tmp") != wire or not _write_revision_matches(expected,bounded): return "not_renamed"
	if not expected.is_empty():
		if installed:
			# Replace the backup directory entry, never write through an incumbent
			# backup hardlink. Its payload is the exactly matched prior revision.
			var stage: String=INSTALLED_PATH+INSTALLED_BACKUP_STAGE
			var prior_wire:=_installed_wire(expected)
			var backup:=FileAccess.open(stage,FileAccess.WRITE)
			if backup==null or prior_wire.is_empty(): return "not_renamed"
			backup.store_buffer(prior_wire); backup.flush()
			var staged: bool=backup.get_error()==OK
			backup.close()
			if not staged or FileAccess.get_file_as_bytes(stage)!=prior_wire or DirAccess.rename_absolute(stage,INSTALLED_PATH+".bak")!=OK: return "not_renamed"
		elif DirAccess.copy_absolute(slot_path(),slot_path()+".bak")!=OK: return "not_renamed"
	if not _write_revision_matches(expected,bounded): return "not_renamed"
	if DirAccess.rename_absolute(slot_path()+".tmp",slot_path()) != OK: return "not_renamed"
	var readback_ok: bool=_write_revision_matches(bytes,true) if bounded else raw()==bytes
	if not readback_ok: return "renamed_unverified"
	last_error = ""
	return "saved"

# The bounded cross path never reads a replacement of arbitrary length and
# refuses linked slot objects. The stock write protocol is otherwise unchanged.
static func _write_revision_matches(bytes: PackedByteArray,bounded: bool) -> bool:
	if not bounded or LeagueAdmission.installed_career_feature(): return _matches(bytes)
	var dir:=DirAccess.open("user://")
	if dir==null or dir.is_link(slot_path().get_file()): return false
	var f:=FileAccess.open(slot_path(),FileAccess.READ)
	if f==null or f.get_length()!=bytes.size(): return false
	var same: bool=f.get_buffer(bytes.size())==bytes and f.get_length()==bytes.size()
	f.close()
	return same

static func _matches_raw(value: RefCounted, snapshot: Dictionary) -> bool:

# Pure refusal predicate, never a writer or an admission token. Routine actions
# cannot replace either session or provenance through the generic candidate API.
static func _routine_pins(candidate: RefCounted) -> bool:
	if candidate.active!=_active or not RawGuard.same(candidate.paused,_paused) or candidate.imported_sha256!=_imported_sha256: return false
	if not RawGuard.same(candidate.migration_v2,_metadata.get("migration_v2")): return false
	for name: String in ["common","ordinary_state","migration_v3","workshop_draft","workshop_close_history","workshop_chair_purchase","workshop_applied"]:
		if not RawGuard.same(candidate.get(name),_metadata.get(name)): return false
	if _active=="ordinary": return RawGuard.same(Codec.data_fields(candidate,Owner.META_FIELDS),_metadata)
	return candidate.birth_seed==_metadata.get("birth_seed")

static func _paired_owner(value: RefCounted) -> bool:
	return value!=null and value.get_script()==Owner and value.paired_draft_only()

static func _paired_settled(value: RefCounted) -> bool:
	return _paired_owner(value) and value.session.round_index!=0

static func prepare(owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> RefCounted:
	forget_projection()
	if saving or _ticket != null or candidate == null or candidate == owner or _paired_settled(owner): return null
	if candidate.get_script()!=Owner or candidate.session==null or _paired_settled(candidate): return null
	if not _starter_owner_allowed(candidate): return null
	if owner == null:
		if candidate.ordinary_state.has("archive") or Owner.Common.cross_proposal_kind(
				candidate.ordinary_state.get("proposal",{}).get("kind")): return null
		if not expected.is_empty() or not _matches(expected) or not candidate.workshop_draft.is_empty() or not candidate.workshop_close_history.is_empty() or not candidate.workshop_chair_purchase.is_empty() or not candidate.workshop_applied.is_empty(): return null
	else:
		if not match_owner(owner,expected) or candidate.session==owner.session: return null
	# Only this private boundary creates the writer's canonical payload. Callers
	# can receive detached diagnostics, never supply a substitute validated value.
	var data: Dictionary = {}
	if owner != null:
		if not _routine_pins(candidate): return null
		# Dormant alternate was strictly admitted on read/promotion. Exact live raw
		# owner pins were checked above; no caller validation flag or payload.
		if Owner.MixedAppend.marked_state(candidate.ordinary_state):
			data=Owner.MixedAppend.pick_payload(owner,candidate)
		elif Owner.Mixed.marked_state(candidate.ordinary_state):
			data=Owner.Mixed.draft_pick_payload(owner,candidate)
		else:
			var current: Dictionary = Adapter.capture(candidate.active,candidate.session)
			if not current.is_empty():
				data = Owner.pack_current(candidate,current)
	else:
		data = Owner.capture(candidate)
	if data.is_empty(): return null
	var snapshot: Dictionary = RawGuard.capture(candidate)
	if snapshot.is_empty(): return null
	if owner == null:
		_owner = null; _owner_session = null; _owner_raw = {}; _owner_bytes = expected.duplicate()
	_ticket = RefCounted.new()
	_candidate = candidate; _candidate_session = candidate.session; _candidate_raw = snapshot; _candidate_data = data
	_expected = expected.duplicate(); _written = PackedByteArray()
	return _ticket

## One synchronous Store-owned admission/write stack. No token or caller flag
## permits skipping later checks. prepare strictly captures the candidate and
## matches the live owner; no callbacks or external handoff intervene here.
## On obstruction the ordinary private ticket survives, and retry stays strict.
static func prepare_and_write(owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> RefCounted:
	var ticket: RefCounted = prepare(owner,candidate,expected)
	if ticket == null: return null
	_write_admitted()
	return ticket

## Fixed synchronous producer: authenticate before constructing any candidate.
## Only named runtime/build factories admitted by match_owner can execute here.
## No UI callbacks, awaits, caller candidate or unchecked payload are accepted.
static func prepare_shop_and_write(owner: RefCounted, expected: PackedByteArray, command: Dictionary) -> RefCounted:
	forget_projection()
	last_error = ""
	if saving or _ticket != null or not Codec.primitive(command) or not SHOP_PRODUCER_OPS.has(str(command.get("op",""))): return null
	if not match_owner(owner,expected): return null
	var session: SeasonState = owner.session
	var mixed:=mixed_session_scope(owner)
	var appended:=mixed_append_shop_scope(owner)
	if (not owner.playable() and not appended) or (not Adapter.scope(owner.active,session) and not mixed and not appended) \
			or not session.physical.pending.is_empty() or not session.shop_available(): return null
	if (mixed or appended) and not Owner.Mixed.shop_command_allowed(session,command): return null
	var request_id: Variant = command.get("id")
	if request_id is String and not request_id.is_empty() and session.build._requests.has(request_id): return null
	var next_session: SeasonState = session.fork_live()
	if next_session == null: return null
	var next_build: SeasonBuild = next_session.build.candidate(command.duplicate(true))
	if next_build == null:
		last_error = next_session.build.last_error
		return null
	next_session.adopt_build(next_build)
	if next_session.build != next_build: return null
	var envelope: RefCounted = Owner.new()
	envelope.session = next_session
	if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
	# Keep this registration inside the checked producer, rather than exposing
	# an unchecked owner/candidate helper (underscore methods are callable).
	if not _routine_pins(envelope): return null
	var current: Dictionary = {} if mixed or appended else Adapter.capture(envelope.active,next_session)
	var data: Dictionary = Owner.MixedAppend.payload(envelope) if appended else (
		Owner.Mixed.compact_payload(envelope) if mixed else (
			Owner.pack_current(envelope,current) if not current.is_empty() else {}))
	if data.is_empty():
		last_error = "The candidate could not be represented consistently. Saved progress is unchanged."
		return null
	var snapshot: Dictionary = RawGuard.capture(envelope)
	if snapshot.is_empty(): return null
	_ticket = RefCounted.new()
	_candidate = envelope; _candidate_session = next_session; _candidate_raw = snapshot; _candidate_data = data
	_expected = expected.duplicate(); _written = PackedByteArray()
	_write_admitted()
	return _ticket

static func prepare_mixed_round_human(app: Variant) -> RefCounted:

static func _prepare_mixed_round(owner: RefCounted, expected: PackedByteArray, op: String, native_input: Variant=null) -> Dictionary:
	forget_projection()
	if saving or _ticket!=null or not mixed_round_scope(owner) or not match_owner(owner,expected): return {}
	var current: SeasonState=owner.session
	if current.phase in [SeasonState.Phase.DRAFT,SeasonState.Phase.COMPLETE] or current.build.pack_pending(): return {}
	var next: SeasonState
	match op:
		"human":
			if not native_input is Object or not is_instance_valid(native_input) \
					or native_input.get_script()!=load("res://src/presentation/season/successor_dynasty_owner_v1/app.gd") \
					or native_input.career_owner!=owner or native_input.season!=current or native_input.saved_bytes!=expected \
					or not mixed_round_entry(owner): return {}
			var human: Array=Adapter.OrdinaryCapture.capture(native_input)
			if human.size()!=13 or human[0]!=current.pending_fixture().get("id"): return {}
			next=current.fork_live()
			if next==null: return {}
			next.physical.pending={"human":human.duplicate(true),"reports":[]}
			if Adapter.OrdinaryPending.project(next).has("error"): return {}
		"report":
			if not native_input is Dictionary or current.physical.pending.is_empty(): return {}
			next=Adapter.OrdinaryPending.stage_report(current,native_input).get("candidate")
		"settle":
			if native_input!=null or current.physical.pending.is_empty(): return {}
			var projected: Dictionary=Adapter.OrdinaryPending.project(current)
			if projected.has("fixture") or projected.has("error"): return projected
			next=projected.get("candidate")
		_: return {}
	if next==null: return {}
	var envelope: RefCounted=Owner.new()
	envelope.session=next
	if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)) \
			or not _routine_pins(envelope): return {}
	var data: Dictionary=Owner.MixedAppend.payload(envelope) if Owner.MixedAppend.marked_state(owner.ordinary_state) \
		else Owner.Mixed.compact_payload(envelope)
	var snapshot: Dictionary=RawGuard.capture(envelope) if not data.is_empty() else {}
	if snapshot.is_empty() or not match_owner(owner,expected): return {}
	_ticket=RefCounted.new()
	_candidate=envelope;_candidate_session=next;_candidate_raw=snapshot;_candidate_data=data
	_expected=expected.duplicate();_written=PackedByteArray()
	_write_admitted()
	return {"ticket":_ticket}

## Fixed producer invokes the named pending implementation itself. The optional
## terminal result is retained only after this exact immediate checkpoint write.
## A failed write/retry keeps the ordinary ticket, but deliberately loses reuse.
static func prepare_report_and_write(owner: RefCounted, expected: PackedByteArray, report: Dictionary) -> RefCounted:
	if mixed_round_scope(owner): return _prepare_mixed_round(owner,expected,"report",report).get("ticket")
	forget_projection()
	if saving or _ticket != null or not match_owner(owner,expected): return null
	if not owner.playable() or not Adapter.scope(owner.active,owner.session) or owner.session.physical.pending.is_empty(): return null
	var staged: Dictionary = Adapter.pending(owner.active).stage_report(owner.session,report)
	var current: SeasonState = staged.get("candidate")
	if current == null: return null
	var envelope: RefCounted = Owner.new()
	envelope.session = current
	if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
	var settled: RefCounted
	var snapshot: Dictionary = {}
	if staged.get("settled") is SeasonState:
		settled = Owner.new(); settled.session = staged.settled
		if not Codec.install_fields(settled,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
		if not _routine_pins(settled) or not settled.session.physical.pending.is_empty() \
			or not settled.session.physical.needed.is_empty() or settled.session.physical.projecting: return null
		snapshot = RawGuard.capture(settled)
		if snapshot.is_empty(): return null
	var ticket: RefCounted = prepare_and_write(owner,envelope,expected)
	if ticket != null and not _written.is_empty() and settled != null:
		_projection_ticket = ticket; _projection_checkpoint = envelope
		_projection_value = settled; _projection_raw = snapshot
	return ticket

static func _bind_visit6_fresh_profile(build: SeasonBuild) -> bool:
	var id: String=Adapter.Ordinary.VISIT6_RECRUIT

	var profile: Dictionary=build.player(id)
	var original: Dictionary=SeasonPlayerCatalog.profile(id)
	if original.is_empty() or not Adapter.Ordinary.same(profile,original): return false
	var before: String=_canonical_shop_bytes(Adapter.Ordinary.Builds.capture(build))

		profile[field]=original[field].duplicate(true)
	profile.capacity=original.capacity
	if profile!=original: return false
	build._book._players[id]=profile
	var after: Dictionary=Adapter.Ordinary.Builds.capture(build)
	return before==_canonical_shop_bytes(after)

		return result if not result.is_empty() else {"error":"The saved round could not advance."}
	if saving or _ticket != null:
		forget_projection(); return {"error":"A checkpoint is already retained."}
	if not match_owner(owner,expected): return {"error":"The current owner changed."}
	if not owner.playable() or not Adapter.scope(owner.active,owner.session) or owner.session.physical.pending.is_empty():
		forget_projection(); return {"error":"No current round is pending."}
	var envelope: RefCounted
	if _projection_owner == owner:
		if _projection_value == null or not RawGuard.matches(_projection_value,_projection_raw):
			forget_projection(); return {"error":"The retained terminal projection changed."}
		envelope = _projection_value
	forget_projection()
	if envelope == null:
		var projected: Dictionary = Adapter.pending(owner.active).project(owner.session)
		if projected.has("error") or projected.has("fixture"): return projected
		var current: SeasonState = projected.get("candidate")
		if current == null: return {"error":"The current round did not produce a candidate."}
		envelope = Owner.new(); envelope.session = current
		if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)):
			return {"error":"The round metadata could not be preserved."}
	var ticket: RefCounted = prepare_and_write(owner,envelope,expected)
	return {"ticket":ticket} if ticket != null else {"error":"The settled candidate could not be admitted."}

static func _prepare_paired_metadata(owner: RefCounted, expected: PackedByteArray, action: String, seed_value: int=-1) -> RefCounted:

# Deterministic reconstruction only; this helper never installs Store authority.
# The child separately compares the full serialized current session to this
# native result after strict archive replay. No other phases or modes enter.
static func _cross_draft_owner(value: Variant) -> RefCounted:
	if not value is Dictionary or value.get("format")!=Owner.FORMAT or value.get("version")!=4 \
		or value.get("active")!="ordinary" or value.get("dynasty")!={}:
		return null
	var state: Variant=value.get("ordinary_state")
	var current: Variant=value.get("ordinary")
	if not state is Dictionary or not current is Dictionary \
		or state.get("status")!="active" or state.get("ordinal")!=2 or state.get("last_close")!={} \
		or state.get("proposal")!={} or not state.get("last_transition") is Dictionary:
		return null
	var t: Dictionary=state.last_transition
	var basis: Variant=t.get("basis")
	if t.get("kind")!=Owner.Common.ARCHIVE_TRANSITION or t.get("league")!="dugout" \
		or not basis is Dictionary or basis.get("version")!=2 or not Owner.Common.Start.valid(basis): return null
	var view: Variant=current.get("state")
	if not view is Dictionary or view.get("phase")!=SeasonState.Phase.DRAFT \
		or not view.get("picks") is Array or view.picks.size()>1: return null
	for name: String in ["workshop_draft","workshop_close_history","workshop_chair_purchase","workshop_applied","migration_v3"]:
		if value.get(name,{})!={}: return null
	var owner: RefCounted=Owner.new()
	var fields: Dictionary={}
	for name: String in Owner.META_FIELDS: fields[name]=value.get(name,{})
	if not Codec.install_fields(owner,Owner.META_FIELDS,fields) or not Owner._missing_dynasty(owner): return null
	owner.active="ordinary"
	owner.session=Adapter.OrdinaryRuntime.create_common(basis,"dugout")
	if owner.session==null or (view.picks.size()==1 and (not view.picks[0] is String or not owner.session.choose_player(view.picks[0]))):
		return null
	return owner if Owner.Common.valid(owner) and Owner.Common.archive_play_scope(owner.session) else null

static func _cross_bound(owner: RefCounted,expected: PackedByteArray) -> bool:
	return not saving and _ticket==null and owner!=null and owner==_owner \
		and owner.session==_owner_session and expected==_owner_bytes and not expected.is_empty() \
		and _write_revision_matches(expected,true)

static func _discard_cross_activation() -> void:
	_cross_starter=false
	_cross_pending=PackedByteArray()
	_cross_owner=null
	_cross_expected=PackedByteArray()

static func cancel_cross_activation(owner: RefCounted) -> void:
	if owner!=null and owner==_cross_owner: _discard_cross_activation()

static func retry_cross_activation(owner: RefCounted,expected: PackedByteArray) -> String:
	if _cross_pending.is_empty() or owner!=_cross_owner or expected!=_cross_expected: return "refused"
	var enabled: bool=LeagueAdmission.starter_player() if _cross_starter else cross_league_preview()
	if not enabled or not _cross_bound(owner,expected) or (_cross_starter and (_starter_slot_linked() or not _matches_raw(owner,_owner_raw))):
		_discard_cross_activation()
		return "stale"
	var dir:=DirAccess.open("user://")
	if dir==null: return "obstructed"
	for suffix: String in ["", ".bak", ".tmp"]:
		if dir.is_link((slot_path()+suffix).get_file()) or DirAccess.dir_exists_absolute(slot_path()+suffix): return "obstructed"
	if FileAccess.file_exists(slot_path()+".tmp"): return "obstructed"
	saving=true
	var outcome: String=_write_exact_result(_cross_pending,expected,true)
	saving=false
	if outcome in ["saved","renamed_unverified"]:
		# Once renamed, never retry or publish old owner authority even when
		# verifying the new primary failed. Strict Continue decides what exists.
		_invalidate()
		return outcome
	if _write_revision_matches(expected,true): return "obstructed"
	_discard_cross_activation()
	return "stale"

static func switch_available(owner: RefCounted) -> bool:

# A proof bundle, never a write ticket. Live creation can only originate from
# this exact admitted slot and ordinary Base championship.
static func prepare_standard_d1_witness(owner: RefCounted, expected: PackedByteArray, seed_value: int) -> Dictionary:
	var transition:=prepare_standard_d1_transition(owner,expected,seed_value)
	if transition.is_empty(): return {}
	var contract=preload("standard_d1_transition.gd")
	var witness: Dictionary=contract._pack_closed_source(expected)
	if witness.is_empty() or not match_owner(owner,expected): return {}
	return {"transition":transition,"witness":witness}

# Actual Common CAS preparation: materialize only from an already strict Store
# admission. No IPC packet, caller's draft, or serialized success grants entry.
# This intentionally cannot arm a generic ticket while Continue still cannot
# adopt the witnessed dialect.
static func _prepare_standard_d1_slot(owner: RefCounted, expected: PackedByteArray,
		seed_value: int) -> PackedByteArray:
	var contract=preload("standard_d1_transition.gd")
	var bundle:=prepare_standard_d1_witness(owner,expected,seed_value)
	if bundle.is_empty(): return PackedByteArray()
	var draft: SeasonState=Adapter.OrdinaryRuntime.create_d1_draft(bundle.transition)
	if draft==null or draft.phase!=SeasonState.Phase.DRAFT or not draft.picks.is_empty():
		return PackedByteArray()
	var snapshot: Dictionary=Owner.Previous.capture_d1_draft(draft)
	if snapshot.is_empty(): return PackedByteArray()
	var bytes: PackedByteArray=contract._durable_candidate(bundle.witness,snapshot)
	if contract._durable_shape(bytes).is_empty() or saving or _ticket!=null or not match_owner(owner,expected):
		return PackedByteArray()
	return bytes

static func prepare_session_and_write(owner: RefCounted, expected: PackedByteArray, operation: String, seed_value: int = 0, league: String = "", reason: String = "") -> RefCounted:
	forget_projection()
	last_error=""
	if saving or _ticket!=null or operation not in ["resume","create","attach119","close","propose","next"]: return null
	if LeagueAdmission.starter_feature() and operation in ["resume","create","attach119"]: return null
	if not match_owner(owner,expected): return null
	if Owner.MixedAppend.marked_state(owner.ordinary_state):
		if operation!="close" or reason!="complete" or seed_value!=0 or not league.is_empty(): return null
		return _prepare_mixed_append_completed_close(owner,expected)
	if Owner.Mixed.marked_state(owner.ordinary_state):
		if operation!="close" or reason!="complete" or seed_value!=0 or not league.is_empty(): return null
		return _prepare_mixed_completed_close(owner,expected)
	if owner.ordinary_state.has("archive"):
		if Owner.Common.StandardChain.marked(owner.session):
			if operation!="close" or reason!="abandon" or not (LeagueAdmission.starter_player() or private_standard_chain_opening()) \
					or not Owner.Common.StandardChain.tagged_abandon_scope(owner,owner.session): return null
		elif Owner.Common.StandardArchive.marked(owner.session):
			if operation!="close" and (not LeagueAdmission.starter_player() or operation not in ["propose","next"] \
				or not Owner.Common.StandardChain.eligible(owner)): return null
		elif LeagueAdmission.starter_player() and Owner.Common.starter_dugout_chain_custody(owner):
			if operation!="close" or reason!="complete" \
					or not Owner.Common.starter_dugout_complete_scope(owner,owner.session): return null
		elif LeagueAdmission.starter_player() and Owner.Common.starter_dugout_archive(owner):
			if operation!="close" or (reason!="abandon" and (reason!="complete" \
					or not Owner.Common.starter_dugout_complete_scope(owner,owner.session))): return null
		elif operation!="close" or reason!="abandon" or not cross_league_preview(): return null
	last_error=league_block(owner,operation,league)
	if not last_error.is_empty(): return null
	if operation in ["resume","create","attach119"] and not switch_available(owner): return null
	if operation in ["propose","next"] and owner.active=="ordinary" \
		and Owner.Common.cross_proposal_kind(owner.ordinary_state.proposal.get("kind")): return null
	if operation not in ["create","propose"] and seed_value!=0: return null
	if operation!="create" and not league.is_empty(): return null
	if (operation=="close" and reason not in ["complete","abandon"]) or (operation!="close" and not reason.is_empty()): return null
	var imported: PackedByteArray=PackedByteArray()
	const SOURCE119: String="user://wiffaltro-materialized-play-v1.json"
	if operation=="attach119":
		if owner.active!="dynasty" or not owner.paused.is_empty() or not FileAccess.file_exists(SOURCE119): return null
		var source: FileAccess=FileAccess.open(SOURCE119,FileAccess.READ)
		if source==null or source.get_length()>MAX_BYTES: return null
		imported=source.get_buffer(source.get_length());source.close()
	if LeagueAdmission.starter_player() and operation in ["propose","next"] and Owner.Common.StandardChain.eligible(owner):
		return _prepare_starter_standard_chain(owner,expected,operation,seed_value)
	if LeagueAdmission.starter_player() and operation=="next" and Owner.Common.StandardArchive.first_next(owner):
		return _prepare_starter_standard_next(owner,expected)
	var candidate: RefCounted=owner.fork()
	if candidate==null: return null
	var changed: bool=false
	if operation=="resume": changed=candidate.switch_existing()
	elif operation=="create": changed=candidate.create_missing(seed_value,league)
	elif operation=="attach119": changed=candidate.import_119(imported)
	elif operation=="close": changed=candidate.close_current(reason)
	elif operation=="propose": changed=candidate.propose_next(seed_value)
	elif operation=="next": changed=candidate.commit_next()
	if not changed: return null
	# This rare explicit boundary strictly admits BOTH sides and all metadata;
	# routine operations still pin the dormant side and validate current only.
	var data: Dictionary=Owner.capture(candidate)
	if data.is_empty(): return null
	var snapshot: Dictionary=RawGuard.capture(candidate)
	if snapshot.is_empty(): return null
	if operation=="attach119" and FileAccess.get_file_as_bytes(SOURCE119)!=imported: return null
	_ticket=RefCounted.new()
	_candidate=candidate;_candidate_session=candidate.session;_candidate_raw=snapshot;_candidate_data=data
	_expected=expected.duplicate();_written=PackedByteArray()
	_write_admitted()
	return _ticket

	if data.is_empty() or snapshot.is_empty() or not match_owner(owner,expected): return null
	_ticket=RefCounted.new()
	_candidate=candidate;_candidate_session=candidate.session;_candidate_raw=snapshot;_candidate_data=data
	_expected=expected.duplicate();_written=PackedByteArray()
	_write_admitted()
	return _ticket

static func _prepare_standard_chain(owner: RefCounted, expected: PackedByteArray, operation: String, seed_value: int, opening: bool) -> RefCounted:
	if saving or _ticket!=null or not match_owner(owner,expected) or not Owner.Common.StandardChain.eligible(owner):
		return null
	var candidate: RefCounted
	if operation=="propose" and owner.ordinary_state.proposal.is_empty():
		candidate=owner.fork()
		if candidate==null or not candidate.propose_next(seed_value): return null
	elif operation=="next" and seed_value==0:
		# Choose exactly once from the already-owned predecessor. A failed
		# tagged birth must not silently start its untagged counterpart.
		var choose_opening: bool=opening or Owner.Common.StandardChain._opening_predecessor(owner)
		candidate=Owner.Common.StandardChain.opening_candidate(owner,expected) if choose_opening \
			else Owner.Common.StandardChain.candidate(owner,expected)
	else: return null
	if candidate==null or not _starter_owner_allowed(candidate): return null
	var data: Dictionary=Owner.capture(candidate)
	var snapshot: Dictionary=RawGuard.capture(candidate)
	if data.is_empty() or snapshot.is_empty() or not match_owner(owner,expected): return null
	_ticket=RefCounted.new()
	_candidate=candidate;_candidate_session=candidate.session;_candidate_raw=snapshot;_candidate_data=data
	_expected=expected.duplicate();_written=PackedByteArray()
	_write_admitted()
	return _ticket

static func _prepare_private_standard_flat(owner: RefCounted, expected: PackedByteArray, op: String, argument: String, app: Variant=null) -> RefCounted:
	if not private_standard_chain_opening() or saving or _ticket!=null or not _match_bound_owner(owner,expected):
		return null
	var candidate: RefCounted
	var close_data: Dictionary={}
	if op=="birth":
		if not argument.is_empty() or not Flat._scope(owner): return null
		var draft:=Flat._from_authenticated(owner,expected,Owner._hash(expected))
		candidate=Flat._local_owner(owner,draft)
	elif op=="abandon":
		if not argument.is_empty() or not private_standard_flat_close_available(app) or app.career_owner!=owner: return null
		candidate=Owner.new()
		if not Codec.install_fields(candidate,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
		candidate.session=owner.session.fork_live()
		if candidate.session==null: return null
		close_data=Flat.capture_abandon(candidate,Flat._capture_local(candidate))
		if close_data.is_empty(): return null
	elif op in ["npc","settle"]:
		if not argument.is_empty() or not Flat._storage_scope(owner) or owner.session.physical.pending.is_empty(): return null
		var pending: Dictionary=owner.session.physical.pending
		var current: SeasonState
		if op=="npc":
			if not _private_flat_npc_matches(app,owner,expected) or pending.reports.size()>=2: return null
			var report: Dictionary=_flat_npc.native_report()
			if report.is_empty(): return null
			var staged: Dictionary=Adapter.OrdinaryPending.stage_report(owner.session,report)
			current=staged.get("candidate")
			if current==null or not _private_flat_npc_matches(app,owner,expected): return null
			var row: Dictionary=current.physical.pending.reports[-1]
			if row.get("fixture")!=_flat_npc_fixture or row.get("request")!=_flat_npc_request: return null
		else:
			if not _private_flat_app(app) or app.career_owner!=owner or app.season!=owner.session \
					or app.saved_bytes!=expected or pending.reports.size()!=2 or is_instance_valid(_flat_npc): return null
			var settled:=Flat.project_regular(owner)
			if settled.is_empty(): return null
			current=settled.candidate
		candidate=Owner.new()
		if not Codec.install_fields(candidate,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
		candidate.session=current
		if op=="settle":
			var record: Dictionary={"version":1,"human":pending.human.duplicate(true)}
			if owner.session.round_index==0:
				candidate.ordinary_state[Flat.ROUND1]=record
			elif owner.session.round_index==1:
				record["property"]=pending.property
				candidate.ordinary_state[Flat.ROUND2]=record
			else: return null
	elif op in ["pick","opening","human"]:
		var fixture: Dictionary=Flat.human_fixture(owner) if op=="human" else {}
		if (fixture.is_empty() if op=="human" else not Flat._draft_scope(owner)): return null
		var next: SeasonState=owner.session.fork_live()
		if next==null: return null
		candidate=Owner._new()
		if not Codec.install_fields(candidate,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
		candidate.session=next
		if op=="pick":
			if next.phase!=SeasonState.Phase.DRAFT or not next.choose_player(argument): return null
			if next.picks.size()==4 and not Flat._enter_opening(candidate): return null
		elif op=="opening":
			if not Flat._lock_opening(candidate,argument): return null
		else:
			if not argument.is_empty() or not _private_flat_play_matches(app,owner,expected): return null
			var human:=observe_private_standard_flat_human(app)
			if human.size()!=13 or human[0]!=fixture.id: return null
			candidate.session.physical.pending={"human":human.duplicate(true),"reports":[]}
			var property: String=Owner.Common.StandardWorkshop.Home.key(owner.session,fixture)
			if fixture.home==0:
				if property.is_empty(): return null
				candidate.session.physical.pending["property"]=property
			elif not property.is_empty() or not Flat._validate_human_pending(candidate): return null
			# Property-bound human proof is independently replayed by ordinary
			# Codec.capture below, exactly once. No trusted evidence flag.
			if not _private_flat_play_matches(app,owner,expected): return null
	else: return null
	var data:=close_data if op=="abandon" else Flat._capture_local(candidate)
	if data.is_empty() or not _match_bound_owner(owner,expected): return null
	var snapshot:=RawGuard.capture(candidate)
	if snapshot.is_empty(): return null
	_ticket=RefCounted.new();_flat_ticket=_ticket
	_candidate=candidate;_candidate_session=candidate.session;_candidate_raw=snapshot;_candidate_data=data
	_expected=expected.duplicate();_written=PackedByteArray()
	_write_admitted()
	return _ticket

class PrivateFlatOpeningPlay:
	var match_state: MatchState
	var field_id: StringName
	var _field: FieldDefinition
	var _fixture: Dictionary
	var _request: String
	var _stadium: Dictionary

# Unbound/private launch data only. The normal App/Return writers still
# refuse this owner. No report, game result or disk change can be supplied.
static func private_standard_flat_opening_play(owner: RefCounted, expected: PackedByteArray) -> PrivateFlatOpeningPlay:
	if not private_standard_chain_opening() or saving or _ticket!=null \
			or not _match_bound_owner(owner,expected): return null
	var fixture:=Flat.human_fixture(owner)
	if fixture.is_empty(): return null
	var detached: SeasonState=owner.session.fork_live()
	if detached==null: return null
	var state: MatchState=detached.make_match()
	var field: FieldDefinition=Adapter.OrdinaryRuntime.field_for_fixture(fixture,detached)
	var stadium: Dictionary=Owner.Common.StandardWorkshop.Home.fixture(detached,fixture)
	if state==null or field==null or not Codec.same(state.stadium,stadium) \
			or stadium.is_empty()!=(fixture.home!=0) or not state.sure_shot.enable_current_evidence() \
			or not _match_bound_owner(owner,expected): return null
	var result := PrivateFlatOpeningPlay.new()
	result.match_state=state
	result.field_id=field.id
	result._field=field
	result._fixture=fixture.duplicate(true)
	result._stadium=stadium.duplicate(true)
	result._request=Adapter.OrdinaryFixtures.request_key(detached,fixture)
	return result if not result._request.is_empty() else null

static func _clear_private_flat_play() -> void:
	_flat_play=null; _flat_play_app=null; _flat_play_lab=null

static func _private_flat_app(app: Variant) -> bool:
	return app is SeasonApp and app.get_script()!=null and app.get_script().resource_path=="res://src/presentation/season/successor_dynasty_owner_v1/app.gd"

# Fixed native App-only launch. Store drives the scene from the exact own
# request and records the actual Lab it opened, not caller-supplied tokens.
static func open_private_standard_flat_human(app: Variant) -> bool:
	if not _private_flat_app(app) or is_instance_valid(app.lab) or app.career_owner==null \
			or app.season!=app.career_owner.session: return false
	var owner: RefCounted=app.career_owner
	var choice: Dictionary=owner.ordinary_state.get(Owner.Common.StandardWorkshop.OPENING,{})
	if choice.get("choice")!="chair": return false
	var play:=private_standard_flat_opening_play(owner,app.saved_bytes)
	if play==null: return false
	_clear_private_flat_play()
	app._fixture_id=int(play._fixture.id)
	app._result_recorded=false
	app._result_saved=false
	app._open_match(play.match_state,play._fixture.home==0,true,play.field_id)
	if not is_instance_valid(app.lab): return false
	_flat_play=play
	_flat_play_app=weakref(app)
	_flat_play_lab=weakref(app.lab)
	if _private_flat_play_matches(app,owner,app.saved_bytes): return true
	_clear_private_flat_play()
	app._close_match(false)
	app.notice="The private authored field could not be verified. Saved progress was unchanged."
	return false

static func _private_flat_play_matches(app: Variant, owner: RefCounted, expected: PackedByteArray) -> bool:
	if not private_standard_chain_opening() or not _private_flat_app(app) or _flat_play==null \
			or _flat_play_app.get_ref()!=app or not is_instance_valid(app.lab) or _flat_play_lab.get_ref()!=app.lab \
			or not _match_bound_owner(owner,expected) or app.career_owner!=owner or app.season!=owner.session \
			or app.saved_bytes!=expected or app._fixture_id!=_flat_play._fixture.id: return false
	var lab: PitchBatLab=app.lab
	var state: MatchState=lab._match_state
	var fixture:=Flat.human_fixture(owner)
	return Codec.same(fixture,_flat_play._fixture) and lab._managed_match and lab._match_mode \
		and lab._player_home==(fixture.home==0) and state==_flat_play.match_state \
		and lab._field_definition==_flat_play._field and lab._field_id==_flat_play.field_id \
		and Codec.same(state.league_rules,owner.session.league_rules) and state.pitch_catalog==owner.session.build.pitch_catalog_signature() \
		and _private_flat_geometry(lab,owner.session,fixture) \
		and Adapter.OrdinaryFixtures.request_key(owner.session,fixture)==_flat_play._request

static func _private_flat_geometry(lab: PitchBatLab, season: SeasonState, fixture: Dictionary) -> bool:
	var workshop = Owner.Common.StandardWorkshop
	var native: Dictionary=workshop.Home.fixture(season,fixture)
	if not Codec.same(native,_flat_play._stadium) or not Codec.same(lab._match_state.stadium,native): return false
	if native.is_empty(): return not lab.has_meta("stadium_property")
	if fixture.home!=0 or fixture.get("neutral",false) or not lab._player_home \
			or not Codec.same(lab.get_meta("stadium_snapshot",{}),native): return false
	var property: Variant=lab.get_meta("stadium_property",null)
	if not is_instance_valid(property) or not property is StadiumPropertyGeometry \
			or native.layout.receiver!=workshop.COPY or not property.objects.has(native.layout.receiver): return false
	var receiver: StadiumPitchReceiver=property.get_node_or_null("PitchReceiverPresentation")
	return receiver!=null and receiver.chair==property.objects[native.layout.receiver] \
		and lab._pitch_actor.receiver_router==receiver.route

# Construct the native job here. The completion callback takes runner
# identity only; an external report argument cannot reach the ticket.
static func start_private_standard_flat_npc(app: Variant) -> bool:
	if not private_standard_chain_opening() or not _private_flat_app(app) or saving or _ticket!=null \
			or is_instance_valid(_flat_npc) or app._busy or is_instance_valid(app.lab) or is_instance_valid(app.ai_runner): return false
	var owner: RefCounted=app.career_owner
	var expected: PackedByteArray=app.saved_bytes
	if not _match_bound_owner(owner,expected) or app.season!=owner.session or not Flat._storage_scope(owner) \
			or owner.session.physical.pending.is_empty() or owner.session.physical.pending.reports.size()>=2: return false
	var projected:=Flat.project_regular(owner)
	if projected.is_empty() or not projected.has("fixture"): return false
	var fixture: Dictionary=projected.fixture
	var day: SeasonState=projected.candidate
	var state: MatchState=Adapter.OrdinaryFixtures.match_for(day,fixture)
	var field: FieldDefinition=Adapter.OrdinaryRuntime.field_for_fixture(fixture,day)
	var request: String=Adapter.OrdinaryFixtures.request_key(day,fixture)
	if state==null or not state.stadium.is_empty() or field==null or request.is_empty() \
			or not _match_bound_owner(owner,expected): return false
	var runner:=FlatRunner.new()
	app.add_child(runner)
	_flat_npc=runner; _flat_npc_app=weakref(app); _flat_npc_expected=expected.duplicate()
	_flat_npc_fixture=int(fixture.id); _flat_npc_request=request
	app.ai_runner=runner; app._busy=true
	runner.native_ready.connect(_finish_private_flat_npc.bind(runner),CONNECT_DEFERRED)
	runner.failed.connect(_fail_private_flat_npc.bind(runner))
	if runner.start(state,Adapter.OrdinaryFixtures.seed_for(day,fixture),field.id): return true
	_fail_private_flat_npc("The authored private native fixture could not start.",runner)
	return false

static func _finish_private_flat_npc(runner: Node) -> void:

	var ticket: RefCounted=_prepare_private_standard_flat(app.career_owner,app.saved_bytes,"npc","",app)
	_clear_private_flat_npc()
	app._private_flat_native_npc_finished(runner,ticket)

static func _fail_private_flat_npc(reason: String, runner: Node) -> void:
	if not is_instance_valid(runner) or runner!=_flat_npc: return
	var app: Variant=_flat_npc_app.get_ref()
	_clear_private_flat_npc()
	if _private_flat_app(app): app._ai_failed(reason,runner)

# Identity handoff only. Later retry/promotion still authenticate this exact
# live candidate against the private snapshot; no returned value authorizes IO.
static func prepared_candidate(ticket: RefCounted) -> RefCounted:
	return _candidate if ticket != null and ticket == _ticket else null

# Cheap handoff identity only, never raw admission or authority to write/promote.
static func prepared_binding(ticket: RefCounted, owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> bool:
	if ticket == null or ticket != _ticket or owner != _owner or candidate != _candidate or expected != _expected: return false
	return candidate.session == _candidate_session and (owner == null or owner.session == _owner_session)

static func prepared_matches(ticket: RefCounted, owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> bool:
	if not prepared_binding(ticket,owner,candidate,expected):
		forget_projection(); return false
	var valid: bool = (owner == null or _matches_raw(owner,_owner_raw)) and _matches_raw(candidate,_candidate_raw)
	if not valid: forget_projection()
	return valid

static func prepared_snapshot(ticket: RefCounted) -> Dictionary:
	return _candidate_data.duplicate(true) if ticket != null and ticket == _ticket else {}

# A renamed-but-unverified ticket is never promoted or retried. Only its
# original App handoff may release stale native authority and hand on the pin.
static func take_rollover_close_recovery(ticket: RefCounted, owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> PackedByteArray:
	if saving or not private_mixed_append() or not _rollover_close_unverified or ticket==null \
		or ticket!=_rollover_close_ticket or ticket!=_ordinary_encoded_ticket \
		or not prepared_binding(ticket,owner,candidate,expected): return PackedByteArray()
	var revision:=_ordinary_encoded.duplicate()
	if revision.is_empty(): return revision
	_invalidate()
	_rollover_recovery_expected = revision
	return revision.duplicate()

static func write_prepared(ticket: RefCounted) -> bool:
	forget_projection()
	if saving or ticket == null or ticket != _ticket or not _written.is_empty() or _rollover_close_unverified: return false
	if not prepared_matches(ticket,_owner,_candidate,_expected): return false
	return _write_admitted()

static func _ordinary_prepared_revision() -> bool:
	return _ticket!=null and _ticket!=_flat_ticket and _candidate!=null and _candidate.active=="ordinary"

# Private immediate continuation only; callers use prepare_and_write or the
# fully checked write_prepared retry. Disk CAS/readback remain in every write.
static func _write_admitted() -> bool:
	if _rollover_close_unverified or not _rollover_recovery_expected.is_empty(): return false
	if _flat_ticket!=null and _flat_ticket==_ticket:
		if not private_standard_chain_opening() or not Flat._storage_scope(_candidate): return false
	elif not _starter_owner_allowed(_candidate): return false
	last_error = "The exact candidate is retained. Resolve the save obstruction, then retry."
	if not _matches(_expected):
		last_error = "The saved file changed. It was not overwritten; your candidate remains retained."
		return false
	for suffix: String in ["", ".bak", ".tmp"]:
		if DirAccess.dir_exists_absolute(slot_path()+suffix): return false
	if FileAccess.file_exists(slot_path()+".tmp") or (_expected.is_empty() and FileAccess.file_exists(slot_path()+".bak")): return false
	if _ordinary_prepared_revision():
		if _ordinary_encoded_ticket!=_ticket:
			_ordinary_encoded_ticket = _ticket
			_ordinary_encoded = _native_encoded(_candidate_data)
		if _ordinary_encoded.is_empty(): return false
		saving = true
		var result: String = _write_exact_result(_ordinary_encoded,_expected,false)
		saving = false
		if result=="renamed_unverified" and _rollover_close_ticket==_ticket:
			_rollover_close_unverified = true
			forget_projection()
			last_error="The season save was replaced, but its contents could not be verified. Use Continue to check it."
		if result!="saved": return false
		# Keep precisely the bytes whose immediate readback passed, not a
		# second unbound sample from a possibly replaced slot.
		_written = _ordinary_encoded.duplicate()
		return true
	saving = true
	var legacy_result: bool = _write_data(_candidate_data,_expected)
	saving = false
	if legacy_result: _written = raw()
	return legacy_result

static func written_bytes(ticket: RefCounted) -> PackedByteArray:
	return _written.duplicate() if ticket != null and ticket == _ticket else PackedByteArray()

static func promote(ticket: RefCounted, owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> bool:
	if _rollover_close_unverified or not _rollover_recovery_expected.is_empty(): return false
	if ticket!=null and ticket==_flat_ticket and not private_standard_chain_opening(): return false
	if _ordinary_prepared_revision():
		# Raw snapshots may be expensive. Check the same pinned revision only
		# after all of them pass, just before promoting native ownership.
		if _written.is_empty() or not prepared_matches(ticket,owner,candidate,expected) or raw()!=_written:
			forget_projection(); return false
	elif _written.is_empty() or raw()!=_written or not prepared_matches(ticket,owner,candidate,expected):
		forget_projection(); return false
	var arm: bool = _projection_ticket == ticket and _projection_checkpoint == candidate and _projection_value != null
	if arm:
		_projection_owner = candidate; _projection_ticket = null; _projection_checkpoint = null
	else: forget_projection()
	_owner = _candidate; _owner_session = _candidate_session; _owner_raw = _candidate_raw; _owner_bytes = _written.duplicate()
	_paused = _candidate.paused.duplicate(true); _active = _candidate.active
	_metadata = Codec.data_fields(_candidate,Owner.META_FIELDS); _imported_sha256 = _candidate.imported_sha256
	_clear_private_flat_play()
	_flat_ticket = null; _ticket = null; _candidate = null; _candidate_session = null; _candidate_raw = {}; _candidate_data = {}
	_expected = PackedByteArray(); _written = PackedByteArray()
	_ordinary_encoded_ticket = null; _ordinary_encoded = PackedByteArray()
	_rollover_close_ticket = null; _rollover_close_unverified = false

	return true

# Snapshot distinguishes absent from present. No unbounded text reads or
# following slot/object links. Only fixed names and checked SHA names call it.
static func _base_archive_member(path: String) -> Dictionary:
	var dir: DirAccess=DirAccess.open(path.get_base_dir())
	if dir==null or dir.is_link(path.get_file()) or DirAccess.dir_exists_absolute(path): return {}
	if not FileAccess.file_exists(path): return {"present":false,"bytes":PackedByteArray()}
	var file: FileAccess=FileAccess.open(path,FileAccess.READ)
	if file==null or file.get_length()<=0 or file.get_length()>MAX_BYTES: return {}
	var size: int=file.get_length()
	var bytes: PackedByteArray=file.get_buffer(size)
	if bytes.size()!=size or file.get_length()!=size: return {}
	return {"present":true,"bytes":bytes}

static func _base_archive_pair(reader: bool=false) -> Dictionary:
	if not _base_archive_root_ok(reader): return {}
	var slot: String=BASE_ARCHIVE_READER_SLOT if reader else slot_path()
	var primary:=_base_archive_member(slot)
	var backup:=_base_archive_member(slot+".bak")
	if primary.is_empty() or backup.is_empty(): return {}
	return {"primary":primary,"backup":backup}

static func _base_archive_absent(path: String) -> bool:
	return _base_archive_member(path).get("present",true)==false

static func _base_archive_temps_clear(reader: bool=false) -> bool:
	var slot: String=BASE_ARCHIVE_READER_SLOT if reader else slot_path()
	return _base_archive_absent(slot+".tmp") and _base_archive_absent(slot+BASE_ARCHIVE_BACKUP_STAGE)

static func _base_archive_source_valid(owner: RefCounted, pair: Dictionary) -> bool:
	if owner==null or pair.is_empty() or not pair.primary.present or owner!=_owner \
			or owner.active!="ordinary" or owner.session!=_owner_session or not Owner._missing_dynasty(owner) \
			or owner.session.league_rules.get("league")!="dugout" or owner.session.league_rules.get("tier")!=0: return false
	var st: Dictionary=owner.ordinary_state
	if st.has("archive") or st.get("status")!="closed" or st.get("last_close",{}).get("reason")!="abandon" \
			or st.get("proposal",{}).get("basis",{}).get("seed")!=BaseArchive._default_next_seed(owner) \
			or not match_owner(owner,pair.primary.bytes): return false
	if not pair.backup.present: return true
	var bare: Dictionary=_native_plain_without_proposal(owner)
	if bare.is_empty(): return false
	return Codec.same(Owner._bytes_value(pair.backup.bytes),bare) \
		and _base_archive_backup_wire_matches(pair.primary.bytes,pair.backup.bytes,st.proposal)

# Project the original stock wire, not a re-encoding of rehydrated numeric
# types. A prepared-primary/closed-backup pair must differ only by inserting
# this exact native nonempty ordinary proposal into an otherwise-empty object.
static func _base_archive_backup_wire_matches(primary: PackedByteArray, backup: PackedByteArray,
		proposal: Dictionary) -> bool:
	var delta: int=primary.size()-backup.size()
	if delta<=0 or backup.is_empty() or primary.size()>MAX_BYTES or proposal.is_empty(): return false
	var offset: int=0
	while offset<backup.size() and primary[offset]==backup[offset]:
		offset+=1
	var tag: PackedByteArray='"proposal":{'.to_utf8_buffer()
	if offset<tag.size() or offset>=backup.size() or backup[offset]!=125 or primary[offset]!=34 \
			or primary.slice(offset-tag.size(),offset)!=tag: return false
	if primary.slice(offset+delta)!=backup.slice(offset): return false
	var inserted: Variant=Owner._bytes_value(primary.slice(offset-1,offset+delta+1))
	return inserted is Dictionary and Codec.same(inserted,proposal)

static func _native_plain_without_proposal(owner: RefCounted) -> Dictionary:
	var fork: RefCounted=owner.fork()
	if fork==null: return {}
	fork.ordinary_state.proposal={}
	return Owner.capture(fork)

	if saving or _ticket!=null or not _base_archive_qa(): return null

			or not _base_archive_temps_clear(): return null

	if not _base_archive_source_valid(owner,pair) or not Codec.same(_base_archive_pair(),pair) \
			or not _base_archive_temps_clear(): return null

	return owner

static func _base_archive_page_path(pin: String) -> String:
	return BASE_ARCHIVE_PAGES.path_join(pin+".json") if BaseArchive._digest(pin) else ""

static func _base_archive_page_root(create: bool) -> bool:

	if dir==null or dir.is_link(BASE_ARCHIVE_PAGES.get_file()) or FileAccess.file_exists(BASE_ARCHIVE_PAGES): return false

static func write_base_archive_next(owner: RefCounted, expected: PackedByteArray) -> bool:
	if saving or _ticket!=null or not _base_archive_qa() or _base_archive_source_pair.is_empty(): return false
	var pair: Dictionary=_base_archive_source_pair
	_base_archive_source_pair={}
	if not pair.primary.present or pair.primary.bytes!=expected or not Codec.same(_base_archive_pair(),pair) \
			or not _base_archive_temps_clear() or not _base_archive_source_valid(owner,pair): return false
	var source: Dictionary=Owner.capture(owner)
	if source.is_empty() or not Codec.same(source,Owner._bytes_value(expected)): return false
	var genesis: Dictionary=owner.session.career.start_basis.get("genesis",{})
	var seed: Variant=genesis.get("seed")
	if not Codec.whole(seed,0,2147483647): return false
	var head: PackedByteArray=BaseArchive.make_head(int(seed),"dugout")
	var page: PackedByteArray=BaseArchive.make_page(head,source.ordinary,owner.ordinary_state.last_close)
	head=BaseArchive.append_head(head,page)
	if head.is_empty(): return false
	var next: RefCounted=owner.fork()
	if next==null or not next.commit_next(): return false
	var next_data: Dictionary=Owner.capture(next)
	var current: PackedByteArray=_native_encoded(next_data)
	var pages: Array[PackedByteArray]=[page]
	var checked: RefCounted=BaseArchive.restore_pages(head,current,pages)
	if checked==null or not Codec.same(Owner.capture(checked),next_data): return false
	var marked: PackedByteArray=BASE_ARCHIVE_HEADER.to_utf8_buffer()
	var prior_pin: String=BaseArchive._hash(expected)
	if not BaseArchive._digest(prior_pin): return false
	marked.append_array(BaseArchive._wire({"head":BaseArchive._head(head),"current":next_data,
		"prior_primary_sha256":prior_pin}))
	if marked.size()>MAX_BYTES or not _base_archive_source_valid(owner,pair) \
			or not Codec.same(_base_archive_pair(),pair) or not _base_archive_temps_clear(): return false
	saving=true
	_base_archive_write_ticket=RefCounted.new()
	var committed: bool=_base_archive_install(_base_archive_write_ticket,pair,page,marked)
	saving=false
	# Never arm routine play on the marked slot. Failed/partial transactions
	# require separately inspecting the exact surviving primary/backup/temp.
	_invalidate()
	return committed

static func _base_archive_install(ticket: RefCounted, pair: Dictionary, page: PackedByteArray,
		marked: PackedByteArray, create_page: bool=true,
		old_head: Dictionary={}, old_pages: Array[PackedByteArray]=[]) -> bool:
	if ticket==null or ticket!=_base_archive_write_ticket: return false
	_base_archive_write_ticket=null
	if not saving or not _base_archive_qa() or not Codec.same(_base_archive_pair(),pair) \
			or not _base_archive_temps_clear() or not _base_archive_pages_live(old_head,old_pages) \
			or not _base_archive_page_root(create_page): return false
	var final_page: String=_base_archive_page_path(BaseArchive._hash(page))
	if final_page.is_empty(): return false
	# Closed over fixed derived names. No caller-supplied path writer.
	var stage:=func(path: String, bytes: PackedByteArray) -> bool:
		if bytes.is_empty() or bytes.size()>MAX_BYTES or not _base_archive_absent(path): return false
		var file: FileAccess=FileAccess.open(path,FileAccess.WRITE)
		if file==null: return false
		file.store_buffer(bytes)
		file.flush()
		var ok: bool=file.get_error()==OK
		file.close()
		return ok and _base_archive_member(path).get("bytes",PackedByteArray())==bytes
	var existing:=_base_archive_member(final_page)
	var page_tmp: String=final_page+".tmp"
	if existing.is_empty() or (not create_page and not existing.present) or not _base_archive_absent(page_tmp): return false
	last_error="The archive transaction was not verified. No interruption file was removed."
	if existing.present:
		if existing.bytes!=page: return false
	else:
		if not stage.call(page_tmp,page) or not _base_archive_absent(final_page) \
				or DirAccess.rename_absolute(page_tmp,final_page)!=OK: return false
	if _base_archive_member(final_page).get("bytes",PackedByteArray())!=page \
			or not Codec.same(_base_archive_pair(),pair) or not _base_archive_temps_clear() \
			or not _base_archive_pages_live(old_head,old_pages): return false
	var backup_tmp: String=slot_path()+BASE_ARCHIVE_BACKUP_STAGE
	if not stage.call(slot_path()+".tmp",marked) or not stage.call(backup_tmp,pair.primary.bytes) \
			or not Codec.same(_base_archive_pair(),pair): return false
	if DirAccess.rename_absolute(backup_tmp,slot_path()+".bak")!=OK: return false
	last_error="The prior primary was rotated, but the new archive head was not verified. Inspect the exact surviving files."
	var rotated: Dictionary={"primary":pair.primary,"backup":pair.primary}
	if not Codec.same(_base_archive_pair(),rotated) or _base_archive_member(slot_path()+".tmp").get("bytes",PackedByteArray())!=marked \
			or _base_archive_member(final_page).get("bytes",PackedByteArray())!=page \
			or not _base_archive_pages_live(old_head,old_pages) \
			or DirAccess.rename_absolute(slot_path()+".tmp",slot_path())!=OK: return false
	var final: Dictionary=_base_archive_pair()
	if final.is_empty() or final.primary.get("bytes")!=marked or final.backup.get("bytes")!=pair.primary.bytes: return false
	last_error=""
	return true

static func read_base_archive_qualification() -> RefCounted:
	return _base_archive_load().get("owner")

static func _base_archive_wrapper(bytes: PackedByteArray) -> Dictionary:
	if bytes.size()>MAX_BYTES or bytes.slice(0,BASE_ARCHIVE_HEADER.length()).get_string_from_ascii()!=BASE_ARCHIVE_HEADER:
		return {}
	var wrapped: Dictionary=BaseArchive._decode(bytes.slice(BASE_ARCHIVE_HEADER.length()),MAX_BYTES)
	var fields: Array[String]=["head","current","prior_primary_sha256"]
	if wrapped.has("generation"):
		if wrapped.generation not in [BASE_ARCHIVE_DRAFT_GENERATION,BASE_ARCHIVE_HUMAN_GENERATION,
				BASE_ARCHIVE_REPORT_GENERATION,BASE_ARCHIVE_SETTLED_GENERATION,BASE_ARCHIVE_SHOP_GENERATION,
				BASE_ARCHIVE_NEXT_HUMAN_GENERATION,BASE_ARCHIVE_REGULAR_REPORT_GENERATION,
				BASE_ARCHIVE_REGULAR_SETTLED_GENERATION,BASE_ARCHIVE_LAST_REGULAR_REPORT_GENERATION,
				BASE_ARCHIVE_POSTSEASON_GENERATION,BASE_ARCHIVE_SEMI_HUMAN_GENERATION,
				BASE_ARCHIVE_SEMI_REPORT_GENERATION,BASE_ARCHIVE_SEMI_SETTLED_GENERATION,BASE_ARCHIVE_COMPLETED_GENERATION,
				BASE_ARCHIVE_ABANDONED_GENERATION,BASE_ARCHIVE_COMPLETED_PREPARED_GENERATION,
				BASE_ARCHIVE_FINAL_HUMAN_GENERATION,BASE_ARCHIVE_FINAL_SETTLED_GENERATION,BASE_ARCHIVE_FINAL_CLOSED_GENERATION,
				BASE_ARCHIVE_PREPARED_GENERATION,BASE_ARCHIVE_NEXT_BIRTH_GENERATION,BASE_ARCHIVE_EARNED_D1_GENERATION]: return {}
		fields.append("generation")
	if not Codec.keys(wrapped,fields) or not wrapped.head is Dictionary or not wrapped.current is Dictionary \
			or not BaseArchive._digest(wrapped.prior_primary_sha256): return {}
	return wrapped

static func _base_archive_load(reader: bool=false) -> Dictionary:
	if saving or _ticket!=null or not (base_archive_reader_profile() if reader else _base_archive_qa()): return {}
	var pair:=_base_archive_pair(reader)
	if pair.is_empty() or not pair.primary.present or not _base_archive_temps_clear(reader): return {}
	var wrapped: Dictionary=_base_archive_wrapper(pair.primary.bytes)
	if wrapped.is_empty(): return {}
	# The new tier has a dedicated related reader. No birth enters the
	# legacy archive App or ordinary save writer through this path.
	if wrapped.get("generation")==BASE_ARCHIVE_EARNED_D1_GENERATION: return {}
	var draft: bool=wrapped.has("generation")
	if (not draft and not pair.backup.present) or (pair.backup.present \
			and BaseArchive._hash(pair.backup.bytes)!=wrapped.prior_primary_sha256): return {}
	var head_bytes: PackedByteArray=BaseArchive._wire(wrapped.head)
	var head: Dictionary=BaseArchive._head(head_bytes)
	if head.is_empty() or head.pages.is_empty() or head.pages.size()>BASE_ARCHIVE_QA_MAX_PAGES \
			or (not draft and head.pages.size()!=1) or not _base_archive_page_root(false): return {}
	var pages: Array[PackedByteArray]=[]
	for pin: String in head.pages:
		var path: String=_base_archive_page_path(pin)
		if path.is_empty() or not _base_archive_absent(path+".tmp"): return {}
		var data: Dictionary=_base_archive_member(path)
		if data.is_empty() or not data.present or BaseArchive._hash(data.bytes)!=pin: return {}
		pages.append(data.bytes)
	var result: RefCounted
	var related: RefCounted
	if pair.backup.present and _base_archive_related_role(wrapped.get("generation","")):
		var prior: Dictionary=_base_archive_wrapper(pair.backup.bytes)
		if prior.is_empty(): return {}
		var prior_head: PackedByteArray=BaseArchive._wire(prior.head)
		if wrapped.generation!=BASE_ARCHIVE_NEXT_BIRTH_GENERATION and prior_head!=head_bytes: return {}
		if wrapped.generation==BASE_ARCHIVE_NEXT_BIRTH_GENERATION and prior_head==head_bytes: return {}
		var current_bytes: PackedByteArray=BaseArchive._wire(wrapped.current)
		var prior_current_bytes: PackedByteArray=BaseArchive._wire(prior.current)
		var verified: Dictionary
		if wrapped.generation in [BASE_ARCHIVE_COMPLETED_GENERATION,BASE_ARCHIVE_COMPLETED_PREPARED_GENERATION,
				BASE_ARCHIVE_FINAL_CLOSED_GENERATION]:
			var required_prior: String=BASE_ARCHIVE_COMPLETED_GENERATION
			if wrapped.generation==BASE_ARCHIVE_COMPLETED_GENERATION:
				required_prior=BASE_ARCHIVE_SEMI_SETTLED_GENERATION
			elif wrapped.generation==BASE_ARCHIVE_FINAL_CLOSED_GENERATION:
				required_prior=BASE_ARCHIVE_FINAL_SETTLED_GENERATION
			if prior.get("generation")!=required_prior: return {}
			# Same-head equality is mandatory above. Do not fall back if the
			# exact full images or either native completed-state tail disagree.
			verified=BaseArchive.restore_completed_related_pages(head_bytes,current_bytes,pages,prior_current_bytes)
		else:
			verified=BaseArchive.restore_related_pages(head_bytes,current_bytes,pages,prior_head,prior_current_bytes)
		if verified.is_empty(): return {}
		result=verified.primary
		related=verified.sibling
	else:
		result=BaseArchive.restore_pages(head_bytes,BaseArchive._wire(wrapped.current),pages)
	if result==null: return {}
	if draft:
		match wrapped.generation:
			BASE_ARCHIVE_HUMAN_GENERATION:
				if not _base_archive_human_image(result): return {}
			BASE_ARCHIVE_NEXT_HUMAN_GENERATION:
				if not _base_archive_next_human_image(result): return {}
			BASE_ARCHIVE_REPORT_GENERATION:
				if _base_archive_opening_report_count(result)<1: return {}
			BASE_ARCHIVE_REGULAR_REPORT_GENERATION:
				if _base_archive_last_regular(result) or _base_archive_regular_report_count(result)<1: return {}
			BASE_ARCHIVE_LAST_REGULAR_REPORT_GENERATION:
				if not _base_archive_last_regular(result) or _base_archive_regular_report_count(result)<1: return {}
			BASE_ARCHIVE_POSTSEASON_GENERATION:
				if not _base_archive_postseason_image(result): return {}
			BASE_ARCHIVE_SEMI_HUMAN_GENERATION:
				if _base_archive_semifinal_count(result)!=0: return {}
			BASE_ARCHIVE_SEMI_REPORT_GENERATION:
				if _base_archive_semifinal_count(result)<1: return {}
			BASE_ARCHIVE_SEMI_SETTLED_GENERATION:
				if not _base_archive_semifinal_settled(result): return {}
			BASE_ARCHIVE_FINAL_HUMAN_GENERATION:
				if not _base_archive_final_human(result): return {}
			BASE_ARCHIVE_FINAL_SETTLED_GENERATION:
				if not _base_archive_final_settled(result): return {}
			BASE_ARCHIVE_FINAL_CLOSED_GENERATION:
				if not _base_archive_final_closed(result): return {}
			BASE_ARCHIVE_REGULAR_SETTLED_GENERATION:
				if not _base_archive_regular_settled_image(result): return {}
			BASE_ARCHIVE_SETTLED_GENERATION:
				if not _base_archive_unshopped_settled_image(result): return {}
			BASE_ARCHIVE_SHOP_GENERATION:
				if not _base_archive_shop_image(result): return {}
			BASE_ARCHIVE_COMPLETED_GENERATION:
				if not _base_archive_completed_closed(result): return {}
			BASE_ARCHIVE_COMPLETED_PREPARED_GENERATION:
				if not _base_archive_completed_closed(result,true): return {}
			BASE_ARCHIVE_ABANDONED_GENERATION:
				if not _base_archive_abandoned_image(result,false): return {}
			BASE_ARCHIVE_PREPARED_GENERATION:
				if not _base_archive_abandoned_image(result,true): return {}
			BASE_ARCHIVE_NEXT_BIRTH_GENERATION:
				if not _base_archive_draft_image(result,0) or not result.session.picks.is_empty() or pages.size()<2: return {}
			_:
				if not _base_archive_draft_image(result,1): return {}
		if pair.backup.present and not _base_archive_marked_previous_matches(pair.backup.bytes,wrapped,result,head_bytes,pages,related): return {}
	elif not _base_archive_previous_matches(pair.backup.bytes,pages[0],wrapped.current): return {}
	if not Codec.same(_base_archive_pair(reader),pair) or not _base_archive_temps_clear(reader) \
			or not _base_archive_pages_live(head,pages): return {}
	return {"owner":result,"head":wrapped.head,"pair":pair,"pages":pages}

static func _base_archive_draft_image(owner: RefCounted, minimum_picks: int, human: bool=false) -> bool:
	if owner==null or owner.active!="ordinary" or owner.lifecycle_status()!="active" or owner.session==null:
		return false
	var s: SeasonState=owner.session
	var count: int=s.picks.size()
	return count>=minimum_picks and count<=4 and s.round_index==0 \
		and s.phase==(SeasonState.Phase.DRAFT if count<4 else SeasonState.Phase.REGULAR) \
		and s.results.is_empty() and s.player_results.is_empty() and s.physical.pending.is_empty()!=human \
		and s.physical.reports.is_empty() and s.physical.cursor==0 and not s.build.pack_pending()

static func _base_archive_human_image(owner: RefCounted) -> bool:
	if not _base_archive_draft_image(owner,4,true): return false
	var s: SeasonState=owner.session
	var pending: Dictionary=s.physical.pending
	return Codec.keys(pending,["human","reports"]) and pending.reports==[] \
		and s.pending_fixture().get("id",-1)==0 and Adapter.OrdinaryPending.shape(s)

# Stock regular phase and fixture define the next managed game. The
# qualification still loads only the previous admitted settled/Shop
# generations; these predicates do not give authority to other saved roles.
static func _base_archive_paid_regular(owner: RefCounted, allow_closed: bool=false) -> bool:
	if owner==null or owner.active!="ordinary" or owner.session==null \
			or (owner.lifecycle_status()!="active" and (not allow_closed or owner.lifecycle_status()!="closed")):
		return false
	var s: SeasonState=owner.session
	var n: int=s.round_index
	if not Adapter.scope("ordinary",s) or s.phase!=SeasonState.Phase.REGULAR or n<1 or s.picks.size()!=4 \
			or s.league_rules.get("league")!="dugout" or s.league_rules.get("tier")!=0 \
			or s.results.size()!=3*n or s.player_results.size()!=n or s.physical.reports.size()!=2*n \
			or s.physical.cursor!=s.physical.reports.size() or not s.physical.needed.is_empty() \
			or not s.physical.error.is_empty() or s.physical.projecting or s.build.pack_pending(): return false
	var fixture: Dictionary=s.pending_fixture()
	return not fixture.is_empty() and fixture.get("round")==n \
		and (fixture.get("home",-1)==0 or fixture.get("away",-1)==0)

static func _base_archive_managed_ready(owner: RefCounted) -> bool:
	return _base_archive_final_ready(owner) or _base_archive_semifinal_ready(owner) or _base_archive_draft_image(owner,4) or (_base_archive_paid_regular(owner) \
		and owner.session.physical.pending.is_empty() and (_base_archive_last_regular(owner) \
			or _base_archive_has_later_regular(owner.session)))

static func _base_archive_next_human_image(owner: RefCounted) -> bool:
	if not _base_archive_paid_regular(owner): return false
	var s: SeasonState=owner.session
	var pending: Dictionary=s.physical.pending
	return Codec.keys(pending,["human","reports"]) and pending.reports==[] and pending.human is Array \
		and pending.human.size()==13 and pending.human[0]==s.pending_fixture().get("id") \
		and Adapter.OrdinaryPending.shape(s)

static func _base_archive_related_role(generation: String) -> bool:
	return generation in [BASE_ARCHIVE_NEXT_HUMAN_GENERATION,BASE_ARCHIVE_SHOP_GENERATION,
		BASE_ARCHIVE_SETTLED_GENERATION,BASE_ARCHIVE_REGULAR_REPORT_GENERATION,BASE_ARCHIVE_REGULAR_SETTLED_GENERATION,
		BASE_ARCHIVE_ABANDONED_GENERATION,BASE_ARCHIVE_PREPARED_GENERATION,BASE_ARCHIVE_NEXT_BIRTH_GENERATION,
		BASE_ARCHIVE_LAST_REGULAR_REPORT_GENERATION,BASE_ARCHIVE_POSTSEASON_GENERATION,
		BASE_ARCHIVE_SEMI_HUMAN_GENERATION,BASE_ARCHIVE_SEMI_REPORT_GENERATION,BASE_ARCHIVE_SEMI_SETTLED_GENERATION,
		BASE_ARCHIVE_COMPLETED_GENERATION,BASE_ARCHIVE_COMPLETED_PREPARED_GENERATION,
		BASE_ARCHIVE_FINAL_HUMAN_GENERATION,BASE_ARCHIVE_FINAL_SETTLED_GENERATION,BASE_ARCHIVE_FINAL_CLOSED_GENERATION]

# A rotating sibling is evidence of the last single native offer, not proof
# of the archive. The caller has already restored the primary/page strictly.
static func _base_archive_marked_previous_matches(bytes: PackedByteArray, wrapped: Dictionary, current: RefCounted,
		head_bytes: PackedByteArray, pages: Array[PackedByteArray], related: RefCounted) -> bool:
	var prior: Dictionary=_base_archive_wrapper(bytes)
	if prior.is_empty(): return false
	if wrapped.generation==BASE_ARCHIVE_NEXT_BIRTH_GENERATION:
		return _base_archive_read_lifecycle_sibling(prior,wrapped,head_bytes,pages,related)
	if not Codec.same(prior.head,wrapped.head): return false
	if wrapped.generation in [BASE_ARCHIVE_ABANDONED_GENERATION,BASE_ARCHIVE_PREPARED_GENERATION,
			BASE_ARCHIVE_COMPLETED_GENERATION,BASE_ARCHIVE_COMPLETED_PREPARED_GENERATION]:
		return _base_archive_read_lifecycle_sibling(prior,wrapped,head_bytes,pages,related)
	if wrapped.generation in [BASE_ARCHIVE_FINAL_HUMAN_GENERATION,BASE_ARCHIVE_FINAL_SETTLED_GENERATION,
			BASE_ARCHIVE_FINAL_CLOSED_GENERATION]:
		return _base_archive_final_sibling(prior,wrapped,current,related)
	if wrapped.generation in [BASE_ARCHIVE_SEMI_HUMAN_GENERATION,BASE_ARCHIVE_SEMI_REPORT_GENERATION,
			BASE_ARCHIVE_SEMI_SETTLED_GENERATION]:
		return _base_archive_semifinal_sibling(prior,wrapped,current,related)
	if wrapped.generation==BASE_ARCHIVE_NEXT_HUMAN_GENERATION:
		if prior.get("generation") not in [BASE_ARCHIVE_SETTLED_GENERATION,BASE_ARCHIVE_SHOP_GENERATION,
				BASE_ARCHIVE_REGULAR_SETTLED_GENERATION]: return false
		var checkpoint: RefCounted=related
		if not _base_archive_paid_regular(checkpoint) or not checkpoint.session.physical.pending.is_empty() \
				or not Codec.same(checkpoint.session.pending_fixture(),current.session.pending_fixture()): return false
		if prior.generation==BASE_ARCHIVE_SETTLED_GENERATION and not _base_archive_unshopped_settled_image(checkpoint):
			return false
		if prior.generation==BASE_ARCHIVE_SHOP_GENERATION and not _base_archive_shop_image(checkpoint): return false
		if prior.generation==BASE_ARCHIVE_REGULAR_SETTLED_GENERATION and not _base_archive_regular_settled_image(checkpoint):
			return false
		# The primary's H13 already passed strict native projection. Verify it
		# is the only difference from the exact independently restored sibling.
		checkpoint.session.physical.pending=current.session.physical.pending.duplicate(true)
		return Codec.same(Owner.capture(checkpoint),wrapped.current)
	if wrapped.generation in [BASE_ARCHIVE_REGULAR_REPORT_GENERATION,BASE_ARCHIVE_REGULAR_SETTLED_GENERATION,
			BASE_ARCHIVE_LAST_REGULAR_REPORT_GENERATION,BASE_ARCHIVE_POSTSEASON_GENERATION]:
		return _base_archive_regular_sibling(prior,wrapped,current,related)
	if wrapped.generation==BASE_ARCHIVE_SHOP_GENERATION:
		if prior.get("generation") not in [BASE_ARCHIVE_SETTLED_GENERATION,BASE_ARCHIVE_SHOP_GENERATION]: return false
		var checkpoint: RefCounted=related
		if not _base_archive_shop_scope(checkpoint): return false
		if prior.generation==BASE_ARCHIVE_SETTLED_GENERATION and not _base_archive_unshopped_settled_image(checkpoint):
			return false
		if prior.generation==BASE_ARCHIVE_SHOP_GENERATION and not _base_archive_shop_image(checkpoint): return false
		var before: Array=checkpoint.session.build._events
		var after: Array=current.session.build._events
		if after.size()!=before.size()+1 or not Codec.same(after.slice(0,before.size()),before): return false
		var operation: RefCounted=_base_archive_shop_candidate(checkpoint,after.back())
		return operation!=null and Codec.same(Owner.capture(operation),wrapped.current)
	if wrapped.generation==BASE_ARCHIVE_SETTLED_GENERATION:
		if prior.get("generation")!=BASE_ARCHIVE_REPORT_GENERATION: return false
		# Establish the exact typed old checkpoint from the same authenticated
		# head/pages, not from the plain V4 snapshot hydrator. Neither the
		# sibling nor its type restoration supplies authority to the primary.
		var checkpoint: RefCounted=related
		if _base_archive_opening_report_count(checkpoint)!=2: return false
		var stock: Dictionary=Adapter.OrdinaryPending.project(checkpoint.session)
		if stock.has("error") or stock.has("fixture") or not stock.get("candidate") is SeasonState: return false
		checkpoint.session=stock.candidate
		return _base_archive_settled_image(checkpoint) and Codec.same(Owner.capture(checkpoint),wrapped.current)
	var previous: RefCounted=Owner.restore(prior.current)
	if wrapped.generation==BASE_ARCHIVE_REPORT_GENERATION:
		var old_count: int=_base_archive_opening_report_count(previous)
		if old_count<0 or _base_archive_opening_report_count(current)!=old_count+1 \
				or prior.get("generation")!=(BASE_ARCHIVE_HUMAN_GENERATION if old_count==0 else BASE_ARCHIVE_REPORT_GENERATION):
			return false
		var next_pending: Dictionary=current.session.physical.pending
		if not Codec.same(previous.session.physical.pending.human,next_pending.human) \
				or not Codec.same(previous.session.physical.pending.reports,next_pending.reports.slice(0,old_count)):
			return false
		previous.session.physical.pending=next_pending.duplicate(true)
		return Codec.same(Owner.capture(previous),wrapped.current)
	var prior_birth: bool=not prior.has("generation") or prior.generation==BASE_ARCHIVE_NEXT_BIRTH_GENERATION
	if prior.get("generation")==BASE_ARCHIVE_NEXT_BIRTH_GENERATION and pages.size()<2: return false
	if not _base_archive_draft_image(previous,0 if prior_birth else 1): return false
	var count: int=previous.session.picks.size()
	if wrapped.generation==BASE_ARCHIVE_HUMAN_GENERATION:
		if prior.get("generation")!=BASE_ARCHIVE_DRAFT_GENERATION or count!=4: return false
		previous.session.physical.pending=current.session.physical.pending.duplicate(true)
		return Codec.same(Owner.capture(previous),wrapped.current)
	if (prior_birth and count!=0) or count+1!=current.session.picks.size(): return false
	var id: String=current.session.picks.back()
	return previous.session.offers().has(id) and previous.session.choose_player(id) \
		and Codec.same(Owner.capture(previous),wrapped.current)

static func _base_archive_draft_page_live(pin: String, page: PackedByteArray) -> bool:
	var path: String=_base_archive_page_path(pin)
	return not path.is_empty() and _base_archive_page_root(false) and _base_archive_absent(path+".tmp") \
		and _base_archive_member(path).get("bytes",PackedByteArray())==page

static func _base_archive_draft_pin_live(pin: Dictionary) -> bool:
	return not pin.is_empty() and _base_archive_qa() and _base_archive_temps_clear() \
		and Codec.same(_base_archive_pair(),pin.pair) and pin.owner.session==pin.session \
		and RawGuard.matches(pin.owner,pin.snapshot) and _base_archive_pages_live(pin.head,pin.pages)

# Detached option only; no proposal is stored and no new writer or public
# difficulty is admitted before that same source can be preserved at rollover.
static func earned_dugout_d1_offer(owner: RefCounted, expected: PackedByteArray) -> Dictionary:
	if not base_archive_app_binding(owner,expected) or not _base_archive_final_closed(owner): return {}
	var wrapped:=_base_archive_wrapper(expected)
	if wrapped.get("generation")!=BASE_ARCHIVE_FINAL_CLOSED_GENERATION: return {}
	var offer:=preload("earned_dugout_d1.gd").from_closed_image(wrapped.current,wrapped.head)
	return offer if not offer.is_empty() and base_archive_app_binding(owner,expected) else {}

# Kept off both the normal Store writer and the public launch route.
# Native installed-only read. The pure title stage remains detached and
# never serves as a Store ticket or a caller-supplied unlock assertion.
static func inspect_installed_earned_dugout_d1(owner: RefCounted, expected: PackedByteArray) -> Dictionary:
	if not OS.has_feature(LeagueAdmission.CAREER_FORWARD_FEATURE) \
			or not LeagueAdmission.installed_career_owned() or not installed_owner_shape(owner) \
			or owner.lifecycle_status()!="closed" or owner.lifecycle_reason()!="complete" \
			or owner.session.phase!=SeasonState.Phase.COMPLETE or owner.session.champion!=0 \
			or not match_owner(owner,expected): return {}
	var earned=preload("earned_dugout_d1.gd")
	var source: Dictionary=earned.first_installed_source(owner)
	if source.is_empty(): return {}
	var offer: Dictionary=earned.from_closed_image(source.image,source.head)
	return offer if not offer.is_empty() and match_owner(owner,expected) else {}

# A new installed-only save shape, distinct from both ordinary Base and
# the private legacy archive format. A head shape alone confers no authority.
static func _installed_dugout_d1_parts(value: Variant) -> Dictionary:
	if not Codec.keys(value,["kind","version","head","current"]) \
			or not value.kind is String or value.kind!=INSTALLED_DUGOUT_D1_KIND or not Codec.whole(value.version,1,3) \
			or not value.head is Dictionary or not value.current is Dictionary \
			or not value.head.get("kind") is String or not value.head.get("league") is String \
			or not Codec.whole(value.head.get("version"),1,1): return {}
	var closed: bool=value.version==2
	var tier: int=2 if value.version==3 else 1
	var ordinal: int=tier+1
	var head_bytes: PackedByteArray=BaseArchive._wire(value.head)
	var head:=BaseArchive._head(head_bytes)
	var current: Dictionary=value.current
	if head.is_empty() or head.ordinal!=ordinal or head.pages.size()!=tier \
			or not current.get("active") is String or current.get("active")!="ordinary" or not current.get("ordinary_state") is Dictionary \
			or not Codec.whole(current.ordinary_state.get("ordinal"),ordinal,ordinal) \
			or not current.ordinary_state.get("status") is String or current.ordinary_state.get("status")!=("closed" if closed else "active") \
			or not current.get("ordinary") is Dictionary or not current.ordinary.get("career") is Dictionary: return {}
	var basis: Variant=current.ordinary.career.get("start_basis")
	var starts: Script=Adapter.OrdinaryRuntime.CommonStart
	if not basis is Dictionary or not starts.valid(basis) or basis.ordinal!=ordinal \
			or (not starts.detached_earned_dugout_d2(basis) if tier==2 else not starts.detached_earned_dugout_d1(basis)) \
			or not Codec.same(basis.archive_origin.ordered_sha256,head.pages) \
			or basis.archive_origin.offer.target_head_sha256!=BaseArchive._hash(BaseArchive._wire(head)): return {}
	var pin: String=head.pages[0] if tier==1 else BaseArchive._hash(BaseArchive._wire(head))
	return {"head":head,"current":current,"closed":closed,"tier":tier,"pin":pin}

static func _installed_dugout_d1_lane() -> bool:
	return OS.has_feature(LeagueAdmission.CAREER_FORWARD_FEATURE) \
		and LeagueAdmission.installed_career_owned() and not _installed_paths_blocked() \
		and not FileAccess.file_exists(INSTALLED_PATH+".tmp") \
		and not FileAccess.file_exists(INSTALLED_PATH+INSTALLED_BACKUP_STAGE)

static func _installed_dugout_d1_page_root(create: bool) -> bool:
	var dir:=DirAccess.open("user://")
	if dir==null or dir.is_link(INSTALLED_DUGOUT_D1_PAGES.get_file()) \
			or FileAccess.file_exists(INSTALLED_DUGOUT_D1_PAGES): return false
	if not DirAccess.dir_exists_absolute(INSTALLED_DUGOUT_D1_PAGES):
		if not create or DirAccess.make_dir_absolute(INSTALLED_DUGOUT_D1_PAGES)!=OK: return false
	return not dir.is_link(INSTALLED_DUGOUT_D1_PAGES.get_file()) and DirAccess.open(INSTALLED_DUGOUT_D1_PAGES)!=null

static func _installed_dugout_d1_page_path(id: String) -> String:
	return INSTALLED_DUGOUT_D1_PAGES.path_join(id+".json") if BaseArchive._digest(id) else ""

# Only the bound installed transaction calls this. Publish an immutable
# exact page before the mutable head. An interruption can leave an orphan
# page, never a new head whose required page has not passed full readback.
static func _keep_installed_dugout_d1_page(page: PackedByteArray) -> bool:
	if not saving or page.is_empty() or page.size()>BaseArchive.MAX_PAGE \
			or not _installed_dugout_d1_lane() or not _installed_dugout_d1_page_root(true): return false
	var path:=_installed_dugout_d1_page_path(BaseArchive._hash(page))
	var tmp:=path+".tmp"
	if path.is_empty() or not _base_archive_absent(tmp): return false
	var prior:=_base_archive_member(path)
	if prior.is_empty(): return false
	if prior.present: return prior.bytes==page
	var file:=FileAccess.open(tmp,FileAccess.WRITE)
	if file==null: return false
	file.store_buffer(page); file.flush()
	var ok: bool=file.get_error()==OK
	file.close()
	if not ok or _base_archive_member(tmp).get("bytes",PackedByteArray())!=page \
			or not _base_archive_absent(path) or DirAccess.rename_absolute(tmp,path)!=OK: return false
	return _base_archive_member(path).get("bytes",PackedByteArray())==page

# Only the dedicated installed draft view uses this. Never authorize a new
# installed head from a file-supplied Champion assertion.
static func write_installed_dugout_d1_birth(owner: RefCounted, expected: PackedByteArray) -> bool:
	if saving or _ticket!=null or not _installed_dugout_d1_lane(): return false
	var offer:=inspect_installed_earned_dugout_d1(owner,expected)
	if offer.is_empty(): return false
	var earned=preload("earned_dugout_d1.gd")
	var source: Dictionary=earned.first_installed_source(owner)
	if source.is_empty() or not match_owner(owner,expected): return false
	var staged: Dictionary=earned.stage_birth(source.image,source.head,offer)
	if staged.is_empty(): return false
	var value: Dictionary={"kind":INSTALLED_DUGOUT_D1_KIND,"version":1,"head":staged.head,"current":staged.current}
	if _installed_dugout_d1_parts(value).is_empty(): return false
	var draft:=_installed_dugout_d1_local_draft(staged.current.ordinary)
	var body:=_native_encoded(value)
	if draft==null or body.is_empty() or not match_owner(owner,expected): return false
	saving=true
	var wrote_page:=_keep_installed_dugout_d1_page(staged.page)
	var saved: bool=wrote_page and _installed_dugout_d1_lane() and _matches(expected) and _write_exact_bytes(body,expected)
	saving=false
	if saved:
		_invalidate()
		_installed_dugout_d1_bound={"bytes":body,"page":staged.head.pages[0],"draft":draft}
	return saved

# Fixed read-only child. It cannot enter raw(), any writer, or become
# the installed owner. Only the named bootstrap dispatches this worker.
static func _installed_dugout_d1_read_worker() -> bool:
	return OS.has_feature(LeagueAdmission.CAREER_FORWARD_FEATURE) \
		and LeagueAdmission.installed_career_worker()

# Snapshot only this fixed revision and its one immutable page. Never fall
# back to .bak, an unbound path, or another saved-career dialect.
static func _installed_dugout_d1_source(worker: bool=false) -> Dictionary:
	if saving or _ticket!=null: return {}
	if worker:
		if not _installed_dugout_d1_read_worker() or _installed_paths_blocked() \
			or not _base_archive_absent(INSTALLED_PATH+".tmp") \
			or not _base_archive_absent(INSTALLED_PATH+INSTALLED_BACKUP_STAGE): return {}
	elif not _installed_dugout_d1_lane(): return {}
	var stored:=_base_archive_member(INSTALLED_PATH)
	if stored.is_empty() or not stored.present: return {}
	var before:=_installed_payload(stored.bytes)
	var parts:=_installed_dugout_d1_parts(Owner._bytes_value(before))
	if parts.is_empty() or not _installed_dugout_d1_page_root(false): return {}
	var pages: Array[PackedByteArray]=[]
	for id: String in parts.head.pages:
		var path:=_installed_dugout_d1_page_path(id)
		if path.is_empty() or not _base_archive_absent(path+".tmp"): return {}
		var member:=_base_archive_member(path)
		if member.is_empty() or not member.present or BaseArchive._hash(member.bytes)!=id: return {}
		pages.append(member.bytes)
	return {"bytes":before,"head":parts.head,"current":parts.current,"pages":pages,"closed":parts.closed,
		"tier":parts.tier,"pin":parts.pin}

static func _restore_installed_dugout_d1_source(source: Dictionary) -> SeasonState:
	if source.is_empty(): return null
	var pages: Array[PackedByteArray]=source.pages
	var head: PackedByteArray=BaseArchive._wire(source.head)
	if source.tier==2: return preload("earned_dugout_d2.gd").draft_from_ordered_pages(head,pages,source.current)
	var earned: Script=preload("earned_dugout_d1.gd")
	if source.closed: return earned.restore_completed_from_pages(head,pages,source.current)
	return earned.restore_draft_from_pages(head,pages,source.current,true)

static func read_installed_dugout_d1_draft() -> Dictionary:
	var source:=_installed_dugout_d1_source()
	var restored:=_restore_installed_dugout_d1_source(source)
	if restored==null: return {}
	var again:=_installed_dugout_d1_source()
	if again.is_empty() or again.bytes!=source.bytes or again.pages!=source.pages: return {}
	return {"bytes":source.bytes,"head":source.head,"current":source.current,"session":restored}

# Only the no-game native draft dialect may use the warm draft binding.
# A held view never carries a mutable SeasonState or grants general play.
static func _installed_dugout_d1_local_draft(image: Dictionary) -> SeasonState:
	var draft: SeasonState=Adapter.Ordinary.restore_common(image)
	if draft==null or not Codec.same(draft.league_rules,LeagueRules.capture_tier("dugout",1)) \
			or not Adapter.OrdinaryRuntime.CommonStart.detached_earned_dugout_d1(draft.career.start_basis) \
			or draft.round_index!=0 or not draft.results.is_empty() or not draft.player_results.is_empty() \
			or not draft.physical.pending.is_empty() or not draft.physical.reports.is_empty() \
			or not Codec.same(Adapter.Ordinary.capture(draft),image): return null
	if draft.phase==SeasonState.Phase.DRAFT and draft.picks.size()<4: return draft
	return draft if draft.phase==SeasonState.Phase.REGULAR and draft.picks.size()==4 else null

static func _installed_dugout_d1_regular_limit() -> int:
	return preload("earned_dugout_d1.gd").LAST_INSTALLED_REGULAR_ROUND

# Native ordinary regular season has one managed fixture followed by two CPU
# fixtures per round. This checks chronology, never grants write authority.
static func _installed_dugout_d1_history_shape(session: SeasonState) -> bool:
	if session==null or session.phase!=SeasonState.Phase.REGULAR or session.picks.size()!=4: return false
	var played: int=session.round_index
	return played>=0 and played<=_installed_dugout_d1_regular_limit() \
		and session.pending_fixture().get("id")==played*3 and session.results.size()==played*3 \
		and session.player_results.size()==played and session.physical.reports.size()==played*2

# Hydrate a current image only after the revision-bound child replays the
# championship and opening. This parent shape check grants no replay authority.
static func _installed_dugout_d1_verified_current(image: Dictionary) -> SeasonState:
	var current:=_installed_dugout_d1_local_draft(image)
	if current!=null: return current
	current=Adapter.Ordinary.restore_common(image)
	if current==null or current.picks.size()!=4 \
			or not Codec.same(current.league_rules,LeagueRules.capture_tier("dugout",1)) \
			or not Adapter.OrdinaryRuntime.CommonStart.detached_earned_dugout_d1(current.career.start_basis) \
			or not Codec.same(Adapter.Ordinary.capture(current),image): return null
	var earned: Script=preload("earned_dugout_d1.gd")
	if not earned.final_regular_destination(current).is_empty() or earned.semifinal_pending(current) \
			or not earned.settled_semifinal(current).is_empty() or earned.managed_final_pending(current) \
			or not earned.managed_final_destination(current).is_empty(): return current
	if not _installed_dugout_d1_history_shape(current): return null
	var pending: Dictionary=current.physical.pending
	if pending.is_empty(): return current if current.round_index>0 else null
	if current.round_index>=_installed_dugout_d1_regular_limit(): return null
	return current if Codec.keys(pending,["human","reports"]) and pending.human is Array \
		and pending.human.size()==13 and pending.human[0]==current.round_index*3 \
		and pending.reports is Array and pending.reports.size()<=preload("earned_dugout_d1.gd").regular_report_cap(current.round_index) else null


# Keep the request number types from the actual authenticated in-memory
# producer when it becomes the next warm save binding. Ordinary's materialized
# image alone cannot retain them, even before a disk round trip.
static func _installed_dugout_d1_rebind(image: Dictionary, producer: SeasonState) -> SeasonState:
	var is_d2: bool=Adapter.OrdinaryRuntime.CommonStart.detached_earned_dugout_d2(producer.career.start_basis)
	var next:=_installed_dugout_d2_local_draft(image) if is_d2 else _installed_dugout_d1_verified_current(image)
	if next==null: return null
	var earned: Script=preload("earned_dugout_d1.gd")
	if not earned.needs_wallet_types(next): return next
	if not earned.restore_wallet_types(next,earned.wallet_types(producer)) \
			or not Codec.same(Adapter.Ordinary.capture(next),image): return null
	if not next.physical.pending.is_empty() and _installed_dugout_d1_project_pending(next).is_empty(): return null
	return next


static func _bound_installed_dugout_d1(expected: PackedByteArray) -> Dictionary:
	if _installed_dugout_d1_bound.is_empty() or expected.is_empty() \
			or _installed_dugout_d1_bound.bytes!=expected: return {}
	var source:=_installed_dugout_d1_source()
	if source.is_empty() or source.bytes!=expected or source.pin!=_installed_dugout_d1_bound.page: return {}
	var draft: SeasonState=_installed_dugout_d1_bound.draft
	if draft==null or not Codec.same(Adapter.Ordinary.capture(draft),source.current.ordinary): return {}
	return source

static func _installed_dugout_d2_local_draft(image: Dictionary) -> SeasonState:
	var draft: SeasonState=Adapter.Ordinary.restore_common(image)
	var d2: Script=preload("earned_dugout_d2.gd")
	if (not d2.draft_boundary(draft) and d2.regular_boundary(draft).is_empty()) \
			or not Codec.same(Adapter.Ordinary.capture(draft),image): return null
	return draft

static func installed_dugout_d1_view(expected: PackedByteArray) -> Dictionary:
	var source:=_bound_installed_dugout_d1(expected)
	if source.is_empty(): return {}
	var draft: SeasonState=_installed_dugout_d1_bound.draft
	if source.tier==2:
		var d2_state: String="draft" if draft.picks.size()<4 else preload("earned_dugout_d2.gd").regular_boundary(draft)
		if d2_state.is_empty(): return {}
		var earned_d1: Script=preload("earned_dugout_d1.gd")
		var finish_d2: String=earned_d1.final_regular_destination(draft)
		var count_d2: int=draft.physical.pending.reports.size() if d2_state=="unpaid" else 0
		var total_d2: int=2
		if d2_state=="unpaid" and draft.round_index==_installed_dugout_d1_regular_limit()-1 and count_d2>=2:
			var pending_d2:=_installed_dugout_d1_pending(expected)
			if pending_d2.is_empty(): return {}
			if pending_d2.projection.has("fixture") or count_d2>2: total_d2=5
		var d2_scores: Array=[]
		if d2_state=="paid":
			var start_d2: int=draft.results.size()-3 if finish_d2.is_empty() else 27
			for row: Dictionary in draft.results.slice(start_d2):
				d2_scores.append([int(row.id),int(row.away_runs),int(row.home_runs)])
		return {"bytes":expected.duplicate(),"tier":2,"picks":draft.picks.duplicate(),"offers":draft.offers(),
			"credits":int(source.current.common.aggregate.currency),"cash":draft.cash(),"state":d2_state,"closed":false,
			"cpu_reports":count_d2,"cpu_total":total_d2,"finish":finish_d2,
			"scores":d2_scores,"shop_available":d2_state=="paid" and draft.shop_available(),
			"round":draft.round_index+int(d2_state!="paid"),
			"next_available":preload("earned_dugout_d2.gd").next_regular_available(draft)}
	var state: String="draft"
	if draft.picks.size()==4:
		if not draft.physical.pending.is_empty(): state="unpaid"
		else: state="ready" if draft.round_index==0 else "paid"
	var earned: Script=preload("earned_dugout_d1.gd")
	if earned.semifinal_pending(draft): state="semifinal_unpaid"
	elif earned.managed_final_pending(draft): state="final_unpaid"
	var reports: int=draft.physical.pending.reports.size() if state in ["unpaid","semifinal_unpaid"] else 0
	var finish: String=earned.final_regular_destination(draft)
	if finish.is_empty(): finish=earned.settled_semifinal(draft)
	if finish.is_empty(): finish=earned.managed_final_destination(draft)
	var completion: Dictionary={}
	if source.closed:
		if state!="paid" or earned.close_destination(draft).is_empty(): return {}
		completion=source.current.ordinary_state.last_close.receipt.duplicate(true)
	var cpu_total: int=earned.semifinal_jobs(draft) if state=="semifinal_unpaid" else (0 if state=="final_unpaid" else 2)
	if state=="unpaid" and draft.round_index==_installed_dugout_d1_regular_limit()-1 and reports>=2:
		var plan:=_installed_dugout_d1_pending(expected)
		if plan.is_empty(): return {}
		if plan.projection.has("fixture") or reports>2: cpu_total=5
	var scores: Array=[]
	var playoff_club_id: int=-1
	if state=="paid":
		var settled_semi: bool=finish in ["semifinal_exit","final_ready","final_champion","final_runner_up"]
		if settled_semi: playoff_club_id=int(draft.player_results[10].id)
		var start: int=30 if settled_semi else (27 if not finish.is_empty() else draft.results.size()-3)
		for row: Dictionary in draft.results.slice(start):
			scores.append([int(row.id),int(row.away_runs),int(row.home_runs)])
	return {"bytes":expected.duplicate(),"picks":draft.picks.duplicate(),"offers":draft.offers(),
		"credits":int(source.current.common.aggregate.currency),"cash":draft.cash(),"state":state,
		"cpu_reports":reports,"cpu_total":cpu_total,"scores":scores,"round":draft.round_index if state in ["paid","semifinal_unpaid","final_unpaid"] else draft.round_index+1,
		"next_available":draft.round_index<_installed_dugout_d1_regular_limit(),"finish":finish,"playoff_club_id":playoff_club_id,
		"shop_available":state=="paid" and not source.closed and draft.shop_available(),"closed":source.closed,
		"close_available":not source.closed and state=="paid" and not earned.close_destination(draft).is_empty(),
		"completion":completion,"tier":1,
		"start_d2_available":source.closed and finish=="final_champion" and completion.get("finish")=="champion"}

static func pick_installed_dugout_d1(expected: PackedByteArray,id: String) -> Dictionary:

	if source.closed or old.phase!=SeasonState.Phase.DRAFT or old.picks.size()>=4 or not old.offers().has(id): return {}
	var candidate: SeasonState=old.fork_live()

			or candidate.picks.back()!=id or not Codec.same(old.picks,candidate.picks.slice(0,-1)): return {}
	var recorded: Dictionary=Adapter.Ordinary.capture(candidate)
	if recorded.is_empty(): return {}
	var current: Dictionary=source.current.duplicate(true)
	current.ordinary=recorded
	var value: Dictionary={"kind":INSTALLED_DUGOUT_D1_KIND,"version":3 if source.tier==2 else 1,"head":source.head,"current":current}
	var native:=_installed_dugout_d2_local_draft(recorded) if source.tier==2 else _installed_dugout_d1_local_draft(recorded)
	if _installed_dugout_d1_parts(value).is_empty() or native==null: return {}
	var body:=_native_encoded(value)
	var again:=_installed_dugout_d1_source()
	if body.is_empty() or again.is_empty() or again.bytes!=expected or again.pages!=source.pages: return {}
	saving=true
	var saved: bool=_write_exact_bytes(body,expected)
	saving=false
	if not saved: return {}
	_invalidate()
	_installed_dugout_d1_bound={"bytes":body,"page":source.pin,"draft":candidate}
	return installed_dugout_d1_view(body)

# Only the exact installed App can open the authored opening fixture. The
# lease holds the native match and its one lab, not a caller-provided score.
static func open_installed_dugout_d1_match(app: Variant) -> bool:
	if not is_instance_valid(app) or app.get_script()!=load("res://src/presentation/season/successor_dynasty_owner_v1/app.gd") \
			or app.career_owner!=null or app.season!=null or is_instance_valid(app.lab): return false
	var expected: PackedByteArray=app.saved_bytes
	var source:=_bound_installed_dugout_d1(expected)
	if source.is_empty(): return false
	var bound: SeasonState=_installed_dugout_d1_bound.draft
	if source.closed or not bound.physical.pending.is_empty(): return false
	if source.tier==2 and not preload("earned_dugout_d2.gd").next_regular_available(bound): return false
	var earned: Script=preload("earned_dugout_d1.gd")
	var semifinal: bool=earned.final_regular_destination(bound)=="semifinal"
	var managed_final: bool=earned.settled_semifinal(bound)=="final_ready"
	if not semifinal and not managed_final and (not _installed_dugout_d1_history_shape(bound) or bound.round_index>=_installed_dugout_d1_regular_limit()):
		return false
	var live: SeasonState=bound.fork_live()
	if live==null: return false
	var fixture: Dictionary=live.pending_fixture()
	if semifinal:
		var pair: Dictionary=earned.semifinal_pair(live)
		if pair.is_empty() or not Codec.same(pair.club,fixture): return false
	elif managed_final:
		if fixture.get("id")!=32 or fixture.get("round")!=11 or fixture.get("neutral")!=true \
				or (fixture.get("away")!=0 and fixture.get("home")!=0): return false
	elif fixture.get("id",-1)!=live.round_index*3 or (fixture.get("away",-1)!=0 and fixture.get("home",-1)!=0): return false
	var state: MatchState=Adapter.OrdinaryFixtures.match_for(live,fixture)
	var field: FieldDefinition=SeasonState.field_for_fixture(fixture,live)
	if state==null or field==null or not state.stadium.is_empty() or not state.sure_shot.enable_current_evidence(): return false
	app.season=live
	app._fixture_id=int(fixture.id)
	app._result_recorded=false
	app._result_saved=false
	app._open_match(state,int(fixture.home)==0,true,field.id)
	if not is_instance_valid(app.lab) or app.lab._match_state!=state or app.season!=live: return false
	_installed_dugout_d1_play={"app":weakref(app),"bytes":expected.duplicate(),"season":live,"state":state,"lab":weakref(app.lab),"field":field,"fixture":int(fixture.id)}
	return installed_dugout_d1_match_bound(app)

static func installed_dugout_d1_match_bound(app: Variant) -> bool:
	var bind: Dictionary=_installed_dugout_d1_play
	if bind.is_empty() or not is_instance_valid(app) or bind.app.get_ref()!=app or bind.bytes!=app.saved_bytes \
			or app.career_owner!=null or app.season!=bind.season or app._fixture_id!=bind.fixture \
			or not is_instance_valid(app.lab) or bind.lab.get_ref()!=app.lab: return false
	var lab: PitchBatLab=app.lab
	var season: SeasonState=bind.season
	return lab._match_state==bind.state and lab._field_definition==bind.field and lab._field_id==bind.field.id \
		and season.pending_fixture().get("id")==bind.fixture and lab._player_home==(int(season.pending_fixture().get("home",-1))==0) \
		and ClubCareer.same(lab._match_state.league_rules,season.league_rules) \
		and lab._match_state.pitch_catalog==season.build.pitch_catalog_signature() \
		and lab._match_state.stadium.is_empty() and season.physical.pending.is_empty()

static func save_installed_dugout_d1_managed(app: Variant) -> Dictionary:
	if not installed_dugout_d1_match_bound(app): return {}
	var expected: PackedByteArray=app.saved_bytes
	var source:=_bound_installed_dugout_d1(expected)
	if source.is_empty(): return {}
	var previous: SeasonState=_installed_dugout_d1_bound.draft
	if not Codec.same(Adapter.Ordinary.capture(app.season),source.current.ordinary): return {}
	var human: Array=Adapter.OrdinaryCapture._capture_bound(app,func(): return installed_dugout_d1_match_bound(app))
	if human.size()!=13 or not installed_dugout_d1_match_bound(app): return {}
	var candidate: SeasonState=previous.fork_live()
	if candidate==null: return {}
	candidate.physical.pending={"human":human.duplicate(true),"reports":[]}
	var checked: Dictionary={}
	var managed_final: bool=candidate.phase==SeasonState.Phase.FINAL
	if managed_final: checked=_installed_dugout_d1_final_projection(candidate)
	elif candidate.phase==SeasonState.Phase.SEMIFINAL: checked=_installed_dugout_d1_semifinal_projection(candidate)
	else: checked=BaseArchive.Core.MixedProof._project_mixed_round(candidate)
	if checked.is_empty(): return {}
	var projection: Dictionary=checked.projection
	if projection.has("error") or projection.has("fixture")!=(not managed_final) or not checked.get("source") is SeasonState: return {}
	var recorded: Dictionary=Adapter.Ordinary.capture(checked.source)
	var native:=_installed_dugout_d1_rebind(recorded,checked.source)
	if native==null or not native.physical.pending.reports.is_empty() \
			or not Codec.same(recorded.state.results,source.current.ordinary.state.results) \
			or not Codec.same(recorded.state.player_results,source.current.ordinary.state.player_results) \
			or not Codec.same(recorded.physical.reports,source.current.ordinary.physical.reports) \
			or not Codec.same(recorded.build,source.current.ordinary.build): return {}
	if not installed_dugout_d1_match_bound(app): return {}
	return _write_installed_dugout_d1_opening(source,expected,recorded,native)

# No command formats enter here. Each fixed producer has already proved the
# exact gameplay transition and independently materialized its canonical save.
static func _write_installed_dugout_d1_opening(source: Dictionary,expected: PackedByteArray,recorded: Dictionary,native: SeasonState) -> Dictionary:
	if source.closed or native==null or not Codec.same(Adapter.Ordinary.capture(native),recorded): return {}
	if source.tier==2 and preload("earned_dugout_d2.gd").regular_boundary(native) not in ["unpaid","paid"]: return {}
	var current: Dictionary=source.current.duplicate(true)
	current.ordinary=recorded
	var value: Dictionary={"kind":INSTALLED_DUGOUT_D1_KIND,"version":3 if source.tier==2 else 1,"head":source.head,"current":current}
	if _installed_dugout_d1_parts(value).is_empty(): return {}
	var body:=_native_encoded(value)
	var again:=_bound_installed_dugout_d1(expected)
	if body.is_empty() or again.is_empty() or again.bytes!=source.bytes or again.pages!=source.pages: return {}
	saving=true
	var saved: bool=_write_exact_bytes(body,expected)
	saving=false
	if not saved: return {}
	_invalidate()
	_installed_dugout_d1_bound={"bytes":body,"page":source.pin,"draft":native}
	return installed_dugout_d1_view(body)

static func _installed_dugout_d1_project_pending(session: SeasonState) -> Dictionary:
	if session!=null and session.career!=null and Adapter.OrdinaryRuntime.CommonStart.detached_earned_dugout_d2(session.career.start_basis) \
			and preload("earned_dugout_d2.gd").regular_boundary(session)!="unpaid": return {}
	if session!=null and session.phase==SeasonState.Phase.SEMIFINAL: return _installed_dugout_d1_semifinal_projection(session)
	if session!=null and session.phase==SeasonState.Phase.FINAL: return _installed_dugout_d1_final_projection(session)
	if not _installed_dugout_d1_history_shape(session) or session.round_index>=_installed_dugout_d1_regular_limit() \
			or session.physical.pending.is_empty(): return {}
	var checked: Dictionary=BaseArchive.Core.MixedProof._project_mixed_round(session)
	var result: Dictionary=checked.projection
	if result.has("error") or not result.get("candidate") is SeasonState or not checked.get("source") is SeasonState: return {}
	var count: int=session.physical.pending.reports.size()
	if not preload("earned_dugout_d1.gd").regular_projection_shape(session.round_index,count,result.has("fixture")): return {}
	return {"projection":result,"session":checked.source,"count":count,
		"round":session.round_index,"managed_id":session.round_index*3}

# Only the actual OrdinaryRunner's completion for the projected fixture can
# append one receipt. No score/fixture selector or fallback save is exposed.
static func start_installed_dugout_d1_cpu(app: Variant) -> bool:
	if not _installed_dugout_d1_round_app(app) or not _installed_dugout_d1_cpu.is_empty() or is_instance_valid(app.ai_runner): return false
	var expected: PackedByteArray=app.saved_bytes
	var round_plan:=_installed_dugout_d1_pending(expected)
	if round_plan.is_empty() or not round_plan.projection.get("fixture") is Dictionary: return false
	var fixture: Dictionary=round_plan.projection.fixture
	var plan: SeasonState=round_plan.projection.candidate
	if fixture.get("away")==0 or fixture.get("home")==0: return false
	if not round_plan.get("semifinal",false):
		var next_id: int=round_plan.managed_id+round_plan.count+1
		var next_round: int=int(round_plan.round) if next_id<30 else (10 if next_id<32 else 11)
		if fixture.get("id")!=next_id or fixture.get("round")!=next_round: return false
	var state: MatchState=Adapter.OrdinaryFixtures.match_for(plan,fixture)
	var field: FieldDefinition=SeasonState.field_for_fixture(fixture,plan)
	if state==null or field==null: return false
	var runner: Node=Adapter.OrdinaryRunner.new()
	_installed_dugout_d1_cpu={"app":weakref(app),"runner":weakref(runner),"bytes":expected.duplicate(),"fixture":int(fixture.id),"count":round_plan.count}
	app.ai_runner=runner
	app.add_child(runner)
	runner.finished.connect(_finish_installed_dugout_d1_cpu.bind(app,runner),CONNECT_ONE_SHOT)
	runner.failed.connect(_fail_installed_dugout_d1_cpu.bind(app,runner),CONNECT_ONE_SHOT)
	if runner.start(state,Adapter.OrdinaryFixtures.seed_for(plan,fixture),field.id): return true
	_installed_dugout_d1_cpu={}
	app.ai_runner=null
	runner.queue_free()
	return false

static func _installed_dugout_d1_cpu_matches(app: Variant,runner: Node) -> bool:
	var lease: Dictionary=_installed_dugout_d1_cpu
	return not lease.is_empty() and _installed_dugout_d1_round_app(app) and is_instance_valid(runner) \
		and lease.app.get_ref()==app and app.ai_runner==runner and lease.runner.get_ref()==runner \
		and lease.bytes==app.saved_bytes

static func cancel_installed_dugout_d1_cpu(app: Variant) -> bool:
	var runner: Node=app.ai_runner if is_instance_valid(app) else null
	if not _installed_dugout_d1_cpu_matches(app,runner): return false
	_installed_dugout_d1_cpu={}
	runner.cancel()
	app._installed_dugout_d1_cpu_done(runner,{},"CPU game canceled. Your saved round is unchanged.")
	return true

static func _fail_installed_dugout_d1_cpu(reason: String,app: Variant,runner: Node) -> void:
	if not _installed_dugout_d1_cpu_matches(app,runner): return
	_installed_dugout_d1_cpu={}
	app._installed_dugout_d1_cpu_done(runner,{},reason)

static func _finish_installed_dugout_d1_cpu(report: Dictionary,app: Variant,runner: Node) -> void:
	if not _installed_dugout_d1_cpu_matches(app,runner): return
	var lease: Dictionary=_installed_dugout_d1_cpu
	_installed_dugout_d1_cpu={}
	var view:=_stage_installed_dugout_d1_cpu(lease,report)
	app._installed_dugout_d1_cpu_done(runner,view,"The CPU result could not be saved. Your previous checkpoint is unchanged." if view.is_empty() else "")

static func _stage_installed_dugout_d1_cpu(lease: Dictionary,report: Dictionary) -> Dictionary:
	var round_plan:=_installed_dugout_d1_pending(lease.bytes)
	if round_plan.is_empty() or round_plan.count!=lease.count or not round_plan.projection.get("fixture") is Dictionary \
			or round_plan.projection.fixture.id!=lease.fixture: return {}
	var admitted: Dictionary=Adapter.OrdinaryPending.stage_report(round_plan.session,report)
	if admitted.has("error") or not admitted.get("candidate") is SeasonState: return {}
	var candidate: SeasonState=admitted.candidate
	var before: SeasonState=round_plan.session
	var pending: Dictionary=candidate.physical.pending
	if pending.reports.size()!=int(lease.count)+1 or not Codec.same(pending.human,before.physical.pending.human) \
			or not Codec.same(pending.reports.slice(0,pending.reports.size()-1),before.physical.pending.reports): return {}
	var recorded: Dictionary=Adapter.Ordinary.capture(candidate)
	var native:=_installed_dugout_d1_rebind(recorded,candidate)
	var prior: Dictionary=round_plan.source.current.ordinary
	if native==null or native.physical.pending.reports.size()!=pending.reports.size() \
			or not Codec.same(recorded.state.results,prior.state.results) \
			or not Codec.same(recorded.state.player_results,prior.state.player_results) \
			or not Codec.same(recorded.physical.reports,prior.physical.reports) \
			or not Codec.same(recorded.build,prior.build): return {}
	return _write_installed_dugout_d1_opening(round_plan.source,lease.bytes,recorded,native)

static func settle_installed_dugout_d1_round(app: Variant) -> Dictionary:
	if not _installed_dugout_d1_round_app(app) or is_instance_valid(app.ai_runner) or not _installed_dugout_d1_cpu.is_empty(): return {}
	var expected: PackedByteArray=app.saved_bytes
	var round_plan:=_installed_dugout_d1_pending(expected)
	if round_plan.is_empty() or round_plan.projection.has("fixture"): return {}
	var candidate: SeasonState=round_plan.projection.candidate
	var prior: Dictionary=round_plan.source.current.ordinary
	var count: int=round_plan.count
	if not candidate.physical.pending.is_empty(): return {}
	if round_plan.source.tier==2 and preload("earned_dugout_d2.gd").regular_boundary(candidate)!="paid": return {}
	if round_plan.get("managed_final",false):
		if count!=0 or preload("earned_dugout_d1.gd").managed_final_destination(candidate).is_empty(): return {}
	elif round_plan.get("semifinal",false):
		var finish: String=preload("earned_dugout_d1.gd").settled_semifinal(candidate)
		if finish.is_empty() or count!=(1 if finish=="final_ready" else 2): return {}
	else:
		if candidate.round_index!=int(round_plan.round)+1: return {}
		if candidate.round_index==_installed_dugout_d1_regular_limit():
			var finish: String=preload("earned_dugout_d1.gd").final_regular_destination(candidate)
			if finish.is_empty() or count!=(2 if finish=="semifinal" else 5): return {}
		elif not _installed_dugout_d1_history_shape(candidate): return {}
	var recorded: Dictionary=Adapter.Ordinary.capture(candidate)
	var native:=_installed_dugout_d1_rebind(recorded,candidate)
	if native==null or native.round_index!=candidate.round_index or native.cash()!=candidate.cash() \
			or not Codec.same(recorded.state.results.slice(0,-(count+1)),prior.state.results) \
			or not Codec.same(recorded.state.player_results.slice(0,-1),prior.state.player_results): return {}
	if count==0:
		if not Codec.same(recorded.physical.reports,prior.physical.reports) or not prior.physical.pending.reports.is_empty(): return {}
	elif not Codec.same(recorded.physical.reports.slice(0,-count),prior.physical.reports) \
			or not Codec.same(recorded.physical.reports.slice(-count),prior.physical.pending.reports): return {}
	return _write_installed_dugout_d1_opening(round_plan.source,expected,recorded,native)

			or not Codec.whole(command.get("rev"),previous.build.revision(),previous.build.revision()): return {}

			or not Codec.same(image.state.player_results,shop_source.source.current.ordinary.state.player_results): return {}

# Qualification-side transaction only. Ordinary installed play remains closed
# until its own authenticated rollover and draft checkpoints are wired.
static func write_base_archive_earned_dugout_d1(owner: RefCounted, expected: PackedByteArray) -> bool:
	var staged:=stage_earned_dugout_d1(owner,expected)
	if staged.is_empty(): return false
	var pin: Dictionary=_base_archive_draft_pin
	if not base_archive_app_binding(owner,expected) or staged.head.pages.size()>BASE_ARCHIVE_QA_MAX_PAGES: return false
	var wire: PackedByteArray=BASE_ARCHIVE_HEADER.to_utf8_buffer()
	wire.append_array(BaseArchive._wire({"head":staged.head,"current":staged.current,
		"prior_primary_sha256":BaseArchive._hash(expected),"generation":BASE_ARCHIVE_EARNED_D1_GENERATION}))
	if wire.size()>MAX_BYTES: return false
	saving=true
	_base_archive_write_ticket=RefCounted.new()
	var saved: bool=_base_archive_install(_base_archive_write_ticket,pin.pair,staged.page,wire,true,pin.head,pin.pages)
	saving=false
	_invalidate()
	return saved

# Separate read-only validation. A literal predecessor is required for this
# first-generation witness; never silently fall back to another career.
static func read_base_archive_earned_dugout_d1() -> Dictionary:
	if saving or _ticket!=null or not _base_archive_qa() or not _base_archive_temps_clear(): return {}
	var pair:=_base_archive_pair()
	if pair.is_empty() or not pair.primary.present or not pair.backup.present: return {}
	var result:=_base_archive_wrapper(pair.primary.bytes)
	var source:=_base_archive_wrapper(pair.backup.bytes)
	if result.get("generation")!=BASE_ARCHIVE_EARNED_D1_GENERATION \
		or source.get("generation")!=BASE_ARCHIVE_FINAL_CLOSED_GENERATION \
		or result.prior_primary_sha256!=BaseArchive._hash(pair.backup.bytes): return {}
	var current_head:=BaseArchive._head(BaseArchive._wire(result.head))
	var prior_head:=BaseArchive._head(BaseArchive._wire(source.head))
	if current_head.is_empty() or prior_head.is_empty() or prior_head.pages.is_empty() \
		or current_head.pages.size()!=prior_head.pages.size()+1 \
		or current_head.pages.size()>BASE_ARCHIVE_QA_MAX_PAGES \
		or current_head.pages.slice(0,-1)!=prior_head.pages or not _base_archive_page_root(false): return {}
	var full: Array[PackedByteArray]=[]
	var prefix: Array[PackedByteArray]=[]
	for index: int in current_head.pages.size():
		var name: String=_base_archive_page_path(current_head.pages[index])
		var item:=_base_archive_member(name)
		if item.is_empty() or not item.present or BaseArchive._hash(item.bytes)!=current_head.pages[index] \
			or not _base_archive_absent(name+".tmp"): return {}
		full.append(item.bytes)
		if index<prior_head.pages.size(): prefix.append(item.bytes)
	if BaseArchive.append_head(BaseArchive._wire(source.head),full.back())!=BaseArchive._wire(result.head): return {}
	var verified: RefCounted=BaseArchive.restore_pages(BaseArchive._wire(source.head),BaseArchive._wire(source.current),prefix)
	if not _base_archive_final_closed(verified): return {}
	var contract=preload("earned_dugout_d1.gd")
	var offer: Dictionary=contract.from_closed_image(source.current,source.head)
	var staged: Dictionary=contract.stage_birth(source.current,source.head,offer)
	if staged.is_empty() or staged.page!=full.back() or not Codec.same(staged.current,result.current) \
		or not Codec.same(staged.head,result.head) or not Codec.same(result.current.common,verified.common): return {}
	var native: SeasonState=Adapter.Ordinary.restore_common(result.current.ordinary)
	if native==null or not Codec.same(Adapter.Ordinary.capture(native),result.current.ordinary) \
		or not _base_archive_temps_clear() or not Codec.same(_base_archive_pair(),pair) \
		or not _base_archive_pages_live(current_head,full): return {}
	return {"current":result.current,"head":current_head,"offer":offer}

# Exact process-verified writer readback, not a resample or external admission.
static func base_archive_app_result(previous: PackedByteArray) -> Dictionary:
	var proof: Dictionary=_base_archive_draft_pin
	if proof.is_empty() or not proof.pair.backup.present or proof.pair.backup.bytes!=previous \
			or not base_archive_app_binding(proof.owner,proof.pair.primary.bytes): return {}
	return {"owner":proof.owner,"bytes":proof.pair.primary.bytes.duplicate()}

static func _read_base_archive_live_source(allow_human: bool) -> RefCounted:
	if saving or _ticket!=null or not _base_archive_qa(): return null
	if _base_archive_draft_pin_live(_base_archive_draft_pin):
		var owner: RefCounted=_base_archive_draft_pin.owner
		return owner if allow_human or _base_archive_draft_image(owner,0) else null
	_base_archive_draft_pin={}
	var proof: Dictionary=_base_archive_load()
	if proof.is_empty() or not (_base_archive_draft_image(proof.owner,0) \
			or (allow_human and (_base_archive_report_count(proof.owner)>=0 or _base_archive_settled_image(proof.owner)
				or _base_archive_next_human_image(proof.owner) or _base_archive_regular_settled_image(proof.owner)
				or _base_archive_postseason_image(proof.owner) or _base_archive_semifinal_settled(proof.owner)
				or _base_archive_completed_closed(proof.owner) or _base_archive_completed_closed(proof.owner,true)
				or _base_archive_final_human(proof.owner) or _base_archive_final_settled(proof.owner)
				or _base_archive_final_closed(proof.owner)
				or _base_archive_abandoned_image(proof.owner,false) or _base_archive_abandoned_image(proof.owner,true)))): return null
	proof.session=proof.owner.session
	proof.snapshot=RawGuard.capture(proof.owner)
	if proof.snapshot.is_empty() or not _base_archive_draft_pin_live(proof): return null
	_base_archive_draft_pin=proof
	return proof.owner

static func write_base_archive_pick(owner: RefCounted, expected: PackedByteArray, id: String) -> bool:
	if saving or _ticket!=null or _base_archive_draft_pin.is_empty() or not _base_archive_qa(): return false
	var proof: Dictionary=_base_archive_draft_pin
	if owner!=proof.owner or expected!=proof.pair.primary.bytes: return false
	if not _base_archive_draft_pin_live(proof):
		_base_archive_draft_pin={}
		return false
	if not _base_archive_draft_image(owner,0) or owner.session.phase!=SeasonState.Phase.DRAFT \
			or not owner.session.offers().has(id): return false
	_base_archive_draft_pin={}
	var candidate: RefCounted=owner.fork()
	if candidate==null or not candidate.session.choose_player(id) or not _base_archive_draft_image(candidate,1):
		return false
	_base_archive_live_op=RefCounted.new()
	return _base_archive_commit_current(_base_archive_live_op,proof,candidate,expected,BASE_ARCHIVE_DRAFT_GENERATION)

static func _base_archive_commit_current(op: RefCounted, proof: Dictionary, candidate: RefCounted,
		expected: PackedByteArray, generation: String, next_head: Dictionary={},
		new_page: PackedByteArray=PackedByteArray()) -> bool:
	if op==null or op!=_base_archive_live_op: return false
	_base_archive_live_op=null
	var data: Dictionary=Owner.capture(candidate)
	var snapshot: Dictionary=RawGuard.capture(candidate)
	if data.is_empty() or snapshot.is_empty(): return false
	var previous: String=BaseArchive._hash(expected)
	if not BaseArchive._digest(previous): return false
	var extending: bool=not next_head.is_empty()
	if extending!=(not new_page.is_empty()): return false
	var bytes: PackedByteArray=BASE_ARCHIVE_HEADER.to_utf8_buffer()
	bytes.append_array(BaseArchive._wire({"head":next_head if extending else proof.head,"current":data,
		"prior_primary_sha256":previous,"generation":generation}))
	if bytes.size()>MAX_BYTES or not _base_archive_draft_pin_live(proof): return false
	saving=true
	_base_archive_write_ticket=RefCounted.new()
	var committed: bool=_base_archive_install(_base_archive_write_ticket,proof.pair,
		new_page if extending else proof.pages.back(),bytes,extending,proof.head,proof.pages)
	saving=false
	_invalidate()
	if not committed: return false
	if extending:
		proof.head=next_head
		proof.pages.append(new_page)
	proof.owner=candidate
	proof.session=candidate.session
	proof.snapshot=snapshot
	proof.pair={"primary":{"present":true,"bytes":bytes},"backup":{"present":true,"bytes":expected}}
	if not _base_archive_draft_pin_live(proof):
		last_error="The career save was replaced but could not be verified. Use the strict qualification reader."
		return false
	_base_archive_draft_pin=proof
	return true

# This previous primary has no admission power. The caller first strictly
# replayed the archive, and we bind the exact backup image/Close to that
# now-verified page before using the stock, non-physical Next recurrence.
static func _base_archive_previous_matches(backup: PackedByteArray, page_bytes: PackedByteArray,
		current: Dictionary) -> bool:
	var previous: Variant=Owner._bytes_value(backup)
	var page: Variant=Owner._bytes_value(page_bytes)
	if not previous is Dictionary or not previous.get("ordinary_state") is Dictionary \
			or not page is Dictionary or not Codec.same(previous.get("ordinary"),page.get("image")): return false
	var state: Dictionary=previous.ordinary_state
	if state.get("status")!="closed" or state.has("archive") or not Codec.same(state.get("last_close"),page.get("close")):
		return false
	var native: RefCounted=Owner.restore(previous)
	if native==null or native.active!="ordinary" or not Owner._missing_dynasty(native) \
			or native.session.league_rules.get("league")!="dugout" or native.session.league_rules.get("tier")!=0 \
			or state.get("proposal",{}).get("basis",{}).get("seed")!=BaseArchive._default_next_seed(native):
		return false
	return native.commit_next() and Codec.same(Owner.capture(native),current)

# Fixed App producer. Caller never supplies H13, a pending state, or a writer
# flag. The qualified stock App, lab, field and owner are all still bound.
static func write_base_archive_human(app: Variant) -> bool:
	if not is_instance_valid(app) or app.get_script()!=load("res://src/presentation/season/successor_dynasty_owner_v1/app.gd") \
			or not base_archive_app_binding(app.career_owner,app.saved_bytes): return false
	var proof: Dictionary=_base_archive_draft_pin
	var owner: RefCounted=proof.owner
	var expected: PackedByteArray=proof.pair.primary.bytes
	var session: SeasonState=proof.session
	if app.season!=session or not _base_archive_managed_ready(owner): return false
	var lab: PitchBatLab=app.lab
	if not is_instance_valid(lab): return false
	var match_state: MatchState=lab._match_state
	var human: Array=Adapter.human_capture("ordinary").capture(app)
	if human.size()!=13 or not is_instance_valid(lab) or app.lab!=lab or lab._match_state!=match_state \
			or app.career_owner!=owner or app.season!=session or app.saved_bytes!=expected \
			or not base_archive_app_binding(owner,expected): return false
	var candidate: RefCounted=owner.fork()
	if candidate==null: return false
	candidate.session.physical.pending={"human":human.duplicate(true),"reports":[]}
	var opening: bool=_base_archive_draft_image(owner,4)
	var semifinal: bool=_base_archive_semifinal_ready(owner)
	var final: bool=_base_archive_final_ready(owner)
	if final:
		if _base_archive_wrapper(expected).get("generation")!=BASE_ARCHIVE_SEMI_SETTLED_GENERATION \
				or not _base_archive_final_human(candidate): return false
	elif semifinal:
		if _base_archive_semifinal_count(candidate)!=0: return false
	elif not (_base_archive_human_image(candidate) if opening else _base_archive_next_human_image(candidate)): return false
	var projection: Dictionary=Adapter.OrdinaryPending.project(candidate.session)
	if projection.has("error") or not base_archive_app_binding(owner,expected): return false
	if final:
		if _base_archive_projected_final(candidate,projection)==null: return false
	elif not projection.has("fixture"): return false
	_base_archive_draft_pin={}
	_base_archive_live_op=RefCounted.new()
	var generation: String=BASE_ARCHIVE_HUMAN_GENERATION if opening else BASE_ARCHIVE_NEXT_HUMAN_GENERATION
	if semifinal: generation=BASE_ARCHIVE_SEMI_HUMAN_GENERATION
	if final: generation=BASE_ARCHIVE_FINAL_HUMAN_GENERATION
	return _base_archive_commit_current(_base_archive_live_op,proof,candidate,expected,generation)

static func _base_archive_opening_report_count(owner: RefCounted) -> int:
	if not _base_archive_draft_image(owner,4,true): return -1
	var s: SeasonState=owner.session
	var pending: Dictionary=s.physical.pending
	if not Codec.keys(pending,["human","reports"]) or not pending.reports is Array or pending.reports.size()>2 \
			or s.pending_fixture().get("id",-1)!=0 or not Adapter.OrdinaryPending.shape(s): return -1
	return pending.reports.size()

# Interior pending rounds have a native later managed fixture. The actual
# next CPU identity is returned by the projector, never inferred from an ID.
static func _base_archive_has_later_regular(s: SeasonState) -> bool:
	for fixture: Dictionary in s.schedule:
		if fixture.get("round",-1)>s.round_index and (fixture.get("home",-1)==0 or fixture.get("away",-1)==0):
			return true
	return false

# The direct authored Dugout regular season has exactly ten managed rounds.
# _paid_regular still binds the current fixture to the real schedule/team 0.
static func _base_archive_last_regular(owner: RefCounted) -> bool:
	if not _base_archive_paid_regular(owner): return false
	var s: SeasonState=owner.session
	return s.round_index==9 and s.schedule.size()==30 and not _base_archive_has_later_regular(s)

static func _base_archive_postseason_image(owner: RefCounted) -> bool:
	return _base_archive_postseason_shape(owner,false)

static func _base_archive_postseason_shape(owner: RefCounted, semifinal_pending: bool) -> bool:
	if owner==null or owner.active!="ordinary" or owner.lifecycle_status()!="active" or owner.session==null: return false
	var s: SeasonState=owner.session
	if not Adapter.scope("ordinary",s) or s.league_rules.get("league")!="dugout" or s.league_rules.get("tier")!=0 \
			or s.round_index!=10 or s.picks.size()!=4 or s.player_results.size()!=10 \
			or (not semifinal_pending and not s.physical.pending.is_empty()) or not s.physical.needed.is_empty() or not s.physical.error.is_empty() \
			or s.physical.projecting or s.build.pack_pending() or s.physical.cursor!=s.physical.reports.size() \
			or s.playoff_seeds.size()!=4 or s.semifinals.size()!=2: return false
	var summary: Dictionary=ClubSeasonRecord.analyze(ClubSeasonRecord.capture(s))
	if summary.is_empty() or summary.games!=10: return false
	if s.phase==SeasonState.Phase.SEMIFINAL:
		var next: Dictionary=s.pending_fixture()
		return summary.finish=="unfinished" and s.playoff_seeds.has(0) and s.results.size()==30 \
			and s.physical.cursor==20 and s.champion==-1 and s.final_fixture.is_empty() \
			and next in s.semifinals and 0 in [next.get("home"),next.get("away")]
	return not semifinal_pending and s.phase==SeasonState.Phase.COMPLETE and summary.finish=="missed" and not s.playoff_seeds.has(0) \
		and s.results.size()==33 and s.physical.cursor==23 and s.champion>0 and s.pending_fixture().is_empty()

static func _base_archive_semifinal_ready(owner: RefCounted) -> bool:
	return owner!=null and owner.session!=null and owner.session.phase==SeasonState.Phase.SEMIFINAL \
		and _base_archive_postseason_image(owner)

static func _base_archive_semifinal_count(owner: RefCounted) -> int:
	if owner==null or owner.session==null or owner.session.phase!=SeasonState.Phase.SEMIFINAL \
			or not _base_archive_postseason_shape(owner,true): return -1
	var s: SeasonState=owner.session
	var pending: Dictionary=s.physical.pending
	if not Codec.keys(pending,["human","reports"]) or not pending.human is Array or pending.human.size()!=13 \
			or pending.human[0]!=s.pending_fixture().get("id") or not pending.reports is Array \
			or pending.reports.size()>2 or not Adapter.OrdinaryPending.shape(s): return -1
	if not pending.reports.is_empty():
		var opponent_fixture: int=-1
		for fixture: Dictionary in s.semifinals:
			if fixture.get("id")!=pending.human[0] and 0 not in [fixture.get("home"),fixture.get("away")]:
				opponent_fixture=int(fixture.get("id",-1))
		if opponent_fixture not in [30,31] or pending.reports[0].fixture!=opponent_fixture: return -1
	if pending.reports.size()==2 and pending.reports[1].fixture!=32: return -1
	return pending.reports.size()

static func _base_archive_semifinal_settled(owner: RefCounted, closed: bool=false, final_pending: bool=false) -> bool:
	if owner==null or owner.active!="ordinary" or owner.session==null \
			or owner.lifecycle_status()!=("closed" if closed else "active"): return false
	var s: SeasonState=owner.session
	if not Adapter.scope("ordinary",s) or s.league_rules.get("league")!="dugout" or s.league_rules.get("tier")!=0 \
			or s.round_index!=10 or s.picks.size()!=4 or s.player_results.size()!=11 \
			or (not final_pending and not s.physical.pending.is_empty()) or not s.physical.needed.is_empty() or not s.physical.error.is_empty() \
			or s.physical.projecting or s.build.pack_pending() or s.physical.cursor!=s.physical.reports.size() \
			or s.playoff_seeds.size()!=4 or not s.playoff_seeds.has(0) or s.semifinals.size()!=2: return false
	var human: Dictionary=s.player_results.back()
	if human.get("id") not in [30,31] or 0 not in [human.get("away"),human.get("home")]: return false
	var summary: Dictionary=ClubSeasonRecord.analyze(ClubSeasonRecord.capture(s))
	if summary.get("games")!=10: return false
	var won: bool=SeasonState._winner(human)==0
	if won:
		var next: Dictionary=s.pending_fixture()
		return s.phase==SeasonState.Phase.FINAL and summary.finish=="unfinished" and s.results.size()==32 \
			and s.physical.cursor==21 and s.champion==-1 and Codec.same(next,s.final_fixture) \
			and next.get("id")==32 and next.get("round")==11 and next.get("neutral")==true \
			and 0 in [next.get("away"),next.get("home")]
	return not final_pending and s.phase==SeasonState.Phase.COMPLETE and summary.finish=="semifinal" and s.results.size()==33 \
		and s.physical.cursor==22 and s.champion>0 and s.pending_fixture().is_empty()

static func _base_archive_semifinal_sibling(prior: Dictionary, wrapped: Dictionary, current: RefCounted,
		previous: RefCounted) -> bool:
	if wrapped.generation==BASE_ARCHIVE_SEMI_HUMAN_GENERATION:
		if prior.get("generation")!=BASE_ARCHIVE_POSTSEASON_GENERATION or not _base_archive_semifinal_ready(previous):
			return false
	else:
		var count: int=_base_archive_semifinal_count(previous)
		if count<0 or prior.get("generation")!=(BASE_ARCHIVE_SEMI_HUMAN_GENERATION if count==0 else BASE_ARCHIVE_SEMI_REPORT_GENERATION):
			return false
		if wrapped.generation==BASE_ARCHIVE_SEMI_SETTLED_GENERATION:
			if count<1: return false
			var stock: Dictionary=Adapter.OrdinaryPending.project(previous.session)
			if stock.has("error") or stock.has("fixture") or not stock.get("candidate") is SeasonState: return false
			var settled: RefCounted=previous.fork()
			if settled==null: return false
			settled.session=stock.candidate
			return _base_archive_settled_after(previous,settled) and Codec.same(Owner.capture(settled),wrapped.current)
		var before: Dictionary=previous.session.physical.pending
		var after: Dictionary=current.session.physical.pending
		if _base_archive_semifinal_count(current)!=count+1 or not Codec.same(before.human,after.human) \
				or not Codec.same(before.reports,after.reports.slice(0,count)): return false
	previous.session.physical.pending=current.session.physical.pending.duplicate(true)
	return Codec.same(Owner.capture(previous),wrapped.current)

static func _base_archive_final_fixture(season: SeasonState) -> bool:
	var fixture: Dictionary=season.final_fixture
	return fixture.get("id")==32 and fixture.get("round")==11 and fixture.get("neutral")==true \
		and 0 in [fixture.get("home"),fixture.get("away")] and fixture.get("home")!=fixture.get("away")

static func _base_archive_final_ready(owner: RefCounted) -> bool:
	return _base_archive_semifinal_settled(owner) and owner.session.phase==SeasonState.Phase.FINAL \
		and _base_archive_final_fixture(owner.session)

static func _base_archive_final_human(owner: RefCounted) -> bool:
	if owner==null or owner.session==null or owner.session.phase!=SeasonState.Phase.FINAL \
			or not _base_archive_semifinal_settled(owner,false,true) or not _base_archive_final_fixture(owner.session):
		return false
	var s: SeasonState=owner.session
	var pending: Dictionary=s.physical.pending
	return Codec.keys(pending,["human","reports"]) and pending.human is Array and pending.human.size()==13 \
		and pending.human[0]==32 and pending.reports==[] and Adapter.OrdinaryPending.shape(s)

static func _base_archive_final_settled(owner: RefCounted,closed: bool=false) -> bool:
	if owner==null or owner.active!="ordinary" or owner.session==null \
			or owner.lifecycle_status()!=("closed" if closed else "active"): return false
	var s: SeasonState=owner.session
	if not Adapter.scope("ordinary",s) or s.league_rules.get("league")!="dugout" or s.league_rules.get("tier")!=0 \
			or s.phase!=SeasonState.Phase.COMPLETE or s.round_index!=10 or s.picks.size()!=4 \
			or s.results.size()!=33 or s.player_results.size()!=12 or s.physical.reports.size()!=21 \
			or s.physical.cursor!=21 or not s.physical.pending.is_empty() or not s.physical.needed.is_empty() \
			or not s.physical.error.is_empty() or s.physical.projecting or s.build.pack_pending() \
			or not _base_archive_final_fixture(s) or not s.pending_fixture().is_empty(): return false
	var human: Dictionary=s.player_results.back()
	if human.get("id")!=32 or human.get("round")!=11 or human.get("home")!=s.final_fixture.home \
			or human.get("away")!=s.final_fixture.away or s.champion!=SeasonState._winner(human): return false
	var summary: Dictionary=ClubSeasonRecord.analyze(ClubSeasonRecord.capture(s))
	return summary.get("games")==10 and summary.get("finish")==("champion" if s.champion==0 else "runner_up")

# The terminal FINAL projection consumes the human evidence, not a CPU row.
# The Store never receives a result from the caller and no projection is
# retained between Return/Settle or accepted as a serialized proof.
static func _base_archive_projected_final(source: RefCounted,projection: Dictionary) -> RefCounted:
	if not _base_archive_final_human(source) or projection.has("error") or projection.has("fixture") \
			or not projection.get("candidate") is SeasonState: return null
	var result: RefCounted=source.fork()
	if result==null: return null
	result.session=projection.candidate
	if not _base_archive_final_settled(result) or not _base_archive_settlement_prefix(source,result,0): return null
	var h: Array=source.session.physical.pending.human
	var final: Dictionary=result.session.player_results.back()
	return result if Codec.same([final.id,final.away_runs,final.home_runs],h.slice(0,3)) else null

static func _base_archive_final_closed(owner: RefCounted) -> bool:
	return _base_archive_final_settled(owner,true) and _base_archive_complete_receipt(owner) \
		and owner.ordinary_state.proposal.is_empty()

static func _base_archive_final_sibling(prior: Dictionary,wrapped: Dictionary,current: RefCounted,
		previous: RefCounted) -> bool:
	if previous==null: return false
	match wrapped.generation:
		BASE_ARCHIVE_FINAL_HUMAN_GENERATION:
			if prior.get("generation")!=BASE_ARCHIVE_SEMI_SETTLED_GENERATION or not _base_archive_final_ready(previous):
				return false
			previous.session.physical.pending=current.session.physical.pending.duplicate(true)
			return Codec.same(Owner.capture(previous),wrapped.current)
		BASE_ARCHIVE_FINAL_SETTLED_GENERATION:
			if prior.get("generation")!=BASE_ARCHIVE_FINAL_HUMAN_GENERATION or not _base_archive_final_human(previous): return false
			var result: RefCounted=_base_archive_projected_final(previous,Adapter.OrdinaryPending.project(previous.session))
			return result!=null and Codec.same(Owner.capture(result),wrapped.current)
		BASE_ARCHIVE_FINAL_CLOSED_GENERATION:
			if prior.get("generation")!=BASE_ARCHIVE_FINAL_SETTLED_GENERATION: return false
			var result: RefCounted=_base_archive_lifecycle_candidate(previous,"complete_final")
			return result!=null and Codec.same(Owner.capture(result),wrapped.current)
	return false

static func settle_base_archive_final(app: Variant) -> bool:
	if not _base_archive_app_valid(app) or not _base_archive_cpu_job.is_empty() or is_instance_valid(app.ai_runner):
		return false
	var proof: Dictionary=_base_archive_draft_pin
	var source: RefCounted=proof.owner
	if not _base_archive_final_human(source) \
			or _base_archive_wrapper(proof.pair.primary.bytes).get("generation")!=BASE_ARCHIVE_FINAL_HUMAN_GENERATION:
		return false
	var result: RefCounted=_base_archive_projected_final(source,Adapter.OrdinaryPending.project(proof.session))
	if result==null or not _base_archive_app_valid(app): return false
	_base_archive_draft_pin={}
	_base_archive_live_op=RefCounted.new()
	return _base_archive_commit_current(_base_archive_live_op,proof,result,app.saved_bytes,BASE_ARCHIVE_FINAL_SETTLED_GENERATION)

static func _base_archive_regular_roles(owner: RefCounted) -> Dictionary:
	return {"reports":BASE_ARCHIVE_LAST_REGULAR_REPORT_GENERATION,"settled":BASE_ARCHIVE_POSTSEASON_GENERATION} \
		if _base_archive_last_regular(owner) else \
		{"reports":BASE_ARCHIVE_REGULAR_REPORT_GENERATION,"settled":BASE_ARCHIVE_REGULAR_SETTLED_GENERATION}

static func _base_archive_regular_report_count(owner: RefCounted) -> int:
	if not _base_archive_paid_regular(owner): return -1
	var s: SeasonState=owner.session
	var final: bool=_base_archive_last_regular(owner)
	if not final and not _base_archive_has_later_regular(s): return -1
	var pending: Dictionary=s.physical.pending
	if not Codec.keys(pending,["human","reports"]) or not pending.human is Array or pending.human.size()!=13 \
			or pending.human[0]!=s.pending_fixture().get("id") or not pending.reports is Array \
			or pending.reports.size()>(5 if final else 2) or not Adapter.OrdinaryPending.shape(s): return -1
	return pending.reports.size()

static func _base_archive_report_count(owner: RefCounted) -> int:
	var opening: int=_base_archive_opening_report_count(owner)
	if opening>=0: return opening
	var semifinal: int=_base_archive_semifinal_count(owner)
	return semifinal if semifinal>=0 else _base_archive_regular_report_count(owner)

static func _base_archive_regular_settled_image(owner: RefCounted) -> bool:
	if not _base_archive_paid_regular(owner) or _base_archive_settled_image(owner) \
			or not owner.session.physical.pending.is_empty(): return false
	var events: Array=owner.session.build._events
	return not events.is_empty() and events.back() is Dictionary \
		and not SeasonBuild.SHOP_OPS.has(str(events.back().get("op","")))

static func _base_archive_settled_after(before: RefCounted, after: RefCounted) -> bool:
	if _base_archive_opening_report_count(before)==2: return _base_archive_settled_image(after)
	var semifinal: int=_base_archive_semifinal_count(before)
	if semifinal>=0:
		if not _base_archive_semifinal_settled(after) or semifinal!=(1 if after.session.phase==SeasonState.Phase.FINAL else 2):
			return false
	else:
		var count: int=_base_archive_regular_report_count(before)
		if _base_archive_last_regular(before):
			if not _base_archive_postseason_image(after) or count!=(2 if after.session.phase==SeasonState.Phase.SEMIFINAL else 5):
				return false
		elif count!=2 or not _base_archive_regular_settled_image(after): return false
	return _base_archive_settlement_prefix(before,after,0 if semifinal>=0 else 1)

static func _base_archive_settlement_prefix(before: RefCounted,after: RefCounted,round_delta: int) -> bool:
	var old: SeasonState=before.session
	var new: SeasonState=after.session
	var events: Array=new.build._events
	var old_events: Array=old.build._events
	if new.round_index!=old.round_index+round_delta or not Codec.same(new.results.slice(0,old.results.size()),old.results) \
			or not Codec.same(new.player_results.slice(0,old.player_results.size()),old.player_results) \
			or not Codec.same(new.physical.reports.slice(0,old.physical.cursor),old.physical.reports) \
			or not Codec.same(new.physical.reports.slice(old.physical.cursor),old.physical.pending.reports) \
			or events.size()<=old_events.size() or not Codec.same(events.slice(0,old_events.size()),old_events) \
			or not Codec.same(before.common,after.common): return false
	for event: Variant in events.slice(old_events.size()):
		if not event is Dictionary or SeasonBuild.SHOP_OPS.has(str(event.get("op",""))): return false
	return true

static func _base_archive_regular_sibling(prior: Dictionary, wrapped: Dictionary, current: RefCounted,
		previous: RefCounted) -> bool:
	var old_count: int=_base_archive_regular_report_count(previous)
	if old_count<0: return false
	var roles: Dictionary=_base_archive_regular_roles(previous)
	if prior.get("generation")!=(BASE_ARCHIVE_NEXT_HUMAN_GENERATION if old_count==0 else roles.reports):
		return false
	if wrapped.generation==roles.settled:
		if old_count<2: return false
		var stock: Dictionary=Adapter.OrdinaryPending.project(previous.session)
		if stock.has("error") or stock.has("fixture") or not stock.get("candidate") is SeasonState: return false
		var next: RefCounted=previous.fork()
		if next==null: return false
		next.session=stock.candidate
		return _base_archive_settled_after(previous,next) and Codec.same(Owner.capture(next),wrapped.current)
	var next_pending: Dictionary=current.session.physical.pending
	if wrapped.generation!=roles.reports or _base_archive_regular_report_count(current)!=old_count+1 \
			or not Codec.same(previous.session.physical.pending.human,next_pending.human) \
			or not Codec.same(previous.session.physical.pending.reports,next_pending.reports.slice(0,old_count)):
		return false
	previous.session.physical.pending=next_pending.duplicate(true)
	return Codec.same(Owner.capture(previous),wrapped.current)

static func _base_archive_settled_image(owner: RefCounted) -> bool:
	if owner==null or owner.active!="ordinary" or owner.lifecycle_status()!="active" or owner.session==null: return false
	var s: SeasonState=owner.session
	return s.phase==SeasonState.Phase.REGULAR and s.round_index==1 and s.picks.size()==4 \
		and s.results.size()==3 and s.player_results.size()==1 and s.player_results[0].get("id") == 0 \
		and s.physical.pending.is_empty() and s.physical.reports.size()==2 and s.physical.cursor==2

static func _base_archive_app_valid(app: Variant) -> bool:
	return is_instance_valid(app) and app.get_script()==load("res://src/presentation/season/successor_dynasty_owner_v1/app.gd") \
		and app.base_archive_qualification_view() and not is_instance_valid(app.lab) \
		and app.season==app.career_owner.session and base_archive_app_binding(app.career_owner,app.saved_bytes)

static func _base_archive_cpu_fixture(owner: RefCounted, projected: SeasonState, fixture: Dictionary, count: int) -> bool:
	if not (fixture.get("home",-1)>0 and fixture.get("away",-1)>0): return false
	if _base_archive_semifinal_count(owner)>=0:
		if count==0:
			var managed: Dictionary=owner.session.pending_fixture()
			return projected.phase==SeasonState.Phase.SEMIFINAL and fixture in owner.session.semifinals \
				and fixture.get("id")!=managed.get("id") and fixture.get("round")==10
		return count==1 and projected.phase==SeasonState.Phase.FINAL and fixture.get("id")==32 \
			and fixture.get("round")==11 and fixture.get("neutral")==true and Codec.same(fixture,projected.final_fixture)
	if count<2: return fixture.get("round")==owner.session.round_index
	# Stock lockstep order for an eliminated club only: CPUs finish both
	# semifinals and their final. The exact projector supplies the fixture.
	if not _base_archive_last_regular(owner) or projected.playoff_seeds.has(0): return false
	if count in [2,3]:
		return projected.phase==SeasonState.Phase.SEMIFINAL and fixture.get("round")==10 \
			and fixture.get("id")==28+count and fixture in projected.semifinals
	return count==4 and projected.phase==SeasonState.Phase.FINAL and fixture.get("round")==11 \
		and fixture.get("id")==32 and Codec.same(fixture,projected.final_fixture)

static func start_base_archive_cpu(app: Variant) -> Node:
	return _base_archive_resume_projection(app,"regular").get("runner")

static func resume_base_archive_last_regular(app: Variant) -> Dictionary:
	return _base_archive_resume_projection(app,"last_regular")

static func resume_base_archive_semifinal(app: Variant) -> Dictionary:
	return _base_archive_resume_projection(app,"semifinal")

static func _base_archive_resume_projection(app: Variant, lane: String) -> Dictionary:
	if not _base_archive_cpu_job.is_empty() or not _base_archive_app_valid(app) or is_instance_valid(app.ai_runner): return {}
	var proof: Dictionary=_base_archive_draft_pin
	var count: int=_base_archive_report_count(proof.owner)
	var semifinal: bool=_base_archive_semifinal_count(proof.owner)>=0
	var final_round: bool=_base_archive_last_regular(proof.owner)
	var exact_lane: String="semifinal" if semifinal else ("last_regular" if final_round else "regular")
	var cap: int=2 if semifinal else (5 if final_round else 1)
	if count<0 or lane!=exact_lane or count>cap: return {}
	var p: Dictionary=Adapter.OrdinaryPending.project(proof.session)
	if p.has("error") or not p.get("candidate") is SeasonState: return {}
	if not p.has("fixture"):
		if (not semifinal and not final_round) or count<(1 if semifinal else 2): return {}
		var terminal: RefCounted=proof.owner.fork()
		if terminal==null: return {}
		terminal.session=p.candidate
		if not _base_archive_settled_after(proof.owner,terminal) or not _base_archive_app_valid(app): return {}
		var snapshot: Dictionary=RawGuard.capture(terminal)
		if snapshot.is_empty(): return {}
		_base_archive_settle_pin={"owner":proof.owner,"bytes":app.saved_bytes,"terminal":terminal,"raw":snapshot}
		return {"settle":true}
	if not p.fixture is Dictionary or (semifinal and count>=2) or count>=5: return {}
	var fixture: Dictionary=p.fixture
	var projected: SeasonState=p.candidate
	var request: String=Adapter.OrdinaryFixtures.request_key(projected,fixture)
	var field: FieldDefinition=Adapter.OrdinaryRuntime.field_for_fixture(fixture,projected)
	var state: MatchState=Adapter.OrdinaryFixtures.match_for(projected,fixture)
	var seed: int=Adapter.OrdinaryFixtures.seed_for(projected,fixture)
	if not BaseArchive._digest(request) or not _base_archive_cpu_fixture(proof.owner,projected,fixture,count) \
			or not Codec.whole(fixture.get("id"),0,32) or field==null or state==null \
			or not _base_archive_app_valid(app): return {}
	var runner: Node=BaseArchiveRunner.new()
	var cookie:=RefCounted.new()
	app.add_child(runner)
	if not runner.arm_base_archive_receipt(cookie):
		runner.queue_free()
		return {}
	var job: Dictionary={"app":app,"runner":runner,"cookie":cookie,"proof":proof,
		"count":count,"fixture":int(fixture.id),"request":request}
	_base_archive_cpu_job=job
	runner.finished.connect(func(_untrusted): _base_archive_cpu_signal(job))
	runner.failed.connect(func(reason): _base_archive_cpu_failed(job,reason))
	if runner.start(state,seed,field.id): return {"runner":runner}
	_base_archive_cpu_job={}
	runner.queue_free()
	return {}

static func _base_archive_cpu_live(job: Dictionary) -> bool:
	return not job.is_empty() and not _base_archive_cpu_job.is_empty() \
		and _base_archive_cpu_job.cookie==job.cookie and is_instance_valid(job.app) \
		and is_instance_valid(job.runner) and job.app.ai_runner==job.runner

static func _base_archive_cpu_failed(job: Dictionary, reason: String) -> void:
	if not _base_archive_cpu_live(job) or not job.runner.base_archive_failure_ready(job.cookie): return
	_base_archive_cpu_job={}
	job.app._base_archive_cpu_result(job.runner,false,reason)

static func _base_archive_cpu_signal(job: Dictionary) -> void:
	if not _base_archive_cpu_live(job): return
	# A forged signal at any time has no usable report. Only actual terminal
	# runner capture populates this one-use receipt before signalling.
	var report: Dictionary=job.runner.take_base_archive_receipt(job.cookie)
	if report.is_empty(): return
	var proof: Dictionary=job.proof
	var app: Variant=job.app
	var runner: Node=job.runner
	_base_archive_cpu_job={}
	# This builder is not callable with a supplied report; it closes only over
	# the consumed engine receipt above.
	var commit:=func() -> bool:
		if not _base_archive_app_valid(app) or not _base_archive_draft_pin_live(proof) \
				or _base_archive_draft_pin.owner!=proof.owner or _base_archive_report_count(proof.owner)!=job.count:
			return false
		var staged: Dictionary=Adapter.OrdinaryPending.stage_report(proof.session,report)
		if not staged.get("candidate") is SeasonState: return false
		var candidate: RefCounted=proof.owner.fork()
		if candidate==null: return false
		candidate.session=staged.candidate
		if _base_archive_report_count(candidate)!=job.count+1: return false
		var row: Dictionary=candidate.session.physical.pending.reports.back()
		if row.get("fixture")!=job.fixture or row.get("request")!=job.request or not Codec.same(row.get("report"),report):
			return false
		var terminal: RefCounted
		var terminal_raw: Dictionary={}
		var is_last: bool=_base_archive_last_regular(proof.owner)
		var semifinal: bool=_base_archive_semifinal_count(proof.owner)>=0
		if staged.has("settled"):
			if (job.count==0 and not semifinal) or not staged.get("settled") is SeasonState: return false
			terminal=proof.owner.fork()
			if terminal==null: return false
			terminal.session=staged.settled
			if not _base_archive_settled_after(candidate,terminal): return false
			terminal_raw=RawGuard.capture(terminal)
			if terminal_raw.is_empty(): return false
		elif job.count>0 and (semifinal or not is_last or job.count>=4): return false
		if not _base_archive_app_valid(app) or not _base_archive_draft_pin_live(proof): return false
		_base_archive_draft_pin={}
		_base_archive_live_op=RefCounted.new()
		var generation: String=BASE_ARCHIVE_REPORT_GENERATION if _base_archive_opening_report_count(proof.owner)>=0 \
			else _base_archive_regular_roles(proof.owner).reports
		if semifinal: generation=BASE_ARCHIVE_SEMI_REPORT_GENERATION
		if not _base_archive_commit_current(_base_archive_live_op,proof,candidate,app.saved_bytes,generation):
			return false
		if terminal!=null:
			_base_archive_settle_pin={"owner":candidate,"bytes":_base_archive_draft_pin.pair.primary.bytes,
				"terminal":terminal,"raw":terminal_raw}
		return true
	var committed: bool=commit.call()
	app._base_archive_cpu_result(runner,committed,
		"Saved the CPU game. No rewards were paid." if committed else "The CPU game could not be verified against this save.")

static func cancel_base_archive_cpu(app: Variant) -> void:
	if _base_archive_cpu_job.is_empty() or _base_archive_cpu_job.app!=app: return
	var runner: Node=_base_archive_cpu_job.runner
	_base_archive_cpu_job={}
	if is_instance_valid(runner): runner.cancel()

static func settle_base_archive_opening(app: Variant) -> bool:
	if not _base_archive_cpu_job.is_empty() or not _base_archive_app_valid(app) \
			or is_instance_valid(app.ai_runner): return false
	var semifinal: bool=_base_archive_semifinal_count(app.career_owner)>=0
	if _base_archive_report_count(app.career_owner)<(1 if semifinal else 2): return false
	var proof: Dictionary=_base_archive_draft_pin
	var next: RefCounted
	var cache: Dictionary=_base_archive_settle_pin
	_base_archive_settle_pin={}
	if not cache.is_empty() and cache.owner==proof.owner and cache.bytes==app.saved_bytes:
		if not RawGuard.matches(cache.terminal,cache.raw): return false
		next=cache.terminal
	else:
		var p: Dictionary=Adapter.OrdinaryPending.project(proof.session)
		if p.has("error") or p.has("fixture") or not p.get("candidate") is SeasonState: return false
		next=proof.owner.fork()
		if next==null: return false
		next.session=p.candidate
	if not _base_archive_settled_after(proof.owner,next) or not _base_archive_app_valid(app): return false
	var generation: String=BASE_ARCHIVE_SETTLED_GENERATION if _base_archive_opening_report_count(proof.owner)==2 \
		else _base_archive_regular_roles(proof.owner).settled
	if semifinal: generation=BASE_ARCHIVE_SEMI_SETTLED_GENERATION
	_base_archive_draft_pin={}
	_base_archive_live_op=RefCounted.new()
	return _base_archive_commit_current(_base_archive_live_op,proof,next,app.saved_bytes,generation)

static func _base_archive_unshopped_settled_image(owner: RefCounted) -> bool:
	if not _base_archive_settled_image(owner): return false
	for event: Variant in owner.session.build._events:
		if not event is Dictionary or SeasonBuild.SHOP_OPS.has(str(event.get("op",""))): return false
	return true

static func _base_archive_shop_scope(owner: RefCounted) -> bool:
	return _base_archive_settled_image(owner) and owner.session.shop_available()

static func _base_archive_shop_image(owner: RefCounted) -> bool:
	if not _base_archive_shop_scope(owner): return false
	var events: Array=owner.session.build._events
	return not events.is_empty() and events.back() is Dictionary \
		and SeasonBuild.SHOP_OPS.has(str(events.back().get("op","")))

# Native quote/payment/ownership/revision semantics remain in Build. This
# detached helper cannot grant authority, save anything, or install an owner.
static func _base_archive_shop_candidate(owner: RefCounted, command: Dictionary) -> RefCounted:
	if not _base_archive_shop_scope(owner) or not Codec.primitive(command) \
			or not SeasonBuild.SHOP_OPS.has(str(command.get("op",""))): return null
	var id: Variant=command.get("id")
	if id is String and not id.is_empty() and owner.session.build._requests.has(id): return null
	var candidate: RefCounted=owner.fork()
	if candidate==null: return null
	var build: SeasonBuild=candidate.session.build.candidate(command.duplicate(true))
	if build==null: return null
	candidate.session.adopt_build(build)
	if candidate.session.build!=build or not _base_archive_shop_image(candidate): return null
	var previous: Array=owner.session.build._events
	var events: Array=build._events
	if events.size()!=previous.size()+1 or not Codec.same(events.slice(0,previous.size()),previous) \
			or not Codec.same(events.back(),command): return null
	return candidate

static func open_base_archive_shop(app: Variant, ticket: RefCounted) -> bool:
	if not _base_archive_app_valid(app) or not app.take_base_archive_shop_open(ticket) \
			or not _base_archive_shop_scope(app.career_owner) or app.season.build._visit.open: return false
	var revision: int=app.season.build.revision()
	_base_archive_shop_op=RefCounted.new()
	return _write_base_archive_shop(_base_archive_shop_op,app,{"id":"shop-open:%d"%revision,"rev":revision,"op":"open"})

static func write_base_archive_shop(app: Variant, window: Variant, ticket: RefCounted) -> bool:
	if not _base_archive_app_valid(app) or not is_instance_valid(window) \
			or window.get_script()!=load("res://src/presentation/season/successor_dynasty_owner_v1/shop_window.gd") \
			or window.app!=app or window.get_parent()!=app.menu or not window.visible or not window._origin_matches():
		return false
	var command: Dictionary=window.take_base_archive_shop(ticket)
	if command.is_empty() or command.get("op")=="open": return false
	_base_archive_shop_op=RefCounted.new()
	return _write_base_archive_shop(_base_archive_shop_op,app,command)

static func _write_base_archive_shop(op: RefCounted, app: Variant, command: Dictionary) -> bool:
	if op==null or op!=_base_archive_shop_op: return false
	_base_archive_shop_op=null
	if not _base_archive_cpu_job.is_empty() or not _base_archive_app_valid(app) \
			or is_instance_valid(app.ai_runner): return false
	var proof: Dictionary=_base_archive_draft_pin
	var expected: PackedByteArray=app.saved_bytes
	var candidate: RefCounted=_base_archive_shop_candidate(proof.owner,command)
	if candidate==null or not _base_archive_app_valid(app) or not _base_archive_draft_pin_live(proof): return false
	_base_archive_draft_pin={}
	_base_archive_live_op=RefCounted.new()
	return _base_archive_commit_current(_base_archive_live_op,proof,candidate,expected,BASE_ARCHIVE_SHOP_GENERATION)

# Private profile bound, not a career ordinal. CP364 qualifies two ordered
# immutable members only; the pure archive format retains its native bounds.
static func _base_archive_pages_live(head: Dictionary, pages: Array[PackedByteArray]) -> bool:
	if head.is_empty(): return pages.is_empty()
	if not head.get("pages") is Array or head.pages.size()!=pages.size(): return false
	for i: int in pages.size():
		if not _base_archive_draft_page_live(head.pages[i],pages[i]): return false
	return true

static func _base_archive_abandoned_image(owner: RefCounted, prepared: bool) -> bool:
	if not _base_archive_paid_regular(owner,true) or owner.lifecycle_status()!="closed" \
			or not owner.session.physical.pending.is_empty(): return false
	var state: Dictionary=owner.ordinary_state
	var close: Dictionary=state.last_close
	if close.get("reason")!="abandon" or close.get("receipt")!={}: return false
	return state.proposal.get("basis",{}).get("seed")==BaseArchive._default_next_seed(owner) \
		if prepared else state.proposal.is_empty()

# Narrow proof for the real completed semifinal-loss season. The existing
# archive graph reader replays and verifies the native fold before this role.
static func _base_archive_completed_closed(owner: RefCounted, prepared: bool=false) -> bool:
	if not _base_archive_semifinal_settled(owner,true) or owner.session.phase!=SeasonState.Phase.COMPLETE:
		return false
	if not _base_archive_complete_receipt(owner): return false
	var close: Dictionary=owner.ordinary_state.last_close
	var proposal: Dictionary=owner.ordinary_state.proposal
	if prepared:
		if proposal.get("league")!="dugout" or proposal.get("tier")!=0 or proposal.get("source")!=close.get("session_sha256") \
				or proposal.get("basis",{}).get("seed")!=BaseArchive._default_next_seed(owner): return false
	elif not proposal.is_empty(): return false
	return true

static func _base_archive_complete_receipt(owner: RefCounted) -> bool:
	var close: Dictionary=owner.ordinary_state.last_close
	if close.get("reason")!="complete" or not close.get("first_title") is bool: return false
	var receipt: Dictionary=ClubSeasonRecord.receipt(ClubSeasonRecord.capture(owner.session),close.first_title,0)
	return not receipt.is_empty() and Codec.same(close.get("receipt"),receipt) \
		and Codec.same(owner.common.last_fold.get("receipt"),receipt)

static func _base_archive_lifecycle_candidate(owner: RefCounted, action: String) -> RefCounted:
	if owner==null: return null
	var next: RefCounted=owner.fork()
	if next==null: return null
	match action:
		"abandon":
			if not _base_archive_paid_regular(owner) or not owner.session.physical.pending.is_empty() \
					or not next.close_current("abandon") or not _base_archive_abandoned_image(next,false): return null
		"complete":
			if not _base_archive_semifinal_settled(owner) or owner.session.phase!=SeasonState.Phase.COMPLETE \
					or not next.close_current("complete") or not _base_archive_completed_closed(next): return null
		"complete_final":
			if not _base_archive_final_settled(owner) or not next.close_current("complete") \
					or not _base_archive_final_closed(next): return null
		"prepare":
			if not _base_archive_abandoned_image(owner,false) or not next.propose_next(BaseArchive._default_next_seed(owner)) \
					or not _base_archive_abandoned_image(next,true): return null
		"prepare_completed":
			if not _base_archive_completed_closed(owner) or not next.propose_next(BaseArchive._default_next_seed(owner)) \
					or not _base_archive_completed_closed(next,true): return null
		"next_completed":
			if not _base_archive_completed_closed(owner,true) or not next.commit_next() \
					or not _base_archive_draft_image(next,0) or not next.session.picks.is_empty(): return null
		"next":
			if not _base_archive_abandoned_image(owner,true) or not next.commit_next() \
					or not _base_archive_draft_image(next,0) or not next.session.picks.is_empty(): return null
		_: return null
	return next

static func _base_archive_read_lifecycle_sibling(prior: Dictionary, wrapped: Dictionary,
		head_bytes: PackedByteArray, pages: Array[PackedByteArray], previous: RefCounted) -> bool:
	if previous==null: return false
	var action: String
	match wrapped.generation:
		BASE_ARCHIVE_ABANDONED_GENERATION:
			if prior.get("generation") not in [BASE_ARCHIVE_SETTLED_GENERATION,BASE_ARCHIVE_SHOP_GENERATION,
					BASE_ARCHIVE_REGULAR_SETTLED_GENERATION]: return false
			action="abandon"
		BASE_ARCHIVE_COMPLETED_GENERATION:
			if prior.get("generation")!=BASE_ARCHIVE_SEMI_SETTLED_GENERATION: return false
			action="complete"
		BASE_ARCHIVE_PREPARED_GENERATION:
			if prior.get("generation")!=BASE_ARCHIVE_ABANDONED_GENERATION: return false
			action="prepare"
		BASE_ARCHIVE_COMPLETED_PREPARED_GENERATION:
			if prior.get("generation")!=BASE_ARCHIVE_COMPLETED_GENERATION: return false
			action="prepare_completed"
		BASE_ARCHIVE_NEXT_BIRTH_GENERATION:
			if prior.get("generation") not in [BASE_ARCHIVE_PREPARED_GENERATION,BASE_ARCHIVE_COMPLETED_PREPARED_GENERATION] \
					or pages.size()<2: return false
			if BaseArchive.append_head(BaseArchive._wire(prior.head),pages.back())!=head_bytes: return false
			action="next_completed" if prior.generation==BASE_ARCHIVE_COMPLETED_PREPARED_GENERATION else "next"
		_: return false
	if action=="abandon":
		if prior.generation==BASE_ARCHIVE_SETTLED_GENERATION and not _base_archive_unshopped_settled_image(previous):
			return false
		if prior.generation==BASE_ARCHIVE_SHOP_GENERATION and not _base_archive_shop_image(previous): return false
		if prior.generation==BASE_ARCHIVE_REGULAR_SETTLED_GENERATION and not _base_archive_regular_settled_image(previous):
			return false
	if action in ["next","next_completed"]:
		var page: Dictionary=BaseArchive._page(pages.back(),prior.head,previous.current_ordinal())
		if page.is_empty() or not Codec.same(page.image,prior.current.ordinary) \
				or not Codec.same(page.close,previous.ordinary_state.last_close): return false
	var result: RefCounted=_base_archive_lifecycle_candidate(previous,action)
	return result!=null and Codec.same(Owner.capture(result),wrapped.current)

static func base_archive_lifecycle_available(owner: RefCounted, expected: PackedByteArray, action: String) -> bool:
	if not base_archive_app_binding(owner,expected): return false
	if action=="complete":
		var source: Dictionary=_base_archive_wrapper(expected)
		if source.get("generation")==BASE_ARCHIVE_FINAL_SETTLED_GENERATION:
			return _base_archive_final_settled(owner)
		return source.get("generation")==BASE_ARCHIVE_SEMI_SETTLED_GENERATION \
			and _base_archive_semifinal_settled(owner) and owner.session.phase==SeasonState.Phase.COMPLETE
	if _base_archive_draft_pin.pages.size()>=BASE_ARCHIVE_QA_MAX_PAGES: return false
	match action:
		"abandon": return _base_archive_paid_regular(owner) and owner.session.physical.pending.is_empty()
		"prepare":
			return _base_archive_abandoned_image(owner,false) or (_base_archive_completed_closed(owner) \
				and _base_archive_wrapper(expected).get("generation")==BASE_ARCHIVE_COMPLETED_GENERATION)
		"next":
			return _base_archive_abandoned_image(owner,true) or (_base_archive_completed_closed(owner,true) \
				and _base_archive_wrapper(expected).get("generation")==BASE_ARCHIVE_COMPLETED_PREPARED_GENERATION)
	return false

static func write_base_archive_lifecycle(app: Variant, ticket: RefCounted) -> bool:
	if not _base_archive_app_valid(app) or not _base_archive_cpu_job.is_empty() or is_instance_valid(app.ai_runner):
		return false
	var action: String=app.take_base_archive_lifecycle(ticket)
	var proof: Dictionary=_base_archive_draft_pin
	var owner: RefCounted=app.career_owner
	var expected: PackedByteArray=app.saved_bytes
	if not base_archive_lifecycle_available(owner,expected,action): return false
	var native_action: String=action
	if action=="complete" and _base_archive_final_settled(owner): native_action="complete_final"
	if action=="prepare" and _base_archive_completed_closed(owner): native_action="prepare_completed"
	if action=="next" and _base_archive_completed_closed(owner,true): native_action="next_completed"
	var candidate: RefCounted=_base_archive_lifecycle_candidate(owner,native_action)
	if candidate==null: return false
	var generation: String=BASE_ARCHIVE_ABANDONED_GENERATION if action=="abandon" else BASE_ARCHIVE_PREPARED_GENERATION
	if action=="complete": generation=BASE_ARCHIVE_COMPLETED_GENERATION
	if native_action=="complete_final": generation=BASE_ARCHIVE_FINAL_CLOSED_GENERATION
	if native_action=="prepare_completed": generation=BASE_ARCHIVE_COMPLETED_PREPARED_GENERATION
	var appended: PackedByteArray
	var head: Dictionary={}
	if action=="next":
		generation=BASE_ARCHIVE_NEXT_BIRTH_GENERATION
		var source: Dictionary=Owner.capture(owner)
		if source.is_empty(): return false
		var old_head: PackedByteArray=BaseArchive._wire(proof.head)
		appended=BaseArchive.make_page(old_head,source.ordinary,owner.ordinary_state.last_close)
		head=BaseArchive._head(BaseArchive.append_head(old_head,appended))
		if head.is_empty() or head.pages.size()>BASE_ARCHIVE_QA_MAX_PAGES: return false
	if not _base_archive_app_valid(app) or not _base_archive_draft_pin_live(proof): return false
	_base_archive_draft_pin={}
	_base_archive_live_op=RefCounted.new()
	return _base_archive_commit_current(_base_archive_live_op,proof,candidate,expected,generation,head,appended)

# The compiled reader returns presentation data only; it must never retain a
# live Store owner, enter a play transaction, consume .bak as primary, or write.
static func base_archive_read_playtest() -> Dictionary:
	if not base_archive_reader_profile(): return {}
	var proof: Dictionary=_base_archive_load(true)
	if proof.is_empty(): return {}
	var history: Array[Dictionary]=_base_archive_history_pages(proof)
	if history.size()!=proof.pages.size() or not Codec.same(_base_archive_pair(true),proof.pair) \
			or not _base_archive_temps_clear(true) or not _base_archive_pages_live(proof.head,proof.pages):
		return {}
	var owner: RefCounted=proof.owner
	return {"season":proof.pages.size()+1,"round":int(owner.session.round_index),
		"cash":int(owner.session.build.cash()),"history":history}

static func base_archive_history(owner: RefCounted, expected: PackedByteArray) -> Array[Dictionary]:
	if not base_archive_app_binding(owner,expected): return []
	var history: Array[Dictionary]=_base_archive_history_pages(_base_archive_draft_pin)
	return history if base_archive_app_binding(owner,expected) else []

static func _base_archive_history_pages(proof: Dictionary) -> Array[Dictionary]:
	var history: Array[Dictionary]=[]
	for i: int in proof.pages.size():
		var page: Dictionary=BaseArchive._page(proof.pages[i],proof.head,i+1)
		if page.is_empty(): return []
		var native: Dictionary=_base_archive_history_record(page)
		if native.is_empty(): return []
		history.append(native)
	return history

# Display-only projection from an already independently replayed archive page.
# Native score proof validates the contiguous authored schedule and standings
# for the whole regular season; do not use Standard's bounded preview helper.
static func _base_archive_history_record(page: Dictionary) -> Dictionary:
	var state: Variant=page.image.get("state")
	var close: Dictionary=page.close
	if not state is Dictionary or not state.get("teams") is Array or state.teams.size()!=6 \
			or not state.get("results") is Array or not state.get("league_rules") is Dictionary: return {}
	var completed: bool=close.reason=="complete"
	if not completed and (state.get("phase")!=SeasonState.Phase.REGULAR \
			or not Codec.whole(state.get("round_index"),1,10) or state.results.size()!=3*int(state.round_index)):
		return {}
	var scores:=SeasonState.new()
	scores.league_rules=LeagueRules.normalized(state.league_rules)
	if scores.league_rules.get("league")!="dugout" or scores.league_rules.get("tier")!=0: return {}
	var names: Array[String]=[]
	for team: Variant in state.teams:
		if not team is Dictionary or not team.get("name") is String \
				or not (team.get("draw") is float or team.get("draw") is int): return {}
		names.append(team.name)
		scores.teams.append({"draw":float(team.draw)})
	for row: Variant in state.results:
		if not row is Dictionary or not Codec.whole(row.get("id"),0,32) \
				or not Codec.whole(row.get("away"),0,5) or not Codec.whole(row.get("home"),0,5) \
				or not Codec.whole(row.get("away_runs"),0,9999) or not Codec.whole(row.get("home_runs"),0,9999):
			return {}
		scores.results.append({"id":int(row.id),"away":int(row.away),"home":int(row.home),
			"away_runs":int(row.away_runs),"home_runs":int(row.home_runs)})
	scores.results.sort_custom(func(a: Dictionary,b: Dictionary) -> bool: return a.id<b.id)
	var proof: Dictionary=ClubSeasonRecord.capture(scores)
	var summary: Dictionary=ClubSeasonRecord.analyze(proof)
	if summary.is_empty() or (summary.finish!="unfinished")!=completed: return {}
	var receipt: Dictionary=ClubSeasonRecord.receipt(proof,close.first_title,0) if completed else {}
	if (completed and receipt.is_empty()) or not Codec.same(receipt,close.receipt): return {}
	return {"ordinal":int(page.ordinal),"teams":names,"summary":summary,"standings":scores.standings(),
		"results":scores.results,"receipt":receipt,"close":close.duplicate(true)}
