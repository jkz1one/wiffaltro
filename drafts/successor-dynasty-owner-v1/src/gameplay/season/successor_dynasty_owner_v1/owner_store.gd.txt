# A display-only memo for one admitted archive. Never consulted by a save,
# restore, league gate or the pending-result projection.
static var _previous_standard_pin: Dictionary = {}

static var previous_standard_projection_calls: int = 0

static func previous_standard_available(owner: RefCounted, expected: PackedByteArray) -> bool:
	if saving or _ticket!=null or owner==null or owner!=_owner or owner.session!=_owner_session \
		or expected.is_empty() or expected!=_owner_bytes:
		return false
	var pin: Variant = _metadata.get("ordinary_state",{}).get("archive")
	return pin is Dictionary and RawGuard.same(owner.ordinary_state.archive,pin) \
		and ((Owner.Common.StandardArchive.marked(owner.session) and Owner.Common.StandardArchive.shape(owner)) \
			or (LeagueAdmission.starter_player() and Owner.Common.starter_dugout_archive(owner)))

static func previous_standard_season(owner: RefCounted, expected: PackedByteArray) -> Dictionary:
	if not previous_standard_available(owner,expected) or _starter_slot_linked() \
		or not _write_revision_matches(expected,true): return {}
	var pin: Dictionary = _metadata.ordinary_state.archive
	if not RawGuard.same(pin,_previous_standard_pin):
		previous_standard_projection_calls += 1
		var view: Dictionary = Owner.Common.StandardArchive._display_scores(Owner._bytes_value(pin.source.to_utf8_buffer()),true) \
			if pin.get("kind")==Owner.Common.STARTER_DUGOUT_ARCHIVE else Owner.Common.StandardArchive.display(pin)
		if not previous_standard_available(owner,expected) or not _write_revision_matches(expected,true): return {}
		_previous_standard_pin = pin.duplicate(true)
		_previous_standard_view = view
	return _previous_standard_view.duplicate(true)

static func standard_history(owner: RefCounted, expected: PackedByteArray) -> Array:
	if not standard_history_available(owner,expected) or _starter_slot_linked() \
		or not _write_revision_matches(expected,true): return []
	var pin: Dictionary = _metadata.ordinary_state.archive
	if not RawGuard.same(pin,_standard_history_pin):
		standard_history_projection_calls += 1
		var mixed: bool=Owner.Mixed.marked_state(owner.ordinary_state)
		var appended: bool=Owner.MixedAppend.marked_state(owner.ordinary_state)
		var view: Array = Owner.MixedAppend.display(owner) if appended else (Owner.Mixed.display(pin) if mixed else (
			_starter_dugout_history(pin) if pin.get("kind")==Owner.Common.STARTER_DUGOUT_CHAIN
			else Owner.Common.StandardChain.display(pin)))
		var expected_count: int=(5 if owner.current_ordinal()==Owner.MixedAppend.ROLLOVER_ORDINAL else 4) if appended else (3 if mixed else 2)
		if view.size()!=expected_count or not standard_history_available(owner,expected) \
			or not RawGuard.same(pin,_metadata.ordinary_state.archive) or not _write_revision_matches(expected,true):
			return []
		_standard_history_pin = pin.duplicate(true)
		_standard_history_view = view
	return _standard_history_view.duplicate(true)

# Display only: the two exact members were natively admitted on cold
# Continue or a fixed Store promotion. No write path consumes these views.
static func _starter_dugout_history(pin: Dictionary) -> Array:
	var second:=Owner.Common.starter_dugout_second_bytes(pin)
	if second.is_empty(): return []
	var first:=Owner.Common.StandardArchive._display_scores(Owner._bytes_value(pin.entries[0].source.to_utf8_buffer()),true)
	var latest:=Owner.Common.StandardArchive._display_scores(Owner._bytes_value(second),true)
	if first.get("ordinal")!=1 or latest.get("ordinal")!=2: return []
	first["league"]="standard"
	latest["league"]="dugout"
	return [first,latest]

# Clear-only invalidation has no admission/write authority.
static func forget_projection() -> void:
	_projection_ticket = null; _projection_checkpoint = null; _projection_owner = null

static func _starter_slot_linked() -> bool:
	if not LeagueAdmission.starter_feature(): return false
	var root: String=OS.get_user_data_dir().trim_suffix("/")
	var parent:=DirAccess.open(root.get_base_dir())
	if parent==null or parent.is_link(root.get_file()): return true
	var dir:=DirAccess.open("user://")
	if dir==null: return true
	for suffix: String in ["", ".bak", ".tmp"]:
		if dir.is_link((PATH+suffix).get_file()): return true
	return false

static func raw() -> PackedByteArray:
	if _starter_slot_linked(): return PackedByteArray()
	return FileAccess.get_file_as_bytes(PATH) if FileAccess.file_exists(PATH) else PackedByteArray()
static func occupied() -> bool:
	if _starter_slot_linked(): return true
	for suffix: String in ["", ".bak", ".tmp"]:
		if FileAccess.file_exists(PATH+suffix) or DirAccess.dir_exists_absolute(PATH+suffix): return true
	return false
# Bounded prefix probe; unsupported family versions remain protected.
static func standard_d1_slot_kind() -> String:
	if _starter_slot_linked(): return ""
	var contract=preload("standard_d1_transition.gd")
	var file:=FileAccess.open(PATH,FileAccess.READ)
	if file==null: return ""
	var prefix: String=file.get_buffer(contract.SLOT_HEADER.length()).get_string_from_ascii()
	var size:=file.get_length()
	file.close()
	if not prefix.begins_with(contract.SLOT_FAMILY): return ""
	return "supported" if prefix==contract.SLOT_HEADER and size<=contract.MAX_SLOT else "unsupported"

static func _starter_owner_allowed(owner: RefCounted) -> bool:
	if owner!=null and Owner.MixedAppend.marked_state(owner.ordinary_state):
		return private_mixed_append() and (Owner.MixedAppend.draft_scope(owner) or Owner.MixedAppend.season_scope(owner) \
			or Owner.MixedAppend.closed_complete_scope(owner) or Owner.MixedAppend.rollover_play_scope(owner) \
			or Owner.MixedAppend.rollover_closed_complete_scope(owner))
	if Flat.is_flat(owner): return false
	if owner!=null and Owner.Mixed.marked_state(owner.ordinary_state):
		return LeagueAdmission.starter_player() and (Owner.Mixed.season_owner_scope(owner) or Owner.Mixed.closed_complete_scope(owner))
	if owner!=null and owner.ordinary_state.get("archive",{}).get("kind")==Owner.Common.STARTER_DUGOUT_ARCHIVE:
		return LeagueAdmission.starter_player() and Owner.Common.starter_dugout_archive(owner) and Owner.Common._archive_shape(owner)
	if owner!=null and owner.ordinary_state.get("archive",{}).get("kind")==Owner.Common.STARTER_DUGOUT_CHAIN:
		return LeagueAdmission.starter_player() and Owner.Common.starter_dugout_chain_scope(owner)
	if owner!=null and Owner.Common.StandardWorkshop.Home.opening_chain(owner.session):
		return (LeagueAdmission.starter_player() or private_standard_chain_opening()) and _starter_opening_owner(owner)
	if not LeagueAdmission.starter_feature(): return true
	if not LeagueAdmission.starter_player() or owner==null or owner.get_script()!=Owner \
		or owner.active!="ordinary" or owner.session==null or not owner.paused.is_empty() \
		or not Owner._missing_dynasty(owner) or not owner.migration_v3.is_empty() \
		or owner.ordinary_state.has(Owner.Common.StandardWorkshop.APPLIED) \
		or (owner.ordinary_state.has("archive") and not Owner.Common.StandardArchive.shape(owner) \
			and not Owner.Common.StandardChain.shape(owner)) \
		or Owner.Common.cross_proposal_kind(owner.ordinary_state.get("proposal",{}).get("kind")): return false
	var current: SeasonState=owner.session
	var workshop=Owner.Common.StandardWorkshop
	if owner.ordinary_state.has(workshop.OPENING) or not current.home_property.is_empty():
		return _starter_opening_owner(owner)
	# Standard Base's authored envelope has no named league rules. This is
	# not a fallback for missing fields; the strict native codec took priority.
	if current.league_rules.is_empty():
		return not current.league_live and current.career!=null \
			and current.career.start_basis.get("kind")!="common-standard-d1-transition-v1"
	return current.league_live and current.league_rules.get("league")=="dugout" \
		and current.league_rules.get("tier",-1)==0

# The strict worker authenticates these fields before the parent sees
# the seal. JSON has erased the distinction between integer and whole float;
# rebuild the same explicitly authored metadata numerics as the native birth
# so the first normal write matches a cold native owner byte for byte.
static func _mixed_continue_authored_metadata(owner: RefCounted) -> bool:
	var state: Dictionary=owner.ordinary_state
	var transition: Variant=state.get("last_transition")
	if not transition is Dictionary or not transition.get("basis") is Dictionary: return false
	var basis: Dictionary=transition.basis
	if not Owner.Common.Start.valid(basis) or not Owner.Common.Start.detached_mixed_flat(basis) \
			or not basis.get("pennants") is Dictionary: return false
	var pennants: Dictionary=basis.pennants
	for row: Array in [[state,["ordinal"]],[transition,["from","to"]],
			[basis,["common_sequence","ordinal","seed","version"]],
			[pennants,["from_season","mastery_from","version"]]]:
		var target: Dictionary=row[0]
		for field: String in row[1]:
			if not Codec.whole(target.get(field),0,2147483647): return false
			target[field]=int(target[field])
	return true

static func accept_mixed_continue(expected: PackedByteArray,child: Node) -> RefCounted:
	var process_script: Script=load("res://src/gameplay/season/successor_dynasty_owner_v1/starter_mixed_continue.gd")
	if not LeagueAdmission.starter_fast_worker() or saving or _owner!=null or _ticket!=null \
			or child==null or child.get_script()!=process_script or expected.is_empty() \
			or expected.size()>MAX_BYTES or _starter_slot_linked() or not _write_revision_matches(expected,true): return null
	var revision:=Owner._hash(expected)
	var seal: Dictionary=child.take_current_seal(revision)
	if seal.is_empty(): return null
	var valid_binding: Callable = func() -> bool:
		return is_instance_valid(child) and child.binding_live(seal.nonce,revision) \
			and not saving and _owner==null and _ticket==null
	if not valid_binding.call(): return null
	var tree:=child.get_tree()
	child.binding_stage="revision"
	var payload: Variant=Owner._bytes_value(expected)
	if not _mixed_continue_shape(payload) or process_script.current_digest(payload.ordinary)!=seal.current: return null
	await tree.process_frame
	if not valid_binding.call(): return null
	child.binding_stage="metadata"
	var owner:=Owner._restore_metadata(payload)
	if owner==null or not _mixed_continue_authored_metadata(owner): return null
	await tree.process_frame
	if not valid_binding.call(): return null
	child.binding_stage="typed"
	owner.session=await Adapter.Ordinary._hydrate_mixed_continue(payload.ordinary,child,valid_binding)
	if owner.session==null or not valid_binding.call() \
			or not Owner.Mixed.season_owner_scope(owner): return null
	child.binding_stage="native bank representation"
	if not process_script.apply_bank_witness(owner.session,seal.banks) or not valid_binding.call(): return null
	child.binding_stage="native bracket representation"
	if not process_script.apply_bracket_witness(owner.session,seal.bracket) or not valid_binding.call(): return null
	await tree.process_frame
	if not valid_binding.call(): return null
	child.binding_stage="capture"
	var current: Dictionary=await Adapter.Ordinary._recapture_mixed_continue(owner.session,child,valid_binding)
	if current.is_empty() or not valid_binding.call(): return null
	child.binding_stage="compare"
	var rebuilt:=Owner._pack_fields(owner,current)
	if not Codec.keys(payload,rebuilt.keys()) or not Codec.keys(payload.ordinary,current.keys()): return null
	for field: String in rebuilt:
		if field!="ordinary" and not Codec.same(rebuilt[field],payload[field]): return null
	for field: String in current:
		await tree.process_frame
		if not valid_binding.call() \
				or not Codec.same(current[field],payload.ordinary[field]): return null
	await tree.process_frame
	if not valid_binding.call(): return null
	child.binding_stage="raw guard"
	var snapshot: Dictionary=await RawGuard.capture_mixed_continue(owner,child,valid_binding)
	if not valid_binding.call(): return null
	if snapshot.is_empty() or saving or _owner!=null or _ticket!=null or _starter_slot_linked() \
			or not _write_revision_matches(expected,true) or not child.binding_live(seal.nonce,revision,true): return null
	_invalidate()
	_owner=owner; _owner_session=owner.session; _owner_raw=snapshot; _owner_bytes=expected.duplicate()
	_paused=owner.paused.duplicate(true); _active=owner.active
	_metadata=Codec.data_fields(owner,Owner.META_FIELDS); _imported_sha256=owner.imported_sha256
	return owner

static func read() -> RefCounted:
	return _read_common_slot(false)

static func _read_common_slot(flat: bool) -> RefCounted:
	if flat and (saving or _ticket!=null or not private_standard_chain_opening()): return null
	forget_projection()
	last_error = ""
	if _starter_slot_linked():
		last_error="This Starter Beta slot points outside its own files. No files were changed."
		return null
	if not standard_d1_slot_kind().is_empty():
		last_error="This save is outside the Starter Beta slot. No files were changed." if LeagueAdmission.starter_feature() else \
			"This Standard D1 slot requires private Continue. No files were changed."
		return null
	if not FileAccess.file_exists(PATH):
		last_error = "No common career is saved in this slot. Earlier saves are unchanged."
		return null
	var file: FileAccess = FileAccess.open(PATH,FileAccess.READ)
	if file == null or file.get_length() > MAX_BYTES:
		last_error = "The save could not be read safely. It has not been changed."
		return null
	var bytes: PackedByteArray = file.get_buffer(file.get_length())
	file.close()
	var parser: JSON = JSON.new()
	if parser.parse(bytes.get_string_from_utf8()) != OK:
		last_error = "The save is unreadable. Original and backup are preserved."
		return null
	if parser.data is Dictionary and Owner.MixedAppend.marked_state(parser.data.get("ordinary_state")) \
			and not private_mixed_append():
		last_error="This test career requires its isolated qualification build. No files were changed."
		return null
	var season: RefCounted = Flat._restore_private(parser.data) if flat else Owner.restore(parser.data)
	if season == null:
		last_error = "This common career has an unsupported version or inconsistent state. No files were changed."
		return null
	if not flat and not _starter_owner_allowed(season):
		last_error="This save is outside the Starter Beta slot. No files were changed."
		return null
	var snapshot: Dictionary = RawGuard.capture(season)
	if snapshot.is_empty() or raw() != bytes:
		last_error = "The save changed during read or could not be admitted. No files were changed."
		return null
	_invalidate()
	_owner = season; _owner_session = season.session; _owner_raw = snapshot; _owner_bytes = bytes.duplicate()
	_paused = season.paused.duplicate(true); _active = season.active
	_metadata = Codec.data_fields(season,Owner.META_FIELDS); _imported_sha256 = season.imported_sha256
	return season

static func save(season: RefCounted, expected: PackedByteArray, expected_snapshot: Dictionary = {}) -> bool:
	forget_projection()
	if saving: return false
	saving = true
	var result: bool = _write(season,expected,expected_snapshot)
	saving = false
	if result: _invalidate()
	return result
static func _matches(expected: PackedByteArray) -> bool:
	if _starter_slot_linked(): return false
	# Generic read/admission/write/retry paths never own the D1 dialect.
	if not standard_d1_slot_kind().is_empty(): return false
	return not DirAccess.dir_exists_absolute(PATH) and ((expected.is_empty() and not FileAccess.file_exists(PATH)) or (not expected.is_empty() and FileAccess.file_exists(PATH) and raw() == expected))
static func _write(season: RefCounted, expected: PackedByteArray, expected_snapshot: Dictionary) -> bool:
	# Legacy unbound writer never modifies the private paired slot.
	if season==null or season.get_script()!=Owner or _paired_owner(season) or _paired_settled(_owner): return false
	if not _starter_owner_allowed(season): return false
	last_error = "The exact candidate is retained. Resolve the save obstruction, then retry."
	if not _matches(expected):
		last_error = "The saved file changed. It was not overwritten; your candidate remains retained."
		return false
	for suffix: String in ["", ".bak", ".tmp"]:
		if DirAccess.dir_exists_absolute(PATH+suffix): return false
	if FileAccess.file_exists(PATH+".tmp") or (expected.is_empty() and FileAccess.file_exists(PATH+".bak")): return false
	# Unbound legacy writes cannot create, replace, or erase workshop metadata.
	var previous: Variant=Owner._bytes_value(expected)
	# The unbound writer cannot erase or manufacture a typed career proposal.
	if season.ordinary_state.has("archive") or Owner.Common.cross_proposal_kind(season.ordinary_state.get("proposal",{}).get("kind")): return false
	if previous is Dictionary and previous.get("ordinary_state") is Dictionary:
		var prior_proposal: Variant=previous.ordinary_state.get("proposal")
		if previous.ordinary_state.has("archive") or (prior_proposal is Dictionary and Owner.Common.cross_proposal_kind(prior_proposal.get("kind"))): return false
	if not season.workshop_draft.is_empty() or not season.workshop_close_history.is_empty() or not season.workshop_chair_purchase.is_empty() or not season.workshop_applied.is_empty(): return false
	if previous is Dictionary and (previous.has("workshop_draft") or previous.has("workshop_close_history") or previous.has("workshop_chair_purchase") or previous.has("workshop_applied")): return false
	var data: Dictionary = Owner.capture(season)
	if data.is_empty() or (not expected_snapshot.is_empty() and not Codec.same(data,expected_snapshot)):
		last_error = "This candidate is inconsistent with the materialized save contract."
		return false
	return _write_data(data,expected)

static func _native_encoded(data: Dictionary) -> PackedByteArray:
	var bytes: PackedByteArray = JSON.stringify(data,"",true,true).to_utf8_buffer()
	if bytes.is_empty() or bytes.size() > MAX_BYTES: return PackedByteArray()
	# Validate the encoded representation against the consistent native value.
	var parser: JSON = JSON.new()
	if parser.parse(bytes.get_string_from_utf8()) != OK or not Codec.same(parser.data,data):
		return PackedByteArray()
	return bytes

static func _write_data(data: Dictionary, expected: PackedByteArray) -> bool:
	var bytes := _native_encoded(data)
	return not bytes.is_empty() and _write_exact_bytes(bytes,expected)

static func _write_exact_bytes(bytes: PackedByteArray, expected: PackedByteArray) -> bool:
	return _write_exact_result(bytes,expected,false)=="saved"

static func _write_exact_result(bytes: PackedByteArray, expected: PackedByteArray, bounded: bool) -> String:
	var file: FileAccess = FileAccess.open(PATH+".tmp",FileAccess.WRITE)
	if file == null: return "not_renamed"
	file.store_buffer(bytes); file.flush()
	var ok: bool = file.get_error() == OK
	file.close()
	if not ok or FileAccess.get_file_as_bytes(PATH+".tmp") != bytes or not _write_revision_matches(expected,bounded): return "not_renamed"
	if not expected.is_empty() and DirAccess.copy_absolute(PATH,PATH+".bak") != OK: return "not_renamed"
	if not _write_revision_matches(expected,bounded): return "not_renamed"
	if DirAccess.rename_absolute(PATH+".tmp",PATH) != OK: return "not_renamed"
	var readback_ok: bool=_write_revision_matches(bytes,true) if bounded else raw()==bytes
	if not readback_ok: return "renamed_unverified"
	last_error = ""
	return "saved"

# The bounded cross path never reads a replacement of arbitrary length and
# refuses linked slot objects. The stock write protocol is otherwise unchanged.
static func _write_revision_matches(bytes: PackedByteArray,bounded: bool) -> bool:
	if not bounded: return _matches(bytes)
	var dir:=DirAccess.open("user://")
	if dir==null or dir.is_link(PATH.get_file()): return false
	var f:=FileAccess.open(PATH,FileAccess.READ)
	if f==null or f.get_length()!=bytes.size(): return false
	var same: bool=f.get_buffer(bytes.size())==bytes and f.get_length()==bytes.size()
	f.close()
	return same

# Pure refusal predicate, never a writer or an admission token. Routine actions
# cannot replace either session or provenance through the generic candidate API.
static func _routine_pins(candidate: RefCounted) -> bool:
	if candidate.active!=_active or not RawGuard.same(candidate.paused,_paused) or candidate.imported_sha256!=_imported_sha256: return false
	if not RawGuard.same(candidate.migration_v2,_metadata.get("migration_v2")): return false
	for name: String in ["common","ordinary_state","migration_v3","workshop_draft","workshop_close_history","workshop_chair_purchase","workshop_applied"]:
		if not RawGuard.same(candidate.get(name),_metadata.get(name)): return false
	if _active=="ordinary": return RawGuard.same(Codec.data_fields(candidate,Owner.META_FIELDS),_metadata)
	return candidate.birth_seed==_metadata.get("birth_seed")

static func _paired_owner(value: RefCounted) -> bool:
	return value!=null and value.get_script()==Owner and value.paired_draft_only()

static func _paired_settled(value: RefCounted) -> bool:
	return _paired_owner(value) and value.session.round_index!=0

static func prepare(owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> RefCounted:
	forget_projection()
	if saving or _ticket != null or candidate == null or candidate == owner or _paired_settled(owner): return null
	if candidate.get_script()!=Owner or candidate.session==null or _paired_settled(candidate): return null
	if not _starter_owner_allowed(candidate): return null
	if owner == null:
		if candidate.ordinary_state.has("archive") or Owner.Common.cross_proposal_kind(
				candidate.ordinary_state.get("proposal",{}).get("kind")): return null
		if not expected.is_empty() or not _matches(expected) or not candidate.workshop_draft.is_empty() or not candidate.workshop_close_history.is_empty() or not candidate.workshop_chair_purchase.is_empty() or not candidate.workshop_applied.is_empty(): return null
	else:
		if not match_owner(owner,expected) or candidate.session==owner.session: return null
	# Only this private boundary creates the writer's canonical payload. Callers
	# can receive detached diagnostics, never supply a substitute validated value.
	var data: Dictionary = {}
	if owner != null:
		if not _routine_pins(candidate): return null
		# Dormant alternate was strictly admitted on read/promotion. Exact live raw
		# owner pins were checked above; no caller validation flag or payload.
		if Owner.MixedAppend.marked_state(candidate.ordinary_state):
			data=Owner.MixedAppend.pick_payload(owner,candidate)
		elif Owner.Mixed.marked_state(candidate.ordinary_state):
			data=Owner.Mixed.draft_pick_payload(owner,candidate)
		else:
			var current: Dictionary = Adapter.capture(candidate.active,candidate.session)
			if not current.is_empty():
				data = Owner.pack_current(candidate,current)
	else:
		data = Owner.capture(candidate)
	if data.is_empty(): return null
	var snapshot: Dictionary = RawGuard.capture(candidate)
	if snapshot.is_empty(): return null
	if owner == null:
		_owner = null; _owner_session = null; _owner_raw = {}; _owner_bytes = expected.duplicate()
	_ticket = RefCounted.new()
	_candidate = candidate; _candidate_session = candidate.session; _candidate_raw = snapshot; _candidate_data = data
	_expected = expected.duplicate(); _written = PackedByteArray()
	return _ticket

## One synchronous Store-owned admission/write stack. No token or caller flag
## permits skipping later checks. prepare strictly captures the candidate and
## matches the live owner; no callbacks or external handoff intervene here.
## On obstruction the ordinary private ticket survives, and retry stays strict.
static func prepare_and_write(owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> RefCounted:
	var ticket: RefCounted = prepare(owner,candidate,expected)
	if ticket == null: return null
	_write_admitted()
	return ticket

## Fixed synchronous producer: authenticate before constructing any candidate.
## Only named runtime/build factories admitted by match_owner can execute here.
## No UI callbacks, awaits, caller candidate or unchecked payload are accepted.
static func prepare_shop_and_write(owner: RefCounted, expected: PackedByteArray, command: Dictionary) -> RefCounted:
	forget_projection()
	last_error = ""
	if saving or _ticket != null or not Codec.primitive(command) or not SHOP_PRODUCER_OPS.has(str(command.get("op",""))): return null
	if not match_owner(owner,expected): return null
	var session: SeasonState = owner.session
	var mixed:=mixed_session_scope(owner)
	var appended:=mixed_append_shop_scope(owner)
	if (not owner.playable() and not appended) or (not Adapter.scope(owner.active,session) and not mixed and not appended) \
			or not session.physical.pending.is_empty() or not session.shop_available(): return null
	if (mixed or appended) and not Owner.Mixed.shop_command_allowed(session,command): return null
	var request_id: Variant = command.get("id")
	if request_id is String and not request_id.is_empty() and session.build._requests.has(request_id): return null
	var next_session: SeasonState = session.fork_live()
	if next_session == null: return null
	var next_build: SeasonBuild = next_session.build.candidate(command.duplicate(true))
	if next_build == null:
		last_error = next_session.build.last_error
		return null
	next_session.adopt_build(next_build)
	if next_session.build != next_build: return null
	var envelope: RefCounted = Owner.new()
	envelope.session = next_session
	if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
	# Keep this registration inside the checked producer, rather than exposing
	# an unchecked owner/candidate helper (underscore methods are callable).
	if not _routine_pins(envelope): return null
	var current: Dictionary = {} if mixed or appended else Adapter.capture(envelope.active,next_session)
	var data: Dictionary = Owner.MixedAppend.payload(envelope) if appended else (
		Owner.Mixed.compact_payload(envelope) if mixed else (
			Owner.pack_current(envelope,current) if not current.is_empty() else {}))
	if data.is_empty():
		last_error = "The candidate could not be represented consistently. Saved progress is unchanged."
		return null
	var snapshot: Dictionary = RawGuard.capture(envelope)
	if snapshot.is_empty(): return null
	_ticket = RefCounted.new()
	_candidate = envelope; _candidate_session = next_session; _candidate_raw = snapshot; _candidate_data = data
	_expected = expected.duplicate(); _written = PackedByteArray()
	_write_admitted()
	return _ticket

static func _prepare_mixed_round(owner: RefCounted, expected: PackedByteArray, op: String, native_input: Variant=null) -> Dictionary:
	forget_projection()
	if saving or _ticket!=null or not mixed_round_scope(owner) or not match_owner(owner,expected): return {}
	var current: SeasonState=owner.session
	if current.phase in [SeasonState.Phase.DRAFT,SeasonState.Phase.COMPLETE] or current.build.pack_pending(): return {}
	var next: SeasonState
	match op:
		"human":
			if not native_input is Object or not is_instance_valid(native_input) \
					or native_input.get_script()!=load("res://src/presentation/season/successor_dynasty_owner_v1/app.gd") \
					or native_input.career_owner!=owner or native_input.season!=current or native_input.saved_bytes!=expected \
					or not mixed_round_entry(owner): return {}
			var human: Array=Adapter.OrdinaryCapture.capture(native_input)
			if human.size()!=13 or human[0]!=current.pending_fixture().get("id"): return {}
			next=current.fork_live()
			if next==null: return {}
			next.physical.pending={"human":human.duplicate(true),"reports":[]}
			if Adapter.OrdinaryPending.project(next).has("error"): return {}
		"report":
			if not native_input is Dictionary or current.physical.pending.is_empty(): return {}
			next=Adapter.OrdinaryPending.stage_report(current,native_input).get("candidate")
		"settle":
			if native_input!=null or current.physical.pending.is_empty(): return {}
			var projected: Dictionary=Adapter.OrdinaryPending.project(current)
			if projected.has("fixture") or projected.has("error"): return projected
			next=projected.get("candidate")
		_: return {}
	if next==null: return {}
	var envelope: RefCounted=Owner.new()
	envelope.session=next
	if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)) \
			or not _routine_pins(envelope): return {}
	var data: Dictionary=Owner.MixedAppend.payload(envelope) if Owner.MixedAppend.marked_state(owner.ordinary_state) \
		else Owner.Mixed.compact_payload(envelope)
	var snapshot: Dictionary=RawGuard.capture(envelope) if not data.is_empty() else {}
	if snapshot.is_empty() or not match_owner(owner,expected): return {}
	_ticket=RefCounted.new()
	_candidate=envelope;_candidate_session=next;_candidate_raw=snapshot;_candidate_data=data
	_expected=expected.duplicate();_written=PackedByteArray()
	_write_admitted()
	return {"ticket":_ticket}

## Fixed producer invokes the named pending implementation itself. The optional
## terminal result is retained only after this exact immediate checkpoint write.
## A failed write/retry keeps the ordinary ticket, but deliberately loses reuse.
static func prepare_report_and_write(owner: RefCounted, expected: PackedByteArray, report: Dictionary) -> RefCounted:
	if mixed_round_scope(owner): return _prepare_mixed_round(owner,expected,"report",report).get("ticket")
	forget_projection()
	if saving or _ticket != null or not match_owner(owner,expected): return null
	if not owner.playable() or not Adapter.scope(owner.active,owner.session) or owner.session.physical.pending.is_empty(): return null
	var staged: Dictionary = Adapter.pending(owner.active).stage_report(owner.session,report)
	var current: SeasonState = staged.get("candidate")
	if current == null: return null
	var envelope: RefCounted = Owner.new()
	envelope.session = current
	if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
	var settled: RefCounted
	var snapshot: Dictionary = {}
	if staged.get("settled") is SeasonState:
		settled = Owner.new(); settled.session = staged.settled
		if not Codec.install_fields(settled,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
		if not _routine_pins(settled) or not settled.session.physical.pending.is_empty() \
			or not settled.session.physical.needed.is_empty() or settled.session.physical.projecting: return null
		snapshot = RawGuard.capture(settled)
		if snapshot.is_empty(): return null
	var ticket: RefCounted = prepare_and_write(owner,envelope,expected)
	if ticket != null and not _written.is_empty() and settled != null:
		_projection_ticket = ticket; _projection_checkpoint = envelope
		_projection_value = settled; _projection_raw = snapshot
	return ticket

# Quick UI and CAS gate only; the child still restores and validates everything.
static func eligible_private_pair_settlement(owner: RefCounted) -> bool:
	if not private_pair_settlement_preview() or owner==null or not owner.paired_draft_only() \
		or owner.active!="ordinary" or owner.lifecycle_status()!="active" or owner.current_ordinal()!=1 \
		or not Owner._missing_dynasty(owner) or not owner.paused.is_empty() or owner.session.round_index not in [1,2,3]: return false
	var pending: Dictionary=owner.session.physical.pending
	var projection: Script=load("res://src/gameplay/season/successor_live_play_v1/paired_first_projection.gd")
	if not projection.bound_shape(pending) or pending.reports.size()!=2: return false
	if owner.session.round_index==2:
		return pending.human[0]==6 and not Adapter.Ordinary.unpurchased_next_pair_kind(owner.session).is_empty()
	if owner.session.round_index==3:
		return pending.human[0]==9 and not Adapter.Ordinary.unpurchased_next_reverse_kind(owner.session).is_empty()
	return pending.human[0]==3 and not Adapter.Ordinary.unpurchased_return_kind(owner.session.build).is_empty()

static func accept_private_pair_settlement(owner: RefCounted,expected: PackedByteArray,child: Node) -> String:
	var process_script: Script=load("res://src/gameplay/season/successor_dynasty_owner_v1/standard_d1_witness_process.gd")
	if child==null or child.get_script()!=process_script or not eligible_private_pair_settlement(owner): return "refused"
	if not _cross_bound(owner,expected): return "stale"
	var bytes: PackedByteArray=child.take_pair_settled_bytes(Owner._hash(expected),owner.session.round_index)
	if bytes.is_empty(): return "refused"
	var dir:=DirAccess.open("user://")
	if dir==null: return "obstructed"
	for suffix: String in ["", ".bak", ".tmp"]:
		if dir.is_link((PATH+suffix).get_file()) or DirAccess.dir_exists_absolute(PATH+suffix): return "obstructed"
	if FileAccess.file_exists(PATH+".tmp"): return "obstructed"
	saving=true
	var outcome: String=_write_exact_result(bytes,expected,true)
	saving=false
	if outcome in ["saved","renamed_unverified"]:
		_invalidate()
		return outcome
	return "unverified_io" if _write_revision_matches(expected,true) else "stale"

# Exact native representation on the detached candidate only. No changed
# semantic/materialized inputs are allowed to reach the frozen signer.
static func _bind_visit6_fresh_profile(build: SeasonBuild) -> bool:
	var id: String=Adapter.Ordinary.VISIT6_RECRUIT
	var outgoing: String=Adapter.Ordinary.VISIT6_OUTGOING
	if not Adapter.Ordinary.visit6_recruit_quote(build,false) or build.cash()<int(build._visit.recruit.price) or build._contracts.has(id) or build.roster().has(id) or build._blocked.has(id) or not build._pool.has(id) or not Adapter.Ordinary.same(build.roster(),build._initial_roster) or not build.roster().has(outgoing) or SeasonSchoolSponsors.scholarship(build).get("player")==outgoing: return false
	var profile: Dictionary=build.player(id)
	var original: Dictionary=SeasonPlayerCatalog.profile(id)
	if original.is_empty() or not Adapter.Ordinary.same(profile,original): return false
	var before: String=_canonical_shop_bytes(Adapter.Ordinary.Builds.capture(build))

	var after: Dictionary=Adapter.Ordinary.Builds.capture(build)
	return before==_canonical_shop_bytes(after)

## Same active owner only. On cold/cache miss, use the normal complete projection.
## Partial projections are returned solely to construct the next real runner.
## Terminal candidates still cross the existing full semantic prepare boundary.
static func prepare_settlement_and_write(owner: RefCounted, expected: PackedByteArray) -> Dictionary:
	if mixed_round_scope(owner):
		var result:=_prepare_mixed_round(owner,expected,"settle")
		return result if not result.is_empty() else {"error":"The saved round could not advance."}
	if saving or _ticket != null:
		forget_projection(); return {"error":"A checkpoint is already retained."}
	if not match_owner(owner,expected): return {"error":"The current owner changed."}
	if not owner.playable() or not Adapter.scope(owner.active,owner.session) or owner.session.physical.pending.is_empty():
		forget_projection(); return {"error":"No current round is pending."}
	var envelope: RefCounted
	if _projection_owner == owner:
		if _projection_value == null or not RawGuard.matches(_projection_value,_projection_raw):
			forget_projection(); return {"error":"The retained terminal projection changed."}
		envelope = _projection_value
	forget_projection()
	if envelope == null:
		var projected: Dictionary = Adapter.pending(owner.active).project(owner.session)
		if projected.has("error") or projected.has("fixture"): return projected
		var current: SeasonState = projected.get("candidate")
		if current == null: return {"error":"The current round did not produce a candidate."}
		envelope = Owner.new(); envelope.session = current
		if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)):
			return {"error":"The round metadata could not be preserved."}
	var ticket: RefCounted = prepare_and_write(owner,envelope,expected)
	return {"ticket":ticket} if ticket != null else {"error":"The settled candidate could not be admitted."}

	else: return null

	_expected=expected.duplicate();_written=PackedByteArray()

	return _ticket

# Narrow read-only worker prerequisite; neither caller data nor the returned
# bytes authorize writing. The ordinary lineage/ordinal derives a fixed
# seed; the private engineering writer still accepts its existing seed.
static func review_starter_dugout(closed_bytes: PackedByteArray) -> Dictionary:
	if closed_bytes.is_empty() or closed_bytes.size()>Owner.Common.ARCHIVE_MAX: return {}
	if closed_bytes.get_string_from_utf8().to_utf8_buffer()!=closed_bytes: return {}
	var input: Variant=Owner._bytes_value(closed_bytes)
	var owner: RefCounted=Owner.restore(input)
	if owner==null or owner.active!="ordinary" or not owner.paused.is_empty() or not Owner._missing_dynasty(owner) \
		or not owner.migration_v3.is_empty() or not owner.workshop_draft.is_empty() \
		or not owner.workshop_applied.is_empty() or not owner.workshop_chair_purchase.is_empty() \
		or not owner.workshop_close_history.is_empty() or owner.ordinary_state.has("archive"):
		return {}
	var state: Dictionary=owner.ordinary_state
	if state.get("ordinal")!=1 or not state.get("lineage") is String: return {}
	var next_ordinal: int=int(state.ordinal)+1
	var seed: int=("ordinary:%s:%d:nextseed"%[state.lineage,next_ordinal]).hash() & 0x7fffffff
	var proposal:=_cross_proposal_projection(owner,closed_bytes,"dugout",seed)
	if proposal.is_empty(): return {}
	var proposed:=_native_encoded(proposal.data)
	if proposed.is_empty() or proposed.size()>Owner.Common.ARCHIVE_MAX: return {}
	var projected:=_cross_activation_projection(proposal.owner,proposed,"dugout")
	if projected.is_empty(): return {}
	var draft:=_native_encoded(projected.data)
	if draft.is_empty(): return {}
	# Check the actual future persisted bytes, not the detached object alone.
	var restored: RefCounted=Owner.restore(Owner._bytes_value(draft))
	if restored==null or restored.current_ordinal()!=2 or not restored.session.picks.is_empty():
		return {}
	return {"proposal":proposed,"draft":draft}

# Deterministic reconstruction only; this helper never installs Store authority.
# The child separately compares the full serialized current session to this
# native result after strict archive replay. No other phases or modes enter.
static func _cross_draft_owner(value: Variant) -> RefCounted:
	if not value is Dictionary or value.get("format")!=Owner.FORMAT or value.get("version")!=4 \
		or value.get("active")!="ordinary" or value.get("dynasty")!={}:
		return null
	var state: Variant=value.get("ordinary_state")
	var current: Variant=value.get("ordinary")
	if not state is Dictionary or not current is Dictionary \
		or state.get("status")!="active" or state.get("ordinal")!=2 or state.get("last_close")!={} \
		or state.get("proposal")!={} or not state.get("last_transition") is Dictionary:
		return null
	var t: Dictionary=state.last_transition
	var basis: Variant=t.get("basis")
	if t.get("kind")!=Owner.Common.ARCHIVE_TRANSITION or t.get("league")!="dugout" \
		or not basis is Dictionary or basis.get("version")!=2 or not Owner.Common.Start.valid(basis): return null
	var view: Variant=current.get("state")
	if not view is Dictionary or view.get("phase")!=SeasonState.Phase.DRAFT \
		or not view.get("picks") is Array or view.picks.size()>1: return null
	for name: String in ["workshop_draft","workshop_close_history","workshop_chair_purchase","workshop_applied","migration_v3"]:
		if value.get(name,{})!={}: return null
	var owner: RefCounted=Owner.new()
	var fields: Dictionary={}
	for name: String in Owner.META_FIELDS: fields[name]=value.get(name,{})
	if not Codec.install_fields(owner,Owner.META_FIELDS,fields) or not Owner._missing_dynasty(owner): return null
	owner.active="ordinary"
	owner.session=Adapter.OrdinaryRuntime.create_common(basis,"dugout")
	if owner.session==null or (view.picks.size()==1 and (not view.picks[0] is String or not owner.session.choose_player(view.picks[0]))):
		return null
	return owner if Owner.Common.valid(owner) and Owner.Common.archive_play_scope(owner.session) else null

static func _cross_draft_identity(session_raw: Dictionary) -> String:

static func accept_cross_draft_read(expected: PackedByteArray,child: Node) -> RefCounted:
	var process_script: Script=load("res://src/gameplay/season/successor_dynasty_owner_v1/standard_d1_witness_process.gd")

	return owner

# Only a live validated child completion can stage candidate bytes here.
# The child prepared the native state; this parent owns compare and write.
static func accept_cross_activation(owner: RefCounted,expected: PackedByteArray,child: Node) -> String:
	var process_script: Script=load("res://src/gameplay/season/successor_dynasty_owner_v1/standard_d1_witness_process.gd")
	if not cross_league_preview() or child==null or child.get_script()!=process_script or not _cross_pending.is_empty():
		return "refused"
	if not _cross_bound(owner,expected): return "stale"
	var kind: String=process_script.ACTIVATE_KIND
	if owner.ordinary_state.get("proposal",{}).get("kind")==Owner.Common.ARCHIVED_NEXT_KIND:
		kind=process_script.ARCHIVE_ACTIVATE_KIND
	var bytes: PackedByteArray=child.take_cross_bytes(Owner._hash(expected),kind)
	if bytes.is_empty(): return "refused"
	_cross_pending=bytes
	_cross_owner=owner
	_cross_expected=expected

static func _cross_bound(owner: RefCounted,expected: PackedByteArray) -> bool:
	return not saving and _ticket==null and owner!=null and owner==_owner \
		and owner.session==_owner_session and expected==_owner_bytes and not expected.is_empty() \
		and _write_revision_matches(expected,true)

static func _discard_cross_activation() -> void:
	_cross_starter=false
	_cross_pending=PackedByteArray()
	_cross_owner=null
	_cross_expected=PackedByteArray()

static func cancel_cross_activation(owner: RefCounted) -> void:
	if owner!=null and owner==_cross_owner: _discard_cross_activation()

static func retry_cross_activation(owner: RefCounted,expected: PackedByteArray) -> String:
	if _cross_pending.is_empty() or owner!=_cross_owner or expected!=_cross_expected: return "refused"
	var enabled: bool=LeagueAdmission.starter_player() if _cross_starter else cross_league_preview()
	if not enabled or not _cross_bound(owner,expected) or (_cross_starter and (_starter_slot_linked() or not _matches_raw(owner,_owner_raw))):
		_discard_cross_activation()
		return "stale"
	var dir:=DirAccess.open("user://")
	if dir==null: return "obstructed"
	for suffix: String in ["", ".bak", ".tmp"]:
		if dir.is_link((PATH+suffix).get_file()) or DirAccess.dir_exists_absolute(PATH+suffix): return "obstructed"
	if FileAccess.file_exists(PATH+".tmp"): return "obstructed"
	saving=true
	var outcome: String=_write_exact_result(_cross_pending,expected,true)
	saving=false
	if outcome in ["saved","renamed_unverified"]:
		# Once renamed, never retry or publish old owner authority even when
		# verifying the new primary failed. Strict Continue decides what exists.
		_invalidate()
		return outcome
	if _write_revision_matches(expected,true): return "obstructed"
	_discard_cross_activation()
	return "stale"

static func prepare_session_and_write(owner: RefCounted, expected: PackedByteArray, operation: String, seed_value: int = 0, league: String = "", reason: String = "") -> RefCounted:
	forget_projection()
	last_error=""
	if saving or _ticket!=null or operation not in ["resume","create","attach119","close","propose","next"]: return null
	if LeagueAdmission.starter_feature() and operation in ["resume","create","attach119"]: return null
	if not match_owner(owner,expected): return null
	if Owner.MixedAppend.marked_state(owner.ordinary_state):
		if operation!="close" or reason!="complete" or seed_value!=0 or not league.is_empty(): return null
		return _prepare_mixed_append_completed_close(owner,expected)
	if Owner.Mixed.marked_state(owner.ordinary_state):
		if operation!="close" or reason!="complete" or seed_value!=0 or not league.is_empty(): return null
		return _prepare_mixed_completed_close(owner,expected)
	if owner.ordinary_state.has("archive"):
		if Owner.Common.StandardChain.marked(owner.session):
			if operation!="close" or reason!="abandon" or not (LeagueAdmission.starter_player() or private_standard_chain_opening()) \
					or not Owner.Common.StandardChain.tagged_abandon_scope(owner,owner.session): return null
		elif Owner.Common.StandardArchive.marked(owner.session):
			if operation!="close" and (not LeagueAdmission.starter_player() or operation not in ["propose","next"] \
				or not Owner.Common.StandardChain.eligible(owner)): return null
		elif LeagueAdmission.starter_player() and Owner.Common.starter_dugout_chain_custody(owner):
			if operation!="close" or reason!="complete" \
					or not Owner.Common.starter_dugout_complete_scope(owner,owner.session): return null
		elif LeagueAdmission.starter_player() and Owner.Common.starter_dugout_archive(owner):
			if operation!="close" or (reason!="abandon" and (reason!="complete" \
					or not Owner.Common.starter_dugout_complete_scope(owner,owner.session))): return null
		elif operation!="close" or reason!="abandon" or not cross_league_preview(): return null
	last_error=league_block(owner,operation,league)
	if not last_error.is_empty(): return null
	if operation in ["resume","create","attach119"] and not switch_available(owner): return null
	if operation in ["propose","next"] and owner.active=="ordinary" \
		and Owner.Common.cross_proposal_kind(owner.ordinary_state.proposal.get("kind")): return null
	if operation not in ["create","propose"] and seed_value!=0: return null
	if operation!="create" and not league.is_empty(): return null
	if (operation=="close" and reason not in ["complete","abandon"]) or (operation!="close" and not reason.is_empty()): return null
	var imported: PackedByteArray=PackedByteArray()
	const SOURCE119: String="user://wiffaltro-materialized-play-v1.json"
	if operation=="attach119":
		if owner.active!="dynasty" or not owner.paused.is_empty() or not FileAccess.file_exists(SOURCE119): return null
		var source: FileAccess=FileAccess.open(SOURCE119,FileAccess.READ)
		if source==null or source.get_length()>MAX_BYTES: return null
		imported=source.get_buffer(source.get_length());source.close()
	if LeagueAdmission.starter_player() and operation in ["propose","next"] and Owner.Common.StandardChain.eligible(owner):
		return _prepare_starter_standard_chain(owner,expected,operation,seed_value)
	if LeagueAdmission.starter_player() and operation=="next" and Owner.Common.StandardArchive.first_next(owner):
		return _prepare_starter_standard_next(owner,expected)
	var candidate: RefCounted=owner.fork()
	if candidate==null: return null
	var changed: bool=false
	if operation=="resume": changed=candidate.switch_existing()
	elif operation=="create": changed=candidate.create_missing(seed_value,league)
	elif operation=="attach119": changed=candidate.import_119(imported)
	elif operation=="close": changed=candidate.close_current(reason)
	elif operation=="propose": changed=candidate.propose_next(seed_value)
	elif operation=="next": changed=candidate.commit_next()
	if not changed: return null
	# This rare explicit boundary strictly admits BOTH sides and all metadata;
	# routine operations still pin the dormant side and validate current only.
	var data: Dictionary=Owner.capture(candidate)
	if data.is_empty(): return null
	var snapshot: Dictionary=RawGuard.capture(candidate)
	if snapshot.is_empty(): return null
	if operation=="attach119" and FileAccess.get_file_as_bytes(SOURCE119)!=imported: return null
	_ticket=RefCounted.new()
	_candidate=candidate;_candidate_session=candidate.session;_candidate_raw=snapshot;_candidate_data=data
	_expected=expected.duplicate();_written=PackedByteArray()
	_write_admitted()
	return _ticket

static func _prepare_private_standard_flat(owner: RefCounted, expected: PackedByteArray, op: String, argument: String, app: Variant=null) -> RefCounted:
	if not private_standard_chain_opening() or saving or _ticket!=null or not _match_bound_owner(owner,expected):
		return null
	var candidate: RefCounted
	var close_data: Dictionary={}
	if op=="birth":
		if not argument.is_empty() or not Flat._scope(owner): return null
		var draft:=Flat._from_authenticated(owner,expected,Owner._hash(expected))
		candidate=Flat._local_owner(owner,draft)
	elif op=="abandon":
		if not argument.is_empty() or not private_standard_flat_close_available(app) or app.career_owner!=owner: return null
		candidate=Owner.new()
		if not Codec.install_fields(candidate,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
		candidate.session=owner.session.fork_live()
		if candidate.session==null: return null
		close_data=Flat.capture_abandon(candidate,Flat._capture_local(candidate))
		if close_data.is_empty(): return null
	elif op in ["npc","settle"]:
		if not argument.is_empty() or not Flat._storage_scope(owner) or owner.session.physical.pending.is_empty(): return null
		var pending: Dictionary=owner.session.physical.pending
		var current: SeasonState
		if op=="npc":
			if not _private_flat_npc_matches(app,owner,expected) or pending.reports.size()>=2: return null
			var report: Dictionary=_flat_npc.native_report()
			if report.is_empty(): return null
			var staged: Dictionary=Adapter.OrdinaryPending.stage_report(owner.session,report)
			current=staged.get("candidate")
			if current==null or not _private_flat_npc_matches(app,owner,expected): return null
			var row: Dictionary=current.physical.pending.reports[-1]
			if row.get("fixture")!=_flat_npc_fixture or row.get("request")!=_flat_npc_request: return null
		else:
			if not _private_flat_app(app) or app.career_owner!=owner or app.season!=owner.session \
					or app.saved_bytes!=expected or pending.reports.size()!=2 or is_instance_valid(_flat_npc): return null
			var settled:=Flat.project_regular(owner)
			if settled.is_empty(): return null
			current=settled.candidate
		candidate=Owner.new()
		if not Codec.install_fields(candidate,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
		candidate.session=current
		if op=="settle":
			var record: Dictionary={"version":1,"human":pending.human.duplicate(true)}
			if owner.session.round_index==0:
				candidate.ordinary_state[Flat.ROUND1]=record
			elif owner.session.round_index==1:
				record["property"]=pending.property
				candidate.ordinary_state[Flat.ROUND2]=record
			else: return null
	elif op in ["pick","opening","human"]:
		var fixture: Dictionary=Flat.human_fixture(owner) if op=="human" else {}
		if (fixture.is_empty() if op=="human" else not Flat._draft_scope(owner)): return null
		var next: SeasonState=owner.session.fork_live()
		if next==null: return null
		candidate=Owner._new()
		if not Codec.install_fields(candidate,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
		candidate.session=next
		if op=="pick":
			if next.phase!=SeasonState.Phase.DRAFT or not next.choose_player(argument): return null
			if next.picks.size()==4 and not Flat._enter_opening(candidate): return null
		elif op=="opening":
			if not Flat._lock_opening(candidate,argument): return null
		else:
			if not argument.is_empty() or not _private_flat_play_matches(app,owner,expected): return null
			var human:=observe_private_standard_flat_human(app)
			if human.size()!=13 or human[0]!=fixture.id: return null
			candidate.session.physical.pending={"human":human.duplicate(true),"reports":[]}
			var property: String=Owner.Common.StandardWorkshop.Home.key(owner.session,fixture)
			if fixture.home==0:
				if property.is_empty(): return null
				candidate.session.physical.pending["property"]=property
			elif not property.is_empty() or not Flat._validate_human_pending(candidate): return null
			# Property-bound human proof is independently replayed by ordinary
			# Codec.capture below, exactly once. No trusted evidence flag.
			if not _private_flat_play_matches(app,owner,expected): return null
	else: return null
	var data:=close_data if op=="abandon" else Flat._capture_local(candidate)
	if data.is_empty() or not _match_bound_owner(owner,expected): return null
	var snapshot:=RawGuard.capture(candidate)
	if snapshot.is_empty(): return null
	_ticket=RefCounted.new();_flat_ticket=_ticket
	_candidate=candidate;_candidate_session=candidate.session;_candidate_raw=snapshot;_candidate_data=data
	_expected=expected.duplicate();_written=PackedByteArray()
	_write_admitted()
	return _ticket

static func _clear_private_flat_play() -> void:
	_flat_play=null; _flat_play_app=null; _flat_play_lab=null

# Fixed native App-only launch. Store drives the scene from the exact own
# request and records the actual Lab it opened, not caller-supplied tokens.
static func open_private_standard_flat_human(app: Variant) -> bool:
	if not _private_flat_app(app) or is_instance_valid(app.lab) or app.career_owner==null \
			or app.season!=app.career_owner.session: return false
	var owner: RefCounted=app.career_owner
	var choice: Dictionary=owner.ordinary_state.get(Owner.Common.StandardWorkshop.OPENING,{})
	if choice.get("choice")!="chair": return false
	var play:=private_standard_flat_opening_play(owner,app.saved_bytes)
	if play==null: return false
	_clear_private_flat_play()
	app._fixture_id=int(play._fixture.id)
	app._result_recorded=false
	app._result_saved=false
	app._open_match(play.match_state,play._fixture.home==0,true,play.field_id)
	if not is_instance_valid(app.lab): return false
	_flat_play=play
	_flat_play_app=weakref(app)
	_flat_play_lab=weakref(app.lab)
	if _private_flat_play_matches(app,owner,app.saved_bytes): return true
	_clear_private_flat_play()
	app._close_match(false)
	app.notice="The private authored field could not be verified. Saved progress was unchanged."
	return false

static func _private_flat_play_matches(app: Variant, owner: RefCounted, expected: PackedByteArray) -> bool:
	if not private_standard_chain_opening() or not _private_flat_app(app) or _flat_play==null \
			or _flat_play_app.get_ref()!=app or not is_instance_valid(app.lab) or _flat_play_lab.get_ref()!=app.lab \
			or not _match_bound_owner(owner,expected) or app.career_owner!=owner or app.season!=owner.session \
			or app.saved_bytes!=expected or app._fixture_id!=_flat_play._fixture.id: return false
	var lab: PitchBatLab=app.lab
	var state: MatchState=lab._match_state
	var fixture:=Flat.human_fixture(owner)
	return Codec.same(fixture,_flat_play._fixture) and lab._managed_match and lab._match_mode \
		and lab._player_home==(fixture.home==0) and state==_flat_play.match_state \
		and lab._field_definition==_flat_play._field and lab._field_id==_flat_play.field_id \
		and Codec.same(state.league_rules,owner.session.league_rules) and state.pitch_catalog==owner.session.build.pitch_catalog_signature() \
		and _private_flat_geometry(lab,owner.session,fixture) \
		and Adapter.OrdinaryFixtures.request_key(owner.session,fixture)==_flat_play._request

static func _private_flat_geometry(lab: PitchBatLab, season: SeasonState, fixture: Dictionary) -> bool:
	var workshop = Owner.Common.StandardWorkshop
	var native: Dictionary=workshop.Home.fixture(season,fixture)
	if not Codec.same(native,_flat_play._stadium) or not Codec.same(lab._match_state.stadium,native): return false
	if native.is_empty(): return not lab.has_meta("stadium_property")
	if fixture.home!=0 or fixture.get("neutral",false) or not lab._player_home \
			or not Codec.same(lab.get_meta("stadium_snapshot",{}),native): return false
	var property: Variant=lab.get_meta("stadium_property",null)
	if not is_instance_valid(property) or not property is StadiumPropertyGeometry \
			or native.layout.receiver!=workshop.COPY or not property.objects.has(native.layout.receiver): return false
	var receiver: StadiumPitchReceiver=property.get_node_or_null("PitchReceiverPresentation")
	return receiver!=null and receiver.chair==property.objects[native.layout.receiver] \
		and lab._pitch_actor.receiver_router==receiver.route

# Read-only native observation for custody or fixed Return. No public replay
# input exists: the writer invokes this on the same still-bound Lab itself.
static func observe_private_standard_flat_human(app: Variant) -> Array:
	if not _private_flat_app(app) or not _private_flat_play_matches(app,app.career_owner,app.saved_bytes): return []
	return Adapter.OrdinaryCapture._capture_bound(app,func(): return _private_flat_play_matches(app,app.career_owner,app.saved_bytes))

static func prepare_private_standard_flat_human(app: Variant) -> RefCounted:
	if not _private_flat_app(app) or not _private_flat_play_matches(app,app.career_owner,app.saved_bytes): return null
	return _prepare_private_standard_flat(app.career_owner,app.saved_bytes,"human","",app)

static func _clear_private_flat_npc() -> void:
	_flat_npc=null; _flat_npc_app=null; _flat_npc_expected=PackedByteArray()
	_flat_npc_fixture=-1; _flat_npc_request=""

static func _private_flat_npc_matches(app: Variant, owner: RefCounted, expected: PackedByteArray) -> bool:
	return private_standard_chain_opening() and _private_flat_app(app) and is_instance_valid(_flat_npc) \
		and _flat_npc.get_script()==FlatRunner and _flat_npc_app.get_ref()==app and _flat_npc.get_parent()==app \
		and app.ai_runner==_flat_npc and app._busy and not is_instance_valid(app.lab) \
		and app.career_owner==owner and app.season==owner.session and app.saved_bytes==expected \
		and _flat_npc_expected==expected and _match_bound_owner(owner,expected)

static func private_flat_npc_details(app: Variant) -> Dictionary:
	if not _private_flat_app(app) or not _private_flat_npc_matches(app,app.career_owner,app.saved_bytes): return {}
	return {"fixture":_flat_npc_fixture,"request":_flat_npc_request,"source":Owner._hash(_flat_npc_expected)}

# Construct the native job here. The completion callback takes runner
# identity only; an external report argument cannot reach the ticket.
static func start_private_standard_flat_npc(app: Variant) -> bool:
	if not private_standard_chain_opening() or not _private_flat_app(app) or saving or _ticket!=null \
			or is_instance_valid(_flat_npc) or app._busy or is_instance_valid(app.lab) or is_instance_valid(app.ai_runner): return false
	var owner: RefCounted=app.career_owner
	var expected: PackedByteArray=app.saved_bytes
	if not _match_bound_owner(owner,expected) or app.season!=owner.session or not Flat._storage_scope(owner) \
			or owner.session.physical.pending.is_empty() or owner.session.physical.pending.reports.size()>=2: return false
	var projected:=Flat.project_regular(owner)
	if projected.is_empty() or not projected.has("fixture"): return false
	var fixture: Dictionary=projected.fixture
	var day: SeasonState=projected.candidate
	var state: MatchState=Adapter.OrdinaryFixtures.match_for(day,fixture)
	var field: FieldDefinition=Adapter.OrdinaryRuntime.field_for_fixture(fixture,day)
	var request: String=Adapter.OrdinaryFixtures.request_key(day,fixture)
	if state==null or not state.stadium.is_empty() or field==null or request.is_empty() \
			or not _match_bound_owner(owner,expected): return false
	var runner:=FlatRunner.new()
	app.add_child(runner)
	_flat_npc=runner; _flat_npc_app=weakref(app); _flat_npc_expected=expected.duplicate()
	_flat_npc_fixture=int(fixture.id); _flat_npc_request=request
	app.ai_runner=runner; app._busy=true
	runner.native_ready.connect(_finish_private_flat_npc.bind(runner),CONNECT_DEFERRED)
	runner.failed.connect(_fail_private_flat_npc.bind(runner))
	if runner.start(state,Adapter.OrdinaryFixtures.seed_for(day,fixture),field.id): return true
	_fail_private_flat_npc("The authored private native fixture could not start.",runner)
	return false

static func _finish_private_flat_npc(runner: Node) -> void:
	if not is_instance_valid(runner) or runner!=_flat_npc: return
	var app: Variant=_flat_npc_app.get_ref()
	if not _private_flat_app(app): return
	var ticket: RefCounted=_prepare_private_standard_flat(app.career_owner,app.saved_bytes,"npc","",app)
	_clear_private_flat_npc()
	app._private_flat_native_npc_finished(runner,ticket)

static func _fail_private_flat_npc(reason: String, runner: Node) -> void:
	if not is_instance_valid(runner) or runner!=_flat_npc: return
	var app: Variant=_flat_npc_app.get_ref()
	_clear_private_flat_npc()
	if _private_flat_app(app): app._ai_failed(reason,runner)

static func prepare_private_standard_flat_settle(app: Variant) -> RefCounted:
	if not _private_flat_app(app) or app._busy or is_instance_valid(app.ai_runner) or is_instance_valid(app.lab): return null
	return _prepare_private_standard_flat(app.career_owner,app.saved_bytes,"settle","",app)

static func private_standard_flat_close_available(app: Variant) -> bool:
	if not private_standard_chain_opening() or not _private_flat_app(app) or saving or _ticket!=null \
			or app._busy or app.retained!=null or is_instance_valid(app.ai_runner) or is_instance_valid(app.lab) \
			or is_instance_valid(_flat_npc) or app.career_owner==null or app.season!=app.career_owner.session: return false
	return _match_bound_owner(app.career_owner,app.saved_bytes) and Flat.can_abandon(app.career_owner)

static func prepare_private_standard_flat_abandon(app: Variant) -> RefCounted:
	if not private_standard_flat_close_available(app): return null
	return _prepare_private_standard_flat(app.career_owner,app.saved_bytes,"abandon","",app)

# Identity handoff only. Later retry/promotion still authenticate this exact
# live candidate against the private snapshot; no returned value authorizes IO.
static func prepared_candidate(ticket: RefCounted) -> RefCounted:
	return _candidate if ticket != null and ticket == _ticket else null

# Cheap handoff identity only, never raw admission or authority to write/promote.
static func prepared_binding(ticket: RefCounted, owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> bool:
	if ticket == null or ticket != _ticket or owner != _owner or candidate != _candidate or expected != _expected: return false
	return candidate.session == _candidate_session and (owner == null or owner.session == _owner_session)

static func prepared_matches(ticket: RefCounted, owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> bool:
	if not prepared_binding(ticket,owner,candidate,expected):
		forget_projection(); return false
	var valid: bool = (owner == null or _matches_raw(owner,_owner_raw)) and _matches_raw(candidate,_candidate_raw)
	if not valid: forget_projection()
	return valid

static func prepared_snapshot(ticket: RefCounted) -> Dictionary:
	return _candidate_data.duplicate(true) if ticket != null and ticket == _ticket else {}

static func write_prepared(ticket: RefCounted) -> bool:
	forget_projection()
	if saving or ticket == null or ticket != _ticket or not _written.is_empty(): return false
	if not prepared_matches(ticket,_owner,_candidate,_expected): return false
	return _write_admitted()

# Private immediate continuation only; callers use prepare_and_write or the
# fully checked write_prepared retry. Disk CAS/readback remain in every write.
static func _write_admitted() -> bool:
	if _flat_ticket!=null and _flat_ticket==_ticket:
		if not private_standard_chain_opening() or not Flat._storage_scope(_candidate): return false
	elif not _starter_owner_allowed(_candidate): return false
	last_error = "The exact candidate is retained. Resolve the save obstruction, then retry."
	if not _matches(_expected):
		last_error = "The saved file changed. It was not overwritten; your candidate remains retained."
		return false
	for suffix: String in ["", ".bak", ".tmp"]:
		if DirAccess.dir_exists_absolute(PATH+suffix): return false
	if FileAccess.file_exists(PATH+".tmp") or (_expected.is_empty() and FileAccess.file_exists(PATH+".bak")): return false
	saving = true
	var result: bool = _write_data(_candidate_data,_expected)
	saving = false
	if result: _written = raw()
	return result

static func written_bytes(ticket: RefCounted) -> PackedByteArray:
	return _written.duplicate() if ticket != null and ticket == _ticket else PackedByteArray()

static func promote(ticket: RefCounted, owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> bool:
	if ticket!=null and ticket==_flat_ticket and not private_standard_chain_opening(): return false
	if _written.is_empty() or raw() != _written or not prepared_matches(ticket,owner,candidate,expected):
		forget_projection(); return false
	var arm: bool = _projection_ticket == ticket and _projection_checkpoint == candidate and _projection_value != null
	if arm:
		_projection_owner = candidate; _projection_ticket = null; _projection_checkpoint = null
	else: forget_projection()
	_owner = _candidate; _owner_session = _candidate_session; _owner_raw = _candidate_raw; _owner_bytes = _written.duplicate()
	_paused = _candidate.paused.duplicate(true); _active = _candidate.active
	_metadata = Codec.data_fields(_candidate,Owner.META_FIELDS); _imported_sha256 = _candidate.imported_sha256
	_clear_private_flat_play()
	_flat_ticket = null; _ticket = null; _candidate = null; _candidate_session = null; _candidate_raw = {}; _candidate_data = {}
	_expected = PackedByteArray(); _written = PackedByteArray()
	return true
