const SHOP_PRODUCER_OPS: Array[String] = ["open", "pack_open", "pack_pick", "pack_skip"]

# Clear-only invalidation has no admission/write authority.
static func forget_projection() -> void:
	_projection_ticket = null; _projection_checkpoint = null; _projection_owner = null
	_projection_value = null; _projection_raw = {}

static func raw() -> PackedByteArray:
	return FileAccess.get_file_as_bytes(PATH) if FileAccess.file_exists(PATH) else PackedByteArray()
static func occupied() -> bool:
	for suffix: String in ["", ".bak", ".tmp"]:
		if FileAccess.file_exists(PATH+suffix) or DirAccess.dir_exists_absolute(PATH+suffix): return true
	return false
static func read() -> RefCounted:
	forget_projection()
	last_error = ""
	if not FileAccess.file_exists(PATH):
		last_error = "No common career is saved in this slot. Earlier saves are unchanged."
		return null
	var file: FileAccess = FileAccess.open(PATH,FileAccess.READ)
	if file == null or file.get_length() > MAX_BYTES:
		last_error = "The save could not be read safely. It has not been changed."
		return null
	var bytes: PackedByteArray = file.get_buffer(file.get_length())
	file.close()
	var parser: JSON = JSON.new()
	if parser.parse(bytes.get_string_from_utf8()) != OK:
		last_error = "The save is unreadable. Original and backup are preserved."
		return null
	var season: RefCounted = Owner.restore(parser.data)
	if season == null:
		last_error = "This common career has an unsupported version or inconsistent state. No files were changed."
		return null
	var snapshot: Dictionary = RawGuard.capture(season)
	if snapshot.is_empty() or raw() != bytes:
		last_error = "The save changed during read or could not be admitted. No files were changed."
		return null
	_invalidate()
	_owner = season; _owner_session = season.session; _owner_raw = snapshot; _owner_bytes = bytes.duplicate()
	_paused = season.paused.duplicate(true); _active = season.active
	_metadata = Codec.data_fields(season,Owner.META_FIELDS); _imported_sha256 = season.imported_sha256
	return season
static func save(season: RefCounted, expected: PackedByteArray, expected_snapshot: Dictionary = {}) -> bool:
	forget_projection()
	if saving: return false
	saving = true
	var result: bool = _write(season,expected,expected_snapshot)
	saving = false
	if result: _invalidate()
	return result
static func _matches(expected: PackedByteArray) -> bool:
	return not DirAccess.dir_exists_absolute(PATH) and ((expected.is_empty() and not FileAccess.file_exists(PATH)) or (not expected.is_empty() and FileAccess.file_exists(PATH) and raw() == expected))
static func _write(season: RefCounted, expected: PackedByteArray, expected_snapshot: Dictionary) -> bool:
	# Legacy unbound writer never modifies the private paired slot.
	if _paired_owner(season) or _paired_settled(_owner): return false
	last_error = "The exact candidate is retained. Resolve the save obstruction, then retry."
	if not _matches(expected):
		last_error = "The saved file changed. It was not overwritten; your candidate remains retained."
		return false
	for suffix: String in ["", ".bak", ".tmp"]:
		if DirAccess.dir_exists_absolute(PATH+suffix): return false
	if FileAccess.file_exists(PATH+".tmp") or (expected.is_empty() and FileAccess.file_exists(PATH+".bak")): return false
	var data: Dictionary = Owner.capture(season)
	if data.is_empty() or (not expected_snapshot.is_empty() and not Codec.same(data,expected_snapshot)):
		last_error = "This candidate is inconsistent with the materialized save contract."
		return false
	return _write_data(data,expected)

static func _write_data(data: Dictionary, expected: PackedByteArray) -> bool:
	var bytes: PackedByteArray = JSON.stringify(data,"",true,true).to_utf8_buffer()
	if bytes.is_empty() or bytes.size() > MAX_BYTES: return false
	# Validate the actual encoded primitive representation against this exact
	# same-call consistent candidate. Cold hydration is independently qualified;
	# do not rerun it and recapture six clubs again inside every button handler.
	var parser: JSON = JSON.new()
	if parser.parse(bytes.get_string_from_utf8()) != OK or not Codec.same(parser.data,data): return false
	var file: FileAccess = FileAccess.open(PATH+".tmp",FileAccess.WRITE)
	if file == null: return false
	file.store_buffer(bytes); file.flush()
	var ok: bool = file.get_error() == OK
	file.close()
	if not ok or FileAccess.get_file_as_bytes(PATH+".tmp") != bytes or not _matches(expected): return false
	if not expected.is_empty() and DirAccess.copy_absolute(PATH,PATH+".bak") != OK: return false
	if not _matches(expected): return false
	if DirAccess.rename_absolute(PATH+".tmp",PATH) != OK or raw() != bytes: return false
	last_error = ""
	return true

static func _invalidate() -> void:
	forget_projection()
	_owner = null; _owner_session = null; _owner_raw = {}; _owner_bytes = PackedByteArray()
	_paused = {}; _active = ""; _metadata = {}; _imported_sha256 = ""
	_ticket = null; _candidate = null; _candidate_session = null; _candidate_raw = {}; _candidate_data = {}
	_expected = PackedByteArray(); _written = PackedByteArray()

static func match_owner(owner: RefCounted, expected: PackedByteArray) -> bool:
	var valid: bool = owner != null and owner == _owner and owner.session == _owner_session and expected == _owner_bytes \
		and _matches(expected) and _matches_raw(owner,_owner_raw)
	if not valid: forget_projection()
	return valid

static func _matches_raw(value: RefCounted, snapshot: Dictionary) -> bool:
	return RawGuard.matches(value,snapshot)

# Pure refusal predicate, never a writer or an admission token. Routine actions
# cannot replace either session or provenance through the generic candidate API.
static func _routine_pins(candidate: RefCounted) -> bool:
	if candidate.active!=_active or not RawGuard.same(candidate.paused,_paused) or candidate.imported_sha256!=_imported_sha256: return false
	if not RawGuard.same(candidate.migration_v2,_metadata.get("migration_v2")): return false
	for name: String in ["common","ordinary_state","migration_v3"]:
		if not RawGuard.same(candidate.get(name),_metadata.get(name)): return false
	if _active=="ordinary": return RawGuard.same(Codec.data_fields(candidate,Owner.META_FIELDS),_metadata)
	return candidate.birth_seed==_metadata.get("birth_seed")

static func _paired_owner(value: RefCounted) -> bool:
	return value!=null and value.get_script()==Owner and value.paired_draft_only()

static func _paired_settled(value: RefCounted) -> bool:
	return _paired_owner(value) and value.session.round_index!=0

static func prepare(owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> RefCounted:
	forget_projection()
	if saving or _ticket != null or candidate == null or candidate == owner or _paired_settled(owner): return null
	if candidate.get_script()!=Owner or candidate.session==null or _paired_settled(candidate): return null
	if owner == null:
		if not expected.is_empty() or not _matches(expected): return null
	else:
		if not match_owner(owner,expected) or candidate.session==owner.session: return null
	# Only this private boundary creates the writer's canonical payload. Callers
	# can receive detached diagnostics, never supply a substitute validated value.
	var data: Dictionary = {}
	if owner != null:
		if not _routine_pins(candidate): return null
		# Dormant alternate was strictly admitted on read/promotion. Exact live raw
		# owner pins were checked above; no caller validation flag or payload.
		var current: Dictionary = Adapter.capture(candidate.active,candidate.session)
		if not current.is_empty():
			data = Owner.pack_current(candidate,current)
	else:
		data = Owner.capture(candidate)
	if data.is_empty(): return null
	var snapshot: Dictionary = RawGuard.capture(candidate)
	if snapshot.is_empty(): return null
	if owner == null:
		_owner = null; _owner_session = null; _owner_raw = {}; _owner_bytes = expected.duplicate()
	_ticket = RefCounted.new()
	_candidate = candidate; _candidate_session = candidate.session; _candidate_raw = snapshot; _candidate_data = data
	_expected = expected.duplicate(); _written = PackedByteArray()
	return _ticket

## One synchronous Store-owned admission/write stack. No token or caller flag
## permits skipping later checks. prepare strictly captures the candidate and
## matches the live owner; no callbacks or external handoff intervene here.
## On obstruction the ordinary private ticket survives, and retry stays strict.
static func prepare_and_write(owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> RefCounted:
	var ticket: RefCounted = prepare(owner,candidate,expected)
	if ticket == null: return null
	_write_admitted()
	return ticket

## Fixed synchronous producer: authenticate before constructing any candidate.
## Only named runtime/build factories admitted by match_owner can execute here.
## No UI callbacks, awaits, caller candidate or unchecked payload are accepted.
static func prepare_shop_and_write(owner: RefCounted, expected: PackedByteArray, command: Dictionary) -> RefCounted:
	forget_projection()
	last_error = ""
	if saving or _ticket != null or not Codec.primitive(command) or not SHOP_PRODUCER_OPS.has(str(command.get("op",""))): return null
	if not match_owner(owner,expected): return null
	var session: SeasonState = owner.session
	if not owner.playable() or not Adapter.scope(owner.active,session) or not session.physical.pending.is_empty() or not session.shop_available(): return null
	var request_id: Variant = command.get("id")
	if request_id is String and not request_id.is_empty() and session.build._requests.has(request_id): return null
	var next_session: SeasonState = session.fork_live()
	if next_session == null: return null
	var next_build: SeasonBuild = next_session.build.candidate(command.duplicate(true))
	if next_build == null:
		last_error = next_session.build.last_error
		return null
	next_session.adopt_build(next_build)
	if next_session.build != next_build: return null
	var envelope: RefCounted = Owner.new()
	envelope.session = next_session
	if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
	# Keep this registration inside the checked producer, rather than exposing
	# an unchecked owner/candidate helper (underscore methods are callable).
	if not _routine_pins(envelope): return null
	var current: Dictionary = Adapter.capture(envelope.active,next_session)
	var data: Dictionary = Owner.pack_current(envelope,current) if not current.is_empty() else {}
	if data.is_empty():
		last_error = "The candidate could not be represented consistently. Saved progress is unchanged."
		return null
	var snapshot: Dictionary = RawGuard.capture(envelope)
	if snapshot.is_empty(): return null
	_ticket = RefCounted.new()
	_candidate = envelope; _candidate_session = next_session; _candidate_raw = snapshot; _candidate_data = data
	_expected = expected.duplicate(); _written = PackedByteArray()
	_write_admitted()
	return _ticket

## Fixed producer invokes the named pending implementation itself. The optional
## terminal result is retained only after this exact immediate checkpoint write.
## A failed write/retry keeps the ordinary ticket, but deliberately loses reuse.
static func prepare_report_and_write(owner: RefCounted, expected: PackedByteArray, report: Dictionary) -> RefCounted:
	forget_projection()
	if saving or _ticket != null or not match_owner(owner,expected): return null
	if not owner.playable() or not Adapter.scope(owner.active,owner.session) or owner.session.physical.pending.is_empty(): return null
	var staged: Dictionary = Adapter.pending(owner.active).stage_report(owner.session,report)
	var current: SeasonState = staged.get("candidate")
	if current == null: return null
	var envelope: RefCounted = Owner.new()
	envelope.session = current
	if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
	var settled: RefCounted
	var snapshot: Dictionary = {}
	if staged.get("settled") is SeasonState:
		settled = Owner.new(); settled.session = staged.settled
		if not Codec.install_fields(settled,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)): return null
		if not _routine_pins(settled) or not settled.session.physical.pending.is_empty() \
			or not settled.session.physical.needed.is_empty() or settled.session.physical.projecting: return null
		snapshot = RawGuard.capture(settled)
		if snapshot.is_empty(): return null
	var ticket: RefCounted = prepare_and_write(owner,envelope,expected)
	if ticket != null and not _written.is_empty() and settled != null:
		_projection_ticket = ticket; _projection_checkpoint = envelope
		_projection_value = settled; _projection_raw = snapshot
	return ticket

static func _prepare_paired_settlement(owner: RefCounted, expected: PackedByteArray, leg: int) -> RefCounted:
	forget_projection()
	last_error=""
	if leg not in [0,1,2,3,4,5] or saving or _ticket!=null or not match_owner(owner,expected): return null
	if not owner.paired_draft_only() or not Adapter.scope(owner.active,owner.session) or owner.session.round_index!=leg: return null
	var projection=load("res://src/gameplay/season/successor_live_play_v1/paired_first_projection.gd")
	var started: int=Time.get_ticks_msec()
	var current: SeasonState=projection.materialize_settlement(owner.session,leg)
	paired_candidate_ms=Time.get_ticks_msec()-started
	if current==null: return null
	var envelope: RefCounted=Owner.new()
	envelope.session=current
	if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)) or not _routine_pins(envelope): return null
	started=Time.get_ticks_msec()
	var captured: Dictionary=Adapter.capture(owner.active,current)
	paired_admission_ms=Time.get_ticks_msec()-started
	var data: Dictionary=Owner.pack_current(envelope,captured) if not captured.is_empty() else {}
	if data.is_empty(): return null
	var snapshot: Dictionary=RawGuard.capture(envelope)
	if snapshot.is_empty() or not match_owner(owner,expected): return null
	_ticket=RefCounted.new()
	_candidate=envelope; _candidate_session=current; _candidate_raw=snapshot; _candidate_data=data
	_expected=expected.duplicate(); _written=PackedByteArray()
	_write_admitted()
	return _ticket

static func _prepare_paired_shop(owner: RefCounted, expected: PackedByteArray, command: Dictionary, visit: int) -> RefCounted:
	forget_projection()
	last_error=""
	if saving or _ticket!=null or not Codec.primitive(command) or not match_owner(owner,expected): return null
	if not _paired_settled(owner) or not Adapter.scope(owner.active,owner.session): return null
	var session: SeasonState=owner.session
	const General=preload("res://src/gameplay/season/successor_league_general_v2/runtime_commit.gd")
	if visit not in [1,6] or session.round_index!=visit or not General._source_public_shop_available(session): return null
	var sign_visit6: bool=visit==6 and Adapter.Ordinary.visit6_sign_command(command)
	if visit==1:
		var step: int=session.build.revision()-2
		if not Adapter.Ordinary.first_shop_command(session.build,command,step): return null
	else:
		if session.player_results[-1].get("id")!=15: return null
		if sign_visit6:
			if session.build.revision()!=16 or session.build._visit.get("open")!=true or not Adapter.Ordinary.visit6_recruit_quote(session.build,false): return null
		elif not Adapter.Ordinary.visit6_open_command(command) or session.build.revision()!=15 or session.build._visit.get("open")!=false: return null
	if session.build._requests.has(command.id): return null
	var started: int=Time.get_ticks_msec()
	var current: SeasonState=session.fork_live()
	if current==null or (sign_visit6 and not _bind_visit6_fresh_profile(current.build)): return null
	var build: SeasonBuild=current.build.candidate(command.duplicate(true))
	if build==null: return null
	current.adopt_build(build)
	if current.build!=build: return null
	paired_shop_candidate_ms=Time.get_ticks_msec()-started
	var envelope: RefCounted=Owner.new()
	envelope.session=current
	if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)) or not _routine_pins(envelope): return null
	started=Time.get_ticks_msec()
	var captured: Dictionary=Adapter.capture(owner.active,current)
	paired_shop_admission_ms=Time.get_ticks_msec()-started
	var data: Dictionary=Owner.pack_current(envelope,captured) if not captured.is_empty() else {}
	if data.is_empty(): return null
	var snapshot: Dictionary=RawGuard.capture(envelope)
	if snapshot.is_empty() or not match_owner(owner,expected): return null
	_ticket=RefCounted.new()
	_candidate=envelope; _candidate_session=current; _candidate_raw=snapshot; _candidate_data=data
	_expected=expected.duplicate(); _written=PackedByteArray()
	_write_admitted()
	return _ticket

# Exact native representation on the detached candidate only. No changed
# semantic/materialized inputs are allowed to reach the frozen signer.
static func _bind_visit6_fresh_profile(build: SeasonBuild) -> bool:
	var id: String=Adapter.Ordinary.VISIT6_RECRUIT
	var outgoing: String=Adapter.Ordinary.VISIT6_OUTGOING
	if not Adapter.Ordinary.visit6_recruit_quote(build,false) or build.cash()<int(build._visit.recruit.price) or build._contracts.has(id) or build.roster().has(id) or build._blocked.has(id) or not build._pool.has(id) or not Adapter.Ordinary.same(build.roster(),build._initial_roster) or not build.roster().has(outgoing) or SeasonSchoolSponsors.scholarship(build).get("player")==outgoing: return false
	var profile: Dictionary=build.player(id)
	var original: Dictionary=SeasonPlayerCatalog.profile(id)
	if original.is_empty() or not Adapter.Ordinary.same(profile,original): return false
	var before: String=_canonical_shop_bytes(Adapter.Ordinary.Builds.capture(build))
	if before.is_empty(): return false
	for field: String in ["base","stats","mastery"]:
		profile[field]=original[field].duplicate(true)
	profile.capacity=original.capacity
	if profile!=original: return false
	build._book._players[id]=profile
	var after: Dictionary=Adapter.Ordinary.Builds.capture(build)
	return before==_canonical_shop_bytes(after)

static func _canonical_shop_bytes(value: Dictionary) -> String:
	if value.is_empty(): return ""
	var parsed: Variant=JSON.parse_string(JSON.stringify(value,"",true,true))
	return JSON.stringify(parsed,"",true,true) if parsed is Dictionary else ""

static func _prepare_paired_human_checkpoint(owner: RefCounted, expected: PackedByteArray, pending: Dictionary, round_index: int) -> RefCounted:
	forget_projection()
	last_error=""
	if saving or _ticket!=null or not Codec.primitive(pending) or not match_owner(owner,expected): return null
	if not _paired_settled(owner) or not Adapter.scope(owner.active,owner.session): return null
	var session: SeasonState=owner.session
	if round_index not in [1,2,3,4,5,6] or session.round_index!=round_index or session.build.revision()!=[0,5,7,9,11,13,17][round_index] or session.build.pack_pending(): return null
	if round_index==6 and not Adapter.Ordinary._signed_visit6_journal(session.build): return null
	var projection=load("res://src/gameplay/season/successor_live_play_v1/paired_first_projection.gd")
	var opening: Dictionary=projection._round_fixtures(round_index).human
	if not projection.bound_shape(pending) or not Codec.whole(pending.human[0],opening.id,opening.id): return null
	var prior: Dictionary=session.physical.pending
	if prior.is_empty():
		if not pending.reports.is_empty(): return null
	elif pending.reports.size()!=prior.reports.size()+1 or not Codec.same(pending.human,prior.human) or not Codec.same(pending.binding,prior.binding) or not Codec.same(pending.reports.slice(0,prior.reports.size()),prior.reports): return null
	var current: SeasonState=session.fork_live()
	if current==null: return null
	current.physical.pending=pending.duplicate(true)
	var envelope: RefCounted=Owner.new()
	envelope.session=current
	if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)) or not _routine_pins(envelope): return null
	var captured: Dictionary=Adapter.capture(owner.active,current)
	var data: Dictionary=Owner.pack_current(envelope,captured) if not captured.is_empty() else {}
	if data.is_empty(): return null
	var snapshot: Dictionary=RawGuard.capture(envelope)
	if snapshot.is_empty() or not match_owner(owner,expected): return null
	_ticket=RefCounted.new()
	_candidate=envelope; _candidate_session=current; _candidate_raw=snapshot; _candidate_data=data
	_expected=expected.duplicate(); _written=PackedByteArray()
	_write_admitted()
	return _ticket

## Same active owner only. On cold/cache miss, use the normal complete projection.
## Partial projections are returned solely to construct the next real runner.
## Terminal candidates still cross the existing full semantic prepare boundary.
static func prepare_settlement_and_write(owner: RefCounted, expected: PackedByteArray) -> Dictionary:
	if saving or _ticket != null:
		forget_projection(); return {"error":"A checkpoint is already retained."}
	if not match_owner(owner,expected): return {"error":"The current owner changed."}
	if not owner.playable() or not Adapter.scope(owner.active,owner.session) or owner.session.physical.pending.is_empty():
		forget_projection(); return {"error":"No current round is pending."}
	var envelope: RefCounted
	if _projection_owner == owner:
		if _projection_value == null or not RawGuard.matches(_projection_value,_projection_raw):
			forget_projection(); return {"error":"The retained terminal projection changed."}
		envelope = _projection_value
	forget_projection()
	if envelope == null:
		var projected: Dictionary = Adapter.pending(owner.active).project(owner.session)
		if projected.has("error") or projected.has("fixture"): return projected
		var current: SeasonState = projected.get("candidate")
		if current == null: return {"error":"The current round did not produce a candidate."}
		envelope = Owner.new(); envelope.session = current
		if not Codec.install_fields(envelope,Owner.DATA_FIELDS,Codec.data_fields(owner,Owner.DATA_FIELDS)):
			return {"error":"The round metadata could not be preserved."}
	var ticket: RefCounted = prepare_and_write(owner,envelope,expected)
	return {"ticket":ticket} if ticket != null else {"error":"The settled candidate could not be admitted."}

## Explicit session boundary producer. Routine candidate APIs above cannot
## alter selector/dormant identity. No caller candidate, payload or callback.
static func switch_available(owner: RefCounted) -> bool:
	if owner==null or owner.get_script()!=Owner: return false
	var closed: bool=owner.status=="closed" if owner.active=="dynasty" else owner.ordinary_state.get("status")=="closed"
	if not owner.playable() and not closed: return false
	var current: SeasonState=owner.session
	return Adapter.scope(owner.active,current) and current.phase!=SeasonState.Phase.DRAFT \
		and (current.phase!=SeasonState.Phase.COMPLETE or closed) and current.physical.pending.is_empty() \
		and current.physical.needed.is_empty() and not current.physical.projecting and not current.build.pack_pending()

static func prepare_session_and_write(owner: RefCounted, expected: PackedByteArray, operation: String, seed_value: int = 0, league: String = "", reason: String = "") -> RefCounted:
	forget_projection()
	last_error=""
	if saving or _ticket!=null or operation not in ["resume","create","attach119","close","propose","next"]: return null
	if not match_owner(owner,expected): return null
	if operation in ["resume","create","attach119"] and not switch_available(owner): return null
	if operation not in ["create","propose"] and seed_value!=0: return null
	if operation!="create" and not league.is_empty(): return null
	if (operation=="close" and reason not in ["complete","abandon"]) or (operation!="close" and not reason.is_empty()): return null
	var imported: PackedByteArray=PackedByteArray()
	const SOURCE119: String="user://wiffaltro-materialized-play-v1.json"
	if operation=="attach119":
		if owner.active!="dynasty" or not owner.paused.is_empty() or not FileAccess.file_exists(SOURCE119): return null
		var source: FileAccess=FileAccess.open(SOURCE119,FileAccess.READ)
		if source==null or source.get_length()>MAX_BYTES: return null
		imported=source.get_buffer(source.get_length());source.close()
	var candidate: RefCounted=owner.fork()
	if candidate==null: return null
	var changed: bool=false
	if operation=="resume": changed=candidate.switch_existing()
	elif operation=="create": changed=candidate.create_missing(seed_value,league)
	elif operation=="attach119": changed=candidate.import_119(imported)
	elif operation=="close": changed=candidate.close_current(reason)
	elif operation=="propose": changed=candidate.propose_next(seed_value)
	elif operation=="next": changed=candidate.commit_next()
	if not changed: return null
	# This rare explicit boundary strictly admits BOTH sides and all metadata;
	# routine operations still pin the dormant side and validate current only.
	var data: Dictionary=Owner.capture(candidate)
	if data.is_empty(): return null
	var snapshot: Dictionary=RawGuard.capture(candidate)
	if snapshot.is_empty(): return null
	if operation=="attach119" and FileAccess.get_file_as_bytes(SOURCE119)!=imported: return null
	_ticket=RefCounted.new()
	_candidate=candidate;_candidate_session=candidate.session;_candidate_raw=snapshot;_candidate_data=data
	_expected=expected.duplicate();_written=PackedByteArray()
	_write_admitted()
	return _ticket

# Identity handoff only. Later retry/promotion still authenticate this exact
# live candidate against the private snapshot; no returned value authorizes IO.
static func prepared_candidate(ticket: RefCounted) -> RefCounted:
	return _candidate if ticket != null and ticket == _ticket else null

# Cheap handoff identity only, never raw admission or authority to write/promote.
static func prepared_binding(ticket: RefCounted, owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> bool:
	if ticket == null or ticket != _ticket or owner != _owner or candidate != _candidate or expected != _expected: return false
	return candidate.session == _candidate_session and (owner == null or owner.session == _owner_session)

static func prepared_matches(ticket: RefCounted, owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> bool:
	if not prepared_binding(ticket,owner,candidate,expected):
		forget_projection(); return false
	var valid: bool = (owner == null or _matches_raw(owner,_owner_raw)) and _matches_raw(candidate,_candidate_raw)
	if not valid: forget_projection()
	return valid

static func prepared_snapshot(ticket: RefCounted) -> Dictionary:
	return _candidate_data.duplicate(true) if ticket != null and ticket == _ticket else {}

static func write_prepared(ticket: RefCounted) -> bool:
	forget_projection()
	if saving or ticket == null or ticket != _ticket or not _written.is_empty(): return false
	if not prepared_matches(ticket,_owner,_candidate,_expected): return false
	return _write_admitted()

# Private immediate continuation only; callers use prepare_and_write or the
# fully checked write_prepared retry. Disk CAS/readback remain in every write.
static func _write_admitted() -> bool:
	last_error = "The exact candidate is retained. Resolve the save obstruction, then retry."
	if not _matches(_expected):
		last_error = "The saved file changed. It was not overwritten; your candidate remains retained."
		return false
	for suffix: String in ["", ".bak", ".tmp"]:
		if DirAccess.dir_exists_absolute(PATH+suffix): return false
	if FileAccess.file_exists(PATH+".tmp") or (_expected.is_empty() and FileAccess.file_exists(PATH+".bak")): return false
	saving = true
	var result: bool = _write_data(_candidate_data,_expected)
	saving = false
	if result: _written = raw()
	return result

static func written_bytes(ticket: RefCounted) -> PackedByteArray:
	return _written.duplicate() if ticket != null and ticket == _ticket else PackedByteArray()

static func promote(ticket: RefCounted, owner: RefCounted, candidate: RefCounted, expected: PackedByteArray) -> bool:
	if _written.is_empty() or raw() != _written or not prepared_matches(ticket,owner,candidate,expected):
		forget_projection(); return false
	var arm: bool = _projection_ticket == ticket and _projection_checkpoint == candidate and _projection_value != null
	if arm:
		_projection_owner = candidate; _projection_ticket = null; _projection_checkpoint = null
	else: forget_projection()
	_owner = _candidate; _owner_session = _candidate_session; _owner_raw = _candidate_raw; _owner_bytes = _written.duplicate()
	_paused = _candidate.paused.duplicate(true); _active = _candidate.active
	_metadata = Codec.data_fields(_candidate,Owner.META_FIELDS); _imported_sha256 = _candidate.imported_sha256
	_ticket = null; _candidate = null; _candidate_session = null; _candidate_raw = {}; _candidate_data = {}
	_expected = PackedByteArray(); _written = PackedByteArray()
	return true
