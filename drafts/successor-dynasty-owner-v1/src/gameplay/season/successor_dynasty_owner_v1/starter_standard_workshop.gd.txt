extends RefCounted
## Draft custody only. Never equips the chair in MatchState or changes Cash.
const FIELD: String = "starter_workshop"
const Home = preload("res://src/gameplay/season/successor_live_play_v1/starter_home_transport.gd")
const COPY: String = Home.COPY
const APPLIED: String = "starter_workshop_applied" # Retained private late-proof dialect only.
const OPENING: String = "starter_workshop_opening"

static func _opening_schedule(season: SeasonState) -> bool:
	if not Home.opening_base(season): return false
	var away:=false
	var first_home_id:=99
	for fixture: Dictionary in season.schedule:
		if fixture.id==0 and fixture.round==0 and fixture.away==0 and fixture.home!=0 and not fixture.get("neutral",false):
			away=true
		if fixture.home==0 and not fixture.get("neutral",false): first_home_id=mini(first_home_id,int(fixture.id))
	return away and first_home_id==3

static func _unplayed_opener(season: SeasonState) -> bool:
	if not _opening_schedule(season) or season.phase!=SeasonState.Phase.REGULAR or season.round_index!=0 \
		or season.picks.size()!=4 or season.build==null or season.build.pack_pending() or season.physical==null:
		return false
	return season.pending_fixture().get("id")==0 and season.results.is_empty() and season.player_results.is_empty() \
		and season.physical.cursor==0 and season.physical.reports.is_empty() and season.physical.pending.is_empty() \
		and season.physical.needed.is_empty() and season.physical.error.is_empty() and not season.physical.projecting

static func opening_window(owner: RefCounted) -> bool:
	return _standard(owner) and owner.paused.is_empty() and not owner.ordinary_state.has(APPLIED) \
		and _unplayed_opener(owner.session)

static func opening_record(state: Dictionary, choice: String, locked: bool, version: int = 1) -> Dictionary:
	return {"version":version,"lineage":state.lineage,"ordinal":state.ordinal,"start":state.start_basis_sha256,
		"opener":0,"first_home":3,"choice":choice,"locked":locked}

static func opening_valid(state: Dictionary, season: SeasonState) -> bool:
	var chain: bool=Home.opening_chain(season)
	if not state.has(OPENING) or state.has(APPLIED) or not _opening_schedule(season) \
		or state.get("ordinal")!=(3 if chain else 2):
		return false
	var row: Variant=state[OPENING]
	if not row is Dictionary or not row.get("locked") is bool or not row.get("choice") is String \
		or not SeasonOwnership._whole(row.get("version"),3 if chain else 1,3 if chain else 2) \
		or not ClubCareer.same(row,opening_record(state,row.get("choice",""),row.locked,int(row.version))):
		return false
	if row.version>=2 and row.choice=="pending":
		return not row.locked and season.home_property.is_empty() \
			and ((season.phase==SeasonState.Phase.DRAFT and season.round_index==0) or _unplayed_opener(season))
	if season.phase==SeasonState.Phase.DRAFT or season.picks.size()!=4: return false
	if row.choice not in ["chair","stock"] or (row.version>=2 and not row.locked): return false
	if row.choice=="chair":
		if not state.has(FIELD): return false
		if row.locked: return not season.home_property.is_empty() and Home.valid(season)
		return season.home_property.is_empty() and _unplayed_opener(season)
	return row.locked and season.home_property.is_empty()

# Only new, authored v2 births get player choice. A round-zero legacy file
# could be the unchanged primary after leaving an unfinished first game.
static func player_opening_window(owner: RefCounted) -> bool:
	if not opening_window(owner): return false
	var state: Dictionary=owner.ordinary_state
	var row: Variant=state.get(OPENING)
	return row is Dictionary and row.get("version")== (3 if Home.opening_chain(owner.session) else 2) and row.get("choice")=="pending" \
		and opening_valid(state,owner.session)

static func player_opening_allowed(owner: RefCounted) -> bool:
	var state: Dictionary=owner.ordinary_state
	if not opening_valid(state,owner.session) or not valid(state): return false
	var row: Dictionary=state[OPENING]
	# The already locked pre-opening rehearsal uses the same authored
	# snapshot; it is readable, but never authorizes a new player choice.
	return row.version>=2 or row.locked

static func opening_unlocked(owner: RefCounted) -> bool:
	if owner==null or not owner.ordinary_state.has(OPENING): return false
	return not owner.ordinary_state[OPENING].get("locked",false)

static func stock_seed(lineage: String) -> int:
	return int(lineage.trim_prefix("ordinary:").left(8).hex_to_int()) & 0x7fffffff

static func ledger(lineage: String) -> ClubAssetLedger:
	return ClubAssetLedger.new(stock_seed(lineage),[0],[],{})

static func record(lineage: String) -> Dictionary:
	# Reuse the original zero-cost native freeze as the layout validation
	# oracle, not as a persisted Apply record or a gameplay permission.
	var snapshot: Dictionary = Home.snapshot(lineage)
	var layout: Variant = snapshot.get("layout")
	if not layout is Dictionary or layout.get("receiver") != COPY or layout.get("obstacles") != []: return {}
	return {"version":1,"lineage":lineage,"layout":layout}

static func valid(state: Dictionary) -> bool:
	if not state.has(FIELD): return true
	var authored := record(state.lineage)
	return not authored.is_empty() and ClubCareer.same(state[FIELD],authored)

static func _standard(owner: RefCounted) -> bool:
	if owner==null or owner.active!="ordinary" or owner.session==null: return false
	var season: SeasonState = owner.session
	return not season.league_live and season.league_rules.is_empty() and season.career!=null \
		and season.career.start_basis.get("kind")=="common-ordinary-start-v1" \
		and owner.ordinary_state.get("status")=="active"

static func visible(owner: RefCounted) -> bool:
	return _standard(owner) and (owner.ordinary_state.has(FIELD) or editable(owner))

static func editable(owner: RefCounted) -> bool:
	if not _standard(owner) or not owner.paused.is_empty() or owner.ordinary_state.has(FIELD): return false
	if player_opening_window(owner): return true
	var season: SeasonState = owner.session
	return season.phase==SeasonState.Phase.DRAFT and season.round_index==0 and season.picks.is_empty() \
		and season.home_property.is_empty() and season.results.is_empty() and season.build!=null \
		and not season.build.pack_pending() and season.physical!=null and season.physical.pending.is_empty()


# Hidden first-home family. The application window is an already admitted
# and settled away opener, never a substituted schedule, draft or paid item.
static func first_home(owner: RefCounted) -> bool:
	if not _standard(owner) or not owner.paused.is_empty() or owner.ordinary_state.has(APPLIED) \
		or owner.ordinary_state.has(OPENING): return false
	var season: SeasonState = owner.session
	var f: Dictionary=season.pending_fixture()
	return Home.base(season) and season.phase==SeasonState.Phase.REGULAR and season.round_index==1 \
		and season.results.size()==3 and season.player_results.size()==1 and season.home_property.is_empty() \
		and f.get("home")==0 and f.get("id")==3 and season.picks.size()==4 \
		and season.build!=null and not season.build.pack_pending() and season.physical!=null \
		and season.physical.cursor==2 and season.physical.reports.size()==2 and season.physical.pending.is_empty() \
		and season.physical.needed.is_empty() and season.physical.error.is_empty() and not season.physical.projecting \
		and not opening_digest(season).is_empty()

static func opening_digest(season: SeasonState) -> String:
	if season.results.size()<3 or season.player_results.is_empty() or season.physical.reports.size()<2: return ""
	var games: Array=season.results.slice(0,3)
	var reports: Array=season.physical.reports.slice(0,2)
	if games[0].get("id")!=0 or games[0].get("away")!=0 or games[0].has("property") \
		or games[1].get("id")!=1 or games[2].get("id")!=2 or season.player_results[0].get("id")!=0 \
		or reports[0].get("fixture")!=1 or reports[1].get("fixture")!=2: return ""
	# Normalize known emitted whole numbers across the JSON hydration boundary.
	# Compare the entire already-authoritative settled opening, not just scores.
	var text:=JSON.stringify([games,reports],"",true,true)
	var value: Variant=JSON.parse_string(text)
	return JSON.stringify(value,"",true,true).sha256_text() if value is Array else ""

static func placement(state: Dictionary, season: SeasonState) -> Dictionary:
	if not Home.base(season) or not state.has(FIELD): return {}
	var proof:=opening_digest(season)
	if proof.is_empty(): return {}
	return {"version":1,"lineage":state.lineage,"ordinal":state.ordinal,
		"start":state.start_basis_sha256,"first_home":3,"opening":proof}

static func placement_valid(state: Dictionary, season: SeasonState) -> bool:
	if state.has(OPENING): return opening_valid(state,season)
	if Home.opening_chain(season): return false # Even vacant stock/pending needs its birth-bound marker.
	if not state.has(APPLIED): return season.home_property.is_empty()
	if season.home_property.is_empty() or not Home.valid(season): return false
	var expected:=placement(state,season)
	return not expected.is_empty() and ClubCareer.same(state[APPLIED],expected)
