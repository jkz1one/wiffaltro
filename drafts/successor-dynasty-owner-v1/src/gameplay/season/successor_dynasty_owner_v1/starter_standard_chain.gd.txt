extends RefCounted
## Exact Starter Standard Season 3, following the settled first round and
## native Abandon of a marked Standard Season 2. Two ordered sources only.
const KIND: String = "common-standard-base-two-predecessors-v1"
const CUT_KIND: String = "common-standard-base-second-cutout-v1"
const TRANSITION: String = "common-standard-base-third-transition-v1"
const ORIGIN: String = "standard-base-third-archive-v1"
const MAX_DECODED: int = 24 * 1024 * 1024
const MAX_ENCODED: int = 12 * 1024 * 1024
const CUT_TAG: String = "\"archive\":{\"deflate_base64\":"

static func _owner() -> Script:
	return load("res://src/gameplay/season/successor_dynasty_owner_v1/owner.gd")

static func marked(season: SeasonState) -> bool:
	return season != null and season.career != null and season.league_rules.is_empty() \
		and not season.league_live and season.career.start_basis.get("version") == 3 \
		and season.career.start_basis.get("ordinal") == 3 \
		and season.career.start_basis.get("archive_origin",{}).get("kind") == ORIGIN

static func eligible(owner: RefCounted) -> bool:
	var api := _owner()
	if owner == null or owner.active != "ordinary" or not owner.paused.is_empty() \
		or not api._missing_dynasty(owner) or not owner.migration_v3.is_empty() \
		or not api.Common.StandardArchive.marked(owner.session): return false
	var state: Dictionary = owner.ordinary_state
	var close: Dictionary = state.get("last_close",{})
	var season: SeasonState = owner.session
	var physical = season.physical
	return state.get("ordinal") == 2 and state.get("status") == "closed" \
		and close.get("reason") == "abandon" and close.get("league") == "standard" \
		and close.get("tier") == 0 and close.get("receipt") == {} \
		and owner.common.get("sequence") == 2 and api.Codec.same(owner.common.get("cursors"),{"ordinary":2,"dynasty":0}) \
		and season.phase == SeasonState.Phase.REGULAR and season.round_index == 1 \
		and season.picks.size() == 4 and season.build != null and not season.build.pack_pending() \
		and season.results.map(func(r): return int(r.id)) == [0,1,2] \
		and season.player_results.size() == 1 and season.player_results[0].get("id") == 0 \
		and physical != null and physical.cursor == 2 and physical.reports.size() == 2 \
		and physical.reports[0].get("fixture") == 1 and physical.reports[1].get("fixture") == 2 \
		and physical.pending.is_empty() and physical.needed.is_empty() and physical.error.is_empty() \
		and not physical.projecting

static func seed_for(owner: RefCounted) -> int:
	# Identical to the native ordinary Next button's fixed lineage seed.
	return ("ordinary:%s:3:nextseed"%owner.ordinary_state.lineage).hash() & 0x7fffffff

static func proposal_scope(owner: RefCounted) -> bool:
	if not eligible(owner): return false
	var p: Dictionary = owner.ordinary_state.proposal
	var basis: Variant = p.get("basis")
	return p.get("league") == "standard" and p.get("tier") == 0 \
		and basis is Dictionary and basis.get("version") == 1 and basis.get("ordinal") == 3 \
		and basis.get("seed") == seed_for(owner) and basis.get("common_sequence") == 2

static func ready(owner: RefCounted) -> bool:
	return proposal_scope(owner) and _owner().Common.valid(owner)

static func _blob_shape(row: Variant) -> bool:
	var api := _owner()
	var codec = api.Codec
	var archive = api.Common.StandardArchive
	return codec.keys(row,["kind","version","source_bytes","source_sha256","deflate_base64"]) \
		and row.kind == archive.KIND and codec.whole(row.version,1,1) \
		and codec.whole(row.source_bytes,1,archive.MAX_SOURCE) and api._digest(row.source_sha256) \
		and row.deflate_base64 is String and not row.deflate_base64.is_empty() \
		and row.deflate_base64.length() <= archive.MAX_ENCODED

static func shape(owner: RefCounted) -> bool:
	var api := _owner()
	var state: Dictionary = owner.ordinary_state
	var t: Dictionary = state.last_transition
	var row: Variant = state.get("archive")
	if not api.Codec.keys(row,["kind","entries"]) or row.kind != KIND \
		or not row.entries is Array or row.entries.size() != 2 or not _blob_shape(row.entries[0]):
		return false
	var first: Dictionary = row.entries[0]
	var second: Variant = row.entries[1]
	if not api.Codec.keys(second,["kind","source_bytes","source_sha256","offset","cut"]) \
		or second.kind != CUT_KIND or not _blob_shape(second.cut) \
		or not api.Codec.whole(second.source_bytes,1,api.Common.StandardArchive.MAX_SOURCE) \
		or not api.Codec.whole(second.offset,1,int(second.cut.source_bytes)-1) \
		or not api._digest(second.source_sha256) or first.source_sha256 == second.source_sha256 \
		or second.source_bytes != second.cut.source_bytes+first.deflate_base64.length()+2 \
		or first.source_bytes+second.source_bytes > MAX_DECODED \
		or first.deflate_base64.length()+second.cut.deflate_base64.length() > MAX_ENCODED:
		return false
	return owner.active == "ordinary" and owner.paused.is_empty() and api._missing_dynasty(owner) \
		and owner.migration_v3.is_empty() and state.ordinal == 3 and state.status == "active" \
		and state.proposal.is_empty() and t.get("kind") == TRANSITION and t.get("league") == "standard" \
		and t.basis.get("version") == 3 and t.basis.archive_origin.kind == ORIGIN \
		and api.Codec.same(t.basis.archive_origin.ordered_sha256,[first.source_sha256,second.source_sha256])

# The normal writer orders JSON keys; the archived payload is ASCII and its
# exact quoted value occurs in this one slot. No numeric text gets regenerated.
static func _cut(raw: PackedByteArray, first: Dictionary) -> Dictionary:
	var text := raw.get_string_from_utf8()
	if text.to_utf8_buffer() != raw: return {}
	var token: String = CUT_TAG+JSON.stringify(first.deflate_base64)
	var at: int = text.find(token)
	if at < 0 or text.find(token,at+token.length()) >= 0: return {}
	var prefix := text.substr(0,at+CUT_TAG.length()).to_utf8_buffer()
	var suffix := text.substr(at+token.length()).to_utf8_buffer()
	if prefix.is_empty() or suffix.is_empty(): return {}
	return {"offset":prefix.size(),"parts":prefix+suffix}

static func _pack_second(raw: PackedByteArray, first: Dictionary) -> Dictionary:
	var api := _owner()
	if raw.is_empty() or raw.size() > api.Common.StandardArchive.MAX_SOURCE: return {}
	var split := _cut(raw,first)
	if split.is_empty(): return {}
	var packed: Dictionary = api.Common.StandardArchive._pack(split.parts)
	if packed.is_empty(): return {}
	return {"kind":CUT_KIND,"source_bytes":raw.size(),"source_sha256":api._hash(raw),
		"offset":split.offset,"cut":packed}

static func _unpack_second(row: Dictionary, first: Dictionary) -> PackedByteArray:
	var api := _owner()
	var cut: PackedByteArray = api.Common.StandardArchive._unpack(row.cut)
	if cut.is_empty(): return PackedByteArray()
	var at: int = int(row.offset)
	var raw := cut.slice(0,at)+JSON.stringify(first.deflate_base64).to_utf8_buffer()+cut.slice(at)
	if raw.size() != int(row.source_bytes) or api._hash(raw) != row.source_sha256: return PackedByteArray()
	var check := _cut(raw,first)
	return raw if check.get("offset") == at and check.get("parts") == cut else PackedByteArray()

# Read-only score projection of an already-admitted pinned Store archive.
# Each exact predecessor is decompressed once; no career or owner is restored.
static func display(row: Dictionary) -> Array:
	var api := _owner()
	var first: Dictionary = row.entries[0]
	var complete: Dictionary = api.Common.StandardArchive.display(first)
	if complete.is_empty() or complete.ordinal != 1: return []
	var raw := _unpack_second(row.entries[1],first)
	if raw.is_empty(): return []
	var previous: Variant = api._bytes_value(raw)
	if not previous is Dictionary or not previous.get("ordinary_state") is Dictionary: return []
	var state: Dictionary = previous.ordinary_state
	var close: Variant = state.get("last_close")
	if state.get("status") != "closed" or state.get("ordinal") != 2 or not close is Dictionary \
		or close.get("ordinal") != 2 or close.get("reason") != "abandon" \
		or close.get("tier") != 0 or close.get("league") != "standard" \
		or not api.Codec.same(state.get("archive"),first): return []
	var abandoned: Dictionary = api.Common.StandardArchive._display_scores(previous,false)
	if abandoned.is_empty(): return []
	return [complete,abandoned]

static func candidate(owner: RefCounted, raw: PackedByteArray) -> RefCounted:
	var api := _owner()
	if not ready(owner) or not api.Common.StandardArchive.shape(owner): return null
	var first: Dictionary = owner.ordinary_state.archive
	var second := _pack_second(raw,first)
	if second.is_empty(): return null
	var p: Dictionary = owner.ordinary_state.proposal
	var basis: Dictionary = api.Common.Start.bind_standard_chain(p.basis,first.source_sha256,second.source_sha256)
	var next: SeasonState = api.Adapter.OrdinaryRuntime.create_common(basis,"standard")
	if next == null or not marked(next) or next.phase != SeasonState.Phase.DRAFT or not next.picks.is_empty():
		return null
	var result: RefCounted = api._new()
	if not api.Codec.install_fields(result,api.DATA_FIELDS,api.Codec.data_fields(owner,api.DATA_FIELDS)):
		return null
	result.session = next
	var state: Dictionary = result.ordinary_state
	state.last_transition = {"kind":TRANSITION,"id":p.id,"source":p.source,"from":2,"to":3,
		"league":"standard","basis":basis}
	state.ordinal = 3
	state.status = "active"
	state.last_close = {}
	state.proposal = {}
	state.start_basis_sha256 = api.Common.Life.digest(basis)
	state.archive = {"kind":KIND,"entries":[first.duplicate(true),second]}
	return result if api.Common.valid(result) else null

static func validate(owner: RefCounted) -> bool:
	if not shape(owner) or not marked(owner.session): return false
	var api := _owner()
	var rows: Array = owner.ordinary_state.archive.entries
	var first: Dictionary = rows[0]
	var second: Dictionary = rows[1]
	# First owner is hydrated once, then released; only its compact checked
	# fold/start facts survive to independently verify the second close.
	var facts: RefCounted = api.Common.StandardArchive._checked_first(first)
	if facts == null: return false
	var raw := _unpack_second(second,first)
	if raw.is_empty(): return false
	var value: Variant = api._bytes_value(raw)
	if not value is Dictionary or not value.get("ordinary_state") is Dictionary \
		or not api.Codec.same(value.ordinary_state.get("archive"),first): return false
	# Same canonical owner checks as cold restore, without recursive archive
	# replay. This fixed reader then validates its only allowed exact archive.
	var restored: RefCounted = api._restore_canonical(value)
	if restored == null: return false
	var previous: RefCounted = restored.owner
	if not ready(previous) or not api.Common.StandardArchive._validate_checked_first(previous,restored.current,facts):
		return false
	var p: Dictionary = previous.ordinary_state.proposal
	var t: Dictionary = owner.ordinary_state.last_transition
	var basis: Dictionary = api.Common.Start.bind_standard_chain(p.basis,first.source_sha256,second.source_sha256)
	if t.id != p.id or t.source != p.source or owner.ordinary_state.lineage != previous.ordinary_state.lineage \
		or not api.Codec.same(basis,t.basis): return false
	var metadata: Dictionary = api.Codec.data_fields(owner,api.META_FIELDS)
	var old_metadata: Dictionary = api.Codec.data_fields(previous,api.META_FIELDS)
	metadata.erase("ordinary_state")
	old_metadata.erase("ordinary_state")
	return api.Codec.same(metadata,old_metadata)
