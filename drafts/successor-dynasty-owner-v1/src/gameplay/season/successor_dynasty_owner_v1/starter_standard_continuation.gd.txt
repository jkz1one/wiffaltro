extends RefCounted
## Isolated native draft continuation. Never returns a production Owner or
## writes to a slot; CP287's flat byte custody is not a substitute for restore.
const Custody = preload("starter_standard_custody.gd")

class Draft:
	extends RefCounted
	var season: SeasonState
	var _basis: Dictionary
	var _archive: Dictionary
	var _common: Dictionary
	var _workshop: Dictionary
	var _source: Dictionary
	func basis() -> Dictionary: return _basis.duplicate(true)
	func archive() -> Dictionary: return _archive.duplicate(true)
	func common() -> Dictionary: return _common.duplicate(true)
	func workshop() -> Dictionary: return _workshop.duplicate(true)
	func source() -> Dictionary: return _source.duplicate(true)

static func _api() -> Script:
	return load("res://src/gameplay/season/successor_dynasty_owner_v1/owner.gd")

static func _scope(owner: RefCounted) -> bool:
	var api := _api()
	var work = api.Common.StandardWorkshop
	return owner!=null and owner.active=="ordinary" and owner.ordinary_state.get("status")=="closed" \
		and api.Common.StandardChain.tagged_abandon_scope(owner,owner.session) \
		and work.valid(owner.ordinary_state) and owner.ordinary_state.has(work.FIELD)

# No descriptor/rights/basis may be supplied by a caller. The source digest
# pins exact bytes; Owner.restore remains the sole native evidentiary gate.
static func draft_from_closed(raw: PackedByteArray, expected_sha256: String) -> Draft:
	var api := _api()
	if raw.is_empty() or raw.size()>api.Common.StandardArchive.MAX_SOURCE \
			or not api._digest(expected_sha256) or api._hash(raw)!=expected_sha256: return null
	var value: Variant=api._bytes_value(raw)
	var owner: RefCounted=api.restore(value)
	return _from_authenticated(owner,raw,expected_sha256)

# Store hot birth passes only its exact RawGuard/CAS-pinned admitted owner.
# This method has no admission or IO and cannot mint a Store ticket.
static func _from_authenticated(owner: RefCounted, raw: PackedByteArray, expected_sha256: String) -> Draft:
	var api := _api()
	if not _scope(owner): return null
	var archive: Dictionary=Custody.migrate_legacy_closed(raw)
	if archive.is_empty(): return null
	var state: Dictionary=owner.ordinary_state
	var ordinal: int=archive.rows.size()+1
	if archive.lineage!=state.lineage or state.ordinal!=ordinal-1 or owner.common.sequence!=ordinal-1 \
			or archive.rows[-1].source_sha256!=expected_sha256: return null
	var ordered: Array[String]=[]
	for row: Dictionary in archive.rows: ordered.append(row.source_sha256)
	var common = api.Common
	if ordered.size()!=common.Start.STANDARD_FLAT_ROOTS: return null
	# Same authored ordinary Next seed derivation. No user-supplied seed,
	# alternate league/tier, title, receipt or currency operation.
	var seed_value: int=("ordinary:%s:%d:nextseed"%[state.lineage,ordinal]).hash() & 0x7fffffff
	var native: Dictionary=common.Start.make(seed_value,state.lineage,ordinal,int(owner.common.sequence),
		common.Life.digest(owner.common.aggregate),common.projection(owner))
	var basis: Dictionary=common.Start.bind_standard_flat(native,ordered)
	var next: SeasonState=api.Adapter.OrdinaryRuntime.create_common(basis,"standard")
	if next==null or next.phase!=SeasonState.Phase.DRAFT or not next.picks.is_empty() \
			or next.season_seed!=seed_value or not next.league_rules.is_empty() or next.league_live \
			or not next.home_property.is_empty() or not api.Codec.same(next.career.start_basis,basis): return null
	var result := Draft.new()
	result.season=next
	result._basis=basis
	result._archive=archive
	result._common=owner.common.duplicate(true)
	# Keep only the authored saved draft, not old opening/installed state.
	result._workshop=state[common.StandardWorkshop.FIELD].duplicate(true)
	result._source={"sha256":expected_sha256,"lineage":state.lineage,"ordinal":int(state.ordinal),
		"close":state.last_close.duplicate(true)}
	return result

# Comparison for a detached view/test only, never admission. Re-derive
# custody from exact source again at any future production writer boundary.
static func matches_detached(result: Draft, basis: Variant, archive: Variant) -> bool:
	var api := _api()
	if result==null or result.season==null or result.season.phase!=SeasonState.Phase.DRAFT \
			or not result.season.home_property.is_empty() \
			or not api.Codec.same(result.season.career.start_basis,result._basis) \
			or not api.Codec.same(basis,result._basis): return false
	return api.Codec.same(archive,result._archive)


const TRANSITION: String="common-standard-base-flat-transition-v1"

static func is_flat(owner: RefCounted) -> bool:
	return owner!=null and owner.get_script()==_api() and owner.session!=null and owner.session.career!=null \
		and _api().Common.Start.detached_standard_flat(owner.session.career.start_basis)

static func _local_owner(source: RefCounted, draft: Draft) -> RefCounted:
	var api := _api()
	if draft==null: return null
	var owner: RefCounted=api._new()
	if not api.Codec.install_fields(owner,api.DATA_FIELDS,api.Codec.data_fields(source,api.DATA_FIELDS)):
		return null
	owner.session=draft.season
	var state: Dictionary=owner.ordinary_state
	var work = api.Common.StandardWorkshop
	var basis:=draft.basis()
	var close: Dictionary=state.last_close
	state.last_transition={"kind":TRANSITION,"id":"%s:%d:next"%[state.lineage,state.ordinal],
		"source":close.session_sha256,"from":state.ordinal,"to":basis.ordinal,"league":"standard","basis":basis}
	state.ordinal=basis.ordinal
	state.status="active"
	state.last_close={}
	state.proposal={}
	state.start_basis_sha256=api.Common.Life.digest(basis)
	state.archive=draft.archive()
	state.erase(work.OPENING)
	state.erase(work.APPLIED)
	return owner

const OPENING_VERSION: int=4
const ROUND1: String="starter_flat_round1"
const ROUND2: String="starter_flat_round2"

static func _preopening(season: SeasonState) -> bool:
	if season.phase!=SeasonState.Phase.REGULAR or season.picks.size()!=4: return false
	var opener: Dictionary=season.pending_fixture()
	if opener.get("id")!=0 or opener.get("round")!=0 or opener.get("away")!=0 \
			or opener.get("home",0)==0 or opener.get("neutral",false): return false
	var first_home_id: int=99
	for row: Dictionary in season.schedule:
		if row.get("home")==0 and not row.get("neutral",false): first_home_id=mini(first_home_id,int(row.id))
	return first_home_id==3

static func _opening_valid(state: Dictionary, season: SeasonState) -> bool:
	var api := _api()
	var work = api.Common.StandardWorkshop
	var marker: Variant=state.get(work.OPENING)
	if not marker is Dictionary or not marker.get("locked") is bool: return false
	var choice: Variant=marker.get("choice")
	if choice not in ["pending","stock","chair"] or marker.locked!=(choice!="pending") \
			or not api.Codec.same(marker,work.opening_record(state,choice,marker.locked,OPENING_VERSION)): return false
	if choice!="chair": return season.home_property.is_empty()
	return not season.home_property.is_empty() and work.Home.opening_flat(season) and work.Home.valid(season)

static func _draft_scope(owner: RefCounted) -> bool:
	var api := _api()
	if not is_flat(owner) or owner.active!="ordinary" \
			or not owner.paused.is_empty() or not api._missing_dynasty(owner) or not owner.migration_v3.is_empty(): return false
	var state: Dictionary=owner.ordinary_state
	var season: SeasonState=owner.session
	var work = api.Common.StandardWorkshop
	var basis: Dictionary=season.career.start_basis
	if state.get("status")!="active" or state.get("ordinal")!=basis.get("ordinal") \
			or state.get("archive",{}).get("kind")!=Custody.KIND or state.get("last_transition",{}).get("kind")!=TRANSITION \
			or not work.valid(state) or not state.has(work.FIELD) or state.has(work.APPLIED) or state.has(ROUND1) or state.has(ROUND2) \
			or season.round_index!=0 or not season.results.is_empty() or not season.player_results.is_empty() \
			or season.physical==null or season.physical.cursor!=0 \
			or not season.physical.reports.is_empty() or not season.physical.needed.is_empty() \
			or not season.physical.error.is_empty() or season.physical.projecting: return false
	if season.phase==SeasonState.Phase.DRAFT:
		return season.picks.size()<4 and season.physical.pending.is_empty() and not state.has(work.OPENING) and season.home_property.is_empty()
	return _preopening(season) and _opening_valid(state,season) \
		and (season.physical.pending.is_empty() or _human_pending_shape(season.physical.pending,state[work.OPENING]))

static func _report_rows_shape(reports: Variant, first: int) -> bool:
	var api := _api()
	if not reports is Array or reports.size()>2: return false
	for i: int in reports.size():
		var row: Variant=reports[i]
		if not api.Codec.keys(row,["fixture","request","report"]) or not api.Codec.whole(row.fixture,i+first,i+first) \
				or not api._digest(row.request) or not row.report is Dictionary: return false
	return true

static func _human_pending_shape(pending: Variant, opening: Variant) -> bool:
	var api := _api()
	return opening is Dictionary and opening.get("choice")=="chair" and opening.get("locked")==true \
		and api.Codec.keys(pending,["human","reports"]) and pending.human is Array and pending.human.size()==13 \
		and api.Codec.whole(pending.human[0],0,0) and _report_rows_shape(pending.reports,1)

static func _home_pending_shape(pending: Variant) -> bool:
	var api := _api()
	return api.Codec.keys(pending,["human","reports","property"]) \
		and pending.human is Array and pending.human.size()==13 and api.Codec.whole(pending.human[0],3,3) \
		and _report_rows_shape(pending.reports,4) and api._digest(pending.property)

# Derive the current job from the actual native schedule; the allowed set is
# deliberately just the first two authored regular rounds of this dialect.
static func _jobs(season: SeasonState) -> Array[Dictionary]:
	var rows: Array[Dictionary]=[]
	for row: Dictionary in season.schedule:
		if row.get("round")==season.round_index and 0 not in [row.get("home"),row.get("away")]:
			rows.append(row)
	rows.sort_custom(func(a,b): return a.id<b.id)
	return rows

static func project_regular(owner: RefCounted) -> Dictionary:
	if not _storage_scope(owner) or not owner.session.round_index in [0,1] or owner.session.physical.pending.is_empty(): return {}
	var season: SeasonState=owner.session
	var api := _api()
	var checked: Dictionary=api.Adapter.OrdinaryPending.project(season)
	var day: Variant=checked.get("candidate")
	if checked.has("error") or not day is SeasonState or day==season: return {}
	var count: int=season.physical.pending.reports.size()
	if count==2:
		return checked if not checked.has("fixture") and _paid_prefix(day,season.round_index+1) and day.physical.pending.is_empty() else {}
	var fixture: Variant=checked.get("fixture")
	var jobs:=_jobs(season)
	return checked if jobs.size()==2 and api.Codec.same(fixture,jobs[count]) \
		and day.physical.cursor==season.physical.cursor+count \
		and day.physical.reports.size()==season.physical.reports.size()+count else {}

static func project_opening(owner: RefCounted) -> Dictionary:
	return project_regular(owner) if _draft_scope(owner) else {}

static func _validate_human_pending(owner: RefCounted) -> bool:
	return not project_opening(owner).is_empty()

static func _settled_day(season: SeasonState) -> bool:
	return _opening_paid_prefix(season) and season.physical.pending.is_empty()

static func _opening_paid_prefix(season: SeasonState) -> bool:
	return _paid_prefix(season,1)

static func _paid_prefix(season: SeasonState, count: int) -> bool:
	if count not in [1,2] or season.phase!=SeasonState.Phase.REGULAR or season.round_index!=count or season.picks.size()!=4 \
			or season.physical==null or season.physical.projecting \
			or not season.physical.needed.is_empty() or not season.physical.error.is_empty(): return false
	var api := _api()
	var prefix: Dictionary=api.Common.StandardArchive.settled_regular_prefix(season.results,count)
	var next: Dictionary=season.pending_fixture()
	return not prefix.is_empty() and api.Codec.same(season.player_results.map(func(r): return r.id),prefix.human) \
		and season.physical.cursor==count*2 and api.Codec.same(season.physical.reports.map(func(r): return r.fixture),prefix.cpu) \
		and next.get("id")==count*3 and not next.get("neutral",false) \
		and (next.get("home")==0 if count==1 else next.get("away")==0 and next.get("home")!=0)

static func _storage_scope(owner: RefCounted) -> bool:
	if _draft_scope(owner): return true
	if not is_flat(owner): return false
	var season: SeasonState=owner.session
	var count: int=season.round_index
	if not _paid_prefix(season,count): return false
	var api := _api()
	var work = api.Common.StandardWorkshop
	var state: Dictionary=owner.ordinary_state
	var history: Variant=state.get(ROUND1)
	var rows: Array=season.physical.reports
	if not (owner.active=="ordinary" and owner.paused.is_empty() and api._missing_dynasty(owner) \
		and owner.migration_v3.is_empty() and state.get("status")=="active" and state.get("ordinal")==4 \
		and state.get("archive",{}).get("kind")==Custody.KIND and state.get("last_transition",{}).get("kind")==TRANSITION \
		and work.valid(state) and state.has(work.FIELD) and not state.has(work.APPLIED) \
		and _opening_valid(state,season) and api.Codec.keys(history,["version","human"]) \
		and history.version==1 and _human_pending_shape({"human":history.human,"reports":rows.slice(0,2)},state[work.OPENING])):
		return false
	var pending: Dictionary=season.physical.pending
	if count==1:
		return not state.has(ROUND2) and (pending.is_empty() or (_home_pending_shape(pending) and work.Home.pending_shape(season)))
	var second: Variant=state.get(ROUND2)
	return pending.is_empty() and api.Codec.keys(second,["version","human","property"]) and second.version==1 \
		and _home_pending_shape({"human":second.human,"reports":rows.slice(2),"property":second.property}) \
		and second.property==work.Home.key(season,SeasonVenues.fixture_for_id(season,3))

# One regular human selector, driven by the admitted current authored fixture
# and its exact previous round. No intervening shop/playoffs/other lanes.
static func human_fixture(owner: RefCounted) -> Dictionary:
	if not _storage_scope(owner) or not owner.session.physical.pending.is_empty(): return {}
	if _locked(owner) or _settled_day(owner.session):
		return owner.session.pending_fixture()
	return {}

# Only the actual native fourth choice is allowed to enter the opening window.
static func _enter_opening(owner: RefCounted) -> bool:
	var work = _api().Common.StandardWorkshop
	var state: Dictionary=owner.ordinary_state
	if state.has(work.OPENING) or not owner.session.home_property.is_empty() or not _preopening(owner.session):
		return false
	state[work.OPENING]=work.opening_record(state,"pending",false,OPENING_VERSION)
	return _draft_scope(owner)

static func _lock_opening(owner: RefCounted, choice: String) -> bool:
	if choice not in ["stock","chair"] or not _draft_scope(owner): return false
	var work = _api().Common.StandardWorkshop
	var state: Dictionary=owner.ordinary_state
	if state.get(work.OPENING,{}).get("choice")!="pending": return false
	if choice=="chair":
		owner.session.home_property=work.Home.snapshot(state.lineage)
		if owner.session.home_property.is_empty(): return false
	state[work.OPENING]=work.opening_record(state,choice,true,OPENING_VERSION)
	return _draft_scope(owner)

static func _locked(owner: RefCounted) -> bool:
	return _draft_scope(owner) and owner.ordinary_state.get(_api().Common.StandardWorkshop.OPENING,{}).get("locked",false)

# Used only on a newly derived owner or a fork of a privately admitted owner.
# The generic Owner/Adapter persistence routes continue to reject this dialect.
static func _capture_local(owner: RefCounted) -> Dictionary:
	if not _storage_scope(owner): return {}
	var api := _api()
	var current: Dictionary=api.Adapter.Ordinary.capture(owner.session)
	if current.is_empty(): return {}
	var data: Dictionary=api.Codec.data_fields(owner,api.META_FIELDS)
	for field: String in ["workshop_draft","workshop_close_history","workshop_chair_purchase","workshop_applied"]:
		if not data[field].is_empty(): return {}
		data.erase(field)
	data.merge({"version":4,"format":api.FORMAT,"active":"ordinary","dynasty":{},"ordinary":current})
	return data

# No portable trust flag. Exact ordered roots must reconstruct the old primary;
# stock Owner.restore verifies its two preceding folds, current and all history.
# Then the whole destination is rederived from that raw and checked native picks.
static func _restore_private(value: Variant) -> RefCounted:
	if not value is Dictionary or not value.get("ordinary_state") is Dictionary \
			or not value.get("ordinary") is Dictionary: return null
	var api := _api()
	var state: Dictionary=value.ordinary_state
	var transition: Variant=state.get("last_transition")
	var archive: Variant=state.get("archive")
	if not transition is Dictionary or transition.get("kind")!=TRANSITION \
			or not api.Common.Start.valid(transition.get("basis")) \
			or not api.Common.Start.detached_standard_flat(transition.basis) or not Custody._budget(archive): return null
	var digests: Array=archive.rows.map(func(row): return row.source_sha256)
	if not api.Codec.same(digests,transition.basis.archive_origin.ordered_sha256) \
			or archive.lineage!=transition.basis.lineage or not value.ordinary.get("state") is Dictionary: return null
	var picks: Variant=value.ordinary.state.get("picks")
	if not picks is Array or picks.size()>4: return null
	var work = api.Common.StandardWorkshop
	var opening: Variant=state.get(work.OPENING)
	if picks.size()==4:
		if not opening is Dictionary or opening.get("version")!=OPENING_VERSION \
				or opening.get("choice") not in ["pending","stock","chair"]: return null
	elif state.has(work.OPENING): return null
	if not value.ordinary.get("physical") is Dictionary: return null
	var pending: Variant=value.ordinary.physical.get("pending")
	var paid: bool=state.has(ROUND1)
	var paid_home: bool=state.has(ROUND2)
	if not pending is Dictionary or (not pending.is_empty() and (picks.size()!=4 \
			or (not _home_pending_shape(pending) if paid else not _human_pending_shape(pending,opening)))): return null
	var history: Variant=state.get(ROUND1)
	var history2: Variant=state.get(ROUND2)
	var reports: Variant=value.ordinary.physical.get("reports")
	if paid and (picks.size()!=4 or not api.Codec.keys(history,["version","human"]) \
			or history.version!=1 or not reports is Array or reports.size()!=(4 if paid_home else 2) \
			or not _human_pending_shape({"human":history.human,"reports":reports.slice(0,2)},opening)): return null
	if paid_home and (not paid or not pending.is_empty() or not api.Codec.keys(history2,["version","human","property"]) \
			or history2.version!=1 or not _home_pending_shape({"human":history2.human,"reports":reports.slice(2),"property":history2.property})): return null
	var raw: PackedByteArray
	var roots:=Custody.unpack(archive)
	if roots.is_empty(): return null
	raw=roots[-1]
	roots.clear()
	var prior: RefCounted=api.restore(api._bytes_value(raw))
	var draft:=_from_authenticated(prior,raw,digests[-1])
	if not matches_detached(draft,transition.basis,archive): return null
	var owner:=_local_owner(prior,draft)
	if owner==null: return null
	for id: Variant in picks:
		if not id is String or not owner.session.choose_player(id): return null
	if picks.size()==4 and (not _enter_opening(owner) \
			or (opening.choice!="pending" and not _lock_opening(owner,opening.choice))): return null
	if paid:
		owner.session.physical.pending={"human":history.human.duplicate(true),"reports":reports.slice(0,2).duplicate(true)}
		var projection:=project_regular(owner)
		if projection.is_empty(): return null
		owner.session=projection.candidate
		owner.ordinary_state[ROUND1]=history.duplicate(true)
		if paid_home:
			owner.session.physical.pending={"human":history2.human.duplicate(true),
				"reports":reports.slice(2).duplicate(true),"property":history2.property}
			projection=project_regular(owner)
			if projection.is_empty(): return null
			owner.session=projection.candidate
			owner.ordinary_state[ROUND2]=history2.duplicate(true)
		elif not pending.is_empty(): owner.session.physical.pending=pending.duplicate(true)
	elif not pending.is_empty():
		owner.session.physical.pending=pending.duplicate(true)
		if not _validate_human_pending(owner): return null
	return owner if api.Codec.same(value,_capture_local(owner)) else null
