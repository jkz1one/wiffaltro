extends RefCounted
## Isolated native draft continuation. Never returns a production Owner or
## writes to a slot; CP287's flat byte custody is not a substitute for restore.
const Custody = preload("starter_standard_custody.gd")

class Draft:
	extends RefCounted
	var season: SeasonState
	var _basis: Dictionary
	var _archive: Dictionary
	var _common: Dictionary
	var _workshop: Dictionary
	var _source: Dictionary
	func basis() -> Dictionary: return _basis.duplicate(true)
	func archive() -> Dictionary: return _archive.duplicate(true)
	func common() -> Dictionary: return _common.duplicate(true)
	func workshop() -> Dictionary: return _workshop.duplicate(true)
	func source() -> Dictionary: return _source.duplicate(true)

static func _api() -> Script:
	return load("res://src/gameplay/season/successor_dynasty_owner_v1/owner.gd")

static func _scope(owner: RefCounted) -> bool:
	var api := _api()
	var work = api.Common.StandardWorkshop
	return owner!=null and owner.active=="ordinary" and owner.ordinary_state.get("status")=="closed" \
		and api.Common.StandardChain.tagged_abandon_scope(owner,owner.session) \
		and work.valid(owner.ordinary_state) and owner.ordinary_state.has(work.FIELD)

# No descriptor/rights/basis may be supplied by a caller. The source digest
# pins exact bytes; Owner.restore remains the sole native evidentiary gate.
static func draft_from_closed(raw: PackedByteArray, expected_sha256: String) -> Draft:
	var api := _api()
	if raw.is_empty() or raw.size()>api.Common.StandardArchive.MAX_SOURCE \
			or not api._digest(expected_sha256) or api._hash(raw)!=expected_sha256: return null
	var value: Variant=api._bytes_value(raw)
	var owner: RefCounted=api.restore(value)
	return _from_authenticated(owner,raw,expected_sha256)

# Store hot birth passes only its exact RawGuard/CAS-pinned admitted owner.
# This method has no admission or IO and cannot mint a Store ticket.
static func _from_authenticated(owner: RefCounted, raw: PackedByteArray, expected_sha256: String) -> Draft:
	var api := _api()
	if not _scope(owner): return null
	var archive: Dictionary=Custody.migrate_legacy_closed(raw)
	if archive.is_empty(): return null
	var state: Dictionary=owner.ordinary_state
	var ordinal: int=archive.rows.size()+1
	if archive.lineage!=state.lineage or state.ordinal!=ordinal-1 or owner.common.sequence!=ordinal-1 \
			or archive.rows[-1].source_sha256!=expected_sha256: return null
	var ordered: Array[String]=[]
	for row: Dictionary in archive.rows: ordered.append(row.source_sha256)
	var common = api.Common
	if ordered.size()!=common.Start.STANDARD_FLAT_ROOTS: return null
	# Same authored ordinary Next seed derivation. No user-supplied seed,
	# alternate league/tier, title, receipt or currency operation.
	var seed_value: int=("ordinary:%s:%d:nextseed"%[state.lineage,ordinal]).hash() & 0x7fffffff
	var native: Dictionary=common.Start.make(seed_value,state.lineage,ordinal,int(owner.common.sequence),
		common.Life.digest(owner.common.aggregate),common.projection(owner))
	var basis: Dictionary=common.Start.bind_standard_flat(native,ordered)
	var next: SeasonState=api.Adapter.OrdinaryRuntime.create_common(basis,"standard")
	if next==null or next.phase!=SeasonState.Phase.DRAFT or not next.picks.is_empty() \
			or next.season_seed!=seed_value or not next.league_rules.is_empty() or next.league_live \
			or not next.home_property.is_empty() or not api.Codec.same(next.career.start_basis,basis): return null
	var result := Draft.new()
	result.season=next
	result._basis=basis
	result._archive=archive
	result._common=owner.common.duplicate(true)
	# Keep only the authored saved draft, not old opening/installed state.
	result._workshop=state[common.StandardWorkshop.FIELD].duplicate(true)
	result._source={"sha256":expected_sha256,"lineage":state.lineage,"ordinal":int(state.ordinal),
		"close":state.last_close.duplicate(true)}
	return result

# Comparison for a detached view/test only, never admission. Re-derive
# custody from exact source again at any future production writer boundary.
static func matches_detached(result: Draft, basis: Variant, archive: Variant) -> bool:
	var api := _api()
	if result==null or result.season==null or result.season.phase!=SeasonState.Phase.DRAFT \
			or not result.season.home_property.is_empty() \
			or not api.Codec.same(result.season.career.start_basis,result._basis) \
			or not api.Codec.same(basis,result._basis): return false
	return api.Codec.same(archive,result._archive)


const TRANSITION: String="common-standard-base-flat-transition-v1"

static func is_flat(owner: RefCounted) -> bool:
	return owner!=null and owner.get_script()==_api() and owner.session!=null and owner.session.career!=null \
		and _api().Common.Start.detached_standard_flat(owner.session.career.start_basis)

static func _local_owner(source: RefCounted, draft: Draft) -> RefCounted:
	var api := _api()
	if draft==null: return null
	var owner: RefCounted=api._new()
	if not api.Codec.install_fields(owner,api.DATA_FIELDS,api.Codec.data_fields(source,api.DATA_FIELDS)):
		return null
	owner.session=draft.season
	var state: Dictionary=owner.ordinary_state
	var work = api.Common.StandardWorkshop
	var basis:=draft.basis()
	var close: Dictionary=state.last_close
	state.last_transition={"kind":TRANSITION,"id":"%s:%d:next"%[state.lineage,state.ordinal],
		"source":close.session_sha256,"from":state.ordinal,"to":basis.ordinal,"league":"standard","basis":basis}
	state.ordinal=basis.ordinal
	state.status="active"
	state.last_close={}
	state.proposal={}
	state.start_basis_sha256=api.Common.Life.digest(basis)
	state.archive=draft.archive()
	state.erase(work.OPENING)
	state.erase(work.APPLIED)
	return owner

static func _draft_scope(owner: RefCounted) -> bool:
	var api := _api()
	if not is_flat(owner) or owner.get_script()!=api or owner.active!="ordinary" \
			or not owner.paused.is_empty() or not api._missing_dynasty(owner) or not owner.migration_v3.is_empty(): return false
	var state: Dictionary=owner.ordinary_state
	var season: SeasonState=owner.session
	var work = api.Common.StandardWorkshop
	var basis: Dictionary=season.career.start_basis
	return state.get("status")=="active" and state.get("ordinal")==basis.get("ordinal") \
		and state.get("archive",{}).get("kind")==Custody.KIND and state.get("last_transition",{}).get("kind")==TRANSITION \
		and work.valid(state) and state.has(work.FIELD) and not state.has(work.OPENING) and not state.has(work.APPLIED) \
		and season.phase==SeasonState.Phase.DRAFT and season.picks.size()<4 and season.round_index==0 \
		and season.results.is_empty() and season.player_results.is_empty() and season.home_property.is_empty() \
		and season.physical!=null and season.physical.cursor==0 and season.physical.pending.is_empty() \
		and season.physical.reports.is_empty() and season.physical.needed.is_empty() \
		and season.physical.error.is_empty() and not season.physical.projecting

# Used only on a newly derived owner or a fork of a privately admitted owner.
# The generic Owner/Adapter persistence routes continue to reject this dialect.
static func _capture_local(owner: RefCounted) -> Dictionary:
	if not _draft_scope(owner): return {}
	var api := _api()
	var current: Dictionary=api.Adapter.Ordinary.capture(owner.session)
	if current.is_empty(): return {}
	var data: Dictionary=api.Codec.data_fields(owner,api.META_FIELDS)
	for field: String in ["workshop_draft","workshop_close_history","workshop_chair_purchase","workshop_applied"]:
		if not data[field].is_empty(): return {}
		data.erase(field)
	data.merge({"version":4,"format":api.FORMAT,"active":"ordinary","dynasty":{},"ordinary":current})
	return data

# No portable trust flag. Exact ordered roots must reconstruct the old primary;
# stock Owner.restore verifies its two preceding folds, current and all history.
# Then the whole destination is rederived from that raw and checked native picks.
static func _restore_private(value: Variant) -> RefCounted:
	if not value is Dictionary or not value.get("ordinary_state") is Dictionary \
			or not value.get("ordinary") is Dictionary: return null
	var api := _api()
	var state: Dictionary=value.ordinary_state
	var transition: Variant=state.get("last_transition")
	var archive: Variant=state.get("archive")
	if not transition is Dictionary or transition.get("kind")!=TRANSITION \
			or not api.Common.Start.valid(transition.get("basis")) \
			or not api.Common.Start.detached_standard_flat(transition.basis) or not Custody._budget(archive): return null
	var digests: Array=archive.rows.map(func(row): return row.source_sha256)
	if not api.Codec.same(digests,transition.basis.archive_origin.ordered_sha256) \
			or archive.lineage!=transition.basis.lineage or not value.ordinary.get("state") is Dictionary: return null
	var picks: Variant=value.ordinary.state.get("picks")
	if not picks is Array or picks.size()>=4: return null
	var raw: PackedByteArray
	var roots:=Custody.unpack(archive)
	if roots.is_empty(): return null
	raw=roots[-1]
	roots.clear()
	var prior: RefCounted=api.restore(api._bytes_value(raw))
	var draft:=_from_authenticated(prior,raw,digests[-1])
	if not matches_detached(draft,transition.basis,archive): return null
	var owner:=_local_owner(prior,draft)
	if owner==null: return null
	for id: Variant in picks:
		if not id is String or not owner.session.choose_player(id): return null
	return owner if api.Codec.same(value,_capture_local(owner)) else null
