extends Node
## Fixed read-only Starter child. Only the parent Store may admit an exact verified revision.
signal completed(outcome: String)

const Store = preload("owner_store.gd")
const ARG := "--starter-mixed-continue-v1="
const FRAME := "WIFFALTRO_STARTER_MIXED_CONTINUE_V1 "
# At most seven banks: two masks per MAX_EVENTS and bounded held rows.
# The canonical Base64 packet bound includes full job, digest and framing.
const MAX_FRAME := 80 * 1024
const MAX_HELD := 64
const BANK_COUNT := 7
const EVENT_FIELDS := ["rev","game","amount"]
const PAID_FIELDS := ["paid"]
const BRACKET_FIELDS := ["id","round","away","home"]
const MASK_TOKENS := "0123456789abcdef"
const MAX_WHOLE := 2147483647
const MAX_MS := 300000

var _job: Dictionary
var _pid: int=-1
var _out: FileAccess
var _err: FileAccess
var _output: PackedByteArray
var _started: int
var _next_poll: int
var _completed: Dictionary={}
var _binding: Dictionary={}
var binding_stage: String="verifying"

static func supported() -> bool:
	var exe:=OS.get_executable_path()
	return Store.LeagueAdmission.starter_fast_worker() and exe.is_absolute_path() \
		and FileAccess.file_exists(exe.get_basename()+".pck")

static func _hex(value: Variant) -> bool:
	if not value is String or value.length()!=64: return false
	for token: String in value:
		if not "0123456789abcdef".contains(token): return false
	return true

static func _canonical(value: Dictionary) -> String:
	return JSON.stringify(value,"",true,true)

static func _valid(value: Variant) -> bool:
	return value is Dictionary and SeasonOwnership._keys(value,["revision","nonce"]) \
		and _hex(value.revision) and _hex(value.nonce) and value.revision!=value.nonce

static func parse(args: PackedStringArray) -> Dictionary:
	if args.size()!=1 or args[0].length()>256 or not args[0].begins_with(ARG): return {}
	var encoded:=args[0].trim_prefix(ARG)
	var serialized:=Marshalls.base64_to_utf8(encoded)
	if serialized.is_empty() or Marshalls.utf8_to_base64(serialized)!=encoded: return {}
	var job: Variant=JSON.parse_string(serialized)
	return job if _valid(job) and _canonical(job)==serialized else {}

static func bounded_slot() -> PackedByteArray:
	if Store._starter_slot_linked(): return PackedByteArray()
	var f:=FileAccess.open(Store.PATH,FileAccess.READ)
	if f==null: return PackedByteArray()
	var n:=f.get_length()
	if n<1 or n>Store.MAX_BYTES: return PackedByteArray()
	var raw:=f.get_buffer(n)
	var exact: bool=raw.size()==n and f.get_length()==n
	f.close()
	return raw if exact and not Store._starter_slot_linked() else PackedByteArray()

static func _revision() -> String:
	var raw:=bounded_slot()
	return "" if raw.is_empty() else Store.Owner._hash(raw)

# The fixed worker contributes only the int/whole-float representation of
# existing bank leaves from the already strictly restored native owner.
# Values, row ids, paths, and save authority never cross in this witness.
static func _banks(season: SeasonState) -> Array[SeasonOwnership]:
	var banks: Array[SeasonOwnership]=[]
	if season==null or season.build==null or season.ownership==null or season.opponents==null \
			or not SeasonOwnership._keys(season.opponents.clubs,["1","2","3","4","5"]): return banks
	banks.assign([season.ownership,season.build._bank])
	for id: String in ["1","2","3","4","5"]:
		var club: Variant=season.opponents.clubs[id]
		if not club is Dictionary or not club.get("build") is SeasonBuild: return []
		banks.append(club.build._bank)
	for bank: SeasonOwnership in banks:
		if bank==null or bank.get_script()!=SeasonOwnership: return []
	return banks

static func _counts(n: int, r: int, h: int) -> String:
	return "%d:%d:%d"%[n,r,h]

static func _unpack(bank: Variant) -> Dictionary:
	if not bank is String or bank.length()>MAX_FRAME: return {}
	var parts: PackedStringArray=bank.split(":")
	if parts.size()!=4: return {}
	var counts: Array[int]=[]
	for i: int in range(3):
		if not parts[i].is_valid_int(): return {}
		counts.append(int(parts[i]))
	var n: int=counts[0]
	var r: int=counts[1]
	var h: int=counts[2]
	if n<0 or n>SeasonOwnership.MAX_EVENTS or r!=n or h<0 or h>MAX_HELD \
			or _counts(n,r,h)!=":".join(parts.slice(0,3)): return {}
	# Godot reports an error to stderr when asked to encode an empty byte array.
	# Empty is a canonical, explicit shape here, never a decoder bypass.
	var bytes: PackedByteArray=PackedByteArray()
	if n+r+h==0:
		if not parts[3].is_empty(): return {}
	else:
		# Reject invalid text before calling Godot's logging decoder.
		var encoded: String=parts[3]
		var size: int=n+r+h
		var pad: int=(3-size%3)%3
		if encoded.length()!=4*ceili(float(size)/3.0): return {}
		for i: int in encoded.length():
			var token: String=encoded[i]
			if i>=encoded.length()-pad:
				if token!="=": return {}
			elif not "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/".contains(token): return {}
		bytes=Marshalls.base64_to_raw(encoded)
		if bytes.size()!=n+r+h or Marshalls.raw_to_base64(bytes)!=parts[3]: return {}
	for i: int in bytes.size():
		if bytes[i]>(7 if i<n+r else 1): return {}
	return {"events":n,"requests":r,"held":h,"mask":bytes}

static func _wire_valid(value: Variant) -> bool:
	if not value is Array or value.size()!=BANK_COUNT: return false
	for bank: Variant in value:
		if _unpack(bank).is_empty(): return false
	return true

static func _typed_row(row: Dictionary, fields: Array, mask: int=-1) -> int:
	var bits: int=0
	for i: int in fields.size():
		var field: String=fields[i]
		var bit: int=1<<i
		if not row.has(field):
			if mask>=0 and mask&bit: return -1 # No bit outside the existing leaves.
			continue
		var value: Variant=row[field]
		if not SeasonOwnership._whole(value,0,MAX_WHOLE): return -1
		if mask>=0:
			row[field]=int(value) if mask&bit else float(value)
		if typeof(row[field])==TYPE_INT: bits|=bit
	return bits

static func _bank_mask(bank: SeasonOwnership, supplied: Dictionary={}) -> String:
	var events: Array[Dictionary]=bank._events
	var requests: Dictionary=bank._requests
	var held: Variant=bank._state.get("held")
	if events.size()>SeasonOwnership.MAX_EVENTS or not held is Array or held.size()>MAX_HELD: return "!"
	var ids: Array[String]=[]
	var unique: Dictionary={}
	for row: Dictionary in events:
		var id: Variant=row.get("id")
		if not id is String or id.is_empty() or unique.has(id) or not requests.get(id) is Dictionary \
				or requests[id].get("id")!=id: return "!"
		unique[id]=true
		ids.append(id)
	ids.sort()
	if not supplied.is_empty() and (events.size()!=supplied.events or ids.size()!=supplied.requests \
			or held.size()!=supplied.held): return "!"
	var mask: PackedByteArray=supplied.get("mask",PackedByteArray())
	var out: PackedByteArray=PackedByteArray()
	for row: Dictionary in events:
		var b:=_typed_row(row,EVENT_FIELDS,-1 if supplied.is_empty() else mask[out.size()])
		if b<0: return "!"
		out.append(b)
	for id: String in ids:
		var b:=_typed_row(requests[id],EVENT_FIELDS,-1 if supplied.is_empty() else mask[out.size()])
		if b<0: return "!"
		out.append(b)
	for row: Variant in held:
		if not row is Dictionary or not row.has("paid"): return "!"
		var b:=_typed_row(row,PAID_FIELDS,-1 if supplied.is_empty() else mask[out.size()])
		if b<0: return "!"
		out.append(b)
	return _counts(events.size(),ids.size(),held.size())+":"+("" if out.is_empty() else Marshalls.raw_to_base64(out))

static func bank_witness(season: SeasonState) -> Array:
	var banks:=_banks(season)
	if banks.size()!=BANK_COUNT: return []
	var wire: Array=[]
	for bank: SeasonOwnership in banks:
		var mask:=_bank_mask(bank)
		if mask=="!": return []
		wire.append(mask)
	return wire

static func apply_bank_witness(season: SeasonState, witness: Variant) -> bool:
	if not _wire_valid(witness): return false
	var banks:=_banks(season)
	if banks.size()!=BANK_COUNT: return false
	for i: int in banks.size():
		if _bank_mask(banks[i],_unpack(witness[i]))!=witness[i]: return false
	# Re-emit the parent-applied representation before canonical recapture.
	return bank_witness(season)==witness

# Exact fixed postseason slots only. Stock semifinals omit neutral; the
# authored final carries true. No fixture values or arbitrary paths travel.
static func _bracket_wire_valid(value: Variant) -> bool:
	if not value is String or value.length() not in [0,4,6]: return false
	for base: int in range(0,value.length(),2):
		if not MASK_TOKENS.contains(value[base]) or value[base+1]!=("t" if base==4 else "-"): return false
	return true

static func _bracket_frame(row: Dictionary,index: int,mask: int=-1) -> String:
	var final_slot: bool=index==2
	var fields: Array=BRACKET_FIELDS.duplicate()
	if final_slot: fields.append("neutral")
	if not SeasonOwnership._keys(row,fields) \
			or not SeasonOwnership._whole(row.id,30+index,30+index) \
			or not SeasonOwnership._whole(row.round,11 if final_slot else 10,11 if final_slot else 10) \
			or not SeasonOwnership._whole(row.away,0,5) or not SeasonOwnership._whole(row.home,0,5) \
			or row.away==row.home or (final_slot and (not row.neutral is bool or not row.neutral)): return "!"
	var bits:=_typed_row(row,BRACKET_FIELDS,mask)
	return "!" if bits<0 else MASK_TOKENS[bits]+("t" if final_slot else "-")

static func _bracket_rows(season: SeasonState) -> Array:
	if season==null: return [null]
	var rows: Array=[]
	if season.phase in [SeasonState.Phase.DRAFT,SeasonState.Phase.REGULAR]:
		return rows if season.semifinals.is_empty() and season.final_fixture.is_empty() else [null]
	if season.semifinals.size()!=2: return [null]
	rows.assign(season.semifinals)
	if season.phase==SeasonState.Phase.SEMIFINAL:
		return rows if season.final_fixture.is_empty() else [null]
	if season.phase not in [SeasonState.Phase.FINAL,SeasonState.Phase.COMPLETE] or season.final_fixture.is_empty(): return [null]
	rows.append(season.final_fixture)
	return rows

static func bracket_witness(season: SeasonState) -> String:
	var rows:=_bracket_rows(season)
	var wire: String=""
	for i: int in rows.size():
		if not rows[i] is Dictionary: return "!"
		var frame:=_bracket_frame(rows[i],i)
		if frame=="!": return "!"
		wire+=frame
	return wire

static func apply_bracket_witness(season: SeasonState,witness: Variant) -> bool:
	if not _bracket_wire_valid(witness): return false
	var rows:=_bracket_rows(season)
	if rows.size()*2!=witness.length() or bracket_witness(season)=="!": return false
	for i: int in rows.size():
		if _bracket_frame(rows[i],i,MASK_TOKENS.find(witness[2*i]))!=witness.substr(2*i,2): return false
	return bracket_witness(season)==witness

static func _seal(job: Dictionary,current: String,banks: Array,bracket: String) -> String:
	return _canonical({"job":job,"current":current,"banks":banks,"bracket":bracket}).sha256_text()

static func packet(bytes: PackedByteArray, job: Dictionary, code: int) -> Dictionary:
	if code!=0 or not _valid(job) or bytes.is_empty() or bytes.size()>MAX_FRAME: return {}
	var text:=bytes.get_string_from_utf8()
	var lines:=text.split("\n")
	if text.to_utf8_buffer()!=bytes or lines.size()!=4 or not lines[0].begins_with("Godot Engine v") \
		or (not lines[1].is_empty() and lines[1]!=" ") or not lines[2].begins_with(FRAME) \
		or not lines[3].is_empty(): return {}
	var serialized:=lines[2].trim_prefix(FRAME)
	var value: Variant=JSON.parse_string(serialized)
	if not value is Dictionary or not SeasonOwnership._keys(value,["job","current","banks","bracket","seal"]) \
		or value.job!=job or _canonical(value)!=serialized or not _hex(value.current) or value.current in [job.revision,job.nonce] \
			or not _wire_valid(value.banks) or not _bracket_wire_valid(value.bracket) or not _hex(value.seal) \
			or value.seal!=_seal(job,value.current,value.banks,value.bracket):
		return {}
	return value

func start(expected: PackedByteArray) -> bool:
	if _pid>0 or not supported() or expected.is_empty() or expected.size()>Store.MAX_BYTES \
		or Store._starter_slot_linked() or not Store._write_revision_matches(expected,true): return false
	_job={"revision":Store.Owner._hash(expected),"nonce":Crypto.new().generate_random_bytes(32).hex_encode()}
	if not _valid(_job): return false
	var arg:=ARG+Marshalls.utf8_to_base64(_canonical(_job))
	var child:=OS.execute_with_pipe(OS.get_executable_path(),["--headless","--audio-driver","Dummy","--",arg],false)
	_pid=int(child.get("pid",-1))
	_out=child.get("stdio")
	_err=child.get("stderr")
	if _pid<1 or _out==null or _err==null:
		cancel()
		return false
	process_mode=PROCESS_MODE_ALWAYS
	_started=Time.get_ticks_msec()
	set_process(true)
	return true

func cancel() -> void:
	if _pid>0 and OS.is_process_running(_pid): OS.kill(_pid)
	_pid=-1
	_out=null
	_err=null
	_job={}
	_binding={}
	_output.clear()
	set_process(false)

# The seal exists only inside the exact worker-completion signal. It is
# consumed even when the caller supplies a different revision.
func take_current_seal(revision: String) -> Dictionary:
	var packet_value:=_completed
	_completed={}
	if packet_value.is_empty() or packet_value.job.revision!=revision: return {}
	_binding=packet_value.job
	return {"current":packet_value.current,"nonce":packet_value.job.nonce,"banks":packet_value.banks,"bracket":packet_value.bracket}

func binding_live(nonce: String,revision: String,consume: bool=false) -> bool:
	if is_queued_for_deletion() or not is_inside_tree() or _binding.is_empty() \
			or _binding.nonce!=nonce or _binding.revision!=revision: return false
	if consume: _binding={}
	return true

static func current_digest(value: Dictionary) -> String:
	return JSON.stringify(value,"",true,true).sha256_text()

func _exit_tree() -> void:
	cancel()

func _drain() -> bool:
	if _out==null or _err==null or _err.get_length()!=0: return false
	var size:=_out.get_length()
	if size<0 or size>MAX_FRAME-_output.size(): return false
	if size>0: _output.append_array(_out.get_buffer(size))
	return true

func _process(_delta: float) -> void:
	if _pid<1: return
	var now:=Time.get_ticks_msec()
	if now<_next_poll: return
	_next_poll=now+50
	if now-_started>MAX_MS or not _drain():
		cancel()
		completed.emit("refused")
		return
	if OS.is_process_running(_pid): return
	var code:=OS.get_process_exit_code(_pid)
	var result:=packet(_output,_job,code) if _drain() else {}
	var outcome: String="stale" if _revision()!=_job.revision else ("refused" if result.is_empty() else "verified")
	cancel()
	_completed=result if outcome=="verified" else {}
	completed.emit(outcome)
	_completed={}

# Only the fixed Starter bootstrap calls this. No arbitrary paths or writes.
func worker(job: Dictionary) -> void:
	if not supported() or not _valid(job) or DisplayServer.get_name()!="headless" \
		or not OS.get_cmdline_args().is_empty():
		get_tree().quit(3)
		return
	var raw:=bounded_slot()
	if raw.is_empty() or Store.Owner._hash(raw)!=job.revision:
		get_tree().quit(3)
		return
	var value: Variant=Store.Owner._bytes_value(raw)
	if not Store._mixed_continue_shape(value):
		get_tree().quit(3)
		return
	# The full cold verifier remains authoritative, including all historical
	# archives and each chronological result and Shop event.
	# Same owned authored Resources as the normal App and existing native witness
	# worker. Keeping them alive avoids repeated disk/resource loads during replay.
	var roots: Array[Resource]=preload("res://src/gameplay/season/successor_owner_general_v1/codec.gd")._operation_content_roots()
	var owner: RefCounted=Store.Owner.restore(value)
	roots.clear()
	if owner==null or not Store.Owner.Mixed.season_owner_scope(owner) or bounded_slot()!=raw:
		get_tree().quit(3)
		return
	var banks:=bank_witness(owner.session)
	var bracket:=bracket_witness(owner.session)
	if not _wire_valid(banks) or not _bracket_wire_valid(bracket):
		get_tree().quit(3)
		return
	var current:=current_digest(value.ordinary)
	var frame:=FRAME+_canonical({"job":job,"current":current,"banks":banks,"bracket":bracket,"seal":_seal(job,current,banks,bracket)})
	if frame.length()>MAX_FRAME-256:
		get_tree().quit(3)
		return
	print(frame)
	get_tree().quit(0)
