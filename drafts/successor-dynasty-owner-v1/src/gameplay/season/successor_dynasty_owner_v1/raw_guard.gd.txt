extends RefCounted
## Private live identity guard only. Never save admission or serialized authority.
## No semantic capture, catalog signatures, reconstruction or command execution.
const Codec = preload("codec.gd")
const Builds = preload("build_codec.gd")
const Ordinary = preload("res://src/gameplay/season/successor_live_play_v1/codec.gd")
const Adapter = preload("session_adapter.gd")
const MAX_DEPTH: int = 48
const MAX_NODES: int = 1000000

const Owner = preload("owner.gd")
static func capture(owner: Variant) -> Dictionary:
	var budget: Array = [MAX_NODES,true]
	var result: Dictionary = _object(owner,Owner,Owner.DATA_FIELDS,["session"],budget)
	if not budget[1]: return {}
	if not Adapter.side(owner.active): return {}
	result.session = capture_session(owner.session,owner.active)
	return result if not result.session.is_empty() else {}

static func capture_session(season: SeasonState, side: String = "dynasty") -> Dictionary:
	if not Adapter.side(side): return {}
	var schema: Script=Adapter.codec(side)
	var budget: Array = [MAX_NODES, true]
	var result: Dictionary = _object(season, schema.Runtime, schema.FIELDS, schema.OBJECTS, budget)
	if not budget[1]: return {}
	for name: String in ["_heat_replay_team", "_doubleheader_replay_build", "_pool_replay_ledger"]:
		if season.get(name) != null: return {}
		result[name] = null
	result.ownership = _bank(season.ownership, budget,side)
	result.build = _build(season.build, budget,side)
	result.career = _object(season.career, schema.Career, schema.CAREER_FIELDS, [], budget)
	result.physical = _object(season.physical, schema.Physical, schema.PHYSICAL_FIELDS, [], budget)
	result.opponents = _object(season.opponents, schema.Opponents, ["draft_order", "draft_rule", "_format"], ["clubs"], budget)
	if not budget[1]: return {}
	if season.opponents.clubs.is_typed(): return {}
	var clubs: Dictionary = {}
	for key: Variant in season.opponents.clubs:
		var club:=_club(season,key,budget,side)
		if club.is_empty(): return {}
		clubs[key]=club
	if season.opponents.clubs.is_read_only(): clubs.make_read_only()
	result.opponents["clubs"] = clubs
	return result if budget[1] else {}

# New admission has an entirely detached owner until this typed snapshot
# finishes. The normal synchronous writer guard stays exactly the same.
static func capture_mixed_continue(owner: RefCounted,lease: Node,valid_binding: Callable) -> Dictionary:
	if not Ordinary._lease_live(lease,valid_binding) or owner.active!="ordinary": return {}
	var tree: SceneTree=lease.get_tree()
	var owner_budget: Array=[MAX_NODES,true]
	var result:=_object(owner,Owner,Owner.DATA_FIELDS,["session"],owner_budget)
	if result.is_empty() or not owner_budget[1]: return {}
	var season: SeasonState=owner.session
	var budget: Array=[MAX_NODES,true]
	await tree.process_frame
	if not Ordinary._lease_live(lease,valid_binding): return {}
	var current:=_object(season,Ordinary.Runtime,Ordinary.FIELDS,Ordinary.OBJECTS,budget)
	if not budget[1]: return {}
	for name: String in ["_heat_replay_team","_doubleheader_replay_build","_pool_replay_ledger"]:
		if season.get(name)!=null: return {}
		current[name]=null
	await tree.process_frame
	if not Ordinary._lease_live(lease,valid_binding): return {}
	current.ownership=_bank(season.ownership,budget,"ordinary")
	current.build=_build(season.build,budget,"ordinary")
	current.career=_object(season.career,Ordinary.Career,Ordinary.CAREER_FIELDS,[],budget)
	await tree.process_frame
	if not Ordinary._lease_live(lease,valid_binding): return {}
	current.physical=_object(season.physical,Ordinary.Physical,Ordinary.PHYSICAL_FIELDS,[],budget)
	current.opponents=_object(season.opponents,Ordinary.Opponents,["draft_order","draft_rule","_format"],["clubs"],budget)
	if not budget[1] or season.opponents.clubs.is_typed(): return {}
	var clubs: Dictionary={}
	for key: Variant in season.opponents.clubs:
		await tree.process_frame
		if not Ordinary._lease_live(lease,valid_binding): return {}
		var club:=_club(season,key,budget,"ordinary")
		if club.is_empty(): return {}
		clubs[key]=club
	if season.opponents.clubs.is_read_only(): clubs.make_read_only()
	current.opponents["clubs"]=clubs
	result.session=current
	return result if budget[1] else {}

static func _club(season: SeasonState,key: Variant,budget: Array,side: String) -> Dictionary:
	if not key is String or not season.opponents.clubs[key] is Dictionary: return {}
	var original: Dictionary=season.opponents.clubs[key]
	if original.is_typed() or not original.has("build"): return {}
	var club: Dictionary={}
	for field: Variant in original:
		if not field is String: return {}
		if field!="build": club[field]=_copy(original[field],0,budget)
	club.build=_build(original.build,budget,side)
	if original.is_read_only(): club.make_read_only()
	return club

static func _build(build: Variant, budget: Array, side: String = "dynasty") -> Dictionary:
	if not Adapter.side(side): budget[1]=false; return {}
	var builds: Script=Builds if side=="dynasty" else Ordinary.Builds
	var result: Dictionary = _object(build, builds.GameBuild, builds.BUILD.keys()+["last_error"], builds.OBJECTS, budget)
	if not budget[1]: return {}
	result._bank = _bank(build._bank, budget,side)
	result._book = _object(build._book, builds.Book, builds.BOOK.keys(), [], budget)
	result._gear_progress = _object(build._gear_progress, SeasonGearProgress, builds.GEAR.keys(), [], budget)
	result._sponsor_progress = _object(build._sponsor_progress, SeasonSponsorProgress, builds.SPONSOR.keys(), [], budget)
	result._major = _object(build._major, SeasonDoubleMajor, builds.MAJOR.keys(), [], budget)
	result._abilities = _object(build._abilities, SeasonAbilities, builds.ABILITIES.keys(), [], budget)
	result._copy = _object(build._copy, SeasonCarbonCopy, builds.COPY.keys(), [], budget)
	return result

static func _bank(bank: Variant, budget: Array, side: String = "dynasty") -> Dictionary:
	if not Adapter.side(side): budget[1]=false; return {}
	var builds: Script=Builds if side=="dynasty" else Ordinary.Builds
	return _object(bank, SeasonOwnership, builds.BANK.keys()+["_catalog"], [], budget)

static func _object(value: Variant, factory: Script, fields: Array, children: Array, budget: Array) -> Dictionary:
	if not budget[1]: return {}
	if not value is Object or not is_instance_valid(value) or value.get_script() != factory:
		budget[1] = false
		return {}
	if not Codec.known_variables(value, fields+children):
		budget[1] = false
		return {}
	var result: Dictionary = {}
	for field: String in fields:
		result[field] = _copy(value.get(field), 0, budget)
	return result

static func _copy(value: Variant, depth: int, budget: Array) -> Variant:
	budget[0] -= 1
	if not budget[1] or budget[0] < 0 or depth > MAX_DEPTH:
		budget[1] = false
		return null
	match typeof(value):
		TYPE_NIL, TYPE_BOOL, TYPE_INT, TYPE_STRING, TYPE_STRING_NAME:
			return value
		TYPE_FLOAT:
			if is_finite(value): return value
		TYPE_ARRAY:
			if value.get_typed_builtin() == TYPE_OBJECT: budget[1] = false; return null
			var result: Array = value.duplicate(false)
			for i in range(value.size()):
				var item: Variant = _copy(value[i], depth+1, budget)
				if not budget[1]: return null
				result[i] = item
			if value.is_read_only(): result.make_read_only()
			return result
		TYPE_DICTIONARY:
			if value.get_typed_key_builtin() == TYPE_OBJECT or value.get_typed_value_builtin() == TYPE_OBJECT:
				budget[1] = false; return null
			var result: Dictionary = value.duplicate(false)
			for key: Variant in value:
				if typeof(key) not in [TYPE_INT, TYPE_STRING, TYPE_STRING_NAME]: budget[1] = false; return null
				var item: Variant = _copy(value[key], depth+1, budget)
				if not budget[1]: return null
				result[key] = item
			if value.is_read_only(): result.make_read_only()
			return result
	budget[1] = false
	return null

## Compare the live graph against a private capture without allocating another
## graph. Depth/nodes are counted at the same virtual Dictionary positions as
## capture(owner) followed by same(), including synthetic object wrappers.
static func matches(owner: Variant, snapshot: Dictionary) -> bool:
	var budget: Array = [MAX_NODES]
	if not _match_object(owner,Owner,Owner.DATA_FIELDS,["session"],snapshot,0,budget): return false
	return Adapter.side(owner.active) and _match_session(owner.session,snapshot.session,1,budget,owner.active)

static func _match_object(value: Variant, factory: Script, fields: Array, children: Array, snapshot: Variant, depth: int, budget: Array, string_children: Array = []) -> bool:
	budget[0] -= 1
	if budget[0]<0 or depth>MAX_DEPTH: return false
	if not value is Object or not is_instance_valid(value) or value.get_script()!=factory: return false
	if not Codec.known_variables(value,fields+children): return false
	# Capture fields use String keys; dot-assigned child keys are StringName.
	# Preserve this exact native distinction, including explicit [] child writes.
	if not snapshot is Dictionary or snapshot.is_typed() or snapshot.is_read_only() or snapshot.size()!=fields.size()+children.size(): return false
	for key: Variant in snapshot:
		if typeof(key) not in [TYPE_STRING,TYPE_STRING_NAME]: return false
		var name: String=String(key)
		if name in fields:
			if typeof(key)!=TYPE_STRING: return false
		elif name in children:
			if typeof(key)!=(TYPE_STRING if name in string_children else TYPE_STRING_NAME): return false
		else: return false
	for field: String in fields:
		if not _same(value.get(field),snapshot[field],depth+1,budget): return false
	return true

static func _match_session(season: Variant, snapshot: Variant, depth: int, budget: Array, side: String) -> bool:
	if not Adapter.side(side): return false
	var schema: Script=Adapter.codec(side)
	# Replay pointers are synthetic null fields of the captured wrapper.
	var replay: Array = ["_heat_replay_team","_doubleheader_replay_build","_pool_replay_ledger"]
	if not _match_object(season,schema.Runtime,schema.FIELDS,schema.OBJECTS,snapshot,depth,budget,replay): return false
	for name: String in replay:
		if season.get(name)!=null or not _same(null,snapshot[name],depth+1,budget): return false
	if not _match_bank(season.ownership,snapshot.ownership,depth+1,budget,side): return false
	if not _match_build(season.build,snapshot.build,depth+1,budget,side): return false
	if not _match_object(season.career,schema.Career,schema.CAREER_FIELDS,[],snapshot.career,depth+1,budget): return false
	if not _match_object(season.physical,schema.Physical,schema.PHYSICAL_FIELDS,[],snapshot.physical,depth+1,budget): return false
	if not _match_object(season.opponents,schema.Opponents,["draft_order","draft_rule","_format"],["clubs"],snapshot.opponents,depth+1,budget,["clubs"]): return false
	var clubs: Variant=season.opponents.clubs
	var saved: Variant=snapshot.opponents.clubs
	budget[0]-=1
	if budget[0]<0 or depth+2>MAX_DEPTH or not clubs is Dictionary or not saved is Dictionary: return false
	if clubs.is_typed() or saved.is_typed() or clubs.is_read_only()!=saved.is_read_only() or clubs.size()!=saved.size(): return false
	for key: Variant in saved:
		if typeof(key)!=TYPE_STRING: return false
	for key: Variant in clubs:
		if typeof(key)!=TYPE_STRING or not saved.has(key): return false
		var club: Variant=clubs[key]
		var old: Variant=saved[key]
		budget[0]-=1
		if budget[0]<0 or depth+3>MAX_DEPTH or not club is Dictionary or not old is Dictionary: return false
		if club.is_typed() or old.is_typed() or club.is_read_only()!=old.is_read_only() or club.size()!=old.size() or not club.has("build"): return false
		for field: Variant in old:
			if typeof(field)!=(TYPE_STRING_NAME if field=="build" else TYPE_STRING): return false
		for field: Variant in club:
			if typeof(field)!=TYPE_STRING or not old.has(field): return false
			if field!="build" and not _same(club[field],old[field],depth+4,budget): return false
		if not _match_build(club.build,old.build,depth+4,budget,side): return false
	return true

static func _match_build(build: Variant, snapshot: Variant, depth: int, budget: Array, side: String) -> bool:
	var builds: Script=Builds if side=="dynasty" else Ordinary.Builds
	if not _match_object(build,builds.GameBuild,builds.BUILD.keys()+["last_error"],builds.OBJECTS,snapshot,depth,budget): return false
	return _match_bank(build._bank,snapshot._bank,depth+1,budget,side) \
		and _match_object(build._book,builds.Book,builds.BOOK.keys(),[],snapshot._book,depth+1,budget) \
		and _match_object(build._gear_progress,SeasonGearProgress,builds.GEAR.keys(),[],snapshot._gear_progress,depth+1,budget) \
		and _match_object(build._sponsor_progress,SeasonSponsorProgress,builds.SPONSOR.keys(),[],snapshot._sponsor_progress,depth+1,budget) \
		and _match_object(build._major,SeasonDoubleMajor,builds.MAJOR.keys(),[],snapshot._major,depth+1,budget) \
		and _match_object(build._abilities,SeasonAbilities,builds.ABILITIES.keys(),[],snapshot._abilities,depth+1,budget) \
		and _match_object(build._copy,SeasonCarbonCopy,builds.COPY.keys(),[],snapshot._copy,depth+1,budget)

static func _match_bank(bank: Variant, snapshot: Variant, depth: int, budget: Array, side: String) -> bool:
	var builds: Script=Builds if side=="dynasty" else Ordinary.Builds
	return _match_object(bank,SeasonOwnership,builds.BANK.keys()+["_catalog"],[],snapshot,depth,budget)

static func same(a: Variant, b: Variant) -> bool:
	return _same(a,b,0,[MAX_NODES])

static func _same(a: Variant, b: Variant, depth: int, budget: Array) -> bool:
	budget[0] -= 1
	if budget[0] < 0 or depth > MAX_DEPTH or typeof(a) != typeof(b): return false
	match typeof(a):
		TYPE_NIL, TYPE_BOOL, TYPE_INT, TYPE_STRING, TYPE_STRING_NAME:
			return a == b
		TYPE_FLOAT:
			return is_finite(a) and is_finite(b) and a == b
		TYPE_ARRAY:
			if a.get_typed_builtin()==TYPE_OBJECT: return false
			if a.is_read_only()!=b.is_read_only() or a.size()!=b.size() or a.get_typed_builtin()!=b.get_typed_builtin() or a.get_typed_class_name()!=b.get_typed_class_name() or a.get_typed_script()!=b.get_typed_script(): return false
			for i in range(a.size()):
				if not _same(a[i],b[i],depth+1,budget): return false
			return true
		TYPE_DICTIONARY:
			if a.get_typed_key_builtin()==TYPE_OBJECT or a.get_typed_value_builtin()==TYPE_OBJECT: return false
			if a.is_read_only()!=b.is_read_only() or a.size()!=b.size() or a.get_typed_key_builtin()!=b.get_typed_key_builtin() or a.get_typed_value_builtin()!=b.get_typed_value_builtin() or a.get_typed_key_class_name()!=b.get_typed_key_class_name() or a.get_typed_value_class_name()!=b.get_typed_value_class_name() or a.get_typed_key_script()!=b.get_typed_key_script() or a.get_typed_value_script()!=b.get_typed_value_script(): return false
			# Captures preserve insertion order. Check actual key types positionally
			# for the common unchanged case; reordered equal maps still use the
			# exact typed-key fallback below. Neither path copies nested values.
			var ordered_keys: Array = b.keys()
			var ordered: bool = true
			var index: int = 0
			for key: Variant in a:
				if typeof(key) not in [TYPE_INT,TYPE_STRING,TYPE_STRING_NAME]: return false
				if typeof(key)!=typeof(ordered_keys[index]) or key!=ordered_keys[index]:
					ordered=false; break
				index+=1
			if ordered:
				for key: Variant in a:
					if not _same(a[key],b[key],depth+1,budget): return false
				return true
			# Godot dictionary lookup can conflate StringName and String keys.
			# Compare actual key types, never merely has(key) or JSON key strings.
			var keys_b: Dictionary = {}
			for key: Variant in b:
				if typeof(key) not in [TYPE_INT,TYPE_STRING,TYPE_STRING_NAME]: return false
				keys_b[[typeof(key),str(key)]] = key
			for key: Variant in a:
				if typeof(key) not in [TYPE_INT,TYPE_STRING,TYPE_STRING_NAME]: return false
				var tag: Array = [typeof(key),str(key)]
				if not keys_b.has(tag) or not _same(a[key],b[keys_b[tag]],depth+1,budget): return false
			return true
	return false
