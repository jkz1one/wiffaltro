extends ClubCareer
## One current-session projection, never an unbounded archive or unlock proof.
var start_basis: Dictionary = {}
func active_tier() -> int:
	return 0 if current == 1 and runs.size() == 1 else -1
static func row(season: SeasonState) -> Dictionary:
	return {"id":1,"seed":season.season_seed,"league":"dynasty","tier":0,
		"status":"completed" if season.phase == SeasonState.Phase.COMPLETE else "active",
		"games":season.player_results.size(),"round":season.round_index,"champion":season.champion}
func sync(season: SeasonState) -> bool:
	if season == null or season.career != self or current != 1 or runs.size() != 1: return false
	runs[0] = row(season)
	return true
func fork() -> ClubCareer:
	var result: ClubCareer = get_script().new()
	result.current = current
	result.runs.assign(runs.duplicate(true))
	result._assets = _assets.duplicate(true)
	result._pennants = _pennants.duplicate(true)
	result._plaques = _plaques.duplicate(true)
	result._pitch_catalog = _pitch_catalog
	result.start_basis = start_basis.duplicate(true)
	return result
