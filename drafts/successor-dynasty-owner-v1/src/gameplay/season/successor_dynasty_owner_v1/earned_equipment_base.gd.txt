extends RefCounted
## Native Equipment Base draft derived from an exact closed Dugout Base source.
## This projection is not Store authority. It does not read or write a save.
const Archive = preload("starter_base_archive.gd")
const DugoutTitle = preload("earned_dugout_d1.gd")
const Life = preload("lifecycle.gd")
const Unlocks = preload("res://src/gameplay/season/league_unlocks.gd")
const KIND: String = "earned-equipment-base-offer-v1"
const DRAFT_KIND: String = "installed-earned-equipment-base-draft-v1"
const REGULAR_LIMIT: int = 10

# Use the existing native championship proof and archive bytes. The D1
# destination is deliberately not carried into this independent Base route.
# The caller still has to bind this owner to its exact installed revision.
static func stage(owner: RefCounted) -> Dictionary:
	var source: Dictionary = DugoutTitle.first_installed_source(owner)
	if source.is_empty() or not ("equipment" in Unlocks.from_titles(owner.common.aggregate.titles)):
		return {}
	var proof: Dictionary = DugoutTitle._offer_and_page(source.image, source.head)
	if proof.is_empty(): return {}
	var origin: Dictionary = proof.offer
	var api: Script = Archive._owner()
	var runtime: Script = api.Adapter.OrdinaryRuntime
	var basis: Dictionary = runtime.CommonStart.make(int(origin.seed), origin.lineage, int(origin.next_ordinal),
		int(origin.common_sequence), origin.common_sha256, api.Common.projection(owner))
	if basis.is_empty(): return {}
	var draft: SeasonState = runtime.create_common(basis, "equipment")
	var rules: Dictionary = LeagueRules.capture("equipment")
	if draft == null or not draft.league_live or not LeagueRules.equipment(draft.league_rules) \
		or not ClubCareer.same(draft.league_rules, rules) or not ClubCareer.same(draft.career.start_basis, basis) \
		or draft.phase != SeasonState.Phase.DRAFT or not draft.picks.is_empty() or draft.build.cash() != 0:
		return {}
	var current: Dictionary = api.Adapter.capture("ordinary", draft)
	var head: Dictionary = Archive._head(proof.head)
	if current.is_empty() or head.is_empty() or head.pages != [origin.archived_page_sha256] \
		or Archive._hash(proof.head) != origin.target_head_sha256:
		return {}
	var offer: Dictionary = {
		"kind": KIND, "source_close": origin.source_close, "source_session_sha256": origin.source_session_sha256,
		"previous_head_sha256": origin.source_head_sha256, "head_sha256": origin.target_head_sha256,
		"page_sha256": origin.archived_page_sha256, "source_league": "dugout", "source_tier": 0,
		"league": "equipment", "tier": 0, "basis_sha256": Life.digest(basis),
	}
	return {"offer": offer, "head": head, "page": proof.page, "current": current, "aggregate": owner.common.aggregate.duplicate(true)}

# The archive remains content addressed. The current draft is a separate
# native payload, not a second copy of the title or a writable Common owner.
static func draft_value(staged: Dictionary) -> Dictionary:
	var value: Dictionary = {"kind": DRAFT_KIND, "version": 1, "head": staged.head,
		"offer": staged.offer, "current": staged.current, "aggregate": staged.aggregate}
	if staged.has("opening_paid"): value.opening_paid = staged.opening_paid
	if staged.has("opening_shop"): value.opening_shop = staged.opening_shop
	if staged.has("regular"): value.regular = staged.regular
	if staged.has("playoffs"): value.playoffs = staged.playoffs
	return value

# Cold reconstruction is intentionally separate from file or Store authority.
# A successful result authorizes nothing until the owning process rechecks
# the same unchanged revision and page. Forward is the only supported mode.
static func replay_draft(value: Variant, page: PackedByteArray) -> Dictionary:
	if not OS.has_feature("wiffaltro_catalog_forward_v1") or not value is Dictionary: return {}
	var paid: bool = value.has("opening_paid")
	var has_shop: bool = value.has("opening_shop")
	var fields: Array = ["kind", "version", "head", "offer", "current", "aggregate"]
	if paid: fields.append("opening_paid")
	if has_shop:
		if not paid or not value.opening_shop is Array or value.opening_shop.is_empty() or value.opening_shop.size() > 128: return {}
		fields.append("opening_shop")
	var has_regular: bool = value.has("regular")
	if has_regular:
		if not paid or not value.regular is Array or value.regular.is_empty() or value.regular.size() >= REGULAR_LIMIT: return {}
		fields.append("regular")
	var has_playoffs: bool = value.has("playoffs")
	if has_playoffs:
		if not has_regular or value.regular.size() != REGULAR_LIMIT-1 \
			or not value.playoffs is Array or value.playoffs.is_empty() or value.playoffs.size()>2: return {}
		fields.append("playoffs")
	if not Archive.C.keys(value, fields) \
		or value.kind != DRAFT_KIND or not Archive.C.whole(value.version, 1, 1) \
		or not value.head is Dictionary or not value.offer is Dictionary or not value.current is Dictionary \
		or page.is_empty() or page.size() > Archive.MAX_PAGE:
		return {}
	var offer: Dictionary = value.offer
	if not Archive.C.keys(offer, ["kind", "source_close", "source_session_sha256", "previous_head_sha256",
		"head_sha256", "page_sha256", "source_league", "source_tier", "league", "tier", "basis_sha256"]) \
		or offer.kind != KIND or offer.source_league != "dugout" or offer.league != "equipment" \
		or not Archive.C.whole(offer.source_tier, 0, 0) or not Archive.C.whole(offer.tier, 0, 0):
		return {}
	var head_bytes: PackedByteArray = Archive._wire(value.head)
	var head: Dictionary = Archive._head(head_bytes)
	if head.is_empty() or head.ordinal != 2 or head.pages != [offer.page_sha256] \
		or offer.page_sha256 != Archive._hash(page) or offer.head_sha256 != Archive._hash(head_bytes):
		return {}
	var original: Dictionary = Archive._page(page, head, 1)
	if original.is_empty() or original.close.get("league") != "dugout" \
		or original.close.get("tier") != 0 or original.close.get("reason") != "complete" \
		or not original.close.get("receipt") is Dictionary \
		or original.close.receipt.get("finish") != "champion":
		return {}
	var native: RefCounted = Archive.Core.root(int(head.seed), head.league)
	if native == null or not Archive.Core._apply_closed(native, original.image, original.close, false, true):
		return {}
	var computed: Dictionary = stage(native)
	if computed.is_empty() or computed.page != page: return {}
	var draft: SeasonState = Archive._owner().Adapter.restore("ordinary", computed.current)
	var picks: Variant = value.current.get("state", {}).get("picks") if value.current.get("state") is Dictionary else null
	if draft == null or not picks is Array or picks.size() > 4: return {}
	for pick: Variant in picks:
		if not pick is String or not draft.choose_player(pick): return {}
	var physical: Variant = value.current.get("physical")
	if not physical is Dictionary or not physical.get("pending") is Dictionary: return {}
	if paid and not value.opening_paid is Dictionary: return {}
	var witness: Dictionary = value.opening_paid if paid else physical.pending
	if paid or not witness.is_empty():
		draft.physical.pending = witness.duplicate(true)
		var projection := opening_job(draft)
		if projection.is_empty(): return {}
		if paid:
			if projection.has("fixture"): return {}
			draft = projection.candidate
			computed.opening_paid = witness.duplicate(true)
			if has_shop:
				for command: Variant in value.opening_shop:
					if not command is Dictionary: return {}
					draft = shop_step(draft, witness, command)
					if draft == null: return {}
				computed.opening_shop = value.opening_shop.duplicate(true)
			if has_regular:
				for row: Variant in value.regular:
					if not Archive.C.keys(row, ["pending", "shop"]) or not row.pending is Dictionary \
						or not row.shop is Array or row.shop.size() > 128: return {}
					draft.physical.pending = row.pending.duplicate(true)
					var checked := regular_job(draft, true)
					if checked.is_empty() or checked.has("fixture"): return {}
					draft = checked.candidate
					for command: Variant in row.shop:
						if not command is Dictionary: return {}
						draft = shop_step(draft, row.pending, command)
						if draft == null: return {}
				computed.regular = value.regular.duplicate(true)
			if has_playoffs:
				for playoff: Variant in value.playoffs:
					if not playoff is Dictionary or play_slot(draft) not in ["semifinal", "final"]: return {}
					draft.physical.pending = playoff.duplicate(true)
					var checked := pending_plan(draft, true)
					if checked.is_empty() or checked.projection.has("fixture"): return {}
					draft = checked.projection.candidate
				computed.playoffs = value.playoffs.duplicate(true)
			if not physical.pending.is_empty():
				draft.physical.pending = physical.pending.duplicate(true)
				if pending_plan(draft, true).is_empty(): return {}
	computed.current = Archive._owner().Adapter.capture("ordinary", draft)
	if computed.current.is_empty() or not Archive.C.same(draft_value(computed), value): return {}
	return {"draft": draft, "aggregate": native.common.aggregate.duplicate(true)}

# Authored fixture zero must be played; only its two native peer reports
# can advance this held round to the one exact paid continuation.
static func opening_job(draft: SeasonState) -> Dictionary:
	if draft == null or not LeagueRules.equipment(draft.league_rules) or draft.round_index != 0 \
		or draft.phase != SeasonState.Phase.REGULAR or draft.picks.size() != 4 \
		or not Archive.C.keys(draft.physical.pending, ["human", "reports"]) \
		or not draft.physical.pending.human is Array or draft.physical.pending.human.size() != 13 \
		or not draft.physical.pending.reports is Array or draft.physical.pending.reports.size() > 2:
		return {}
	var reports: Array = draft.physical.pending.reports
	for i: int in reports.size():
		if not reports[i] is Dictionary or reports[i].get("fixture") != i + 1: return {}
	var native: Dictionary = Archive._owner().Adapter.OrdinaryPending.project(draft)
	if native.has("error") or not native.get("candidate") is SeasonState: return {}
	if reports.size() < 2:
		if not native.get("fixture") is Dictionary or native.fixture.get("id") != reports.size() + 1: return {}
	elif native.has("fixture") or not _paid_opening(native.candidate, draft.physical.pending):
		return {}
	return native

static func _paid_opening(draft: SeasonState, witness: Dictionary) -> bool:
	return draft.round_index == 1 and draft.phase == SeasonState.Phase.REGULAR and draft.picks.size() == 4 \
		and draft.results.size() == 3 and draft.player_results.size() == 1 and draft.physical.pending.is_empty() \
		and Archive.C.keys(witness, ["human", "reports"]) and witness.reports is Array and witness.reports.size() == 2 \
		and Archive.C.same(draft.physical.reports, witness.reports)

static func regular_job(season: SeasonState, cold: bool=false) -> Dictionary:
	return _regular_plan(season,cold).get("projection", {})

static func _regular_plan(season: SeasonState, cold: bool=false) -> Dictionary:
	var original:=_regular_native_plan(season)
	if not original.is_empty() or not cold or not _materialization_scope(season) \
			or season.round_index>=REGULAR_LIMIT or season.phase!=SeasonState.Phase.REGULAR \
			or not season.physical.pending.get("reports") is Array or season.physical.pending.reports.is_empty():
		return original
	# A live installed operation signs its native CPU request after loading
	# canonical save JSON. A continuous cold replay can still hold a different
	# native number representation. Recreate that one exact persisted boundary,
	# then repeat every ordinary request, evidence and payout check unchanged.
	var rebound:=_round_materialized_exact(season)
	return {} if rebound==null else _regular_native_plan(rebound)

static func _materialization_scope(season: SeasonState) -> bool:
	return season!=null and season.career!=null and OS.has_feature("wiffaltro_catalog_forward_v1") \
		and season.round_index>=1 and season.round_index<=REGULAR_LIMIT \
		and LeagueRules.equipment(season.league_rules) and season.league_rules.get("tier")==0 \
		and season.career.start_basis.get("ordinal")==2 and season.career.start_basis.get("common_sequence")==1

static func _round_materialized_exact(season: SeasonState) -> SeasonState:
	var api:=Archive._owner()
	var base: SeasonState=season.fork_live()
	if base==null: return null
	var pending: Dictionary=season.physical.pending.duplicate(true)
	base.physical.pending.clear()
	var image: Dictionary=api.Adapter.capture("ordinary",base)
	if image.is_empty(): return null
	var parsed: Variant=api._bytes_value(JSON.stringify(image,"",true,true).to_utf8_buffer())
	if not Archive.C.same(image,parsed): return null
	var rebound: SeasonState=api.Adapter.restore("ordinary",parsed)
	if rebound==null or not Archive.C.same(api.Adapter.capture("ordinary",rebound),image): return null
	rebound.physical.pending=pending
	return rebound

static func _regular_native_plan(season: SeasonState) -> Dictionary:
	if season == null: return {}
	var index: int = season.round_index
	if index == 0:
		var opening:=opening_job(season)
		return {} if opening.is_empty() else {"projection":opening,"session":season}
	if index >= REGULAR_LIMIT or not _regular_boundary(season, index) or not LeagueRules.equipment(season.league_rules) \
		or not Archive.C.keys(season.physical.pending, ["human", "reports"]): return {}
	var witness: Dictionary = season.physical.pending
	if not witness.human is Array or witness.human.size() != 13 or witness.human[0] != index * 3 \
		or not witness.reports is Array or witness.reports.size() > DugoutTitle.regular_report_cap(index): return {}
	var count: int = witness.reports.size()
	for i: int in count:
		if not witness.reports[i] is Dictionary or witness.reports[i].get("fixture") != index * 3 + i + 1: return {}
	var original: Dictionary = Archive.Core.MixedProof._project_mixed_round(season)
	var native: Dictionary = original.get("projection", {})
	if native.is_empty() or native.has("error") or not native.get("candidate") is SeasonState \
		or not original.get("source") is SeasonState \
		or not DugoutTitle.regular_projection_shape(index, count, native.has("fixture")): return {}
	if native.has("fixture"):
		if not native.fixture is Dictionary or native.fixture.get("id") != index * 3 + count + 1: return {}
	else:
		var next: SeasonState = native.candidate
		if not _regular_boundary(next, index+1) or not next.physical.pending.is_empty() \
			or not Archive.C.same(next.physical.reports, season.physical.reports + witness.reports) \
			or not Archive.C.same(next.results.slice(0, season.results.size()), season.results) \
			or not Archive.C.same(next.player_results.slice(0, index), season.player_results): return {}
	# The original native projector may rederive an exact typed source after
	# materialization. Later CPU admission must use that same source.
	return {"projection":native,"session":original.source}

# Each transaction consumes only the fixture chosen by the real season.
# It is never inferred from the caller's score or from a title flag.
static func play_slot(season: SeasonState) -> String:
	if season == null or season.picks.size()!=4 or season.build.pack_pending(): return ""
	if season.round_index < REGULAR_LIMIT and season.phase==SeasonState.Phase.REGULAR: return "regular"
	if season.round_index!=REGULAR_LIMIT: return ""
	if not DugoutTitle.semifinal_pair(season).is_empty(): return "semifinal"
	return "final" if DugoutTitle.settled_semifinal(season,true)=="final_ready" else ""

static func pending_plan(season: SeasonState, cold: bool=false) -> Dictionary:
	if season == null or season.physical.pending.is_empty() or not LeagueRules.equipment(season.league_rules): return {}
	var slot := play_slot(season)
	if slot in ["semifinal","final"]:
		var plan:=_postseason_plan(season,slot)
		if not plan.is_empty() or not cold or not _materialization_scope(season): return plan
		var rebound:=_round_materialized_exact(season)
		return {} if rebound==null or play_slot(rebound)!=slot else _postseason_plan(rebound,slot)
	if slot!="regular": return {}
	var regular:=_regular_plan(season,cold)
	if regular.is_empty(): return {}
	regular.merge({"managed_id":3*season.round_index,"total":DugoutTitle.regular_report_cap(season.round_index)})
	return regular

static func _postseason_plan(season: SeasonState, slot: String) -> Dictionary:
	if slot=="semifinal":
		var checked:=DugoutTitle.project_semifinal(season)
		return {} if checked.is_empty() else {"projection":checked.projection,"session":checked.source,
			"managed_id":checked.fixture.id,"semifinal":true,"total":checked.total}
	if slot=="final":
		var checked:=DugoutTitle.project_managed_final(season)
		return {} if checked.is_empty() else {"projection":checked,"session":season,"managed_id":32,"managed_final":true,"total":0}
	return {}

static func finish(season: SeasonState) -> String:
	if season==null or season.round_index!=REGULAR_LIMIT: return ""
	var initial := DugoutTitle.final_regular_destination(season)
	if not initial.is_empty(): return initial
	var semifinal := DugoutTitle.settled_semifinal(season,true)
	return DugoutTitle.managed_final_destination(season) if semifinal.is_empty() else semifinal

static func _regular_boundary(season: SeasonState, completed: int) -> bool:
	if season == null or completed < 1 or completed > REGULAR_LIMIT or season.round_index != completed \
		or season.picks.size() != 4 or season.player_results.size() != completed: return false
	if completed == REGULAR_LIMIT: return not DugoutTitle.final_regular_destination(season).is_empty()
	return season.phase == SeasonState.Phase.REGULAR and season.results.size() == 3*completed \
		and season.physical.reports.size() == 2*completed

static func shop_window(season: SeasonState, witness: Dictionary) -> bool:
	if season == null or not LeagueRules.equipment(season.league_rules) or season.league_rules.get("tier") != 0 \
		or season.phase != SeasonState.Phase.REGULAR or not season.physical.pending.is_empty() \
		or not season.shop_available() or season.build._visit.number != season.round_index: return false
	return _paid_opening(season, witness) if season.round_index == 1 else _regular_boundary(season, season.round_index)

# One original native shop command, during the same earned first visit.
# Cold uses this exact reducer to replay the authored cash and gear state.
static func shop_step(previous: SeasonState, witness: Dictionary, command: Dictionary) -> SeasonState:
	if not shop_window(previous, witness) or not Archive.C.primitive(command) \
		or not SeasonBuild.SHOP_OPS.has(str(command.get("op", ""))): return null
	var op: String = command.op
	var id: Variant = command.get("id")
	if not id is String or id.is_empty() or previous.build._requests.has(id) \
		or not Archive.C.whole(command.get("rev"), previous.build.revision(), previous.build.revision()) \
		or previous.build._visit.open == (op == "open"): return null
	if op == "open" and (not Archive.C.keys(command, ["id", "rev", "op"]) or id != "shop-open:%d"%previous.build.revision()):
		return null
	var candidate: SeasonState = previous.fork_live()
	if candidate == null: return null
	var next: SeasonBuild = candidate.build.candidate(command.duplicate(true))
	if next == null: return null
	candidate.adopt_build(next)
	var before: Array = previous.build._events
	if candidate.build != next or not shop_window(candidate, witness) or not next._visit.open \
		or next.revision() != before.size()+1 or next._events.size() != before.size()+1 \
		or not Archive.C.same(next._events.slice(0, -1), before) or not Archive.C.same(next._events.back(), command):
		return null
	return candidate

# Only materialize bytes already bound by the producer or native child.
# This is not, by itself, a proof of earned progression.
static func local_draft(value: Dictionary, allow_held: bool = false) -> SeasonState:
	if not value.get("current") is Dictionary or not value.get("aggregate") is Dictionary \
		or not value.get("offer") is Dictionary: return null
	var draft: SeasonState = Archive._owner().Adapter.restore("ordinary", value.current)
	if draft == null or draft.picks.size() > 4 or not LeagueRules.equipment(draft.league_rules) \
		or draft.league_rules.get("tier") != 0: return null
	if draft.round_index < REGULAR_LIMIT and draft.phase != (SeasonState.Phase.DRAFT if draft.picks.size() < 4 else SeasonState.Phase.REGULAR):
		return null
	if value.has("opening_paid"):
		if not allow_held or not value.opening_paid is Dictionary: return null
		var regular: Variant = value.get("regular", [])
		if not regular is Array or regular.size()+1 != draft.round_index: return null
		var playoffs: Variant=value.get("playoffs",[])
		if not playoffs is Array or (value.has("playoffs") and playoffs.is_empty()) or playoffs.size()>2: return null
		if playoffs.is_empty():
			if draft.round_index==REGULAR_LIMIT and draft.phase==SeasonState.Phase.SEMIFINAL:
				if DugoutTitle.semifinal_pair(draft).is_empty(): return null
			elif not _regular_boundary(draft,draft.round_index): return null
		elif draft.round_index!=REGULAR_LIMIT \
			or (playoffs.size()==1 and finish(draft) not in ["final_ready","semifinal_exit"]) \
			or (playoffs.size()==2 and finish(draft) not in ["final_champion","final_runner_up"]):
			return null
		if regular.is_empty() and draft.physical.pending.is_empty() and not _paid_opening(draft, value.opening_paid): return null
	elif draft.round_index != 0 or not draft.results.is_empty() or draft.build.cash() != 0:
		return null
	var basis: Dictionary = draft.career.start_basis
	if basis.get("common_sha256") != Life.digest(value.aggregate) \
		or Life.digest(basis) != value.offer.get("basis_sha256") \
		or not Archive.C.whole(value.aggregate.get("currency"), 0, 2147483647): return null
	if not draft.physical.pending.is_empty() and (not allow_held or pending_plan(draft).is_empty()):
		return null
	return draft
