# Derive the first archive root from the actual native installed life.
# This is a pure head candidate, not Store authority or an earned title.
static func first_installed_source(owner: RefCounted) -> Dictionary:
	var api:=Archive._owner()
	if owner==null or owner.get_script()!=api or owner.active!="ordinary" or owner.session==null \
			or not owner.paused.is_empty() or not api._missing_dynasty(owner) \
			or owner.current_ordinal()!=1 or owner.ordinary_state.has("archive") \
			or not owner.ordinary_state.get("last_transition") is Dictionary \
			or not owner.ordinary_state.last_transition.is_empty() \
			or not owner.ordinary_state.get("proposal") is Dictionary \
			or not owner.ordinary_state.proposal.is_empty(): return {}
	var current: SeasonState=owner.session
	if not current.league_live or current.career==null \
			or not ClubCareer.same(current.league_rules,LeagueRules.capture("dugout")): return {}
	var native: RefCounted=Archive.Core.root(current.season_seed,"dugout")
	if native==null or native.session==null or native.current_ordinal()!=1 \
			or owner.ordinary_state.lineage!=native.ordinary_state.lineage \
			or owner.ordinary_state.start_basis_sha256!=native.ordinary_state.start_basis_sha256 \
			or not ClubCareer.same(current.career.start_basis,native.session.career.start_basis): return {}
	var image: Dictionary=api.capture(owner)
	var head: Dictionary=Archive._head(Archive.make_head(current.season_seed,"dugout"))
	return {} if image.is_empty() or head.is_empty() else {"image":image,"head":head}

static func from_closed_image(image: Dictionary, head: Dictionary) -> Dictionary:
	if image.get("active")!="ordinary" or not image.get("ordinary_state") is Dictionary \
		or not image.get("ordinary") is Dictionary or not image.get("common") is Dictionary: return {}
	var state: Dictionary=image.ordinary_state
	var session: Dictionary=image.ordinary
	var common: Dictionary=image.common
	if state.get("status")!="closed" or state.get("proposal")!={} \
		or not state.get("last_close") is Dictionary or not state.get("lineage") is String \
		or not session.get("state") is Dictionary or not common.get("aggregate") is Dictionary \
		or not common.get("last_fold") is Dictionary or not common.get("cursors") is Dictionary: return {}
	var close: Dictionary=state.last_close
	var fold: Dictionary=common.last_fold
	var aggregate: Dictionary=common.aggregate
	var ordinal: Variant=state.get("ordinal")
	var sequence: Variant=common.get("sequence")
	var title: String=Titles.title_key("dugout",0)
	# Exact preceding championship, not a previously earned badge, a runner-up
	# receipt, a different league's title, or an asserted target difficulty.
	if not SeasonOwnership._whole(ordinal,1,ClubCareer.MAX_RUNS-1) \
		or not SeasonOwnership._whole(sequence,int(ordinal),ClubCareer.MAX_RUNS*2) \
		or common.cursors.get("ordinary")!=ordinal or aggregate.get("closed")!=sequence \
		or not state.lineage.begins_with("ordinary:") or not Archive._digest(state.lineage.trim_prefix("ordinary:")) \
		or not aggregate.get("titles") is Dictionary or aggregate.titles.get(title)!=true: return {}
	if close.get("id")!="%s:%d:close"%[state.lineage,int(ordinal)] or close.get("ordinal")!=ordinal \
		or close.get("reason")!="complete" or close.get("league")!="dugout" or close.get("tier")!=0 \
		or not close.get("first_title") is bool or not Archive._digest(close.get("session_sha256")) \
		or close.session_sha256!=Life.digest(session): return {}
	var aggregate_hash: String=Life.digest(aggregate)
	if fold.get("id")!=close.id or fold.get("source")!=close.session_sha256 or fold.get("side")!="ordinary" \
		or fold.get("ordinal")!=ordinal or fold.get("sequence")!=sequence \
		or fold.get("aggregate_sha256")!=aggregate_hash: return {}
	var stock: Dictionary=session.state
	if stock.get("phase")!=SeasonState.Phase.COMPLETE or stock.get("champion")!=0 \
		or stock.get("league_live")!=true \
		or not ClubCareer.same(stock.get("league_rules"),LeagueRules.capture("dugout")): return {}
	var scores: Dictionary=_score_proof(stock)
	var receipt: Dictionary=ClubSeasonRecord.receipt(scores,close.first_title,0)
	if receipt.get("finish")!="champion" or not ClubCareer.same(receipt,close.get("receipt")) \
		or not ClubCareer.same(receipt,fold.get("receipt")): return {}
	var head_bytes: PackedByteArray=Archive._wire(head)
	var checked_head: Dictionary=Archive._head(head_bytes)
	if checked_head.is_empty() or checked_head.ordinal!=ordinal: return {}
	var page: PackedByteArray=Archive.make_page(head_bytes,session,close)
	var successor_head: PackedByteArray=Archive.append_head(head_bytes,page)
	var rules: Dictionary=LeagueRules.capture_tier("dugout",1)
	if page.is_empty() or successor_head.is_empty() or rules.is_empty(): return {}
	var next_ordinal: int=int(ordinal)+1
	var next_seed: int=("ordinary:%s:%d:nextseed"%[state.lineage,next_ordinal]).hash() & 0x7fffffff
	return {"kind":KIND,"version":1,"source_close":close.id,"source_session_sha256":close.session_sha256,
		"source_head_sha256":Archive._hash(head_bytes),"archived_page_sha256":Archive._hash(page),
		"target_head_sha256":Archive._hash(successor_head),"lineage":state.lineage,"next_ordinal":next_ordinal,
		"seed":next_seed,"common_sequence":int(sequence),"common_sha256":aggregate_hash,"title":title,"rules":rules}

static func _score_proof(state: Dictionary) -> Dictionary:
	if not state.get("results") is Array or state.results.size()!=33 \
		or not state.get("teams") is Array or state.teams.size()!=6: return {}
	var draws: Array = []
	for team: Variant in state.teams:
		if not team is Dictionary or not team.has("draw"): return {}
		draws.append(team.draw)
	var scores: Array = []
	for row: Variant in state.results:
		if not row is Dictionary or not SeasonOwnership._whole(row.get("id"),0,32): return {}
		scores.append([row.id,row.get("away"),row.get("home"),row.get("away_runs"),row.get("home_runs")])
	scores.sort_custom(func(a: Array,b: Array) -> bool: return a[0]<b[0])
	return {"draws":draws,"scores":scores,"version":ClubSeasonRecord.LEAGUE_VERSION,"league_rules":state.league_rules}

# A pure exact write candidate, never Store authority. A caller still has
# to prove source custody, atomically publish the page/head and reopen it.
static func stage_birth(image: Dictionary, head: Dictionary, expected_offer: Dictionary) -> Dictionary:
	var offer: Dictionary=from_closed_image(image,head)
	if offer.is_empty() or not ClubCareer.same(offer,expected_offer): return {}
	var owner_class: Script=Archive._owner()
	var view: RefCounted=owner_class.new()
	for key: String in owner_class.META_FIELDS:
		if image.has(key):
			if view.get(key) is Dictionary and image[key] is Dictionary:
				view.set(key,image[key].duplicate(true))
			elif view.get(key) is int and SeasonOwnership._whole(image[key],0,2147483647):
				view.set(key,int(image[key]))
			elif view.get(key) is String and image[key] is String:
				view.set(key,image[key])
			else: return {}
	view.active="ordinary"
	if not owner_class.Common.valid(view) or not owner_class._missing_dynasty(view) \
		or image.get("dynasty")!={} or not view.migration_v3.is_empty() \
		or not view.workshop_draft.is_empty() or not view.workshop_close_history.is_empty() \
		or not view.workshop_chair_purchase.is_empty() or not view.workshop_applied.is_empty() \
		or not image.ordinary.get("career") is Dictionary \
		or not image.ordinary.career.get("start_basis") is Dictionary \
		or not ClubCareer.same(owner_class._pack_fields(view,image.ordinary),image): return {}
	var runtime: Script=owner_class.Adapter.OrdinaryRuntime
	var start: Script=runtime.CommonStart
	# Only the existing Common rights projection is allowed to seed the draft.
	view.session=SeasonState.new()
	view.session.career=runtime.LiveCareer.new()
	view.session.career.start_basis=image.ordinary.career.start_basis.duplicate(true)
	var seed_basis: Dictionary=start.make(int(offer.seed),offer.lineage,int(offer.next_ordinal),
		int(offer.common_sequence),offer.common_sha256,owner_class.Common.projection(view))
	var previous_head: PackedByteArray=Archive._wire(head)
	var page: PackedByteArray=Archive.make_page(previous_head,image.ordinary,image.ordinary_state.last_close)
	var successor: PackedByteArray=Archive.append_head(previous_head,page)
	var new_head: Dictionary=Archive._head(successor)
	if new_head.is_empty() or Archive._hash(successor)!=offer.target_head_sha256 \
		or Archive._hash(page)!=offer.archived_page_sha256: return {}
	var basis: Dictionary=start.bind_earned_dugout_d1(seed_basis,offer,new_head.pages)
	if basis.is_empty(): return {}
	var draft: SeasonState=runtime.create_earned_dugout_d1_draft(basis)
	if draft==null or draft.phase!=SeasonState.Phase.DRAFT or not draft.picks.is_empty() \
		or draft.build.cash()!=0 or not draft.build.d1_pricing_valid(): return {}
	var ordinal: int=int(view.ordinary_state.ordinal)
	view.ordinary_state.last_transition={"id":"%s:%d:next"%[offer.lineage,ordinal],
		"source":offer.source_session_sha256,"from":ordinal,"to":offer.next_ordinal,"league":"dugout","basis":basis}
	view.ordinary_state.ordinal=offer.next_ordinal
	view.ordinary_state.start_basis_sha256=Life.digest(basis)
	view.ordinary_state.status="active"
	view.ordinary_state.last_close={}
	view.session=draft
	var packed: Dictionary=owner_class.Adapter.Ordinary.capture(draft)
	if packed.is_empty() or not owner_class.Common.valid(view) or not owner_class.capture(view).is_empty(): return {}
	var candidate: Dictionary=owner_class._pack_fields(view,packed)
	return {"head":new_head,"page":page,"current":candidate}


# Cold provenance from ordered immutable pages. The rotating backup is not an
# input. Reconstruct every completed native boundary and do not grant authority
# from structural page shape, a stored offer, or a supplied digest alone.
static func _draft_birth_from_pages(head: PackedByteArray, pages: Array[PackedByteArray], installed_forward_first: bool=false) -> Dictionary:
	var checked: Dictionary=Archive._head(head)
	if checked.is_empty() or pages.is_empty() or pages.size()!=checked.pages.size(): return {}
	if installed_forward_first and (checked.ordinal!=2 or pages.size()!=1 \
			or not OS.has_feature("wiffaltro_catalog_forward_v1")): return {}
	# Authenticate shape/digests before the expensive native replay.
	for i: int in pages.size():
		if Archive._hash(pages[i])!=checked.pages[i] or Archive._page(pages[i],checked,i+1).is_empty(): return {}
	var final_index: int=pages.size()-1
	var last: Dictionary=Archive._page(pages[final_index],checked,final_index+1)
	if last.close.get("reason")!="complete" or last.close.get("league")!="dugout" or last.close.get("tier")!=0: return {}
	var old: Dictionary=checked.duplicate(true)
	old.pages.pop_back()
	old.ordinal=int(old.ordinal)-1
	var authenticated: RefCounted=Archive.Core.root(int(checked.seed),checked.league)
	if authenticated==null: return {}
	for i: int in final_index:
		if not Archive._advance_closed_page(authenticated,pages[i],old,i): return {}
	if authenticated.current_ordinal()!=final_index+1 \
			or not Archive.Core._apply_closed(authenticated,last.image,last.close,false,installed_forward_first): return {}
	var api:=Archive._owner()
	var real: Dictionary=api.capture(authenticated)
	if real.is_empty(): return {}
	var offer:=from_closed_image(real,old)
	var staged:=stage_birth(real,old,offer)
	if staged.is_empty() or staged.page!=pages[final_index] or Archive._wire(staged.head)!=head: return {}
	return staged


const LAST_INSTALLED_REGULAR_ROUND: int=10

# The authoritative bracket chooses the Club and CPU sibling. The Club
# need not occupy fixture 30; its sibling may have the lower identity.
static func semifinal_pair(session: SeasonState) -> Dictionary:
	if session==null or session.phase!=SeasonState.Phase.SEMIFINAL \
			or session.round_index!=LAST_INSTALLED_REGULAR_ROUND or session.picks.size()!=4 \
			or session.results.size()!=30 or session.player_results.size()!=10 or session.physical.reports.size()!=20 \
			or session.playoff_seeds.size()!=4 or session.semifinals.size()!=2 or session.champion!=-1 \
			or not session.final_fixture.is_empty(): return {}
	if not session.playoff_seeds.any(func(seed: Variant) -> bool: return SeasonOwnership._whole(seed,0,0)): return {}
	var club: Dictionary={}
	var cpu: Dictionary={}
	for fixture: Dictionary in session.semifinals:
		if not SeasonOwnership._whole(fixture.get("id"),30,31) or fixture.get("round")!=10: return {}
		if fixture.get("away")==0 or fixture.get("home")==0:
			if not club.is_empty(): return {}
			club=fixture
		else:
			if not cpu.is_empty(): return {}
			cpu=fixture
	if club.is_empty() or cpu.is_empty() or club.id==cpu.id \
			or not ClubCareer.same(club,session.pending_fixture()): return {}
	return {"club":club,"cpu":cpu}

# The managed score determines how many other jobs remain, but neither it
# nor the target phase can waive the native proof or its fixture binding.
static func semifinal_jobs(session: SeasonState) -> int:
	var pair:=semifinal_pair(session)
	if pair.is_empty(): return 0
	var pending: Dictionary=session.physical.pending
	if not SeasonOwnership._keys(pending,["human","reports"]) or not pending.get("human") is Array \
			or pending.human.size()!=13 or pending.human[0]!=pair.club.id or not pending.get("reports") is Array:
		return 0
	var h: Array=pending.human
	if not SeasonOwnership._whole(h[1],0,100000) or not SeasonOwnership._whole(h[2],0,100000) or h[1]==h[2]: return 0
	var count: int=1 if (h[1]>h[2] if pair.club.away==0 else h[2]>h[1]) else 2
	if pending.reports.size()>count: return 0
	for i: int in pending.reports.size():
		var report: Variant=pending.reports[i]
		if not report is Dictionary or report.get("fixture")!=(pair.cpu.id if i==0 else 32): return 0
	return count

static func semifinal_pending(session: SeasonState) -> bool:
	return semifinal_jobs(session)>0

# Never derives progression rights. It recognizes native semifinal settlement
# only, with no active pending transaction or already played managed final.
static func settled_semifinal(session: SeasonState, allow_final_pending: bool=false) -> String:
	if session==null or session.round_index!=10 or session.picks.size()!=4 \
			or session.player_results.size()!=11 \
			or (not session.physical.pending.is_empty() and (not allow_final_pending or session.phase!=SeasonState.Phase.FINAL)) \
			or session.semifinals.size()!=2 or session.playoff_seeds.size()!=4 \
			or not session.playoff_seeds.any(func(seed: Variant) -> bool: return SeasonOwnership._whole(seed,0,0)):
		return ""
	var managed: Dictionary=session.player_results.back()
	if not SeasonOwnership._whole(managed.get("id"),30,31) or managed.away_runs==managed.home_runs \
			or (managed.away!=0 and managed.home!=0): return ""
	var pair_matches: bool=session.semifinals.any(func(f: Dictionary) -> bool:
		return f.get("id")==managed.id and f.get("away")==managed.away and f.get("home")==managed.home)
	if not pair_matches or session.physical.reports.size()<21: return ""
	var sibling: int=31 if managed.id==30 else 30
	if session.physical.reports[20].get("fixture")!=sibling: return ""
	var club_won: bool=managed.away_runs>managed.home_runs if managed.away==0 else managed.home_runs>managed.away_runs
	if session.phase==SeasonState.Phase.FINAL and club_won and session.results.size()==32 \
			and session.physical.reports.size()==21 and session.champion==-1:
		var final: Dictionary=session.pending_fixture()
		if final.get("id")==32 and final.get("round")==11 and final.get("neutral")==true \
				and (final.get("away")==0 or final.get("home")==0):
			return "final_ready"
	if session.phase==SeasonState.Phase.COMPLETE and not club_won and session.results.size()==33 \
			and session.physical.reports.size()==22 and session.physical.reports[21].get("fixture")==32 \
			and session.champion in [1,2,3,4,5] and session.pending_fixture().is_empty():
		return "semifinal_exit"
	return ""

# Same checked native projector for a live pending request and a cold replay.
static func project_semifinal(session: SeasonState, cold_materialized_retry: bool=false) -> Dictionary:
	var total:=semifinal_jobs(session)
	if total==0: return {}
	var pair:=semifinal_pair(session)
	var pending: Dictionary=session.physical.pending
	var count: int=pending.reports.size()
	var source: SeasonState=session
	var projected: Dictionary=Archive._owner().Adapter.OrdinaryPending.project(source)
	if cold_materialized_retry and count>0 \
			and projected.get("error")=="The pending round contains an unexpected physical fixture.":
		source=_materialized_semifinal_fixture_types(session)
		if source==null: return {}
		projected=Archive._owner().Adapter.OrdinaryPending.project(source)
	if projected.has("error") or not projected.get("candidate") is SeasonState: return {}
	if count==total:
		if projected.has("fixture") or settled_semifinal(projected.candidate)!=("final_ready" if total==1 else "semifinal_exit"): return {}
	else:
		if not projected.get("fixture") is Dictionary: return {}
		var fixture: Dictionary=projected.fixture
		if count==0 and not ClubCareer.same(fixture,pair.cpu): return {}
		if count==1 and (total!=2 or fixture.get("id")!=32 or fixture.get("round")!=11 \
				or fixture.get("away")==0 or fixture.get("home")==0): return {}
	return {"projection":projected,"fixture":pair.club,"count":count,"total":total,"source":source}

# The installed cold replay creates integer bracket fields, while the live
# parent may have reloaded those four fields through JSON before signing the
# CPU request. Try only that exact materialized representation of the
# locally rederived bracket. The original digest and evidence still have to
# pass the same native projector; neither the save nor its reports change.
# The reverse is confined to the installed D3 parent after its child has
# proven the complete ordered replay with the authentic integer bracket.
static func _materialized_semifinal_fixture_types(session: SeasonState,to_float: bool=true) -> SeasonState:
	if not OS.has_feature("wiffaltro_catalog_forward_v1") or not semifinal_pending(session): return null
	var api:=Archive._owner()
	var d1: bool=api.Codec.same(session.league_rules,LeagueRules.capture_tier("dugout",1))
	var starts: Script=api.Adapter.OrdinaryRuntime.CommonStart
	var basis: Dictionary=session.career.start_basis if session.career!=null else {}
	var d2: bool=starts.detached_earned_dugout_d2(basis) \
		and api.Codec.same(session.league_rules,LeagueRules.capture_tier("dugout",2))
	var d3: bool=starts.detached_earned_dugout_d3(basis) \
		and api.Codec.same(session.league_rules,LeagueRules.capture_tier("dugout",3))
	if (not to_float and not d3) or (not d1 and not d2 and not d3): return null
	var shape: Array[String]=["id","round","away","home"]
	for row: Dictionary in session.semifinals:
		if not SeasonOwnership._keys(row,shape): return null
		for key: String in shape:
			if (to_float and not row[key] is int) or (not to_float and not row[key] is float): return null
	var alternative: SeasonState=session.fork_live()
	if alternative==null: return null
	for row: Dictionary in alternative.semifinals:
		for key: String in shape:
			row[key]=float(row[key]) if to_float else int(row[key])
	return alternative if ClubCareer.same(alternative.semifinals,session.semifinals) else null


# A managed Final is one authentic human game. There is no CPU job and
# no entitlement fold at this boundary.
static func managed_final_pending(session: SeasonState) -> bool:
	if settled_semifinal(session,true)!="final_ready": return false
	var pending: Dictionary=session.physical.pending
	return SeasonOwnership._keys(pending,["human","reports"]) and pending.get("human") is Array \
		and pending.human.size()==13 and pending.human[0]==32 and pending.get("reports") is Array and pending.reports.is_empty()

static func managed_final_destination(session: SeasonState) -> String:
	if session==null or session.phase!=SeasonState.Phase.COMPLETE or session.round_index!=10 \
			or session.picks.size()!=4 or not session.physical.pending.is_empty() or session.results.size()!=33 \
			or session.player_results.size()!=12 or session.physical.reports.size()!=21 or not session.pending_fixture().is_empty():
		return ""
	var final: Dictionary=session.final_fixture
	var row: Dictionary=session.player_results.back()
	if final.get("id")!=32 or final.get("round")!=11 or final.get("neutral")!=true \
			or row.get("id")!=32 or row.get("away")!=final.get("away") or row.get("home")!=final.get("home") \
			or (row.get("away")!=0 and row.get("home")!=0) or row.away_runs==row.home_runs:
		return ""
	var winner: int=int(row.away if row.away_runs>row.home_runs else row.home)
	if session.champion!=winner: return ""
	return "final_champion" if winner==0 else "final_runner_up"

static func project_managed_final(session: SeasonState) -> Dictionary:
	if not managed_final_pending(session): return {}
	var result: Dictionary=Archive._owner().Adapter.OrdinaryPending.project(session)
	if result.has("error") or result.has("fixture") or not result.get("candidate") is SeasonState \
			or managed_final_destination(result.candidate).is_empty(): return {}
	return result


static func final_regular_destination(session: SeasonState) -> String:
	if session==null or session.round_index!=LAST_INSTALLED_REGULAR_ROUND \
			or session.picks.size()!=4 or session.player_results.size()!=10 \
			or not session.physical.pending.is_empty() or session.playoff_seeds.size()!=4: return ""
	if not semifinal_pair(session).is_empty(): return "semifinal"
	# Godot's Array.has distinguishes 0 from the JSON-decoded 0.0.
	var club_qualified: bool=session.playoff_seeds.any(
		func(seed: Variant) -> bool: return SeasonOwnership._whole(seed,0,0))
	if session.phase==SeasonState.Phase.COMPLETE and not club_qualified \
			and session.results.size()==33 and session.physical.reports.size()==23 \
			and session.champion in [1,2,3,4,5] and session.pending_fixture().is_empty():
		return "missed"
	return ""


static func regular_report_cap(index: int) -> int:
	return 5 if index==LAST_INSTALLED_REGULAR_ROUND-1 else 2

static func regular_projection_shape(index: int,count: int, needs_cpu: bool) -> bool:
	var cap:=regular_report_cap(index)
	return count>=0 and count<=cap and (count<cap if needs_cpu else (count==2 or count==cap))

# Numeric types are part of the native CPU request hash, while the JSON
# save preserves their values but not their native integer/float distinction.
# A bounded read witness carries ONLY types from the authenticated replay.
const WALLET_TYPES_MAX: int=1536

static func needs_wallet_types(session: SeasonState) -> bool:
	if session==null: return false
	return (session.phase==SeasonState.Phase.REGULAR and session.round_index==LAST_INSTALLED_REGULAR_ROUND-1) \
		or not semifinal_pair(session).is_empty() or settled_semifinal(session,true)=="final_ready"

static func valid_wallet_types(witness: String) -> bool:
	var parts: PackedStringArray=witness.split(".")
	if parts.size()!=6: return false
	for part: String in parts:
		if part.is_empty() or part.length()>WALLET_TYPES_MAX: return false
		for digit: String in part:
			if digit not in ["i","f"]: return false
	return true

static func _wallets(session: SeasonState) -> Array[SeasonOwnership]:
	var banks: Array[SeasonOwnership]=[]
	if not needs_wallet_types(session) or session.build==null or session.opponents==null: return banks
	banks.append(session.build._bank)
	for club: int in range(1,6):
		var item: Variant=session.opponents.clubs.get(str(club))
		if not item is Dictionary or not item.get("build") is SeasonBuild: return []
		banks.append(item.build._bank)
	return banks

static func _type_tree(value: Variant, job: Dictionary, depth: int=0) -> Variant:
	if not job.ok: return value
	if depth>12:
		job.ok=false
	elif value is Dictionary:
		var keys: Array=value.keys()
		for key: Variant in keys:
			if not key is String:
				job.ok=false
				return value
		keys.sort()
		for key: String in keys:
			value[key]=_type_tree(value[key],job,depth+1)
	elif value is Array:
		for index: int in value.size():
			value[index]=_type_tree(value[index],job,depth+1)
	elif (value is int or value is float) and SeasonOwnership._whole(value,-9007199254740991,9007199254740991):
		var cursor: int=job.cursor
		if cursor>=WALLET_TYPES_MAX:
			job.ok=false
		elif not job.apply:
			job.tape+=("i" if value is int else "f")
		elif cursor>=job.tape.length():
			job.ok=false
		else:
			if job.tape[cursor]=="i": value=int(value)
			else: value=float(value)
		job.cursor+=1
	return value

# The CPU request includes the journal and effective held purchase receipts.
# Preserve types from both parts; the scalar values already exist locally.
static func _wallet_number_scope(bank: SeasonOwnership) -> Dictionary:
	return {"events":bank._events.duplicate(true),"state":bank._state.duplicate(true)}

static func wallet_types(session: SeasonState) -> String:
	var banks:=_wallets(session)
	if banks.size()!=6: return ""
	var out: PackedStringArray=[]
	for bank: SeasonOwnership in banks:
		if bank==null: return ""
		var job: Dictionary={"cursor":0,"tape":"","ok":true,"apply":false}
		_type_tree(_wallet_number_scope(bank),job)
		if not job.ok: return ""
		out.append(job.tape)
	var witness: String=".".join(out)
	return witness if valid_wallet_types(witness) else ""

# Change no journal values or saved bytes. Compare the complete native
# projection before and after this detached, exact source-bound type repair.
# The original physical verifier still decides whether every CPU report fits.
static func restore_wallet_types(session: SeasonState, witness: String) -> bool:
	if not valid_wallet_types(witness): return false
	var banks:=_wallets(session)
	if banks.size()!=6: return false
	var parts: PackedStringArray=witness.split(".")
	var changed: Array[Dictionary]=[]
	var codec: Script=Archive._owner().Codec
	for index: int in banks.size():
		var bank: SeasonOwnership=banks[index]
		if bank==null: return false
		var job: Dictionary={"cursor":0,"tape":parts[index],"ok":true,"apply":true}
		var typed: Dictionary=_type_tree(_wallet_number_scope(bank),job)
		if not job.ok or job.cursor!=parts[index].length() \
				or not codec.same(typed.events,bank._events) or not codec.same(typed.state,bank._state): return false
		changed.append(typed)
	for index: int in banks.size():
		banks[index]._events.assign(changed[index].events)
		banks[index]._state=changed[index].state
	return wallet_types(session)==witness


# Rebuild the separately authorized installed first-title history in order.
# A later Club game uses the exact authored build after earlier rewards and
# Shop actions. Never promote a current pending snapshot into a paid result.
static func _restore_regular_prefix(draft: SeasonState, expected: Dictionary, value: Dictionary) -> SeasonState:
	var api:=Archive._owner()
	var image: Dictionary=value.ordinary
	if draft.phase!=SeasonState.Phase.REGULAR or draft.picks.size()!=4 \
			or draft.round_index!=0 or draft.pending_fixture().get("id")!=0 \
			or not image.get("physical") is Dictionary or not image.physical.get("pending") is Dictionary \
			or not image.physical.get("reports") is Array or not image.state.get("player_results") is Array \
			or not image.state.get("results") is Array or not image.get("build") is Dictionary \
			or not image.build.get("build") is Dictionary: return null
	var pending: Dictionary=image.physical.pending
	var reports: Array=image.physical.reports
	var humans: Array=image.state.player_results
	var events: Variant=image.build.build.get("_events")
	var managed_count: int=humans.size()
	var visited: int=mini(managed_count,LAST_INSTALLED_REGULAR_ROUND)
	var paid_semifinal: bool=managed_count>=LAST_INSTALLED_REGULAR_ROUND+1
	var paid_final: bool=managed_count==LAST_INSTALLED_REGULAR_ROUND+2
	if managed_count>LAST_INSTALLED_REGULAR_ROUND+2 or (visited==0 and pending.is_empty()) \
			or image.state.get("round_index")!=visited \
			or not events is Array or events.size()>SeasonBuild.MAX_EVENTS: return null
	var final_regular: bool=visited==LAST_INSTALLED_REGULAR_ROUND
	if paid_final:
		if not pending.is_empty() or image.state.results.size()!=33 or reports.size()!=21 \
				or image.state.get("phase")!=SeasonState.Phase.COMPLETE: return null
	elif paid_semifinal:
		if not [[32,21],[33,22]].has([image.state.results.size(),reports.size()]): return null
		if not pending.is_empty() and image.state.get("phase")!=SeasonState.Phase.FINAL: return null
	elif final_regular:
		if not [[30,20],[33,23]].has([image.state.results.size(),reports.size()]): return null
		if not pending.is_empty() and image.state.get("phase")!=SeasonState.Phase.SEMIFINAL: return null
	elif image.state.results.size()!=visited*3 or reports.size()!=visited*2: return null
	var proven:=draft
	for index: int in visited:
		var fixture:=proven.pending_fixture()
		if fixture.get("id")!=index*3: return null
		var human:=Archive.Core._human(humans[index],fixture)
		if human.is_empty(): return null
		var end_report: int=reports.size() if index==LAST_INSTALLED_REGULAR_ROUND-1 and not paid_semifinal else 2*index+2
		proven.physical.pending={"human":human,"reports":reports.slice(2*index,end_report)}
		var checked: Dictionary=Archive.Core.MixedProof._project_mixed_round(proven)
		var result: Dictionary=checked.projection
		if result.has("error") or result.has("fixture") or not result.get("candidate") is SeasonState: return null
		proven=result.candidate
		# Only authored Shop operations after the exact native reward prefix
		# can be replayed before constructing the following managed fixture.
		if proven.round_index!=index+1 or not Archive.Core._shop(proven,events,0): return null
		if index==LAST_INSTALLED_REGULAR_ROUND-1 and final_regular_destination(proven).is_empty(): return null
	if paid_semifinal:
		var pair:=semifinal_pair(proven)
		if pair.is_empty(): return null
		var human:=Archive.Core._human(humans[10],pair.club)
		if human.is_empty(): return null
		proven.physical.pending={"human":human,"reports":reports.slice(20)}
		var checked:=project_semifinal(proven,true)
		if checked.is_empty() or checked.projection.has("fixture"): return null
		proven=checked.projection.candidate
		if not Archive.Core._shop(proven,events,0): return null
		if paid_final:
			var final_human:=Archive.Core._human(humans[11],proven.pending_fixture())
			if final_human.is_empty(): return null
			proven.physical.pending={"human":final_human,"reports":[]}
			var last:=project_managed_final(proven)
			if last.is_empty(): return null
			proven=last.candidate
			if not Archive.Core._shop(proven,events,0): return null
		elif not pending.is_empty():
			proven.physical.pending=pending.duplicate(true)
			if project_managed_final(proven).is_empty(): return null
	elif not pending.is_empty() and final_regular:
		proven.physical.pending=pending.duplicate(true)
		var checked:=project_semifinal(proven,true)
		if checked.is_empty(): return null
		proven=checked.source
	elif not pending.is_empty():
		if visited>=LAST_INSTALLED_REGULAR_ROUND or not api.Codec.keys(pending,["human","reports"]) \
				or not pending.reports is Array or pending.reports.size()>regular_report_cap(visited) \
				or proven.pending_fixture().get("id")!=visited*3: return null
		proven.physical.pending=pending.duplicate(true)
		var checked: Dictionary=Archive.Core.MixedProof._project_mixed_round(proven)
		var result: Dictionary=checked.projection
		if result.has("error") or not result.get("candidate") is SeasonState \
				or not regular_projection_shape(visited,pending.reports.size(),result.has("fixture")) \
				or not checked.get("source") is SeasonState: return null
		proven=checked.source
	expected.ordinary=api.Adapter.Ordinary.capture(proven)
	return proven if not expected.ordinary.is_empty() and api.Codec.same(expected,value) else null

# Earlier archive readers remain draft-only. Exact equality includes all
# inherited Club Credits and title rights, with no fallback to a predecessor.
static func _restore_after_picks(staged: Dictionary, value: Dictionary, installed_forward_first: bool=false) -> SeasonState:
	var api:=Archive._owner()
	if not staged.get("current") is Dictionary or not value.get("ordinary") is Dictionary: return null
	var expected: Dictionary=staged.current.duplicate(true)
	if not expected.get("ordinary") is Dictionary or not value.ordinary.get("state") is Dictionary: return null
	var picks: Variant=value.ordinary.state.get("picks")
	if not picks is Array or picks.size()>4: return null
	var draft: SeasonState=api.Adapter.Ordinary.restore_common(expected.ordinary)
	if draft==null or draft.phase!=SeasonState.Phase.DRAFT or not draft.picks.is_empty(): return null
	for id: Variant in picks:
		if not id is String or not draft.choose_player(id): return null
	expected.ordinary=api.Adapter.Ordinary.capture(draft)
	if expected.ordinary.is_empty(): return null
	if api.Codec.same(expected,value): return draft
	return _restore_regular_prefix(draft,expected,value) if installed_forward_first else null


static func restore_draft_from_pages(head: PackedByteArray, pages: Array[PackedByteArray], value: Dictionary, installed_forward_first: bool=false) -> SeasonState:
	var staged:=_draft_birth_from_pages(head,pages,installed_forward_first)
	return null if staged.is_empty() else _restore_after_picks(staged,value,installed_forward_first)


# The active parent must already be bound to its archive and authenticated
# games. This pure projection is also recomputed after full cold replay.
static func close_destination(session: SeasonState) -> String:
	if session==null or session.phase!=SeasonState.Phase.COMPLETE or session.build==null \
			or session.physical==null or not session.physical.pending.is_empty() \
			or not session.physical.needed.is_empty() or not session.physical.error.is_empty() \
			or session.physical.projecting or session.build.pack_pending(): return ""
	if final_regular_destination(session)=="missed": return "missed"
	if settled_semifinal(session)=="semifinal_exit": return "semifinal"
	var finish:=managed_final_destination(session)
	if finish=="final_runner_up": return "runner_up"
	return "champion" if finish=="final_champion" else ""

static func completed_current(active: Dictionary, session: SeasonState) -> Dictionary:
	return completed_current_for_tier(active,session,1)

# Reuse the exact completion reducer only for these installed ordered careers.
static func completed_current_for_tier(active: Dictionary, session: SeasonState, tier: int) -> Dictionary:
	if tier not in [1,2,3]: return {}
	var ordinal: int=tier+1
	var finish:=close_destination(session)
	if finish.is_empty() or session.career==null: return {}
	var api:=Archive._owner()
	var c: Script=api.Codec
	if active.get("active")!="ordinary" or not active.get("ordinary_state") is Dictionary \
			or not active.get("common") is Dictionary or not active.get("ordinary") is Dictionary: return {}
	var state: Dictionary=active.ordinary_state
	var common: Dictionary=active.common
	if state.get("status")!="active" or not c.whole(state.get("ordinal"),ordinal,ordinal) or state.get("proposal")!={} \
			or state.get("last_close")!={} or not state.get("lineage") is String \
			or not c.whole(common.get("sequence"),tier,tier) or not common.get("aggregate") is Dictionary \
			or not c.same(common.get("cursors"),{"ordinary":tier,"dynasty":0}): return {}
	if not c.same(api.Adapter.Ordinary.capture(session),active.ordinary) \
			or not api.Common._starter_dugout_paid_terminal(session): return {}
	var basis: Dictionary=session.career.start_basis
	if basis.get("lineage")!=state.lineage or basis.get("ordinal")!=ordinal or basis.get("common_sequence")!=tier \
			or state.get("start_basis_sha256")!=Life.digest(basis) \
			or basis.get("common_sha256")!=Life.digest(common.aggregate): return {}
	var id: String="%s:%d:close"%[state.lineage,ordinal]
	var folded:=Progress.fold_installed_dugout_complete(common.aggregate,session,id,ordinal,tier)
	if folded.is_empty() or folded.receipt.get("finish")!=finish or folded.receipt.get("version")!=ordinal: return {}
	var image: Dictionary=active.duplicate(true)
	var source: String=Life.digest(active.ordinary)
	var first_title: bool=not common.aggregate.titles.get(Titles.title_key("dugout",tier),false)
	image.ordinary_state.last_close={"id":id,"ordinal":ordinal,"reason":"complete","session_sha256":source,
		"receipt":folded.receipt,"first_title":first_title,"league":"dugout","tier":tier}
	image.ordinary_state.status="closed"
	image.common.aggregate=folded.aggregate
	image.common.cursors.ordinary=ordinal
	image.common.sequence=ordinal
	image.common.last_fold={"id":id,"side":"ordinary","ordinal":ordinal,"sequence":ordinal,"source":source,
		"aggregate_sha256":Life.digest(folded.aggregate),"receipt":folded.receipt.duplicate(true)}
	return image

static func restore_completed_from_pages(head: PackedByteArray, pages: Array[PackedByteArray], value: Dictionary) -> SeasonState:
	var staged:=_draft_birth_from_pages(head,pages,true)
	if staged.is_empty() or not value.get("ordinary") is Dictionary: return null
	var active: Dictionary=staged.current.duplicate(true)
	active.ordinary=value.ordinary
	var proven:=_restore_after_picks(staged,active,true)
	if proven==null: return null
	var expected:=completed_current(active,proven)
	return proven if not expected.is_empty() and Archive._owner().Codec.same(expected,value) else null
