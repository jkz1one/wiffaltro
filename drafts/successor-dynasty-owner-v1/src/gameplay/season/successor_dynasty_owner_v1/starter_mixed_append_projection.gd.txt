extends RefCounted
## Only the first append has a qualified owner lineage. Transport shape never
## grants authority: a cold read reconstructs exact v5 Close and all its roots.
const Custody=preload("starter_mixed_append_custody.gd")
const Base=preload("starter_mixed_custody.gd")
const ORDINAL: int=5
const ROLLOVER_ORDINAL: int=6
const ROLLOVER_TRANSITION: String="common-starter-dugout-season-six-draft-v1"

static func _api() -> Script:
	return load("res://src/gameplay/season/successor_dynasty_owner_v1/owner.gd")

static func marked_state(state: Variant) -> bool:
	return state is Dictionary and state.get("archive") is Dictionary and state.archive.get("kind")==Custody.KIND

# Reject impossible Base receipts before replaying older history. This is
# a refusal only: matching these public numbers never admits a closed owner.
static func _base_close_receipt_shape(value: Variant, first_title: bool) -> bool:
	var api:=_api()
	if not api.Codec.keys(value,["version","finish","wins","finish_award","record_bonus","first_bonus","total"]) \
		or value.version!=1 or not value.finish is String or not api.Codec.whole(value.wins,0,10): return false
	var awards: Dictionary={"missed":35,"semifinal":65,"runner_up":95,"champion":180}
	if not awards.has(value.finish): return false
	var award:int=awards[value.finish]
	var record:int=floori(1.5*int(value.wins))
	var bonus:int=25 if value.finish=="champion" and first_title else 0
	return value.finish_award==award and value.record_bonus==record and value.first_bonus==bonus \
		and value.total==award+record+bonus

static func _close_meta_shape(state: Dictionary, close: Variant) -> bool:
	var api:=_api()
	return api.Codec.keys(close,["id","ordinal","reason","session_sha256","receipt","first_title","league","tier"]) \
		and close.id=="%s:%d:close"%[state.lineage,ORDINAL] and close.ordinal==ORDINAL \
		and close.reason=="complete" and close.league=="dugout" and close.tier==0 \
		and close.first_title is bool and _base_close_receipt_shape(close.receipt,close.first_title) \
		and Base._sha(close.session_sha256)

static func _archive_state(state: Dictionary, closed: bool=false) -> bool:
	var api:=_api()
	if not api.Codec.keys(state,api.Common.ORDINARY_FIELDS+["archive"]) or not marked_state(state) \
			or state.status!=("closed" if closed else "active") or state.ordinal!=ORDINAL or not state.lineage is String or state.proposal!={} \
			or not api.Codec.keys(state.archive,["kind","wire"]) \
			or not state.archive.wire is String or state.archive.wire.is_empty() or state.archive.wire.length()>Custody.MAX_WIRE: return false
	if not (_close_meta_shape(state,state.last_close) if closed else state.last_close=={}): return false
	var t: Variant=state.last_transition
	return api.Codec.keys(t,["kind","id","source","from","to","league","basis"]) \
		and t.kind==Custody.TRANSITION and t.league=="dugout" and t.from==ORDINAL-1 and t.to==ORDINAL \
		and t.id=="%s:%d:next"%[state.lineage,ORDINAL-1] and Base._sha(t.source) \
		and api.Common.Start.valid(t.basis) and api.Common.Start.detached_mixed_append(t.basis) \
		and t.basis.lineage==state.lineage and api.Common.Life.digest(t.basis)==state.start_basis_sha256

static func _rollover_archive_state(state: Dictionary) -> bool:
	var api:=_api()
	if not api.Codec.keys(state,api.Common.ORDINARY_FIELDS+["archive"]) or not marked_state(state) \
			or state.status!="active" or state.ordinal!=ROLLOVER_ORDINAL or not state.lineage is String \
			or state.proposal!={} or state.last_close!={} or not api.Codec.keys(state.archive,["kind","wire"]) \
			or not state.archive.wire is String or state.archive.wire.is_empty() \
			or state.archive.wire.length()>Custody.MAX_WIRE: return false
	var t: Variant=state.last_transition
	return api.Codec.keys(t,["kind","id","source","from","to","league","basis"]) \
		and t.kind==ROLLOVER_TRANSITION and t.league=="dugout" and t.from==ORDINAL and t.to==ROLLOVER_ORDINAL \
		and t.id=="%s:%d:next"%[state.lineage,ORDINAL] and Base._sha(t.source) \
		and api.Common.Start.valid(t.basis) and api.Common.Start.detached_mixed_rollover(t.basis) \
		and t.basis.lineage==state.lineage and api.Common.Life.digest(t.basis)==state.start_basis_sha256

# Detached structural scope only. Producing or publishing always requires the
# exact Store bound owner, and cold restore independently compares everything.
static func _native_owner_scope(owner: RefCounted, closed: bool=false, rollover: bool=false) -> bool:
	var api:=_api()
	if rollover and closed: return false
	var previous: int=ORDINAL if closed or rollover else ORDINAL-1
	if owner==null or owner.get_script()!=api or owner.active!="ordinary" or not owner.paused.is_empty() \
			or not api._missing_dynasty(owner) or not owner.migration_v3.is_empty() \
			or not owner.workshop_draft.is_empty() or not owner.workshop_close_history.is_empty() \
			or not owner.workshop_chair_purchase.is_empty() or not owner.workshop_applied.is_empty() \
			or not (_rollover_archive_state(owner.ordinary_state) if rollover else _archive_state(owner.ordinary_state,closed)) \
			or owner.common.get("sequence")!=previous \
			or not api.Codec.same(owner.common.get("cursors"),{"ordinary":previous,"dynasty":0}): return false
	var s: SeasonState=owner.session
	var b: Dictionary=owner.ordinary_state.last_transition.basis
	if s==null or s.get_script()!=api.Adapter.OrdinaryRuntime or s.career==null or s.build==null or s.physical==null \
			or not s.physical.error.is_empty() or not s.physical.needed.is_empty() or s.physical.projecting \
			or not s.home_property.is_empty() or not s.league_live or s.league_rules.get("league")!="dugout" \
			or s.league_rules.get("tier")!=0 or s.season_seed!=b.seed or not api.Codec.same(s.career.start_basis,b) \
			or s.career.current!=1 or s.career.runs.size()!=1 or s.career.runs[0].get("id")!=1: return false
	return true

static func draft_scope(owner: RefCounted) -> bool:
	if not _native_owner_scope(owner) and not _native_owner_scope(owner,false,true): return false
	var s: SeasonState=owner.session
	if s.round_index!=0 or not s.results.is_empty() or not s.player_results.is_empty() \
			or not s.physical.pending.is_empty() or not s.physical.reports.is_empty() or s.physical.cursor!=0 \
			or s.build.pack_pending(): return false
	return (s.phase==SeasonState.Phase.DRAFT and s.picks.size()<4 and s.cash()==0) \
		or (s.phase==SeasonState.Phase.REGULAR and s.picks.size()==4)

# Bound the disclosed journal and refuse unrelated Shop operations up
# front. Existing native chronological replay still must regenerate each
# game event and replay each approved command from its exact cash/visit.
static func _shop_journal(events: Variant) -> bool:
	if not events is Array or events.size()>SeasonBuild.MAX_EVENTS: return false
	var admitted: Array[String]=_api().Mixed.SHOP_OPS
	for command: Variant in events:
		if not command is Dictionary: return false
		var op: String=str(command.get("op",""))
		if SeasonBuild.SHOP_OPS.has(op) and not admitted.has(op): return false
	return true

# Native state shape is reusable at every authored phase, separate from
# Store's physical writer and from any future Shop/Close authorization.
static func season_scope(owner: RefCounted) -> bool:
	if not _native_owner_scope(owner): return false
	var s: SeasonState=owner.session
	return s.phase!=SeasonState.Phase.DRAFT and _api().Mixed._season_phase(s) \
		and _shop_journal(s.build._events)

# Season 6 owns all regular rounds and the two paid bracket landings. Keep
# this separate from season_scope (v6 Shop/Close) and round_scope (launch).
static func rollover_play_scope(owner: RefCounted) -> bool:
	if not _native_owner_scope(owner,false,true): return false
	var s: SeasonState=owner.session
	return not s.build.pack_pending() \
		and _rollover_phase_shape(s.phase,s.round_index,s.player_results.size(),s.physical.pending,s.build._events) \
		and _api().Mixed._season_phase(s)

static func _rollover_phase_shape(phase: Variant,index: Variant,managed_count: int,pending: Dictionary,events: Variant) -> bool:
	var api:=_api()
	var runtime: Script=api.Adapter.OrdinaryRuntime
	var regular_rounds: int=runtime.MIXED_ROLLOVER_REGULAR_ROUNDS
	if not events is Array or events.size()>SeasonBuild.MAX_EVENTS: return false
	for command: Variant in events:
		if not command is Dictionary or SeasonBuild.SHOP_OPS.has(str(command.get("op",""))): return false
	if phase==SeasonState.Phase.REGULAR:
		if not api.Codec.whole(index,0,regular_rounds-1): return false
		if pending.is_empty(): return true
		var reports: Variant=pending.get("reports")
		return reports is Array and reports.size()<=runtime.mixed_rollover_regular_report_cap(int(index))
	# Ten managed games distinguishes regular elimination from any paid
	# playoff loss or victory. No pending playoff job is admitted here.
	return api.Codec.whole(phase,SeasonState.Phase.SEMIFINAL,SeasonState.Phase.COMPLETE) \
		and int(phase) in [SeasonState.Phase.SEMIFINAL,SeasonState.Phase.COMPLETE] \
		and index==regular_rounds and managed_count==regular_rounds and pending.is_empty()

static func complete_scope(owner: RefCounted) -> bool:
	return season_scope(owner) and owner.session.phase==SeasonState.Phase.COMPLETE

# Closed display/admission grants no round, Shop or next-season authority.
static func closed_complete_scope(owner: RefCounted) -> bool:
	if not _native_owner_scope(owner,true) or owner.session.phase!=SeasonState.Phase.COMPLETE: return false
	var api:=_api()
	var s: SeasonState=owner.session
	if not api.Mixed._season_phase(s) or not _shop_journal(s.build._events): return false
	var c: Dictionary=owner.ordinary_state.last_close
	var row: Variant=owner.common.get("last_fold")
	return api.Codec.keys(row,["id","side","ordinal","sequence","source","aggregate_sha256","receipt"]) \
		and row.id==c.id and row.side=="ordinary" and row.ordinal==ORDINAL and row.sequence==ORDINAL \
		and row.source==c.session_sha256 and api.Common.Progress.valid(owner.common.aggregate) \
		and row.aggregate_sha256==api.Common.Life.digest(owner.common.aggregate) and api.Codec.same(row.receipt,c.receipt)

static func fold_complete(owner: RefCounted, id: String, sequence: int) -> Dictionary:
	if not complete_scope(owner): return {}
	var api:=_api()
	var before: Dictionary=owner.common.aggregate
	if not api.Common.Progress.valid(before) or not api.Common.Progress._close_id(id) \
		or id!="%s:%d:close"%[owner.ordinary_state.lineage,ORDINAL] \
		or sequence!=ORDINAL or sequence!=before.closed+1 or sequence!=owner.common.sequence+1: return {}
	return api.Common.Progress._fold_native(before,"ordinary",owner.session,id,sequence,"complete")

# Only the fixed Store producer and strict replay-backed reader call this.
# Reuse the exact authenticated current without accepting a caller payout.
static func completed_close(owner: RefCounted,current: Dictionary) -> RefCounted:
	if not complete_scope(owner) or current.is_empty(): return null
	var api:=_api()
	var candidate: RefCounted=api._new()
	candidate.session=owner.session
	if not api.Codec.install_fields(candidate,api.DATA_FIELDS,api.Codec.data_fields(owner,api.DATA_FIELDS)) \
		or not api.Common._fold_close_state(candidate,"complete",current) \
		or not closed_complete_scope(candidate): return null
	return candidate

static func _active_play_phase(owner: RefCounted) -> bool:
	return season_scope(owner) and owner.session.phase in \
		[SeasonState.Phase.REGULAR,SeasonState.Phase.SEMIFINAL,SeasonState.Phase.FINAL]

static func round_scope(owner: RefCounted) -> bool:
	return (_active_play_phase(owner) and not owner.session.build.pack_pending()) \
		or (rollover_play_scope(owner) and owner.session.phase==SeasonState.Phase.REGULAR)

static func shop_scope(owner: RefCounted) -> bool:
	return _active_play_phase(owner) and owner.session.physical.pending.is_empty() \
		and owner.session.shop_available()

static func game_entry_scope(owner: RefCounted) -> bool:
	return round_scope(owner) and owner.session.physical.pending.is_empty()

# Cheap typed bounds ahead of decompressing prior history. These reject
# inconsistent shape but never authenticate a result; chronological replay
# and complete recomputed envelope equality below are the authority.
static func _current_shape(current: Variant) -> bool:
	if not current is Dictionary or not current.get("state") is Dictionary \
			or not current.get("physical") is Dictionary or not current.get("career") is Dictionary: return false
	var s: Dictionary=current.state
	var p: Dictionary=current.physical
	var c: Dictionary=current.career
	var codec: Script=_api().Codec
	if not s.get("picks") is Array or s.picks.size()>4 or not s.get("results") is Array or s.results.size()>33 \
			or not s.get("player_results") is Array or s.player_results.size()>12 \
			or not p.get("reports") is Array or p.reports.size()>23 or p.get("cursor")!=p.reports.size() \
			or not p.get("pending") is Dictionary or p.get("needed")!={} or p.get("error")!="" or p.get("projecting")!=false \
			or c.get("current")!=1 or not c.get("runs") is Array or c.runs.size()!=1 \
			or not c.runs[0] is Dictionary or c.runs[0].get("id")!=1 \
			or not current.get("build") is Dictionary or not current.build.get("build") is Dictionary \
			or not _shop_journal(current.build.build.get("_events")): return false
	var phase: Variant=s.get("phase")
	var round_index: Variant=s.get("round_index")
	if s.picks.size()<4:
		return phase==SeasonState.Phase.DRAFT and round_index==0 and s.results.is_empty() \
			and s.player_results.is_empty() and p.reports.is_empty() and p.pending.is_empty()
	if not codec.whole(phase,SeasonState.Phase.REGULAR,SeasonState.Phase.COMPLETE) \
			or not codec.whole(round_index,0,10) or s.results.size()!=s.player_results.size()+p.reports.size(): return false
	if not p.pending.is_empty() and (phase==SeasonState.Phase.COMPLETE or not codec.keys(p.pending,["human","reports"]) \
			or not p.pending.human is Array or p.pending.human.size()!=13 \
			or not p.pending.reports is Array or p.pending.reports.size()>5): return false
	match int(phase):
		SeasonState.Phase.REGULAR:
			return round_index<10 and s.results.size()==3*round_index and s.player_results.size()==round_index
		SeasonState.Phase.SEMIFINAL:
			return round_index==10 and s.results.size()==30 and s.player_results.size()==10
		SeasonState.Phase.FINAL:
			return round_index==10 and s.results.size()==32 and s.player_results.size()==11
		SeasonState.Phase.COMPLETE:
			return round_index==10 and s.results.size()==33 and s.player_results.size() in [10,11,12]
	return false

static func payload(owner: RefCounted) -> Dictionary:
	if not draft_scope(owner) and not season_scope(owner) and not rollover_play_scope(owner): return {}
	var api:=_api()
	var native: Script=api.Adapter.Ordinary
	if not native._structural_valid(owner.session): return {}
	var current: Dictionary=native._payload(owner.session)
	return api._pack_fields(owner,current) if not current.is_empty() else {}

static func _next(prior: RefCounted, wire: PackedByteArray, ordered: Array[String], rollover: bool=false) -> RefCounted:
	var api:=_api()
	var ordinal:int=ROLLOVER_ORDINAL if rollover else ORDINAL
	if not (closed_complete_scope(prior) if rollover else api.Mixed.closed_complete_scope(prior)) \
			or ordered.size()!=ordinal-1 \
			or wire.is_empty() or wire.size()>Custody.MAX_WIRE: return null
	var state: Dictionary=prior.ordinary_state
	var lineage: String=state.lineage
	var seed: int=api.Common.starter_dugout_seed(lineage,ordinal)
	var native: Dictionary=api.Common.Start.make(seed,lineage,ordinal,int(prior.common.sequence),
		api.Common.Life.digest(prior.common.aggregate),api.Common.projection(prior))
	var pin:=Base._hash(wire)
	var basis: Dictionary=api.Common.Start.bind_mixed_rollover(native,ordered,pin) if rollover \
		else api.Common.Start.bind_mixed_append(native,ordered,pin)
	var session: SeasonState=api.Adapter.OrdinaryRuntime.create_common(basis,"dugout")
	if session==null: return null
	var next: RefCounted=api._new()
	if not api.Codec.install_fields(next,api.DATA_FIELDS,api.Codec.data_fields(prior,api.DATA_FIELDS)): return null
	next.session=session
	next.ordinary_state={"status":"active","ordinal":ordinal,"lineage":lineage,"last_close":{},"proposal":{},
		"start_basis_sha256":api.Common.Life.digest(basis),
		"last_transition":{"kind":ROLLOVER_TRANSITION if rollover else Custody.TRANSITION,"id":"%s:%d:next"%[lineage,ordinal-1],
			"source":state.last_close.session_sha256,"from":ordinal-1,"to":ordinal,"league":"dugout","basis":basis},
		"archive":{"kind":Custody.KIND,"wire":wire.get_string_from_utf8()}}
	return next if draft_scope(next) and session.phase==SeasonState.Phase.DRAFT and session.picks.is_empty() else null

# Called only by fixed Store CAS with the actual admitted closed owner/raw.
static func from_bound_close(prior: RefCounted, raw: PackedByteArray) -> RefCounted:
	var api:=_api()
	var rollover:=closed_complete_scope(prior)
	if not (rollover or api.Mixed.closed_complete_scope(prior)) or raw.is_empty() or raw.size()>Base.MAX_MEMBER: return null
	var value: Variant=api._bytes_value(raw)
	var native: Script=api.Adapter.Ordinary
	if not native._structural_valid(prior.session) \
			or not api.Codec.same(api._pack_fields(prior,native._payload(prior.session)),value): return null
	var state: Dictionary=prior.ordinary_state
	var prefix: PackedByteArray=state.archive.wire.to_utf8_buffer()
	var pin: String=state.last_transition.basis.archive_origin.manifest_sha256
	var ordered: Array[String]=[]
	ordered.assign(state.last_transition.basis.archive_origin.ordered_sha256)
	var digest:=Base._hash(raw)
	var wire:=Custody.append(prefix,pin,raw,digest)
	ordered.append(digest)
	return _next(prior,wire,ordered,rollover)

static func _one_tail(wire: PackedByteArray, sha256: String) -> PackedByteArray:
	if not Base._sha(sha256) or Base._hash(wire)!=sha256: return PackedByteArray()
	var manifest:=Custody._manifest(wire)
	if manifest.is_empty() or manifest.rows.size()!=1: return PackedByteArray()
	var tail: Array[PackedByteArray]=[]
	var collect:=func(ordinal:int,raw:PackedByteArray) -> bool:
		if ordinal!=ORDINAL-1: return false
		tail.append(raw)
		return true
	return tail[0] if not Custody._walk(manifest,collect).is_empty() and tail.size()==1 else PackedByteArray()

# Obtain row five only after the entire exact two-row ordered transport
# succeeds. Transport visitors do not confer native authority.
static func _rollover_tail(wire: PackedByteArray, sha256: String) -> PackedByteArray:
	if not Base._sha(sha256) or Base._hash(wire)!=sha256: return PackedByteArray()
	var manifest:=Custody._manifest(wire)
	if manifest.is_empty() or manifest.rows.size()!=2: return PackedByteArray()
	var tail: Array[PackedByteArray]=[]
	var seen: Array[int]=[]
	var collect:=func(ordinal:int,raw:PackedByteArray) -> bool:
		if ordinal!=4+seen.size() or ordinal not in [4,5]: return false
		seen.append(ordinal)
		if ordinal==5: tail.append(raw)
		return true
	return tail[0] if not Custody._walk(manifest,collect).is_empty() and seen==[4,5] and tail.size()==1 else PackedByteArray()

static func _restore_rollover(value: Dictionary, state: Dictionary) -> RefCounted:
	if not _rollover_archive_state(state): return null
	var current: Variant=value.get("ordinary")
	if not _current_shape(current): return null
	var played: bool=not current.state.results.is_empty() or not current.physical.pending.is_empty()
	if current.state.picks.size()==4 and not _rollover_phase_shape(current.state.phase,current.state.round_index, \
			current.state.player_results.size(),current.physical.pending,current.build.build._events): return null
	var api:=_api()
	var wire: PackedByteArray=state.archive.wire.to_utf8_buffer()
	var raw:=_rollover_tail(wire,state.last_transition.basis.archive_origin.manifest_sha256)
	if raw.is_empty(): return null
	# The bounded Season 5 reader authenticates S1–S4 and rederives S5's
	# complete Close exactly once. Do not replay S4 separately here.
	var prior:=restore_owned(api._bytes_value(raw))
	if not closed_complete_scope(prior): return null
	var ordered: Array[String]=[]
	ordered.assign(prior.ordinary_state.last_transition.basis.archive_origin.ordered_sha256)
	ordered.append(Base._hash(raw))
	var candidate:=_next(prior,wire,ordered,true)
	if candidate==null: return null
	for id: Variant in current.state.picks:
		if not id is String or not candidate.session.offers().has(id) or not candidate.session.choose_player(id): return null
	if played:
		# Reuse the already compared current image; never accept caller-supplied
		# results or advance an authenticated but still-unpaid pending round.
		var replay: RefCounted=api.Mixed._replay_current_image(candidate.session,current)
		if replay==null: return null
		candidate.session=replay.session
		return candidate if rollover_play_scope(candidate) \
			and api.Codec.same(api._pack_fields(candidate,replay.image),value) else null
	var rebuilt:=payload(candidate)
	return candidate if not rebuilt.is_empty() and api.Codec.same(rebuilt,value) else null

static func restore_owned(value: Variant) -> RefCounted:
	var api:=_api()
	# Inspect outer types first; neither an incoming basis nor a receipt is a
	# trust root. The transport is bounded before accessing its native tail.
	var state: Variant=value.get("ordinary_state") if value is Dictionary else null
	if not state is Dictionary: return null
	if state.get("ordinal")==ROLLOVER_ORDINAL: return _restore_rollover(value,state)
	var closed: bool=state.get("status")=="closed"
	if not _archive_state(state,closed): return null
	var current: Variant=value.get("ordinary")
	if not _current_shape(current): return null
	var drafted: bool=current.state.picks.size()==4
	if closed and (not drafted or current.state.phase!=SeasonState.Phase.COMPLETE): return null
	var wire: PackedByteArray=state.archive.wire.to_utf8_buffer()
	var raw:=_one_tail(wire,state.last_transition.basis.archive_origin.manifest_sha256)
	if raw.is_empty(): return null
	var prior: RefCounted=api.Mixed.restore_owned(api._bytes_value(raw))
	if not api.Mixed.closed_complete_scope(prior): return null
	var ordered: Array[String]=[]
	ordered.assign(prior.ordinary_state.last_transition.basis.archive_origin.ordered_sha256)
	ordered.append(Base._hash(raw))
	var candidate:=_next(prior,wire,ordered)
	if candidate==null: return null
	for id: Variant in current.state.picks:
		if not id is String or not candidate.session.offers().has(id) or not candidate.session.choose_player(id): return null
	if closed:
		# Recreate and authenticate the active terminal session from its own
		# predecessor, then derive the Close metadata and compare the whole
		# envelope. Incoming close/receipt/aggregate are not inputs to the fold.
		var replay: RefCounted=api.Mixed._replay_current_image(candidate.session,current)
		if replay==null: return null
		candidate.session=replay.session
		var folded:=completed_close(candidate,replay.image)
		return folded if folded!=null and api.Codec.same(api._pack_fields(folded,replay.image),value) else null
	if drafted:
		candidate.session=api.Mixed._replay_current(candidate.session,current)
		if not season_scope(candidate): return null
	var rebuilt:=payload(candidate)
	return candidate if not rebuilt.is_empty() and api.Codec.same(rebuilt,value) else null

static func pick_payload(owner: RefCounted, candidate: RefCounted) -> Dictionary:
	if not draft_scope(owner) or not draft_scope(candidate) or owner.session.phase!=SeasonState.Phase.DRAFT \
			or candidate.session.picks.size()!=owner.session.picks.size()+1: return {}
	var replay: SeasonState=owner.session.fork_live()
	var id: String=candidate.session.picks[-1]
	if replay==null or not replay.offers().has(id) or not replay.choose_player(id): return {}
	var api:=_api()
	var data:=payload(candidate)
	return data if not data.is_empty() and api.Codec.same(api.Adapter.Ordinary._payload(replay),data.ordinary) else {}

# Display projection only from Store's previously authenticated archive pin.
static func display(owner: RefCounted) -> Array:
	if not draft_scope(owner) and not season_scope(owner) and not closed_complete_scope(owner) \
			and not rollover_play_scope(owner): return []
	var api:=_api()
	var state: Dictionary=owner.ordinary_state
	var wire: PackedByteArray=state.archive.wire.to_utf8_buffer()
	var pin: String=state.last_transition.basis.archive_origin.manifest_sha256
	if state.ordinal==ROLLOVER_ORDINAL:
		var fifth_raw:=_rollover_tail(wire,pin)
		if fifth_raw.is_empty(): return []
		var fifth_owner: Variant=api._bytes_value(fifth_raw)
		if not fifth_owner is Dictionary: return []
		var archive5: Dictionary=fifth_owner.ordinary_state.archive
		var pin5: String=fifth_owner.ordinary_state.last_transition.basis.archive_origin.manifest_sha256
		var rows5:=_display_four(archive5.wire.to_utf8_buffer(),pin5)
		var fifth: Dictionary=api.Common.StandardArchive._display_scores(fifth_owner,true)
		if rows5.size()!=4 or fifth.get("ordinal")!=ORDINAL or fifth.results.size()!=33: return []
		fifth["league"]="dugout"
		rows5.append(fifth)
		return rows5
	return _display_four(wire,pin)

static func _display_four(wire: PackedByteArray, pin: String) -> Array:
	var api:=_api()
	var raw:=_one_tail(wire,pin)
	if raw.is_empty(): return []
	var prior: Variant=api._bytes_value(raw)
	var rows: Array=api.Mixed.display(prior.ordinary_state.archive)
	var fourth: Dictionary=api.Common.StandardArchive._display_scores(prior,true)
	if rows.size()!=3 or fourth.get("ordinal")!=ORDINAL-1 or fourth.results.size()!=33: return []
	fourth["league"]="dugout"
	rows.append(fourth)
	return rows
