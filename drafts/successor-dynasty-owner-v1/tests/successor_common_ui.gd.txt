func button(node: Node,text: String) -> Button:
	if node is Button and not node.is_queued_for_deletion() and node.text==text:return node
	for child in node.get_children():
		var found: Button=button(child,text)
		if found!=null:return found
	return null

	var control: Button=button(node,text)
	if not check(control!=null and not control.disabled,"actual control: "+text):return false
