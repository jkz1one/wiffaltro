func rewards(season: SeasonState) -> bool:
	for club in range(6):
		var build: SeasonBuild=season.build if club==0 else season.opponents.clubs[str(club)].build
		var expected: Dictionary={}
		for game: Dictionary in season.results:
			if int(game.home)!=club and int(game.away)!=club:continue
			expected[str(int(game.id))]=(int(game.home)==club)==(game.home_runs>game.away_runs)
		if not check(Codec.same(build._bank.view().rewards,expected),"all actual club%d fixture reward identities and wins exactly once"%club):return false
		var seen: Dictionary={}
		for event: Dictionary in build._bank._events:
			if event.op!="reward":continue
			var id: String=str(int(event.game))
			if not check(not seen.has(id) and expected.has(id) and event.win==expected[id],"reward journal matches actual unique fixture"):return false
			seen[id]=true
		if not check(seen.size()==expected.size(),"no missing or duplicate actual reward"):return false
	return true
func terminal(app: SeasonApp) -> bool:
	var season: SeasonState=app.season
	if not check(season.phase==SeasonState.Phase.COMPLETE and season.results.size()==33 and season.player_results.size()==11 and season.physical.reports.size()==22 and season.physical.pending.is_empty(),"genuine elimination fullseason33 with11human22NPC outcomes"):return false
	if not rewards(season):return false
	var final: Dictionary=season.results[-1]
	if not check(int(final.id)==32 and final.neutral and season.champion==SeasonState._winner(final),"actual final winner recorded as champion"):return false
	return check(app.career_owner.status=="active" and app.career_owner.last_close.is_empty() and app.career_owner.proposal.is_empty(),"terminal gameplay only, no career closure/rollover performed")
func actual_final(app: SeasonApp) -> void:

	app.resume_round()

	var receivers=runner.get_signal_connection_list("finished")
	if not check(receivers.size()==1,"one original final receiver"):return

	runner.finished.connect(func(report: Dictionary):preserve_actual("actual120-npc-32",{"report":report,"fixture":projection.fixture,"before":before,"owner_before":envelope}))
	runner.connect("finished",receiver,int(receivers[0].flags))

		if not app._busy:break

	finish_step(app,"actual-final32-pending")
