func save_artifact(path: String,bytes: PackedByteArray) -> bool:
	if FileAccess.file_exists(path):return check(FileAccess.get_file_as_bytes(path)==bytes,"artifact already exact")
	var file=FileAccess.open(path,FileAccess.WRITE)
	if not check(file!=null,"isolated artifact writable"):return false
	file.store_buffer(bytes);file.flush();file.close();return check(FileAccess.get_file_as_bytes(path)==bytes,"artifact exact")
func by_focus(node: Node,key: String) -> Control:
	if node is Control and node.get_meta("lineup_focus","")==key:return node
	for child in node.get_children():
		var found: Control=by_focus(child,key)
		if found!=null:return found
	return null
