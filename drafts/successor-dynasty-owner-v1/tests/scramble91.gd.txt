	return ok

func authentic(season: SeasonState,seed_value: int) -> bool:
	var rules: Dictionary=LeagueRules.capture("scramble",0,seed_value)
	if not check(season.season_seed==seed_value and ClubCareer.same(season.league_rules,rules),"exact root seed binds Scramble"):return false
	var clubs: Array=[season.build]
	for club: Dictionary in season.opponents.clubs.values():clubs.append(club.build)
	for build: SeasonBuild in clubs:
		if not check(ClubCareer.same(build._league_rules,rules) and build._book.context_matches(rules),"original root in each build and book"):return false
		for id: String in SeasonPlayerCatalog.ids():
			var expected: Dictionary=SeasonScramble.profile(id,rules)
			var actual: Dictionary=build.player(id)
			if not ClubCareer.same(actual.base,expected.base) or not ClubCareer.same(actual.stats,expected.stats):
				return check(false,"canonical pristine transformed vector for "+id)
	return check(true,"all drafted clubs authenticate every pristine seeded vector")
