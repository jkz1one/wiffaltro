func controls(node: Node) -> Array:
	var result: Array=[]
	if node is Button: result.append({"kind":"button","text":node.text,"disabled":node.disabled,"tooltip":node.tooltip_text})
	elif node is Label: result.append({"kind":"label","text":node.text})
	for child: Node in node.get_children():
		if not child.is_queued_for_deletion(): result.append_array(controls(child))
	return result
