extends Node
const App=preload("res://src/presentation/season/successor_dynasty_owner_v1/app.gd")
const A=preload("res://src/gameplay/season/successor_dynasty_owner_v1/session_adapter.gd")
const S=preload("res://src/gameplay/season/successor_dynasty_owner_v1/owner_store.gd")
const Builds=preload("res://src/gameplay/season/successor_live_play_v1/build_codec.gd")
const PATH="user://equipment92-detached-policy-only.json"
var roots: Array[Resource]=[]
func check(ok: bool,msg: String) -> bool:
	print(("CHECK: " if ok else "FAIL: ")+msg)
	if not ok:get_tree().quit(1)
	return ok
func _ready() -> void:
	call_deferred("run")
func command(build: SeasonBuild, operation: String, extra: Dictionary={}) -> Dictionary:
	var value: Dictionary={"id":"equipment92:%s:%d"%[operation,build.revision()],"rev":build.revision(),"op":operation}
	value.merge(extra)
	return value
func run() -> void:
	var args: PackedStringArray=OS.get_cmdline_user_args()
	if not check(OS.get_environment("WIFFALTRO_ISOLATED_TEST")=="1" and OS.get_environment("WIFFALTRO_DYNASTY_ENGINEERING")=="1" and args.size()==1 and args[0] in ["birth","cold","next-cold"],"isolated Equipment gate"): return
	var app=App.new();add_child(app);await get_tree().process_frame
	if args[0]=="birth":
		if not check(S.raw().is_empty() and not FileAccess.file_exists(PATH),"no real or preexisting save"): return
		app.select_league("equipment");app.create_fresh(1001)
		if not check(app.season!=null and app._owned() and LeagueRules.equipment(app.season.league_rules),"actual named Equipment owner"): return
		for idx in range(4):
			var offers: Array=app.season.offers()
			if not check(not offers.is_empty(),"actual Equipment offers"): return
			app.draft(offers[0])
			if not check(app.retained==null and app.season.picks.size()==idx+1,"actual Equipment draft persisted"): return
		if not check(app.season.opponents.clubs.size()==5 and A.OrdinaryFixtures.match_for(app.season,app.season.pending_fixture())!=null,"all five AI and first match builds only"): return
		var original: PackedByteArray=S.raw()
		var build: SeasonBuild=app.season.build._fork()
		var bank: SeasonOwnership=build._bank
		# Synthetic rewards, shop and stock only on a detached build.
		for game in range(2):
			if not check(bank.commit(OwnershipFixtures.request(bank,"reward",{"game":game,"win":true})).ok,"synthetic detached funding, not played games"): return
		build._visit={"number":1,"open":false}
		if not check(build.commit(command(build,"open")).ok,"detached between-game shop opens"): return
		build._visit.offers["active-test"]="BALL-MOV-01"
		build._visit.offers["spare-test"]="BALL-HYB-01"
		var cash: int=build.cash()
		if not check(build.commit(command(build,"equip",{"offer":"active-test","replace":""})).ok and build.cash()==cash-10,"active purchase pays authored price"): return
		var active: Dictionary=build._bank.view().gear.ball
		if not check(build.commit(command(build,"spare_buy",{"offer":"spare-test"})).ok,"paid inactive spare purchase"): return
		var wallet: Dictionary=build._bank.view();var spare: Dictionary=wallet.spare
		if not check(wallet.gear.ball==active and wallet.cash==cash-22 and spare.item=="BALL-HYB-01" and spare.paid==12,"one paid spare leaves active receipt unchanged"): return
		if not check(not build.commit(command(build,"spare_swap",{"spare":spare.id,"active":"wrong"})).ok and build._bank.view()==wallet,"wrong receipt rejected atomically"): return
		var packed: Dictionary=Builds.capture(build)
		if not check(not packed.is_empty(),"current materializer captures Equipment wallet"): return
		var out: FileAccess=FileAccess.open(PATH,FileAccess.WRITE)
		if not check(out!=null,"write isolated detached snapshot"): return
		out.store_string(JSON.stringify(packed,"",true,true));out.close()
		if not check(S.raw()==original and app.season.results.is_empty(),"detached equipment left actual branch untouched"): return
	elif args[0]=="cold":
		var original: PackedByteArray=S.raw();app.reload_save()
		if not check(app.season!=null and app._owned() and app.season.picks.size()==4 and S.raw()==original,"actual Equipment draft cold exact"): return
		var build: SeasonBuild=Builds.restore(JSON.parse_string(FileAccess.get_file_as_string(PATH)))
		if not check(build!=null and SeasonEquipment.between_games(build),"detached paid spare cold restores in allowed window"): return
		var before: Dictionary=build._bank.view();var active: Dictionary=before.gear.ball;var spare: Dictionary=before.spare
		if not check(build.commit(command(build,"spare_swap",{"spare":spare.id,"active":active.id})).ok,"exact active/spare swap accepted"): return
		var after: Dictionary=build._bank.view()
		if not check(after.cash==before.cash and after.gear.ball==spare and after.spare==active and S.raw()==original,"swap free and exact receipts exchanged"): return
		app.close_current("abandon");app.propose_next(1002);app.commit_next()
		if not check(app.career_owner.current_ordinal()==2 and app.career_owner.common.sequence==1 and LeagueRules.equipment(app.season.league_rules),"actual shared Equipment close and next"): return
	else:
		var original: PackedByteArray=S.raw();app.reload_save()
		if not check(app.season!=null and app.career_owner.current_ordinal()==2 and LeagueRules.equipment(app.season.league_rules) and S.raw()==original,"actual Equipment Next cold exact"): return
	print("PASS: shared Equipment 92 ",args[0])
	get_tree().quit(0)
