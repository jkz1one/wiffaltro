func check(ok: bool,msg: String) -> bool:
	print(("CHECK: " if ok else "FAIL: ")+msg)
	if not ok:get_tree().quit(1)
	return ok
func _ready() -> void:
	call_deferred("run")
func command(build: SeasonBuild, operation: String, extra: Dictionary={}) -> Dictionary:
	var value: Dictionary={"id":"academy90:%s:%d"%[operation,build.revision()],"rev":build.revision(),"op":operation}
	value.merge(extra)
	return value
func choice(build: SeasonBuild) -> Dictionary:
	for card: String in build._pack_choices():
		var target: Array[Dictionary]=build.targets(card)
		if not target.is_empty():
			return command(build,"pack_pick",{"item":card,"player":target[0].player,"pitch":target[0].pitch,"replace":target[0].replace})
	return {}
