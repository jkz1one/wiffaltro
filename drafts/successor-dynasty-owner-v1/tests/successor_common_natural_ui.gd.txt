func oracle(season: SeasonState,baseline: Dictionary) -> Dictionary:
	if season.phase!=SeasonState.Phase.COMPLETE or season.results.size()!=33 or not season.league_rules.is_empty():return {}
	var rows: Array=season.results.duplicate();rows.sort_custom(func(a,b):return a.id<b.id)
	var scores: Array=[];var regular_wins: int=0;var games: int=0;var wins: int=0;var ours_total: int=0;var theirs_total: int=0
	var finish: String="missed"
	for index in range(33):
		var game: Dictionary=rows[index]
		if game.id!=index or game.away_runs==game.home_runs:return {}
		scores.append([game.id,game.away,game.home,game.away_runs,game.home_runs])
		if game.away!=0 and game.home!=0:continue
		var ours: int=int(game.home_runs if game.home==0 else game.away_runs)
		var theirs: int=int(game.away_runs if game.home==0 else game.home_runs)
		games+=1;wins+=int(ours>theirs);ours_total+=ours;theirs_total+=theirs
		if index<30:regular_wins+=int(ours>theirs)
		elif index<32:finish="semifinal"
		else:finish="champion" if ours>theirs else "runner_up"
	var first: bool=not baseline.titles.get(LeagueCareer.title_key("standard",0),false)
	var award: int={"missed":35,"semifinal":65,"runner_up":95,"champion":180}[finish]
	var bonus: int=int(15*regular_wins/10.0);var first_bonus: int=25 if finish=="champion" and first else 0
	var receipt: Dictionary={"version":1,"finish":finish,"wins":regular_wins,"finish_award":award,"record_bonus":bonus,"first_bonus":first_bonus,"total":award+bonus+first_bonus}
	var proof: Dictionary=ClubSeasonRecord.capture(season)
	if not C.same(proof.scores,scores) or not C.same(ClubSeasonRecord.receipt(proof,first,0),receipt):return {}
	var stat: int=0;var mastery: int=0;var lessons: Array=[]
	for event: Dictionary in season.build._book._events:
		if event.op=="stat":stat+=1
		elif event.op in ["mastery","round_out"]:mastery+=1
		elif event.op=="learn" and not lessons.has(event.target):lessons.append(event.target)
	return {"proof":proof,"receipt":receipt,"first_title":first,"games":games,"wins":wins,"runs_for":ours_total,"runs_against":theirs_total,"stat":stat,"mastery":mastery,"lessons":lessons}
