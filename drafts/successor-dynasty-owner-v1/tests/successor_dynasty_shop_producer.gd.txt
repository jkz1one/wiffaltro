class OpProbe:
	extends RefCounted
	var conversions: int=0
	func _to_string() -> String:
		conversions+=1
		return "pack_open"
func command_for(owner,op: String) -> Dictionary:
	var build: SeasonBuild=owner.session.build
	var rev: int=build.revision()
	var command: Dictionary={"id":"shop83:%s:%d"%[op,rev],"rev":rev,"op":op}
	if op=="pack_pick":
		for item: String in build._pack_choices():
			for target: Dictionary in build.targets(item):
				var choice: Dictionary=command.duplicate(true);choice.item=item;choice.merge(target)
				if build._fork().preview(choice).ok:return choice
		return {}
	return command
func available_op(owner) -> String:
	if not owner.session.build._visit.get("open",false):return "open"
	return "pack_pick" if owner.session.build.pack_pending() else "pack_open"
