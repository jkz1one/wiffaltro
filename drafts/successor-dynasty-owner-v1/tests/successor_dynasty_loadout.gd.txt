func inspect(ui,app,raw: Dictionary,disk: Dictionary) -> bool:
	ui.entry.grab_focus();ui.entry.pressed.emit()
	if not check(ui.shade.visible and ui._binding() and app._loadout_modal==ui,"actual Equipped entry binds current owner"):return false
	if not check(ui.shade is ColorRect and ui.shade.mouse_filter==Control.MOUSE_FILTER_STOP
		and ui.panel.mouse_filter==Control.MOUSE_FILTER_STOP,"original-style shade/panel block underlying pointer input"):return false
	for tab: Button in ui._tab_buttons:
		tab.pressed.emit()
		if not check(not ui.can_sell() and not ui.inventory_sales_enabled(),"inspection never enables legacy inventory writer"):return false
	var ticket=app._lineup_ticket
	app.open_lineup();app.play_next();app.open_pitch_shop()
	app.lineup_swap(0,1,ticket)
	if not check(not app.commit_inventory({"op":"sell"}) and not app.commit_shop({"id":"blocked","rev":app.season.build.revision(),"op":"open"})
		and app.lab==null and app._loadout_modal==ui and Raw.matches(app.career_owner,raw) and exact_disk(disk),"ColorRect modal also blocks underlying command dispatch and legacy inventory persistence"):return false
	ui.close_button.pressed.emit()
	var focus_ok: bool=ui.get_viewport().gui_get_focus_owner()==ui.entry
	var owner_ok: bool=Raw.matches(app.career_owner,raw)
	var disk_ok: bool=exact_disk(disk)
	print("LOADOUT_CLOSE_PREDICATES ",{"hidden":not ui.shade.visible,"modal_cleared":app._loadout_modal==null,
		"focus_ui_viewport":focus_ok,"owner":owner_ok,"disk":disk_ok})
	return check(not ui.shade.visible and app._loadout_modal==null and focus_ok
		and owner_ok and disk_ok,"close restores exact entry focus without owner/save mutation")
