	if not v:get_tree().quit(1)
	return v

	if not check(OS.get_environment("WIFFALTRO_ISOLATED_TEST")=="1","guard before reading"):return
	roots=preload("res://src/gameplay/season/successor_owner_general_v1/codec.gd")._operation_content_roots()

	if not check(not snap.is_empty() and R.matches(owner,snap),"named raw guard captures earned birth"):return
