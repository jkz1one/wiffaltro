func independent_fold(source,expected: Dictionary) -> bool:
	var wins: int=0;var rf: int=0;var ra: int=0
	for row: Dictionary in source.session.player_results:
		var ours: int=int(row.home_runs if row.home==0 else row.away_runs)
		var theirs: int=int(row.away_runs if row.home==0 else row.home_runs)
		wins+=int(ours>theirs);rf+=ours;ra+=theirs
	if not check(expected.games-source.ledger.games==source.session.player_results.size() and expected.wins-source.ledger.wins==wins
		and expected.runs_for-source.ledger.runs_for==rf and expected.runs_against-source.ledger.runs_against==ra and wins==8 and rf==96 and ra==62,"independent actual11game8wins96for62against totals equal fold"):return false
	for index in range(6):
		var key: String="club:%d"%index
		var build: SeasonBuild=source.session.build if index==0 else source.session.opponents.clubs[str(index)].build
		var stats: int=0;var mastery: int=0;var lessons: Array=[];var abilities: int=0
		for event: Dictionary in build._book._events:
			if event.op=="stat":stats+=1
			elif event.op in ["mastery","round_out"]:mastery+=1
			elif event.op=="learn" and not lessons.has(event.target):lessons.append(event.target)
		for id: String in build.roster():
			for receipt: Dictionary in build._abilities.learned.get(id,[]):
				var inherited: bool=false
				for origin: Dictionary in build._dynasty_start.players[id].abilities:
					var old: Dictionary=origin.duplicate(true);old.erase("source_season")
					if C.same(old,receipt):inherited=true
				if not inherited:abilities+=1
		var prior: Dictionary=source.ledger.development[key]
		var actual: Dictionary=expected.development[key]
		var combined: Array=prior.lessons.duplicate()
		for recipe: String in lessons:
			if not combined.has(recipe):combined.append(recipe)
		combined.sort()
		if not check(actual.stat==prior.stat+stats and actual.mastery==prior.mastery+mastery and actual.abilities==prior.abilities+abilities
			and C.same(actual.lessons,combined),"independent current-only growth fold "+key):return false
	return true
